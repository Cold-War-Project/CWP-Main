# The Great Stink: the foresighted government may act before the crisis.
# Taking it gives the later journal entry a head start (gs_survey_head_start in the opening event).

gs_sanitary_survey_decision = {
	is_shown = {
		NOT = { has_variable = gs_survey_done }
		NOT = { has_variable = gs_stink_done }
		NOT = { has_journal_entry = je_gs_great_stink }
		has_technology_researched = urban_planning
		capital = {
			any_scope_building = {
				is_building_type = building_urban_center
				level >= 10
			}
		}
	}

	possible = {
		is_at_war = no
		in_default = no
	}

	when_taken = {
		set_variable = gs_survey_done
		add_modifier = {
			name = gs_sanitary_survey
			multiplier = money_amount_multiplier_very_small
			days = short_modifier_time
		}
		custom_tooltip = gs_sanitary_survey_decision_tt
	}

	ai_chance = {
		value = 5
		if = {
			limit = { taking_loans = yes }
			multiply = 0
		}
	}
}
