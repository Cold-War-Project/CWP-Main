# The Great Stink: buttons on je_gs_great_stink. root = country.

# Repeatable. The cooldown equals the length of the loan modifier, and `possible` refuses a second loan.
gs_vote_funds_button = {
	name = "gs_vote_funds_button"
	desc = "gs_vote_funds_button_desc"

	visible = { always = yes }

	possible = {
		in_default = no
		NOT = { has_modifier = gs_sewer_loan }
	}

	effect = {
		add_modifier = {
			name = gs_sewer_loan
			multiplier = money_amount_multiplier_medium
			days = short_modifier_time
		}
		custom_tooltip = gs_bar_plus_5_tt
		custom_tooltip = gs_bar_funded_tt
		je:je_gs_great_stink ?= {
			add_progress = { value = 5 name = gs_sewer_bar }
		}
	}

	cooldown = { days = short_modifier_time }

	ai_chance = {
		value = 10
		if = {
			limit = { taking_loans = yes }
			multiply = 0
		}
		if = {
			limit = { is_at_war = yes }
			multiply = 0
		}
	}
}

# Once. Disappears after use. The requirement is a small goal: the know-how, or a government that backs reform.
gs_appoint_engineer_button = {
	name = "gs_appoint_engineer_button"
	desc = "gs_appoint_engineer_button_desc"

	visible = {
		NOT = { has_variable = gs_engineer_appointed }
	}

	possible = {
		OR = {
			has_technology_researched = modern_sewerage
			ig:ig_intelligentsia ?= { is_in_government = yes }
			ig:ig_industrialists ?= { is_in_government = yes }
		}
	}

	effect = {
		set_variable = gs_engineer_appointed
		custom_tooltip = gs_engineer_appointed_tt
		ig:ig_petty_bourgeoisie ?= {
			add_modifier = {
				name = gs_vestries_overruled
				days = normal_modifier_time
			}
		}
	}

	ai_chance = {
		value = 100 # the works crawl without it; use as soon as available
	}
}
