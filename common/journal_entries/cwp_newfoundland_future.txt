# Cold War Project: Newfoundland's future after the Commission of Government.
# The National Convention debates, the bar follows interest groups, living standards and campaigning,
# and the referendum decides between Confederation with Canada and Responsible Government.

je_cwp_newfoundland_future = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_internal_affairs

	activation_description = { always = yes }

	is_shown_when_inactive = {
		cwp_nf_is_candidate = yes
	}

	possible = {
		cwp_nf_is_candidate = yes
	}

	immediate = {
		set_variable = { name = cwp_nf_vote_current_var value = 38 }
		trigger_event = { id = cwp_nf.1 popup = yes }
	}

	scripted_progress_bar = cwp_nf_confederation_bar
	scripted_progress_bar = cwp_nf_convention_bar

	scripted_button = cwp_nf_smallwood_button
	scripted_button = cwp_nf_responsible_government_button
	scripted_button = cwp_nf_delegations_button

	on_monthly_pulse = {
		effect = {
			set_variable = { name = cwp_nf_vote_current_var value = "scope:journal_entry.scripted_bar_progress(cwp_nf_confederation_bar)" }
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(cwp_nf_convention_bar)" >= 100
					}
					NOT = { has_variable = cwp_nf_vote_pending_var }
				}
				set_variable = cwp_nf_vote_pending_var
				trigger_event = { id = cwp_nf.2 days = 30 popup = yes }
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_newfoundland_future_status_confederation
				trigger = { var:cwp_nf_vote_current_var > 50 }
			}
			desc = je_cwp_newfoundland_future_status_responsible
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_cwp_newfoundland_future_settled_desc
		color = "gold_positive"
		trigger = {
			has_variable = cwp_nf_settled_var
		}
		effect = {
			remove_variable = cwp_nf_vote_current_var
			remove_variable = cwp_nf_campaign_var
			remove_variable = cwp_nf_swing_var
			remove_variable = cwp_nf_vote_pending_var
			remove_variable = cwp_nf_delegation_var
		}
	}

	# Independence, annexation or Canada's collapse took the question away.
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_newfoundland_future_settled_desc
		trigger = {
			NOT = { has_variable = cwp_nf_settled_var }
			OR = {
				is_subject = no
				NOT = { exists = c:CAN }
			}
		}
		effect = {
			set_variable = cwp_nf_settled_var
		}
	}

	weight = 1000
	transferable = no
	can_revolution_inherit = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
