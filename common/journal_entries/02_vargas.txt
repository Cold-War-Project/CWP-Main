	group = je_group_brazil
	dlc = dlc008

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		c:BRZ ?= this
		OR = {
			has_technology_researched = socialism
			has_technology_researched = anarchism
			has_technology_researched = corporatism
		}
		has_dlc_feature = rp1_content
	}

	possible = {
		has_technology_researched = political_agitation
		ig:ig_landowners ?= {
			is_in_government = yes
			trigger_if = {
				limit = {
					is_in_government = yes
				}
				is_strongest_ig_in_government = yes
			}
		}
		ig:ig_petty_bourgeoisie ?= {
			ig_counts_as_marginal = no
		}
		OR = {
			AND = {
				has_law = law_type:law_monarchy
				custom_tooltip = {
					text = completed_coffee_and_milk_je_tt
					OR = {
						always = no
						always = no
					}
				}
			}
			has_law_or_variant = law_type:law_presidential_republic
			has_law_or_variant = law_type:law_parliamentary_republic
		}
		OR = {
			has_law_or_variant = law_type:law_wealth_voting
			has_law_or_variant = law_type:law_landed_voting
		}
	}

	scripted_progress_bar = populist_unrest_progress_bar

	immediate = {
		ig:ig_landowners ?= {
			save_scope_as = landowners_ig
		}
		ig:ig_petty_bourgeoisie ?= {
			save_scope_as = pb_ig
		}
		set_populism_vars = yes
		show_as_tooltip = {
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_modernizer
						}
					}
				}
				create_political_movement = { type = movement_modernizer }
			}
		}
		trigger_event = { id = BRZ_populism.1 days = 1 popup = yes }
	}

	on_weekly_pulse = {
		effect = {
			calculate_populism_progress = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			100 = 0
			10 = BRZ_populism.2
			10 = BRZ_populism.4
			10 = BRZ_populism.5
			10 = BRZ_populism.6
			100 = BRZ_populism.7
			10 = BRZ_populism.8
			10 = BRZ_populism.9
			10 = bp1_misc.2
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = BRZ_populism_100_tt
				scope:journal_entry = {
					"scripted_bar_progress(populist_unrest_progress_bar)" >= 100
				}
			}
		}
		effect = {
			clear_populism_vars = yes
			trigger_event = { id = BRZ_vargas.1 popup = yes }
			show_as_tooltip = {
				activate_law = law_type:law_autocracy
				if = {
					limit = {
						has_law = law_type:law_monarchy
					}
					activate_law = law_type:law_social_monarchy
				}
				else_if = {
					limit = {
						NOR = {
							has_law = law_type:law_monarchy
							has_law = law_type:law_social_monarchy
							has_law = law_type:law_presidential_republic
							has_law = law_type:law_corporate_state
						}
					}
					activate_law = law_type:law_presidential_republic
				}
				custom_tooltip = {
					text = add_je_new_republic_tt
					add_journal_entry = {
						type = je_new_republic
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = BRZ_populism_0_tt
				scope:journal_entry = {
					"scripted_bar_progress(populist_unrest_progress_bar)" <= 0
				}
			}
		}
		effect = {
			clear_populism_vars = yes
			trigger_event = { id = BRZ_populism.3 popup = yes }
		}
	}

	weight = 100


	should_be_pinned_by_default_uninvolved_or_context = yes
}

	}

	immediate = {

			limit = {
				NOR = {

			}
			ruler = {
				save_scope_as = vargas_scope
				set_character_immortal = yes
				interest_group ?= {
					save_scope_as = pb_scope
				}
			}
		}
		else = {
			ig:ig_petty_bourgeoisie ?= {
				save_scope_as = pb_scope
				leader = {
					save_scope_as = vargas_scope
					set_character_immortal = yes
				}
			}
		}
		root = {
			save_scope_as = brazil_scope
		}
	}

	on_monthly_pulse = {
		random_events = {
			100 = 0
			10 = BRZ_vargas.4
			10 = BRZ_vargas.5
			10 = BRZ_vargas.6
			10 = BRZ_vargas.7
			10 = BRZ_vargas.8
		}
	}

			trigger_if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_social_monarchy

				}

					text = vargas_not_pm_tt

							is_in_government = yes
						}
					}

			}

				custom_tooltip = {
					text = vargas_out_tt
					always = no

						ruler = {
							interest_group ?= {
								is_interest_group_type = ig_petty_bourgeoisie
							}
						}
					}
				}
			}
		}
		effect = {
			trigger_event = { id = BRZ_vargas.3 popup = yes }
			hidden_effect = {
				scope:vargas_scope = {
					set_character_immortal = no
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = all_socialists_marginalized_tt
				NOT = {
					any_interest_group = {
						AND = {
							ig_counts_as_marginal = no
							leader = {
								has_socialist_ideology = yes
							}
						}
					}
				}
			}
			OR = {
				has_law_or_variant = law_type:law_regulatory_bodies
				has_law_or_variant = law_type:law_worker_protections
			}
			has_law_or_variant = law_type:law_compulsory_primary_school
			has_law_or_variant = law_type:law_censorship
			has_law_or_variant = law_type:law_autocracy

					}
				}

			}
			trigger_if = {
				limit = {
					NOR = {
						has_law_or_variant = law_type:law_monarchy
						has_law_or_variant = law_type:law_social_monarchy
					}
				}
				has_law_or_variant = law_type:law_corporate_state
			}
		}
		effect = {
			trigger_event = { id = BRZ_vargas.2 popup = yes }
			hidden_effect = {
				scope:vargas_scope = {
					set_character_immortal = no
				}
			}
		}
	}

	weight = 100

		
	should_be_pinned_by_default_uninvolved_or_context = yes
