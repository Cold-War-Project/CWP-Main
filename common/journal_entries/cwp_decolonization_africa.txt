# Cold War Project: decolonization in Africa (design: docs/agents/areas/decolonization.md)
# je_cwp_gold_coast         Britain and the Gold Coast: the negotiated road
# je_cwp_kenya              Britain, the Kenya settlers and Land and Freedom
# je_cwp_belgian_congo      Belgium and the Congo: preparation or a rushed exit
# je_cwp_portuguese_africa  the Estado Novo and its overseas provinces
# Contextless; each is opened by cwp_decolonization_monthly_pulse when its cwp_<x>_can_open trigger holds,
# sets cwp_<x>_active in immediate and clears it in every outcome. No outcome is scheduled by date.

### Gold Coast

je_cwp_gold_coast = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:GBR ?= this
			c:GLC ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_africa }
	}

	immediate = {
		set_global_variable = cwp_gold_active
		c:GBR ?= { trigger_event = { id = cwp_africa.1 popup = yes } }
		c:GLC ?= { trigger_event = { id = cwp_africa.1 popup = yes } }
	}

	# No bar: three milestones (constitution, a won election, the Asian precedent), then Britain decides.
	current_value = {
		value = 0
		if = {
			limit = { has_global_variable = cwp_gold_constitution_adopted }
			add = 1
		}
		if = {
			limit = { has_global_variable = cwp_gold_election_won }
			add = 1
		}
		if = {
			limit = {
				c:HND ?= { is_subject = no }
				has_global_variable = cwp_tide_tier
				global_var:cwp_tide_tier >= 2
			}
			add = 1
		}
	}
	goal_add_value = {
		value = 3
	}
	progressbar = yes

	scripted_button = cwp_gold_constitution_button
	scripted_button = cwp_gold_election_button
	scripted_button = cwp_gold_detentions_button
	scripted_button = cwp_gold_independence_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_gold_coast_status_ready
				trigger = { cwp_gold_milestones_met = yes }
			}
			triggered_desc = {
				desc = je_cwp_gold_coast_status_campaign
				trigger = { has_global_variable = cwp_gold_campaign_running }
			}
			triggered_desc = {
				desc = je_cwp_gold_coast_status_waiting
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_gold_coast_independence_desc
		trigger = {
			c:GLC ?= { is_subject_of = c:GBR }
			has_global_variable = cwp_gold_independence_granted
		}
		effect = {
			remove_global_variable = cwp_gold_active
			c:GLC = { cwp_decolonization_negotiated_independence = { METROPOLE = c:GBR } }
			c:GBR ?= { trigger_event = { id = cwp_africa.2 popup = yes } }
			c:GLC ?= { trigger_event = { id = cwp_africa.2 popup = yes } }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			OR = {
				NOT = { exists = c:GLC }
				NOT = { exists = c:GBR }
				c:GLC ?= { NOT = { is_subject_of = c:GBR } }
			}
		}
		effect = {
			remove_global_variable = cwp_gold_active
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}

### Kenya

je_cwp_kenya = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:GBR ?= this
			c:KKY ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_africa }
	}

	immediate = {
		set_global_variable = cwp_kenya_active
	}

	# One bar, settler power. The rising is Kenya's own liberty desire: it breaks out at 40 and becomes the
	# engine's independence war at 90; three Emergencies break the forest fighters, but not the politics.
	scripted_progress_bar = cwp_kenya_settler_bar

	scripted_button = cwp_kenya_emergency_button
	scripted_button = cwp_kenya_land_reform_button
	scripted_button = cwp_kenya_conference_button
	scripted_button = cwp_kenya_forests_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_kenya_status_crushed
				trigger = { has_global_variable = cwp_kenya_rising_crushed }
			}
			triggered_desc = {
				desc = je_cwp_kenya_status_rising
				trigger = { has_global_variable = cwp_kenya_rising }
			}
			triggered_desc = {
				desc = je_cwp_kenya_status_quiet
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_variable = cwp_kenya_rising }
					c:KKY ?= { liberty_desire >= 40 }
				}
				set_global_variable = cwp_kenya_rising
				c:GBR ?= { trigger_event = { id = cwp_africa.10 popup = yes } }
				c:KKY ?= { trigger_event = { id = cwp_africa.10 popup = yes } }
			}
			if = {
				limit = {
					has_global_variable = cwp_kenya_rising
					NOT = { has_global_variable = cwp_kenya_rising_crushed }
					has_global_variable = cwp_kenya_emergencies
					global_var:cwp_kenya_emergencies >= 3
				}
				set_global_variable = cwp_kenya_rising_crushed
				c:KKY ?= { add_liberty_desire = -20 }
				c:GBR ?= {
					cwp_decolonization_add_setback = yes
					trigger_event = { id = cwp_africa.11 popup = yes }
				}
			}
		}
	}

	# Lancaster House: majority rule and independence
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_kenya_independence_desc
		trigger = {
			c:KKY ?= { is_subject_of = c:GBR }
			has_global_variable = cwp_kenya_conference_held
		}
		effect = {
			remove_global_variable = cwp_kenya_active
			c:KKY = { cwp_decolonization_negotiated_independence = { METROPOLE = c:GBR } }
			c:GBR ?= { trigger_event = { id = cwp_africa.12 popup = yes } }
			c:KKY ?= { trigger_event = { id = cwp_africa.12 popup = yes } }
		}
	}

	# The forest war (the engine's independence war at liberty desire 90) drives Britain out
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_kenya_won_desc
		trigger = {
			c:KKY ?= { is_subject = no }
		}
		effect = {
			remove_global_variable = cwp_kenya_active
			c:GBR ?= { trigger_event = { id = cwp_africa.13 popup = yes } }
			c:KKY ?= { trigger_event = { id = cwp_africa.13 popup = yes } }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			OR = {
				NOT = { exists = c:KKY }
				NOT = { exists = c:GBR }
				c:KKY ?= {
					is_subject = yes
					NOT = { is_subject_of = c:GBR }
				}
			}
		}
		effect = {
			remove_global_variable = cwp_kenya_active
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}

### Belgian Congo

je_cwp_belgian_congo = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:BEL ?= this
			c:CNG ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_africa }
	}

	immediate = {
		set_global_variable = cwp_congo_active
		set_global_variable = { name = cwp_congo_cadres value = 0 }
		c:CNG ?= { add_modifier = { name = cwp_congo_paternalism } }
	}

	# No bar: a race. The counter is how much Belgium has prepared a Congolese administration; the clock is the
	# outside world (Ghana free, the tide, the Congo's own liberty desire), which decides when Belgium must go.
	current_value = {
		value = 0
		if = {
			limit = { has_global_variable = cwp_congo_cadres }
			add = global_var:cwp_congo_cadres
		}
	}
	goal_add_value = {
		value = 3
	}
	progressbar = yes

	scripted_button = cwp_congo_cadres_button
	scripted_button = cwp_congo_force_publique_button
	scripted_button = cwp_congo_round_table_button
	scripted_button = cwp_congo_riots_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_belgian_congo_status_awake
				trigger = { NOT = { c:CNG ?= { has_modifier = cwp_congo_paternalism } } }
			}
			triggered_desc = {
				desc = je_cwp_belgian_congo_status_paternalism
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_global_variable = cwp_tide_tier
					global_var:cwp_tide_tier >= 2
					c:CNG ?= { has_modifier = cwp_congo_paternalism }
				}
				c:CNG = { remove_modifier = cwp_congo_paternalism }
				c:BEL ?= { trigger_event = { id = cwp_africa.20 popup = yes } }
				c:CNG ?= { trigger_event = { id = cwp_africa.20 popup = yes } }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_belgian_congo_independence_desc
		trigger = {
			c:CNG ?= { is_subject_of = c:BEL }
			has_global_variable = cwp_congo_round_table_held
		}
		effect = {
			remove_global_variable = cwp_congo_active
			c:CNG = { remove_modifier = cwp_congo_paternalism }
			cwp_congo_independence_effect = yes
			c:BEL ?= { trigger_event = { id = cwp_africa.21 popup = yes } }
			c:CNG ?= { trigger_event = { id = cwp_africa.21 popup = yes } }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			OR = {
				NOT = { exists = c:CNG }
				NOT = { exists = c:BEL }
				c:CNG ?= { NOT = { is_subject_of = c:BEL } }
			}
		}
		effect = {
			remove_global_variable = cwp_congo_active
			c:CNG ?= { remove_modifier = cwp_congo_paternalism }
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}

### Portuguese Africa

je_cwp_portuguese_africa = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:POR ?= this
			c:OVM ?= this
			c:MKU ?= this
			c:GNB ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_africa }
	}

	immediate = {
		set_global_variable = cwp_portugal_active
		c:POR ?= { trigger_event = { id = cwp_africa.30 popup = yes } }
	}

	# One bar, Portugal's exhaustion, fed by every colony actually in revolt (liberty desire 50+). The liberation
	# wars themselves are the colonies' liberty desire, raised by bases abroad (cwp_portugal_fronts_pulse).
	scripted_progress_bar = cwp_portugal_exhaustion_bar

	scripted_button = cwp_portugal_counter_insurgency_button
	scripted_button = cwp_portugal_colonatos_button
	scripted_button = cwp_portugal_decolonize_button

	on_monthly_pulse = {
		effect = {
			cwp_portugal_fronts_pulse = yes
		}
	}

	# Lisbon lets the provinces go
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_portuguese_africa_decolonized_desc
		trigger = {
			cwp_portugal_holds_africa = yes
			c:POR ?= { has_variable = cwp_portugal_decolonize }
		}
		effect = {
			remove_global_variable = cwp_portugal_active
			c:POR = { remove_variable = cwp_portugal_decolonize }
			cwp_portugal_release_effect = yes
			c:POR ?= {
				trigger_event = { id = cwp_africa.31 popup = yes }
				add_modifier = {
					name = cwp_end_of_empire
					years = 5
					is_decaying = yes
				}
			}
		}
	}

	# The colonial wars bring down the regime
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_portuguese_africa_revolution_desc
		trigger = {
			cwp_portugal_holds_africa = yes
			NOT = { c:POR ?= { has_variable = cwp_portugal_decolonize } }
			scope:journal_entry = { "scripted_bar_progress(cwp_portugal_exhaustion_bar)" >= 100 }
		}
		effect = {
			remove_global_variable = cwp_portugal_active
			c:POR = {
				add_modifier = {
					name = cwp_portugal_carnation
					years = 5
					is_decaying = yes
				}
				add_radicals = {
					value = 0.15
					interest_group = ig:ig_armed_forces
				}
				trigger_event = { id = cwp_africa.32 popup = yes }
			}
			cwp_portugal_release_effect = yes
		}
	}

	# The colonies have won their own wars
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_portuguese_africa_won_desc
		trigger = {
			custom_tooltip = {
				text = cwp_portugal_no_colonies_tt
				NOT = { cwp_portugal_holds_africa = yes }
			}
		}
		effect = {
			remove_global_variable = cwp_portugal_active
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			NOT = { exists = c:POR }
		}
		effect = {
			remove_global_variable = cwp_portugal_active
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}
