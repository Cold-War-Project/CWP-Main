# Cold War Project: decolonization (design: docs/agents/areas/decolonization.md)
# je_cwp_winds_of_change           the global anti-colonial tide (proposal option C)
# je_cwp_india_transfer_of_power   India and the partition question (option B)
# je_cwp_indonesian_revolution     the Netherlands and the Republic of Indonesia (option B)
# je_cwp_indochina_war             France, the Viet Minh and the superpowers (option B)
# je_cwp_algerian_question         France and Algeria (option B)
# Contextless JEs do not open from `possible` alone: cwp_decolonization_monthly_pulse
# (common/on_actions/cwp_decolonization_on_actions.txt) adds each one with add_contextless_journal_entry.
# No outcome is scheduled by date. Every one is reached through the bars, the buttons, the colony's liberty
# desire or a war.

### The Winds of Change

je_cwp_winds_of_change = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:USA ?= this
			c:RUS ?= this
			has_variable = cwp_colony
			AND = {
				has_variable = cwp_colonial_power
				cwp_has_tracked_colonies = yes
			}
		}
	}

	should_show_when_not_involved = {
		has_variable = cwp_former_colony
	}

	immediate = {
		set_global_variable = cwp_woc_active
		cwp_decolonization_init_globals = yes
		cwp_decolonization_register_colonies = yes
	}

	scripted_progress_bar = cwp_tide_bar

	# The tally of new nations, against the colonies there were in 1946.
	current_value = {
		value = 0
		if = {
			limit = { has_global_variable = cwp_free_former_colonies }
			add = global_var:cwp_free_former_colonies
		}
	}
	goal_add_value = {
		value = 0
		if = {
			limit = { has_global_variable = cwp_colonies_at_start }
			add = global_var:cwp_colonies_at_start
		}
		min = 1
	}
	progressbar = yes

	scripted_button = cwp_woc_champion_self_determination_button
	scripted_button = cwp_woc_guard_allied_empires_button
	scripted_button = cwp_woc_development_funds_button
	scripted_button = cwp_woc_promise_self_government_button
	scripted_button = cwp_woc_hold_the_line_button
	scripted_button = cwp_woc_managed_withdrawal_button
	scripted_button = cwp_woc_nationalist_campaign_button
	scripted_button = cwp_woc_appeal_to_superpowers_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_winds_of_change_status_4
				trigger = { global_var:cwp_tide_tier >= 4 }
			}
			triggered_desc = {
				desc = je_cwp_winds_of_change_status_3
				trigger = { global_var:cwp_tide_tier >= 3 }
			}
			triggered_desc = {
				desc = je_cwp_winds_of_change_status_2
				trigger = { global_var:cwp_tide_tier >= 2 }
			}
			triggered_desc = {
				desc = je_cwp_winds_of_change_status_1
				trigger = { global_var:cwp_tide_tier >= 1 }
			}
			triggered_desc = {
				desc = je_cwp_winds_of_change_status_0
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			cwp_decolonization_monthly_update = yes
		}
	}

	# Ends only when no colony is left for the tide to act on.
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_winds_of_change_end_desc
		trigger = {
			custom_tooltip = {
				text = je_cwp_winds_of_change_end_tt
				has_global_variable = cwp_tide_tier
				NOT = {
					any_country = {
						OR = {
							has_variable = cwp_colony
							cwp_qualifies_as_colony = yes
						}
					}
				}
			}
		}
		effect = {
			remove_global_variable = cwp_woc_active
			set_global_variable = cwp_woc_concluded
			custom_tooltip = je_cwp_winds_of_change_end_effect_tt
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}

### India: the Transfer of Power
# One bar, communal tension (a real continuous quantity the game has no number for). Independence itself reads
# India's liberty desire: Britain sets a date once India is restless enough, or cannot hold on at 85.

je_cwp_india_transfer_of_power = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:GBR ?= this
			c:HND ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_india }
	}

	immediate = {
		set_global_variable = cwp_india_active
		c:GBR ?= { trigger_event = { id = cwp_india.1 popup = yes } }
		c:HND ?= { trigger_event = { id = cwp_india.1 popup = yes } }
	}

	scripted_progress_bar = cwp_india_communal_bar

	scripted_button = cwp_india_cabinet_mission_button
	scripted_button = cwp_india_restore_order_button
	scripted_button = cwp_india_set_date_button
	scripted_button = cwp_india_quit_india_button
	scripted_button = cwp_india_unity_talks_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_india_transfer_of_power_status_date
				trigger = { c:GBR ?= { has_variable = cwp_india_withdrawal_announced } }
			}
			triggered_desc = {
				desc = je_cwp_india_transfer_of_power_status_partition
				trigger = { scope:journal_entry = { "scripted_bar_progress(cwp_india_communal_bar)" >= 45 } }
			}
			triggered_desc = {
				desc = je_cwp_india_transfer_of_power_status_united
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_variable = cwp_india_direct_action_fired }
					scope:journal_entry = { "scripted_bar_progress(cwp_india_communal_bar)" >= 65 }
				}
				set_global_variable = cwp_india_direct_action_fired
				c:HND ?= { trigger_event = { id = cwp_india.2 popup = yes } }
			}
			if = {
				limit = {
					NOT = { has_global_variable = cwp_india_naval_mutiny_fired }
					c:GBR ?= { NOT = { has_variable = cwp_india_withdrawal_announced } }
					c:HND ?= { liberty_desire >= 70 }
				}
				set_global_variable = cwp_india_naval_mutiny_fired
				c:HND ?= { trigger_event = { id = cwp_india.3 popup = yes } }
			}
		}
	}

	# A united India
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_india_united_desc
		trigger = {
			custom_tooltip = {
				text = cwp_india_still_british_tt
				c:HND ?= { is_subject_of = c:GBR }
			}
			cwp_india_transfer_due = yes
			custom_tooltip = {
				text = cwp_india_communal_below_45_tt
				scope:journal_entry = { "scripted_bar_progress(cwp_india_communal_bar)" < 45 }
			}
		}
		effect = {
			remove_global_variable = cwp_india_active
			c:HND = { cwp_decolonization_negotiated_independence = { METROPOLE = c:GBR } }
			c:GBR ?= { trigger_event = { id = cwp_india.10 popup = yes } }
			c:HND ?= { trigger_event = { id = cwp_india.10 popup = yes } }
		}
	}

	# Partition
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_india_partition_desc
		trigger = {
			custom_tooltip = {
				text = cwp_india_still_british_tt
				c:HND ?= { is_subject_of = c:GBR }
			}
			cwp_india_transfer_due = yes
			custom_tooltip = {
				text = cwp_india_communal_45_tt
				scope:journal_entry = { "scripted_bar_progress(cwp_india_communal_bar)" >= 45 }
			}
		}
		effect = {
			remove_global_variable = cwp_india_active
			c:HND = { cwp_decolonization_negotiated_independence = { METROPOLE = c:GBR } }
			cwp_india_partition_effect = yes
			c:GBR ?= { trigger_event = { id = cwp_india.11 popup = yes } }
			c:HND ?= { trigger_event = { id = cwp_india.11 popup = yes } }
			c:PAK ?= { trigger_event = { id = cwp_india.11 popup = yes } }
		}
	}

	# Independence won against Britain
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_india_won_desc
		trigger = {
			custom_tooltip = {
				text = cwp_india_independent_tt
				c:HND ?= { is_subject = no }
			}
		}
		effect = {
			remove_global_variable = cwp_india_active
			if = {
				limit = { scope:journal_entry = { "scripted_bar_progress(cwp_india_communal_bar)" >= 45 } }
				cwp_india_partition_effect = yes
			}
			c:GBR ?= { trigger_event = { id = cwp_india.12 popup = yes } }
			c:HND ?= { trigger_event = { id = cwp_india.12 popup = yes } }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			custom_tooltip = {
				text = cwp_india_invalid_tt
				OR = {
					NOT = { exists = c:HND }
					NOT = { exists = c:GBR }
					c:HND ?= {
						is_subject = yes
						NOT = { is_subject_of = c:GBR }
					}
				}
			}
		}
		effect = {
			remove_global_variable = cwp_india_active
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}

### Indonesia: the National Revolution
# No bar: a mission, then a button. The Republic must survive on Java and Sumatra, Washington must lean on
# The Hague and the guns must fall silent; then the Netherlands can call the Round Table. The war itself is
# fought by the engine, with war support shaped by cwp_decolonization_war_support.
# 1946 start: the Republic (IDS) is independent and at war with the Netherlands, which holds the outer islands.

je_cwp_indonesian_revolution = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:NET ?= this
			c:IDS ?= this
			c:USA ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_south_east_asia }
	}

	immediate = {
		set_global_variable = cwp_indonesia_active
		c:NET ?= { trigger_event = { id = cwp_indonesia.1 popup = yes } }
		c:IDS ?= { trigger_event = { id = cwp_indonesia.1 popup = yes } }
	}

	current_value = {
		value = 0
		if = {
			limit = { cwp_indonesia_republic_holds = yes }
			add = 1
		}
		if = {
			limit = { has_global_variable = cwp_indonesia_us_pressure_applied }
			add = 1
		}
		if = {
			limit = { cwp_indonesia_ceasefire = yes }
			add = 1
		}
		if = {
			limit = {
				c:NET ?= { has_variable = cwp_indonesia_police_action_taken }
				has_global_variable = cwp_indonesia_war_months
				global_var:cwp_indonesia_war_months >= 18
			}
			add = 1
		}
	}
	goal_add_value = {
		value = 4
	}
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			c:IDS ?= {
				if = {
					limit = { NOT = { has_modifier = cwp_guerrilla_war } }
					add_modifier = { name = cwp_guerrilla_war }
				}
			}
			# Months of actual fighting: the Round Table needs the war to have been fought.
			if = {
				limit = { c:NET ?= { has_war_with = c:IDS } }
				if = {
					limit = { NOT = { has_global_variable = cwp_indonesia_war_months } }
					set_global_variable = { name = cwp_indonesia_war_months value = 0 }
				}
				change_global_variable = { name = cwp_indonesia_war_months add = 1 }
			}
		}
	}

	scripted_button = cwp_indonesia_police_action_button
	scripted_button = cwp_indonesia_talks_button
	scripted_button = cwp_indonesia_round_table_button
	scripted_button = cwp_indonesia_guerrilla_button
	scripted_button = cwp_indonesia_diplomacy_button
	scripted_button = cwp_indonesia_marshall_pressure_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_indonesian_revolution_status_war
				trigger = { c:IDS ?= { is_at_war = yes } }
			}
			triggered_desc = {
				desc = je_cwp_indonesian_revolution_status_truce
			}
		}
	}

	# The Dutch hold nothing in Indonesia any more: by war, or by the Round Table.
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_indonesia_independence_desc
		trigger = {
			custom_tooltip = {
				text = cwp_indonesia_independent_tt
				c:IDS ?= { is_subject = no }
			}
			custom_tooltip = {
				text = cwp_indonesia_no_dutch_states_tt
				NOT = { cwp_dutch_hold_indonesia = yes }
			}
		}
		effect = {
			remove_global_variable = cwp_indonesia_active
			c:IDS ?= { remove_modifier = cwp_guerrilla_war }
			c:IDS = {
				set_variable = cwp_former_colony
				if = {
					limit = { NOT = { has_modifier = cwp_new_nation } }
					add_modifier = {
						name = cwp_new_nation
						years = 10
						is_decaying = yes
					}
				}
			}
			change_global_variable = { name = cwp_tide_new_independences add = 1 }
			c:NET ?= { trigger_event = { id = cwp_indonesia.10 popup = yes } }
			c:IDS ?= { trigger_event = { id = cwp_indonesia.10 popup = yes } }
		}
	}

	# The Republic is broken: made a Dutch dominion by the war, or destroyed.
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_cwp_indonesia_restoration_desc
		trigger = {
			custom_tooltip = {
				text = cwp_indonesia_dutch_restored_tt
				OR = {
					NOT = { exists = c:IDS }
					c:IDS ?= { is_subject_of = c:NET }
				}
			}
		}
		effect = {
			remove_global_variable = cwp_indonesia_active
			c:IDS ?= { remove_modifier = cwp_guerrilla_war }
			set_global_variable = {
				name = cwp_indonesia_je_cooldown
				years = 10
			}
			c:NET = {
				cwp_decolonization_add_setback = yes
				trigger_event = { id = cwp_indonesia.11 popup = yes }
			}
			c:IDS ?= { trigger_event = { id = cwp_indonesia.11 popup = yes } }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			custom_tooltip = {
				text = cwp_indonesia_invalid_tt
				OR = {
					NOT = { exists = c:NET }
					c:IDS ?= {
						is_subject = yes
						NOT = { is_subject_of = c:NET }
					}
				}
			}
		}
		effect = {
			remove_global_variable = cwp_indonesia_active
			c:IDS ?= { remove_modifier = cwp_guerrilla_war }
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}

### Indochina: the war
# No bar: territory and war support (vanilla je_the_first_carlist_war for the people's war phase).
# 1946 start: DAI independent in Tonkin (or a French dominion after the March 1946 accords); VNM, an FRA puppet,
# holds Annam, the Mekong, Laos and Cambodia. The people's war phase narrows combat and props up war support
# (cwp_decolonization_war_support); a general offensive ends it. France can seek a settlement once the war turns:
# its own war support below 40, or the Viet Minh holding Annam or the delta.

je_cwp_indochina_war = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			c:FRA ?= this
			c:VNM ?= this
			c:DAI ?= this
			c:PRC ?= this
			c:USA ?= this
		}
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_south_east_asia }
	}

	immediate = {
		set_global_variable = cwp_indochina_active
		set_global_variable = cwp_indochina_peoples_war_phase
		set_global_variable = { name = cwp_indochina_months value = 0 }
		c:FRA ?= { trigger_event = { id = cwp_indochina.1 popup = yes } }
		c:DAI ?= { trigger_event = { id = cwp_indochina.1 popup = yes } }
	}

	scripted_button = cwp_indochina_reinforce_button
	scripted_button = cwp_indochina_french_offensive_button
	scripted_button = cwp_indochina_negotiate_button
	scripted_button = cwp_indochina_general_offensive_button
	scripted_button = cwp_indochina_american_aid_button
	scripted_button = cwp_indochina_chinese_aid_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_indochina_war_status_accords
				trigger = {
					c:DAI ?= { is_subject_of = c:FRA }
					NOT = { c:FRA ?= { has_war_with = c:DAI } }
				}
			}
			triggered_desc = {
				desc = je_cwp_indochina_war_status_peoples_war
				trigger = { has_global_variable = cwp_indochina_peoples_war_phase }
			}
			triggered_desc = {
				desc = je_cwp_indochina_war_status_offensive
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = { has_global_variable = cwp_indochina_peoples_war_phase }
				cwp_indochina_peoples_war_pulse = yes
			}
			# The DRV inside the French Union (the March 1946 accords) keeps pressing for full independence.
			# The weekly modifier alone was swamped by the engine's liberty desire factors (run e32f10826:
			# 64 to 80 in 19 months), so the Viet Minh's political organisation also adds 3 a month at peace.
			c:DAI ?= {
				if = {
					limit = {
						is_subject_of = c:FRA
						is_at_war = no
					}
					add_liberty_desire = 3
				}
			}
			c:DAI ?= {
				if = {
					limit = {
						is_subject_of = c:FRA
						NOT = { has_modifier = cwp_indochina_viet_minh_resistance }
					}
					add_modifier = { name = cwp_indochina_viet_minh_resistance }
				}
				else_if = {
					limit = {
						is_subject = no
						has_modifier = cwp_indochina_viet_minh_resistance
					}
					remove_modifier = cwp_indochina_viet_minh_resistance
				}
			}
			if = {
				limit = { c:FRA ?= { has_war_with = c:DAI } }
				change_global_variable = { name = cwp_indochina_months add = 1 }
				# Only a war of independence counts: the DRV fighting France while it is a French subject.
				# The 1946 confrontation that ends with the DRV inside the French Union is the March 1946
				# accords, not a French victory (observer run fa0825cb7 closed the JE in mid-1946 that way).
				if = {
					limit = { c:DAI ?= { is_subject_of = c:FRA } }
					set_global_variable = cwp_indochina_war_fought
				}
			}
			c:DAI ?= {
				if = {
					limit = { NOT = { has_modifier = cwp_guerrilla_war } }
					add_modifier = { name = cwp_guerrilla_war }
				}
			}
			if = {
				limit = { NOT = { has_global_variable = cwp_indochina_open_months } }
				set_global_variable = { name = cwp_indochina_open_months value = 0 }
			}
			change_global_variable = { name = cwp_indochina_open_months add = 1 }
			# The accords break down (the Haiphong pattern): once the DRV is restless enough inside the French
			# Union, it decides whether to go to war. A choice, not a date; it can recur after a year.
			if = {
				limit = {
					NOT = { has_global_variable = cwp_indochina_haiphong_cd }
					c:DAI ?= {
						is_at_war = no
						OR = {
							AND = {
								is_subject_of = c:FRA
								liberty_desire >= 70
							}
							AND = {
								is_subject = no
								has_global_variable = cwp_indochina_open_months
								global_var:cwp_indochina_open_months >= 6
							}
						}
					}
				}
				set_global_variable = {
					name = cwp_indochina_haiphong_cd
					months = 12
				}
				c:DAI = { trigger_event = { id = cwp_indochina.3 popup = yes } }
			}
		}
	}

	# A conference partitions Vietnam
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_indochina_partition_desc
		trigger = {
			custom_tooltip = {
				text = cwp_indochina_negotiations_tt
				c:FRA ?= { has_variable = cwp_indochina_negotiations }
			}
		}
		effect = {
			remove_global_variable = cwp_indochina_active
			cwp_indochina_end_war = yes
			cwp_indochina_release_kingdoms = yes
			c:DAI ?= {
				if = {

### Algeria
# One bar, the insurgency. France's side runs through politics: once the war is going badly, a crisis of the
# Republic (cwp_algeria.2) lets the Armed Forces impose a strongman, who alone can then offer self-determination
# from the right; a left-leaning government can offer it too.

je_cwp_algerian_question = {
	group = je_group_cwp_decolonization
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	activation_description = {
		always = yes
	}

	should_be_involved = {
		c:FRA ?= this
	}

	should_show_when_not_involved = {
		capital ?= { is_in_geographic_region = geographic_region_north_africa }
	}

	immediate = {
		set_global_variable = cwp_algeria_active
		c:FRA ?= { trigger_event = { id = cwp_algeria.1 popup = yes } }
	}

	scripted_progress_bar = cwp_algeria_insurgency_bar

	scripted_button = cwp_algeria_pacification_button
	scripted_button = cwp_algeria_constantine_plan_button
	scripted_button = cwp_algeria_referendum_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_algerian_question_status_strongman
				trigger = { c:FRA ?= { has_variable = cwp_algeria_strongman } }
			}
			triggered_desc = {
				desc = je_cwp_algerian_question_status_war
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_variable = cwp_algeria_crisis_fired }
					scope:journal_entry = { "scripted_bar_progress(cwp_algeria_insurgency_bar)" >= 50 }
				}
				set_global_variable = cwp_algeria_crisis_fired
				c:FRA ?= { trigger_event = { id = cwp_algeria.2 popup = yes } }
			}
		}
	}

	# Self-determination by referendum
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_algeria_referendum_desc
		trigger = {
			custom_tooltip = {
				text = cwp_algeria_self_determination_tt
				c:FRA ?= { has_variable = cwp_algeria_self_determination }
			}
		}
		effect = {
			remove_global_variable = cwp_algeria_active
			cwp_algeria_independence_effect = yes
			c:FRA = {
				remove_variable = cwp_algeria_self_determination
				change_relations = {
					country = c:ALD
					value = 20
				}
				add_modifier = {
					name = cwp_end_of_empire
					years = 5
					is_decaying = yes
				}
				# The generals who brought the strongman to power feel betrayed.
				if = {
					limit = { has_variable = cwp_algeria_strongman }
					add_radicals = {
						value = 0.1
						interest_group = ig:ig_armed_forces
					}
				}
				trigger_event = { id = cwp_algeria.10 popup = yes }
			}
			c:ALD ?= { trigger_event = { id = cwp_algeria.10 popup = yes } }
		}
	}

	# The FLN wins on the ground
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_cwp_algeria_fln_victory_desc
		trigger = {
			custom_tooltip = {
				text = cwp_algeria_no_self_determination_tt
				NOT = { c:FRA ?= { has_variable = cwp_algeria_self_determination } }
			}
			custom_tooltip = {
				text = cwp_algeria_insurgency_100_tt
				scope:journal_entry = { "scripted_bar_progress(cwp_algeria_insurgency_bar)" >= 100 }
			}
		}
		effect = {
			remove_global_variable = cwp_algeria_active
			cwp_algeria_independence_effect = yes
			c:FRA = {
				change_relations = {
					country = c:ALD
					value = -50
				}
				create_bidirectional_truce = {
					country = c:ALD
					months = 60
				}
				add_modifier = {
					name = cwp_algeria_war_weariness
					years = 10
					is_decaying = yes
				}
				add_radicals = {
					value = 0.1
					interest_group = ig:ig_armed_forces
				}
				trigger_event = { id = cwp_algeria.12 popup = yes }
			}
			c:ALD ?= { trigger_event = { id = cwp_algeria.12 popup = yes } }
		}
	}

	# Algérie française
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_cwp_algeria_held_desc
		trigger = {
			custom_tooltip = {
				text = cwp_algeria_insurgency_0_tt
				scope:journal_entry = { "scripted_bar_progress(cwp_algeria_insurgency_bar)" <= 0 }
			}
		}
		effect = {
			remove_global_variable = cwp_algeria_active
			set_global_variable = {
				name = cwp_algeria_je_cooldown
				years = 10
			}
			remove_global_variable = cwp_algeria_crisis_fired
			c:FRA = {
				cwp_decolonization_add_setback = yes
				trigger_event = { id = cwp_algeria.13 popup = yes }
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_cwp_decolonization_invalid_desc
		trigger = {
			custom_tooltip = {
				text = cwp_algeria_invalid_tt
				OR = {
					NOT = { exists = c:FRA }
					NOT = { exists = s:STATE_ALGIERS.region_state:FRA }
				}
			}
		}
		effect = {
			remove_global_variable = cwp_algeria_active
		}
	}

	transferable = no
	can_deactivate = no
	weight = 100
}
