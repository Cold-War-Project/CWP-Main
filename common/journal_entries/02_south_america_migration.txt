# Cold War Project: same-name overwrite of the vanilla file. Only references to removed Victorian-era variables, journal entries and keys were stripped (2026-11-14 zero-error pass).
je_south_america_migration = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		is_in_geographic_region = geographic_region_south_america
		any_primary_culture = {
			has_discrimination_trait_group = heritage_group_european
		}
	}

	possible = {
		has_technology_researched = civilizing_mission
		country_rank >= rank_value:minor_power
		average_sol >= 10
		is_at_war = no
		is_subject = no
		OR = {
			has_law_or_variant = law_type:law_racial_segregation
			has_law_or_variant = law_type:law_cultural_exclusion
			has_law_or_variant = law_type:law_multicultural
		}
		NOR = {
			has_law_or_variant = law_type:law_closed_borders
			has_law_or_variant = law_type:law_isolationism
		}
	}

	immediate = {
		set_variable = {
			name = south_america_incorporated_population
			value = 0
		}
		set_variable = {
			name = south_america_incorporated_states
			value = 0
		}
		every_scope_state = {
			limit = {
				is_incorporated = yes
				is_in_geographic_region = geographic_region_south_america
			}
			ROOT = {
				change_variable = {
					name = south_america_incorporated_population
					add = PREV.state_population
				}
				change_variable = {
					name = south_america_incorporated_states
					add = 1
				}
			}
		}
		set_variable = {
			name = south_america_migration_goal
			value = var:south_america_incorporated_states
		}
		change_variable = {
			name = south_america_migration_goal
			multiply = 1500000
		}
		trigger_event = { id = sa_migration.2 popup = yes }
	}

	on_weekly_pulse = {
		effect = {
			set_variable = {
				name = south_america_incorporated_population
				value = 0
			}
			every_scope_state = {
				limit = {
					is_incorporated = yes
					is_in_geographic_region = geographic_region_south_america
				}
				ROOT = {
					change_variable = {
						name = south_america_incorporated_population
						add = PREV.state_population
					}
				}
			}
			set_variable = {
				name = south_america_incorporated_states
				value = 0
			}
			every_scope_state = {
				limit = {
					is_incorporated = yes
					is_in_geographic_region = geographic_region_south_america
				}
				ROOT = {
					change_variable = {
						name = south_america_incorporated_states
						add = 1
					}
				}
			}
			set_variable = {
				name = south_america_migration_goal
				value = var:south_america_incorporated_states
			}
			change_variable = {
				name = south_america_migration_goal
				multiply = 1500000
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = sa_migration_completion_tt
				trigger_if = {
					limit = { exists = var:south_america_incorporated_population }
					var:south_america_incorporated_population >= var:south_america_migration_goal
					var:south_america_incorporated_population > 0
					var:south_america_migration_goal > 0
				}
				trigger_else = {
					always = no
				}
			}
		}
		effect = {
			trigger_event = { id = sa_migration.3 popup = yes }
			show_as_tooltip = {
				add_modifier = {
					name = populated_southamerica
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			always = no
		}
		effect = {
			je_south_america_migration_cleanup = yes
		}
	}

	scripted_button = southamerica_migration_button

	weight = 100


	should_be_pinned_by_default_uninvolved_or_context = yes
}
