# Cold War Project: the United Nations, first organization on the international organization template.
# Design: docs/agents/areas/intl-orgs.md. Machinery: cwp_intl_org_effects.txt / _triggers.txt.
#
# je_cwp_un                    the organization itself; involved countries = members
# je_cwp_un_admission          an admission vote on one applicant (Security Council veto applies)
# je_cwp_un_res_human_rights   General Assembly resolution: Universal Declaration of Human Rights
# je_cwp_un_res_law_of_the_sea General Assembly resolution: Geneva Conventions on the Law of the Sea
#
# Proposal JEs are active only while their vote is open (can_deactivate = yes), so several
# can run at once and a failed one can be tabled again after its cooldown.

	group = je_group_cwp_international_organizations

	is_shown_when_inactive = { always = yes }
	possible = { always = yes }

	should_be_involved = {
		cwp_io_is_member = { ORG = un }
	}

	}

		value = cwp_un_law_of_the_sea_in_force
	}

		cwp_io_setup_org = { ORG = un THRESHOLD = 67 ROTATING_PRESIDENCY = 0 }

		# country variable, so a revolution that takes over the country inherits it.
		c:USA ?= { set_variable = cwp_io_veto_un }
		c:RUS ?= { set_variable = cwp_io_veto_un }

		cwp_io_count_members = { ORG = un }
		cwp_io_sync_involved = { ORG = un JE = je_cwp_un }
	}

	on_monthly_pulse = {
		effect = {
			cwp_io_count_members = { ORG = un }
			cwp_io_sync_involved = { ORG = un JE = je_cwp_un }
			# An applicant that ceased to exist mid-vote: drop the vote.
			if = {
				limit = {
					has_global_variable = cwp_io_un_admission_open
					NOT = { exists = global_var:cwp_un_admission_target }
				}
				cwp_io_close_proposal = { PROP = un_admission }
			}
			# Bring the next applicant to a vote when none is pending.
			if = {
				limit = {
					NOT = { has_global_variable = cwp_io_un_admission_open }
					NOT = { has_global_variable = cwp_io_un_admission_cooldown }
					any_country = { has_variable = cwp_io_applicant_un }
				}
				random_country = {
					limit = { has_variable = cwp_io_applicant_un }
					set_global_variable = { name = cwp_un_admission_target value = this }
				}
				cwp_io_open_proposal = { PROP = un_admission }
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_un_status_rotating
				trigger = { cwp_io_has_rotating_presidency = { ORG = un } }
			}
			desc = je_cwp_un_status
		}
	}

	weight = 50
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_cwp_un_admission = {
	icon = "gfx/interface/icons/cwp_icons/united_nations.dds"
	group = je_group_cwp_international_organizations

	activation_description = { always = yes }

	# A contextless JE activates only when is_shown_when_inactive and possible both hold.
	is_shown_when_inactive = { has_global_variable = cwp_io_un_admission_open }
	possible = {
		has_global_variable = cwp_io_un_admission_open
		exists = global_var:cwp_un_admission_target
	}
	can_deactivate = yes

	should_be_involved = {
		cwp_io_is_member = { ORG = un }
	}
	should_show_when_not_involved = {
		global_var:cwp_un_admission_target ?= this
	}

	scripted_button = cwp_un_admission_vote_yes_button
	scripted_button = cwp_un_admission_vote_no_button
	scripted_button = cwp_un_admission_vote_abstain_button
	scripted_progress_bar = cwp_io_support_bar

	current_value = { value = global_var:cwp_io_un_admission_months }
	goal_add_value = { value = 6 }
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			cwp_io_sync_involved = { ORG = un JE = je_cwp_un_admission }
			cwp_io_advance_proposal = { ORG = un PROP = un_admission VETO = cwp_un_p5_veto_check MONTHS = 6 COOLDOWN = 30 }
			scope:journal_entry = {
				set_bar_progress = { name = cwp_io_support_bar value = global_var:cwp_io_un_admission_share }
			}
			# The admission JE never concludes: it settles each applicant and closes the vote.
			if = {
				limit = { cwp_io_proposal_passed = { PROP = un_admission } }
				global_var:cwp_un_admission_target ?= {
					cwp_io_add_member = { ORG = un }
					je:je_cwp_un ?= { add_involved_country = PREV }
					trigger_event = { id = cwp_intl_org_un.1 popup = yes }
				}
				cwp_io_close_proposal = { PROP = un_admission }
			}
			else_if = {
				limit = { global_var:cwp_io_un_admission_result ?= flag:failed }
				global_var:cwp_un_admission_target ?= {
					remove_variable = cwp_io_applicant_un
					set_variable = { name = cwp_io_rejected_un years = 3 }
					if = {
						limit = { has_variable = cwp_io_un_rejected_before }
						set_variable = cwp_io_un_rejected_twice
					}
					set_variable = cwp_io_un_rejected_before
					trigger_event = { id = cwp_intl_org_un.2 popup = yes }
				}
			}
		}
	}

	status_desc = je_cwp_un_admission_status

	weight = 60
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
