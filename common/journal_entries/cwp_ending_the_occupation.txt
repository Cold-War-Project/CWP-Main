# Cold War Project: ending the occupations of Japan and Austria.
# The case for sovereignty builds with stability, reform and the occupier's own needs; the occupier decides.

je_cwp_ending_the_occupation = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_foreign_affairs

	activation_description = { always = yes }

	is_shown_when_inactive = {
		cwp_occ_is_candidate = yes
	}

	possible = {
		cwp_occ_is_candidate = yes
	}

	immediate = {
		overlord ?= { save_scope_as = cwp_occ_occupier }
	}

	scripted_progress_bar = cwp_occ_sovereignty_bar

	scripted_button = cwp_occ_neutrality_button

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(cwp_occ_sovereignty_bar)" >= 100
					}
					NOT = { has_variable = cwp_occ_pending_var }
				}
				set_variable = cwp_occ_pending_var
				save_scope_as = cwp_occ_subject
				overlord ?= {
					trigger_event = { id = cwp_occ.1 popup = yes }
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_ending_the_occupation_status_neutral
				trigger = { has_variable = cwp_occ_neutrality_var }
			}
			desc = je_cwp_ending_the_occupation_status
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_cwp_ending_the_occupation_success_desc
		color = "gold_positive"
		trigger = {
			is_subject = no
		}
		effect = {
			set_variable = cwp_occ_settled_var
			remove_variable = cwp_occ_pending_var
		}
	}

	weight = 1000
	transferable = no
	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
