# Cold War Project: ending the occupation of Japan (Austria's State Treaty is in the Germany content).
# Missions, not a bar: meet the occupier's conditions (a democratic constitution, a stable government, no
# communists in the cabinet, a few years of occupation behind you), then ask for a treaty. The occupier decides.

je_cwp_ending_the_occupation = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_foreign_affairs

	activation_description = { always = yes }

	is_shown_when_inactive = {
		cwp_occ_is_candidate = yes
	}

	possible = {
		custom_tooltip = {
			text = cwp_occ_possible_tt
			cwp_occ_is_candidate = yes
		}
	}

	immediate = {
		overlord ?= { save_scope_as = cwp_occ_occupier }
	}

	scripted_button = cwp_occ_request_treaty_button

	# Diagnostics for the observer dumps (which list variables, not legitimacy): what the treaty gates see.
	on_yearly_pulse = {
		effect = {
			set_variable = { name = cwp_occ_probe_legitimacy value = legitimacy }
			set_variable = { name = cwp_occ_probe_radicals value = radical_fraction }
			if = {
				limit = { cwp_communists_in_government = yes }
				set_variable = { name = cwp_occ_probe_communists value = 1 }
			}
			else = {
				set_variable = { name = cwp_occ_probe_communists value = 0 }
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_ending_the_occupation_status_refused
				trigger = { has_variable = cwp_occ_refused_var }
			}
			desc = je_cwp_ending_the_occupation_status
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_cwp_ending_the_occupation_success_desc
		color = "gold_positive"
		trigger = {
			is_subject = no
		}
		effect = {
			set_variable = cwp_occ_settled_var
			remove_variable = cwp_occ_pending_var
			remove_variable = cwp_occ_refused_var
			remove_variable = cwp_occ_probe_legitimacy
			remove_variable = cwp_occ_probe_radicals
			remove_variable = cwp_occ_probe_communists
		}
	}

	weight = 1000
	transferable = no
	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
