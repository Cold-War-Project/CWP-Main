# Cold War Project: the International Monetary Fund on the international organization template.
# Design: docs/agents/areas/intl-orgs.md.
#
# je_cwp_imf            the Fund; involved = members; votes weighted by economic weight (quota)
# je_cwp_imf_admission  a membership vote on one applicant
# je_cwp_imf_loan       a stand-by arrangement for one member in payment difficulties

je_cwp_imf = {
	icon = "gfx/interface/icons/cwp_icons/imf.dds"
	group = je_group_cwp_international_organizations

	activation_description = { always = yes }

	is_shown_when_inactive = { always = yes }
	possible = { always = yes }

	should_be_involved = {
		cwp_io_is_member = { ORG = imf }
	}
	should_show_when_not_involved = {
		NOT = { has_law_or_variant = law_type:law_council_republic }
	}

	scripted_button = cwp_imf_request_standby_button
	scripted_button = cwp_imf_withdraw_button

	immediate = {
		# Votes weighted by quota, approximated by economic weight; ordinary decisions by a
		# majority of the votes cast. No rotating chair (the Managing Director is not a member).
		cwp_io_setup_org = { ORG = imf THRESHOLD = 51 ROTATING_PRESIDENCY = 0 }

		# The Soviet Union attended Bretton Woods but never ratified.

		c:BOL ?= { cwp_io_add_member = { ORG = imf } }
		c:CAN ?= { cwp_io_add_member = { ORG = imf } }
		c:CHL ?= { cwp_io_add_member = { ORG = imf } }
		c:ROC ?= { cwp_io_add_member = { ORG = imf } }
		c:CLM ?= { cwp_io_add_member = { ORG = imf } }
		c:COS ?= { cwp_io_add_member = { ORG = imf } }
		c:CZH ?= { cwp_io_add_member = { ORG = imf } }
		c:DCR ?= { cwp_io_add_member = { ORG = imf } }
		c:ECU ?= { cwp_io_add_member = { ORG = imf } }
		c:EGY ?= { cwp_io_add_member = { ORG = imf } }
		c:ETH ?= { cwp_io_add_member = { ORG = imf } }
		c:FRA ?= { cwp_io_add_member = { ORG = imf } }
		c:GRE ?= { cwp_io_add_member = { ORG = imf } }
		c:GUA ?= { cwp_io_add_member = { ORG = imf } }
		c:HON ?= { cwp_io_add_member = { ORG = imf } }
		c:ICL ?= { cwp_io_add_member = { ORG = imf } }
		c:HND ?= { cwp_io_add_member = { ORG = imf } }
		c:IRQ ?= { cwp_io_add_member = { ORG = imf } }
		c:LUX ?= { cwp_io_add_member = { ORG = imf } }
		c:NET ?= { cwp_io_add_member = { ORG = imf } }
		c:NOR ?= { cwp_io_add_member = { ORG = imf } }
		c:PRG ?= { cwp_io_add_member = { ORG = imf } }
		c:PHI ?= { cwp_io_add_member = { ORG = imf } }
		c:POL ?= { cwp_io_add_member = { ORG = imf } }
		c:SAF ?= { cwp_io_add_member = { ORG = imf } }
		c:GBR ?= { cwp_io_add_member = { ORG = imf } }
		c:USA ?= { cwp_io_add_member = { ORG = imf } }
		c:YUG ?= { cwp_io_add_member = { ORG = imf } }

		cwp_io_count_members = { ORG = imf }
		cwp_io_sync_involved = { ORG = imf JE = je_cwp_imf }
	}

	on_monthly_pulse = {
		effect = {
			cwp_io_count_members = { ORG = imf }
			cwp_io_sync_involved = { ORG = imf JE = je_cwp_imf }
			cwp_io_queue_admission = { ORG = imf PROP = imf_admission TARGET = cwp_imf_admission_target }
			cwp_io_queue_admission = { ORG = imf_loan PROP = imf_loan TARGET = cwp_imf_loan_target }

	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cwp_imf_status_rotating
				trigger = { cwp_io_has_rotating_presidency = { ORG = imf } }
			}
			desc = je_cwp_imf_status
		}
	}

	weight = 45
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_cwp_imf_admission = {
	icon = "gfx/interface/icons/cwp_icons/imf.dds"
	group = je_group_cwp_international_organizations

	activation_description = { always = yes }

	is_shown_when_inactive = { has_global_variable = cwp_io_imf_admission_open }
	possible = {
		has_global_variable = cwp_io_imf_admission_open
		exists = global_var:cwp_imf_admission_target
	}
	can_deactivate = yes

	should_be_involved = {
		cwp_io_is_member = { ORG = imf }
	}
	should_show_when_not_involved = {
		global_var:cwp_imf_admission_target ?= this
	}

	scripted_button = cwp_imf_admission_vote_yes_button
	scripted_button = cwp_imf_admission_vote_no_button
	scripted_button = cwp_imf_admission_vote_abstain_button
	scripted_progress_bar = cwp_io_support_bar

	current_value = { value = global_var:cwp_io_imf_admission_months }
	goal_add_value = { value = 6 }
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			cwp_io_sync_involved = { ORG = imf JE = je_cwp_imf_admission }
			cwp_io_advance_proposal = { ORG = imf PROP = imf_admission VETO = no MONTHS = 6 COOLDOWN = 30 }
			scope:journal_entry = {
				set_bar_progress = { name = cwp_io_support_bar value = global_var:cwp_io_imf_admission_share }
			}
			cwp_io_settle_admission = {
				ORG = imf PROP = imf_admission TARGET = cwp_imf_admission_target JE = je_cwp_imf
				EVENT_YES = cwp_intl_org_imf.1 EVENT_NO = cwp_intl_org_imf.2
			}
		}
	}

	status_desc = je_cwp_imf_admission_status

	weight = 60
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_cwp_imf_loan = {
	icon = "gfx/interface/icons/cwp_icons/imf.dds"
	group = je_group_cwp_international_organizations

	activation_description = { always = yes }

	is_shown_when_inactive = { has_global_variable = cwp_io_imf_loan_open }
	possible = {
		has_global_variable = cwp_io_imf_loan_open
		exists = global_var:cwp_imf_loan_target
	}
	can_deactivate = yes

	should_be_involved = {
		cwp_io_is_member = { ORG = imf }
	}
	should_show_when_not_involved = {
		global_var:cwp_imf_loan_target ?= this
	}

	scripted_button = cwp_imf_loan_vote_yes_button
	scripted_button = cwp_imf_loan_vote_no_button
	scripted_button = cwp_imf_loan_vote_abstain_button
	scripted_progress_bar = cwp_io_support_bar

	current_value = { value = global_var:cwp_io_imf_loan_months }
	goal_add_value = { value = 6 }
	progressbar = yes

	on_monthly_pulse = {
		effect = {
			cwp_io_sync_involved = { ORG = imf JE = je_cwp_imf_loan }
			cwp_io_advance_proposal = { ORG = imf PROP = imf_loan VETO = no MONTHS = 6 COOLDOWN = 30 }
			scope:journal_entry = {
				set_bar_progress = { name = cwp_io_support_bar value = global_var:cwp_io_imf_loan_share }
			}
			if = {
				limit = { cwp_io_proposal_passed = { PROP = imf_loan } }
				global_var:cwp_imf_loan_target ?= {
					remove_variable = cwp_io_applicant_imf_loan
					add_treasury = cwp_imf_standby_amount
					add_modifier = { name = cwp_imf_conditionality_modifier years = 5 }
					set_variable = { name = cwp_io_rejected_imf_loan years = 5 }
					trigger_event = { id = cwp_intl_org_imf.3 popup = yes }
				}
				cwp_io_close_proposal = { PROP = imf_loan }
			}
			else_if = {
				limit = { global_var:cwp_io_imf_loan_result ?= flag:failed }
				global_var:cwp_imf_loan_target ?= {
					remove_variable = cwp_io_applicant_imf_loan
					set_variable = { name = cwp_io_rejected_imf_loan years = 2 }
					trigger_event = { id = cwp_intl_org_imf.4 popup = yes }
				}
			}
		}
	}

	status_desc = je_cwp_imf_loan_status

	weight = 60
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
