		law_national_champions = strongly_approve
		law_corporate_deregulation = disapprove
		law_antitrust_legislation = approve
	}

	country_trigger = {
		game_date >= 1946.1.1
	}

	interest_group_leader_trigger = {
		OR = {
			is_interest_group_type = ig_armed_forces
			is_interest_group_type = ig_petty_bourgeoisie
			is_interest_group_type = ig_industrialists
			is_interest_group_type = ig_intelligentsia
		}
	}

	non_interest_group_leader_trigger = {
		exists = interest_group
		interest_group = {
			OR = {
				is_interest_group_type = ig_armed_forces
				is_interest_group_type = ig_petty_bourgeoisie
				is_interest_group_type = ig_industrialists
				is_interest_group_type = ig_intelligentsia
			}
		}
		# Only explicitly republican agendas exclude generated reigning monarchs and heirs.
		NAND = {
			owner = { country_has_monarchy_law = yes }
			OR = {
				is_ruler_of_own_country = yes
				is_heir_of_own_country = yes
			}
		}
		# This grievance guard uses only strongest-approved laws in this focused agenda.
		NAND = {
			has_role_of_type = agitator
			owner = {
				has_law_or_variant = law_type:law_presidential_republic
				has_law_or_variant = law_type:law_universal_suffrage
				has_law_or_variant = law_type:law_interventionism
				has_law_or_variant = law_type:law_national_champions
			}
		}
	}

	interest_group_leader_weight = {
		value = { add = 10 desc = "base_value" }
		if = {
			limit = { is_interest_group_type = ig_armed_forces }
			add = 5
		}
		if = {
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_presidential_republic
						has_law_or_variant = law_type:law_national_champions
					}
				}
			}
			add = 10
		}
	}

	non_interest_group_leader_weight = {
		value = 10
		if = {
			limit = { interest_group = { is_interest_group_type = ig_armed_forces } }
			add = 5
		}
	}
}
