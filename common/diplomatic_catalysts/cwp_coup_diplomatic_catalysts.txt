# Diplomatic Catalysts for Coups
catalyst_orchestrate_coup_successful = { # Pro-Lobby always exists
	category = cc_orchestrate_coup

	ai_country_goal_recalculation = {
		type = more_friendly
		chance = 1.0
	}
}

catalyst_orchestrate_coup_failed = {
	category = cc_orchestrate_coup

	political_lobby_creation = {
		lobby_anti_country = {
			value = 0.4

			if = {
				limit = { NOT = { has_strategic_adjacency = scope:target_country } }
				multiply = 0.5
			}
		}
	}

	ai_country_goal_recalculation = {
		type = more_hostile
		chance = 1.0
	}
}

catalyst_orchestrate_coup_fizzled = {
	category = cc_orchestrate_coup

	political_lobby_creation = {
		lobby_anti_country = {
			value = 0.05

			if = {
				limit = { NOT = { has_strategic_adjacency = scope:target_country } }
				multiply = 0.5
			}

			if = {
				limit = { "relations(scope:target_country)" > relations_threshold:poor }
				multiply = 0.5
			}
			else_if = {
				limit = { "relations(scope:target_country)" <= relations_threshold:hostile }
				multiply = 1.5
			}
		}
		
		lobby_anti_overlord = {
			value = 0.05
			
			if = {
				limit = { liberty_desire >= 75 }
				multiply = 3
			}
			else_if = {
				limit = { liberty_desire >= 50 }
				multiply = 2
			}

			if = {
				limit = { NOT = { has_strategic_adjacency = scope:target_country } }
				multiply = 0.5
			}
			
			if = {
				limit = { "relations(scope:target_country)" > relations_threshold:poor }
				multiply = 0.5
			}
		}
	}

	ai_country_goal_recalculation = {
		type = more_hostile
		chance = {
			value = 0.1

			if = {
				limit = { "relations(scope:target_country)" <= relations_threshold:cold }
				multiply = 4.0
			}
			else_if = {
				limit = { "relations(scope:target_country)" < relations_threshold:cordial }
				multiply = 3.0
			}
			else_if = {
				limit = { "relations(scope:target_country)" < relations_threshold:friendly }
				multiply = 2.0
			}
		}
	}
}
