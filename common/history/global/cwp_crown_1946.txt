GLOBAL = {
	# Monarchy tradition (cwp_crown_has_monarchy_tradition, cwp_crown_triggers.txt): every 1946 monarchy, plus the
	# kingdoms abolished in 1945-46 whose royal houses kept a claim (Albania: Zog I; Yugoslavia: Peter II;
	# Hungary: kingdom until 1 February 1946). Only these countries can enact or have the monarchy imposed.
	every_country = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_monarchy
				c:ALB ?= this
				c:YUG ?= this
				c:HUN ?= this
			}
		}
		set_variable = cwp_crown_tradition_var
	}
}
