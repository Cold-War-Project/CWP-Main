GLOBAL = {
	# Monarchy tradition (cwp_crown_has_monarchy_tradition, cwp_crown_triggers.txt): every 1946 monarchy (not
	# colonial administrations); the kingdoms abolished in 1945-46 whose royal houses kept a claim (Albania: Zog I;
	# Yugoslavia: Peter II; Hungary: kingdom until 1 February 1946); and native monarchies under colonial rule
	# (Libya's Senussi, the Malay sultans, Basutoland, Swaziland, Zanzibar, Brunei, Buganda, Rwanda).
	# Realms sharing the British crown are left out: they never had restorationists of their own once republics.
	every_country = {
		limit = {
			NOT = { has_variable = cwp_shared_crown_var }
			OR = {
				has_law = law_type:law_monarchy
				c:ALB ?= this
				c:YUG ?= this
				c:HUN ?= this
				c:LBY ?= this
				c:MYY ?= this
				c:BST ?= this
				c:ESW ?= this
				c:ZZB ?= this
				c:BRU ?= this
				c:BUG ?= this
				c:RWD ?= this
			}
		}
		set_variable = cwp_crown_tradition_var
	}
}
