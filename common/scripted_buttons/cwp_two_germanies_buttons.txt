# Cold War Project: buttons for the German question, chapter 2 (the two Germanies).
# Journal entry: common/journal_entries/cwp_two_germanies.txt
# root = the country pressing the button; scope:journal_entry is the chapter.

# GDR: close the border with the West, Berlin included
cwp_tg_wall_button = {
	name = "cwp_tg_wall_button"
	desc = "cwp_tg_wall_button_desc"

	visible = {
		c:DDR ?= this
		NOT = { has_global_variable = cwp_tg_wall }
	}

	possible = {
		custom_tooltip = {
			text = cwp_tg_emigration_active_tt
			cwp_tg_emigration_active = yes
		}
	}

	effect = {
		set_global_variable = cwp_tg_wall
		add_modifier = { name = cwp_tg_wall }
		hidden_effect = {
			every_country = {
				limit = {
					cwp_tg_is_party = yes
					NOT = { this = root }
				}
				trigger_event = { id = cwp_two_germanies.2 days = 1 popup = yes }
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				scope:journal_entry ?= { "scripted_bar_progress(cwp_tg_legitimacy_bar)" <= 25 }
			}
			add = 25
		}
	}
}

# Federal Republic: drop the Hallstein Doctrine and deal with the East
cwp_tg_ostpolitik_button = {
	name = "cwp_tg_ostpolitik_button"
	desc = "cwp_tg_ostpolitik_button_desc"

	visible = {
		c:GER ?= this
		has_global_variable = cwp_tg_hallstein
	}

	possible = {
		custom_tooltip = {
			text = cwp_tg_hallstein_tt
			has_global_variable = cwp_tg_hallstein
		}
	}

	effect = {
		remove_global_variable = cwp_tg_hallstein
		set_global_variable = cwp_tg_ostpolitik
		add_journal_entry = { type = je_cwp_ostpolitik }
	}

	# Ostpolitik followed détente and a closed border (1969): the weight sits on those conditions
	ai_chance = {
		value = 0.05
		if = {
			limit = {
				has_global_variable = cwp_tg_wall
				exists = c:RUS
				exists = c:USA
				c:USA = { "relations(c:RUS)" >= 20 }
			}
			add = 25
		}
		if = {
			limit = {
				has_global_variable = cwp_tg_wall
				ig:ig_trade_unions ?= { is_in_government = yes }
			}
			add = 5
		}
	}
}

# Federal Republic: offer the GDR a treaty on relations between the two states
cwp_tg_basic_treaty_button = {
	name = "cwp_tg_basic_treaty_button"
	desc = "cwp_tg_basic_treaty_button_desc"

	visible = {
		c:GER ?= this
		has_global_variable = cwp_tg_ostpolitik
		NOT = { has_global_variable = cwp_tg_basic_treaty }
	}

	possible = {
		exists = c:DDR
		custom_tooltip = {
			text = cwp_tg_basic_treaty_moscow_treaty_tt
			has_global_variable = cwp_tg_moscow_treaty
		}
		custom_tooltip = {
			text = cwp_tg_basic_treaty_detente_tt
			exists = c:USA
			exists = c:RUS
			c:USA = { "relations(c:RUS)" >= 20 }
		}
		custom_tooltip = {
			text = cwp_tg_basic_treaty_border_tt
			has_global_variable = cwp_tg_wall
		}
		custom_tooltip = {
			text = cwp_tg_basic_treaty_moscow_tt
			NOT = { has_global_variable = cwp_tg_support_withdrawn }
		}
		custom_tooltip = {
			text = cwp_tg_basic_treaty_pending_tt
			NOT = { has_global_variable = cwp_tg_basic_treaty_offered }
		}
	}

	effect = {
		set_global_variable = { name = cwp_tg_basic_treaty_offered days = very_short_modifier_time }
		custom_tooltip = cwp_tg_basic_treaty_button_tt
		hidden_effect = {
			c:DDR = { trigger_event = { id = cwp_two_germanies.3 days = 7 popup = yes } }
		}
	}

	ai_chance = {
		value = 25
	}
}

cwp_tg_withdraw_support_button = {
	name = "cwp_tg_withdraw_support_button"
	desc = "cwp_tg_withdraw_support_button_desc"

	visible = {
		c:RUS ?= this
		NOT = { has_global_variable = cwp_tg_support_withdrawn }
	}

	possible = {
		exists = c:DDR
	}

	effect = {
		set_global_variable = cwp_tg_support_withdrawn
		add_modifier = { name = cwp_tg_support_withdrawn }
		hidden_effect = {
			c:DDR = { trigger_event = { id = cwp_two_germanies.5 days = 1 popup = yes } }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				exists = c:USA
				average_sol < c:USA.average_sol
				c:USA = { "relations(c:RUS)" >= 20 }
			}
			add = 2
		}
		if = {
			limit = { radical_fraction = { value >= 0.2 } }
			add = 5
		}
	}
}
