# Cold War Project: the energy crisis. Country scope; scope:journal_entry = je_cwp_energy_crisis.

# Ask the public to save fuel: lower thermostats, slower speed limits, car-free Sundays.
cwp_ec_conservation_button = {
	name = "cwp_ec_conservation_button"
	desc = "cwp_ec_conservation_button_desc"

	visible = { always = yes }

	possible = {
		NOT = { has_modifier = cwp_ec_conservation }
	}

	effect = {
		add_modifier = {
			name = cwp_ec_conservation
			years = 1
		}
	}

	cooldown = { months = 12 }

	ai_chance = {
		value = 10
	}
}

# Ration fuel: coupons, odd-even days at the pump, priority for essential services.
cwp_ec_rationing_button = {
	name = "cwp_ec_rationing_button"
	desc = "cwp_ec_rationing_button_desc"

	visible = { always = yes }

	possible = {
		NOT = { has_modifier = cwp_ec_rationing }
		cwp_ec_fuel_pricier = { PCT = 0.75 }
	}

	effect = {
		add_modifier = {
			name = cwp_ec_rationing
			years = 2
		}
		ig:ig_petty_bourgeoisie ?= {
			add_modifier = {
				name = cwp_ec_rationing_resented
				years = 2
			}
		}
		ig:ig_industrialists ?= {
			add_modifier = {
				name = cwp_ec_rationing_resented
				years = 2
			}
		}
	}

	cooldown = { months = 24 }

	ai_chance = {
		value = 4
	}
}

# Cap fuel prices and pay the difference from the treasury. While it lasts, there are no fuel riots.
cwp_ec_price_controls_button = {
	name = "cwp_ec_price_controls_button"
	desc = "cwp_ec_price_controls_button_desc"

	visible = { always = yes }

	possible = {
		NOT = { has_modifier = cwp_ec_price_controls }
	}

	effect = {
		add_treasury = cwp_ec_subsidy_cost
		add_modifier = {
			name = cwp_ec_price_controls
			months = 6
		}
	}

	cooldown = { months = 6 }

	ai_chance = {
		value = 4
		if = {
			limit = { government_legitimacy < 60 }
			add = 6
		}
		if = {
			limit = { in_default = yes }
			multiply = 0
		}
	}
}

# Fund a long-term programme to cut dependence on oil: nuclear power, hydro, domestic refining.
cwp_ec_diversify_button = {
	name = "cwp_ec_diversify_button"
	desc = "cwp_ec_diversify_button_desc"

	visible = {
		NOT = { has_variable = cwp_ec_diversify_var }
	}

	effect = {
		set_variable = cwp_ec_diversify_var
		add_treasury = cwp_ec_diversification_cost
	}

	ai_chance = {
		value = 3
		if = {
			limit = { in_default = yes }
			multiply = 0
		}
	}
}

# The mission's last step: once the programme has built the plant, the government declares energy independence.
cwp_ec_declare_independence_button = {
	name = "cwp_ec_declare_independence_button"
	desc = "cwp_ec_declare_independence_button_desc"

	visible = {
		has_variable = cwp_ec_diversify_var
	}

	possible = {
		custom_tooltip = {
			text = cwp_ec_clean_power_tt
			any_scope_building = {
				OR = {
					AND = {
						is_building_type = building_nuclear_power_plant
						level >= 1
					}
					AND = {
						is_building_type = building_hydroelectric_power_plant
						level >= 3
					}
				}
			}
		}
		custom_tooltip = {
			text = cwp_ec_domestic_refining_tt
			any_scope_building = {
				is_building_type = building_oil_refineries
				level >= 3
			}
		}
		custom_tooltip = {
			text = cwp_ec_demand_cut_tt
			OR = {
				has_modifier = cwp_ec_conservation
				has_modifier = cwp_ec_rationing
			}
		}
	}

	effect = {
		remove_variable = cwp_ec_diversify_var
		set_variable = cwp_ec_independence_declared_var
	}

	ai_chance = {
		value = 50
	}
}
