# Cold War Project: the energy crisis. Country scope; scope:journal_entry = je_cwp_energy_crisis.

# Ask the public to save fuel: lower thermostats, slower speed limits, car-free Sundays.
cwp_ec_conservation_button = {
	name = "cwp_ec_conservation_button"
	desc = "cwp_ec_conservation_button_desc"

	visible = { always = yes }

	effect = {
		add_modifier = {
			name = cwp_ec_conservation
			years = 1
		}
		scope:journal_entry = {
			add_progress = { value = -3 name = cwp_ec_anger_bar }
		}
	}

	cooldown = { months = 12 }

	ai_chance = {
		value = 10
	}
}

# Ration fuel: coupons, odd-even days at the pump, priority for essential services.
cwp_ec_rationing_button = {
	name = "cwp_ec_rationing_button"
	desc = "cwp_ec_rationing_button_desc"

	visible = { always = yes }

	possible = {
		NOT = { has_modifier = cwp_ec_rationing }
		cwp_ec_fuel_pricier = { PCT = 0.75 }
	}

	effect = {
		add_modifier = {
			name = cwp_ec_rationing
			years = 2
		}
		ig:ig_petty_bourgeoisie ?= {
			add_modifier = {
				name = cwp_ec_rationing_resented
				years = 2
			}
		}
		ig:ig_industrialists ?= {
			add_modifier = {
				name = cwp_ec_rationing_resented
				years = 2
			}
		}
		scope:journal_entry = {
			add_progress = { value = 5 name = cwp_ec_anger_bar }
		}
	}

	cooldown = { months = 24 }

	ai_chance = {
		value = 4
		if = {
			limit = {
				scope:journal_entry = { "scripted_bar_progress(cwp_ec_anger_bar)" >= 60 }
			}
			multiply = 0.25
		}
	}
}

# Cap fuel prices and pay the difference from the treasury.
cwp_ec_price_controls_button = {
	name = "cwp_ec_price_controls_button"
	desc = "cwp_ec_price_controls_button_desc"

	visible = { always = yes }

	possible = {
		NOT = { has_modifier = cwp_ec_price_controls }
	}

	effect = {
		add_treasury = cwp_ec_subsidy_cost
		add_modifier = {
			name = cwp_ec_price_controls
			months = 6
		}
	}

	cooldown = { months = 6 }

	ai_chance = {
		value = 6
		if = {
			limit = { in_default = yes }
			multiply = 0
		}
	}
}

# A long-term programme to cut dependence on oil: nuclear power, coal, efficiency standards.
cwp_ec_diversify_button = {
	name = "cwp_ec_diversify_button"
	desc = "cwp_ec_diversify_button_desc"

	visible = {
		NOT = { has_variable = cwp_ec_diversify_var }
	}

	effect = {
		set_variable = cwp_ec_diversify_var
		add_treasury = cwp_ec_diversification_cost
	}

	ai_chance = {
		value = 3
		if = {
			limit = { in_default = yes }
			multiply = 0
		}
	}
}
