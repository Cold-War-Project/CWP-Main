# Cold War Project: ending the occupation of Japan. Country scope = the occupied country.

cwp_occ_request_treaty_button = {
	name = "cwp_occ_request_treaty_button"
	desc = "cwp_occ_request_treaty_button_desc"

	visible = {
		NOT = { has_variable = cwp_occ_pending_var }
	}

	possible = {
		custom_tooltip = {
			text = cwp_occ_constitution_tt
			OR = {
				has_variable = cwp_jap_new_constitution_var
				has_law = law_type:law_universal_suffrage
			}
		}
		custom_tooltip = {
			text = cwp_occ_stable_tt
			legitimacy >= 25
		}
		custom_tooltip = {
			text = cwp_occ_no_communists_tt
			cwp_communists_in_government = no
		}
		custom_tooltip = {
			text = cwp_occ_years_tt
			scope:journal_entry = { journal_entry_age >= 1460 }
		}
		custom_tooltip = {
			text = cwp_occ_not_refused_tt
			NOT = { has_variable = cwp_occ_refused_var }
		}
	}

	effect = {
		set_variable = { name = cwp_occ_pending_var days = 90 } # expires if the request is lost (e.g. a new overlord)
		save_scope_as = cwp_occ_subject
		custom_tooltip = cwp_occ_request_treaty_tt
		overlord ?= {
			trigger_event = { id = cwp_occ.1 days = 30 popup = yes }
		}
	}

	ai_chance = { value = 50 }
}
