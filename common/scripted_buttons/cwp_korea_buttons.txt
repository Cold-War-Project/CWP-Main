cwp_korea_commission_concessions_button = {
	name = "cwp_korea_commission_concessions_button"
	desc = "cwp_korea_commission_concessions_button_desc"

	visible = {
		OR = {
			c:USA ?= this
			c:RUS ?= this
		}
		cwp_korea_commission_active = yes
	}

	possible = {
		NOT = { has_modifier = cwp_korea_commission_concessions }
	}

	effect = {
		add_modifier = { name = cwp_korea_commission_concessions days = 365 }
	}

	ai_chance = {
		value = 3
		if = {
			limit = {
				c:USA ?= this
				exists = c:RUS
				"relations(c:RUS)" >= 0
			}
			add = 10
		}
		if = {
			limit = {
				c:RUS ?= this
				exists = c:USA
				"relations(c:USA)" >= 0
			}
			add = 10
		}
	}
}
