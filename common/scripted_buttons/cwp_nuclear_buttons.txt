# Cold War Project: nuclear weapons programme. scope:journal_entry = je_cwp_nuclear_programme.

cwp_nuc_uranium_button = {
	name = "cwp_nuc_uranium_button"
	desc = "cwp_nuc_uranium_button_desc"

	visible = {
		NOT = { has_variable = cwp_nuc_uranium_var }
	}

	possible = { always = yes }

	effect = {
		set_variable = cwp_nuc_uranium_var
		add_modifier = {
			name = cwp_nuc_uranium_cost
			days = normal_modifier_time
			multiplier = cwp_nuc_programme_cost
		}
		custom_tooltip = cwp_nuc_uranium_tt
	}

	ai_chance = { value = 30 }
}

# Spies inside a nuclear power's programme (Fuchs and the Rosenbergs for the Soviet Union).
cwp_nuc_espionage_button = {
	name = "cwp_nuc_espionage_button"
	desc = "cwp_nuc_espionage_button_desc"

	visible = {
		NOT = { has_variable = cwp_nuc_espionage_var }
		any_country = {
			cwp_is_nuclear_power = yes
			relations:root <= relations_threshold:neutral
		}
	}

	possible = {
		has_law = law_type:law_secret_police
	}

	effect = {
		set_variable = cwp_nuc_espionage_var
		custom_tooltip = cwp_nuc_espionage_tt
		random_list = {
			70 = {}
			30 = {
				random_country = {
					limit = {
						cwp_is_nuclear_power = yes
						relations:root <= relations_threshold:neutral
					}
					change_relations = { country = root value = -30 }
				}
			}
		}
	}

	ai_chance = { value = 40 }
}

# A nuclear ally shares its knowledge (the Anglo-American partnership, Soviet aid to China in 1957).
cwp_nuc_foreign_help_button = {
	name = "cwp_nuc_foreign_help_button"
	desc = "cwp_nuc_foreign_help_button_desc"

	visible = {
		NOT = { has_variable = cwp_nuc_foreign_help_var }
	}

	possible = {
		custom_tooltip = {
			text = cwp_nuc_friendly_nuclear_power_tt
			OR = {
				overlord ?= { cwp_is_nuclear_power = yes }
				power_bloc ?= {
					power_bloc_leader ?= {
						NOT = { this = root }
						cwp_is_nuclear_power = yes
						relations:root >= relations_threshold:friendly
					}
				}
			}
		}
	}

	effect = {
		set_variable = cwp_nuc_foreign_help_var
		custom_tooltip = cwp_nuc_foreign_help_tt
	}

	ai_chance = { value = 40 }
}

# Choose a remote state as the test site (Semipalatinsk, Monte Bello, Reggane, Lop Nur): the emptiest state we own.
cwp_nuc_test_site_button = {
	name = "cwp_nuc_test_site_button"
	desc = "cwp_nuc_test_site_button_desc"

	visible = {
		NOT = { has_variable = cwp_nuc_test_site_var }
	}

	possible = {
		custom_tooltip = {
			text = cwp_nuc_has_territory_tt
			any_scope_state = { is_incorporated = yes }
		}
	}

	effect = {
		ordered_scope_state = {
			limit = { is_incorporated = yes }
			order_by = {
				value = 0
				subtract = this.state_population
			}
			save_scope_as = cwp_nuc_site
		}
		set_variable = { name = cwp_nuc_test_site_var value = scope:cwp_nuc_site }
		custom_tooltip = cwp_nuc_test_site_tt
	}

	ai_chance = { value = 50 }
}
