# Cold War Project: Newfoundland's future. Country scope = NEW.

cwp_nf_smallwood_button = {
	name = "cwp_nf_smallwood_button"
	desc = "cwp_nf_smallwood_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_iq_campaign_cap_tt
			OR = {
				NOT = { has_variable = cwp_nf_campaign_var }
				var:cwp_nf_campaign_var < 12
			}
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_variable = cwp_nf_campaign_var } }
			set_variable = { name = cwp_nf_campaign_var value = 0 }
		}
		change_variable = { name = cwp_nf_campaign_var add = 4 }
		custom_tooltip = cwp_nf_smallwood_tt
		ig:ig_industrialists ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 1
			}
		}
	}

	cooldown = { months = 3 }

	ai_chance = {
		value = 0
		if = {
			limit = { cwp_nf_confederation_pivot > 35 }
			add = 10
		}
	}
}

# The Convention's missions: ask London what a self-governing Newfoundland could expect, and Ottawa what terms it offers.
cwp_nf_london_delegation_button = {
	name = "cwp_nf_london_delegation_button"
	desc = "cwp_nf_london_delegation_button_desc"

	visible = {
		NOT = { has_variable = cwp_nf_london_var }
	}

	possible = {
		custom_tooltip = {
			text = cwp_nf_convention_sat_tt
			scope:journal_entry = { journal_entry_age >= 120 }
		}
	}

	effect = {
		set_variable = cwp_nf_london_var
		custom_tooltip = cwp_nf_london_tt
		if = {
			limit = { NOT = { has_variable = cwp_nf_campaign_var } }
			set_variable = { name = cwp_nf_campaign_var value = 0 }
		}
		if = {
			limit = { overlord ?= { in_default = yes } }
			change_variable = { name = cwp_nf_campaign_var add = 2 }
		}
		else = {
			change_variable = { name = cwp_nf_campaign_var add = -2 }
		}
	}

	ai_chance = { value = 30 }
}

# The Convention reports and London sets the ballot.
cwp_nf_convention_report_button = {
	name = "cwp_nf_convention_report_button"
	desc = "cwp_nf_convention_report_button_desc"

	visible = {
		NOT = { has_variable = cwp_nf_vote_pending_var }
	}

	possible = {
		custom_tooltip = {
			text = cwp_nf_london_done_tt
			has_variable = cwp_nf_london_var
		}
		custom_tooltip = {
			text = cwp_nf_debate_done_tt
			scope:journal_entry = { journal_entry_age >= 365 }
		}
	}

	effect = {
		set_variable = cwp_nf_vote_pending_var
		custom_tooltip = cwp_nf_convention_report_tt
		trigger_event = { id = cwp_nf.2 days = 60 popup = yes }
	}

	ai_chance = { value = 20 }
}
