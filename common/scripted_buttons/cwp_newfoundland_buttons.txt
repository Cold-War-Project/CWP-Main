# Cold War Project: Newfoundland's future. Country scope = NEW.

cwp_nf_smallwood_button = {
	name = "cwp_nf_smallwood_button"
	desc = "cwp_nf_smallwood_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_iq_campaign_cap_tt
			OR = {
				NOT = { has_variable = cwp_nf_campaign_var }
				var:cwp_nf_campaign_var < 12
			}
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_variable = cwp_nf_campaign_var } }
			set_variable = { name = cwp_nf_campaign_var value = 0 }
		}
		change_variable = { name = cwp_nf_campaign_var add = 4 }
		custom_tooltip = cwp_nf_smallwood_tt
		ig:ig_industrialists ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 1
			}
		}
	}

	cooldown = { months = 3 }

	ai_chance = {
		value = 0
		if = {
			limit = { var:cwp_nf_vote_current_var > 40 }
			add = 10
		}
	}
}
