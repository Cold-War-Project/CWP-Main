# USA and Britain: agree to merge the American and British zones (both must agree)
cwp_gq_merge_zones_button = {
	name = "cwp_gq_merge_zones_button"
	desc = "cwp_gq_merge_zones_button_desc"

	visible = {
		OR = {
			c:USA ?= this
			c:GBR ?= this
		}
		NOT = { has_global_variable = cwp_gq_bizone }
	}

		exists = c:AOZ
		exists = c:BOZ
		custom_tooltip = {
			text = cwp_gq_already_agreed_tt
			NOT = { has_variable = cwp_gq_agreed_bizone }
		}
	}

		has_variable = cwp_gq_agreed_bizone
	}

	effect = {
		set_variable = cwp_gq_agreed_bizone
		custom_tooltip = cwp_gq_merge_zones_button_tt
	}

	ai_chance = {
		value = 1
		if = {
			limit = {
				NOT = { cwp_gq_four_power_accord = yes }
			}
			add = 10
		}
	}
}

# Any occupying power: back, or stop backing, central German administrations and an all-German
# government (Potsdam). If all four back it before the zones split, Germany is united and neutral.
cwp_gq_central_government_button = {
	name = "cwp_gq_central_government_button"
	desc = "cwp_gq_central_government_button_desc"

	visible = {
		cwp_gq_is_occupier = yes
		NOT = { has_global_variable = cwp_gq_frg_proclaimed }
		NOT = { has_global_variable = cwp_gq_gdr_proclaimed }
	}

	possible = {
		always = yes
	}

	selected = {
		has_variable = cwp_gq_backs_central_government
	}

	effect = {
		if = {
			limit = { has_variable = cwp_gq_backs_central_government }
			remove_variable = cwp_gq_backs_central_government
			custom_tooltip = cwp_gq_central_government_withdraw_tt
		}
		else = {
			set_variable = cwp_gq_backs_central_government
			custom_tooltip = cwp_gq_central_government_back_tt
		}
	}

	# Pressing it toggles the stance. Washington and London give up on four-power government once the
	# Soviet zone is being remade; Paris rarely accepts a central German authority; Moscow keeps backing it.
	ai_chance = {
		value = 0
		if = {
			limit = {
				has_variable = cwp_gq_backs_central_government
				OR = {
					c:USA ?= this
					c:GBR ?= this
				}
				scope:journal_entry ?= { "scripted_bar_progress(cwp_gq_soviet_bar)" >= 25 }
			}
			add = 10
		}
		if = {
			limit = {
				NOT = { has_variable = cwp_gq_backs_central_government }
				c:FRA ?= this
				NOT = { has_global_variable = cwp_gq_french_reservations }
			}
			add = 0.2
		}
		if = {
			limit = {
				NOT = { has_variable = cwp_gq_backs_central_government }
				c:RUS ?= this
			}
			add = 5
		}
	}
}

# Soviet Union: cut the land and water routes to West Berlin
cwp_gq_blockade_button = {
	name = "cwp_gq_blockade_button"
	desc = "cwp_gq_blockade_button_desc"

	visible = {
		c:RUS ?= this
		NOT = { has_global_variable = cwp_gq_blockade }
		NOT = { has_global_variable = cwp_gq_berlin_settled }
	}

	possible = {
		custom_tooltip = {
			text = cwp_gq_needs_currency_reform_tt
			has_global_variable = cwp_gq_currency_reform
		}
		exists = c:DDR
		cwp_gq_soviet_zone_held = yes
	}

	effect = {
		set_global_variable = cwp_gq_blockade
		set_global_variable = { name = cwp_gq_blockade_months value = 0 }
		hidden_effect = {
			if = {
				limit = { exists = c:USA }
				c:USA = { trigger_event = { id = cwp_german_question.5 days = 3 popup = yes } }
			}
			else = {
				c:GBR ?= { trigger_event = { id = cwp_german_question.5 days = 3 popup = yes } }
			}
			every_country = {
				limit = {
					OR = {
						c:GBR ?= this
						c:FRA ?= this
						c:DDR ?= this
						cwp_gq_is_west_german = yes
					}
					NOT = {
						AND = {
							c:GBR ?= this
							NOT = { exists = c:USA }
						}
					}
				}
				trigger_event = { id = cwp_german_question.6 days = 3 popup = yes }
			}
		}
	}

	ai_chance = {
		value = 1
		if = {
			limit = {
				scope:journal_entry ?= { "scripted_bar_progress(cwp_gq_soviet_bar)" >= 40 }
			}
			add = 10
		}
	}
}
