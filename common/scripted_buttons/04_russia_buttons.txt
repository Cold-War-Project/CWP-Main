		exists = c:RUS
		c:RUS = ROOT

			owns_entire_state_region = STATE_OUTER_MANCHURIA

		}
		any_neighbouring_state = {
			OR = {
				state_region = s:STATE_OUTER_MANCHURIA
				state_region = s:STATE_AMUR
			}
		}
		any_country_in_qing_empire = {
			OR = {
				country_has_primary_culture = cu:manchu
				country_has_primary_culture = cu:han
			}
			any_scope_state = {
				state_region = s:STATE_OUTER_MANCHURIA
				is_capital = no
			}
			is_country_type = unrecognized
			is_player = no
		}
	}

	ai_chance = {
		value = 10
		if = {
			limit = {
				always = no
			}
			add = 90
		}
	}

	possible = {
		c:RUS ?= {
			country_rank >= rank_value:great_power
		}
		any_neighbouring_state = {
			state_region = s:STATE_OUTER_MANCHURIA
		}
		any_country_in_qing_empire = {
			any_scope_state = {
				state_region = s:STATE_OUTER_MANCHURIA
				is_capital = no
			}
			country_is_valid_qing_treaty_partner = yes
		}
	}

	effect = {
		random_country_in_qing_empire = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_OUTER_MANCHURIA
					is_capital = no
				}
				country_is_valid_qing_treaty_partner = yes
			}
			save_scope_as = donor_scope
		}
		if = {
			limit = {
				scope:donor_scope = {
					owes_obligation_to = ROOT
				}
			}
			custom_tooltip = obligation_for_outer_manchuria_tt
		}
		else_if = {
			limit = {
				scope:donor_scope = {
					is_subject_of = ROOT
				}
			}
			custom_tooltip = transfer_of_outer_manchuria_tt
		}
		else = {
			if = {
				limit = {
					exists = scope:donor_scope
				}
				c:RUS ?= {
					create_incident = {
						country = scope:donor_scope
						value = 10
					}
				}
			}
			custom_tooltip = transfer_of_outer_manchuria_tt
		}
		set_variable = rus_bought_outer_manchuria
		trigger_event = { id = russo_chinese.1 popup = yes }
	}
}

				is_country_type = decentralized
			}
		}

		trigger_if = {
			limit = {

					is_building_type = building_barrack

				}
			}
		}
		has_technology_researched = colonization
	}

	effect = {
		trigger_event = { id = russian_central_asia.3 popup = yes }

			if = {
				limit = {

				activate_law = law_type:law_frontier_colonization

					name = modifier_turkmen_subjugation
					days = normal_modifier_time
				}
			}
			else = {
				add_modifier = {
					name = modifier_turkmen_subjugation
					days = normal_modifier_time
					multiplier = 2
				}
			}
			add_modifier = {
				name = modifier_turkmen_logistics
				days = normal_modifier_time
				multiplier = money_amount_multiplier_very_large
			}
			add_radicals = {
				culture = cu:turkmen

				limit = { region = sr:region_central_asia }
