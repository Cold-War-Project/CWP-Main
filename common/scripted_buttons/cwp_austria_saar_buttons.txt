# Cold War Project: buttons for the Austrian State Treaty and the Saar.
# Journal entries: common/journal_entries/cwp_austria_saar.txt

# Austria: promise permanent neutrality in return for a State Treaty
cwp_at_neutrality_button = {
	name = "cwp_at_neutrality_button"
	desc = "cwp_at_neutrality_button_desc"

	visible = {
		c:AUS ?= this
		NOT = { has_global_variable = cwp_at_neutrality_pledged }
	}

	possible = {
		always = yes
	}

	effect = {
		set_global_variable = cwp_at_neutrality_pledged
	}

	ai_chance = {
		value = 1
		if = {
			limit = { has_global_variable = cwp_gq_frg_proclaimed }
			add = 5
		}
	}
}

# Four powers: consent to a State Treaty
cwp_at_consent_button = {
	name = "cwp_at_consent_button"
	desc = "cwp_at_consent_button_desc"

	visible = {
		OR = {
			c:USA ?= this
			c:RUS ?= this
		}
		NOT = { has_variable = cwp_at_consent }
	}

	possible = {
		always = yes
	}

	effect = {
		set_variable = cwp_at_consent
	}

	# Moscow's consent follows the German settlement: it accepted a neutral Austria once the Federal
	# Republic was sovereign and bound to the West (1955). Washington consents readily.
	ai_chance = {
		value = 0
		if = {
			limit = { c:USA ?= this }
			add = 15
		}
		if = {
			limit = {
				c:RUS ?= this
				has_global_variable = cwp_at_neutrality_pledged
			}
			add = 0.02
		}
		if = {
			limit = {
				c:RUS ?= this
				has_global_variable = cwp_at_neutrality_pledged
				OR = {
					has_global_variable = cwp_gq_frg_sovereign
					has_global_variable = cwp_gq_neutral_unification
				}
			}
			add = 15
		}
	}
}

# Soviet Union: run the former German assets in its zone for its own benefit
cwp_at_usia_button = {
	name = "cwp_at_usia_button"
	desc = "cwp_at_usia_button_desc"

	visible = {
		c:RUS ?= this
	}

	possible = {
		exists = c:AUS
		NOT = { has_modifier = cwp_at_usia }
	}

	selected = {
		has_modifier = cwp_at_usia
	}

	effect = {
		add_modifier = {
			name = cwp_at_usia
			days = short_modifier_time
		}
		c:AUS = {
			add_modifier = {
				name = cwp_at_usia_loss
				days = short_modifier_time
			}
		}
	}

	ai_chance = {
		value = 1
		if = {
			limit = { has_variable = cwp_at_consent }
			add = -1
		}
	}
}

# France and the Federal Republic: agree to put a European statute to the Saar's voters
cwp_saar_statute_button = {
	name = "cwp_saar_statute_button"
	desc = "cwp_saar_statute_button_desc"

	visible = {
		OR = {
			c:FRA ?= this
			c:GER ?= this
		}
		NOT = { has_variable = cwp_saar_agreed }
	}

	possible = {
		exists = c:FRA
		exists = c:GER
	}

	effect = {
		set_variable = cwp_saar_agreed
		custom_tooltip = cwp_saar_statute_button_tt
	}

	# The Paris Agreements (1954) paired Bonn's sovereignty with a Saar statute put to a vote.
	# In runs (batch 8c95) no button weighted below about 10 was acted on, so the trigger condition carries a full weight.
	ai_chance = {
		value = 1
		if = {
			limit = {
				OR = {
					has_global_variable = cwp_gq_frg_sovereign
					has_global_variable = cwp_gq_neutral_unification
				}
			}
			add = 15
		}
	}
}

# Austria: agree to pay the Soviet Union for the German assets it holds in its zone
cwp_at_buy_back_button = {
	name = "cwp_at_buy_back_button"
	desc = "cwp_at_buy_back_button_desc"

	visible = {
		c:AUS ?= this
		NOT = { has_global_variable = cwp_at_assets_bought }
	}

	possible = {
		exists = c:RUS
		custom_tooltip = {
			text = cwp_at_buy_back_needs_neutrality_tt
			has_global_variable = cwp_at_neutrality_pledged
		}
	}

	effect = {
		cwp_at_buy_back_assets = yes
	}

	ai_chance = {
		value = 0.5
		if = {
			limit = { c:RUS ?= { has_variable = cwp_at_consent } }
			add = 20
		}
	}
}
