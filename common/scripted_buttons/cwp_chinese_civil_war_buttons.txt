cwp_ccw_currency_reform_button = {
	name = "cwp_ccw_currency_reform_button"
	desc = "cwp_ccw_currency_reform_button_desc"

	visible = {
		c:ROC ?= this
		has_modifier = cwp_ccw_hyperinflation
	}

	possible = {
		in_default = no
		scaled_debt < 0.5
	}

	effect = {
		remove_modifier = cwp_ccw_hyperinflation
		add_modifier = { name = cwp_ccw_currency_reform }
		custom_tooltip = cwp_ccw_currency_reform_holds_tt
	}

	# A reform is a gamble taken when the war is going badly (the gold yuan, August 1948).
	ai_chance = {
		value = 0
		if = {
			limit = {
				global_var:cwp_ccw_balance >= 10
				global_var:cwp_ccw_balance < 25
			}
			add = 25
		}
	}
}

cwp_ccw_great_retreat_button = {
	name = "cwp_ccw_great_retreat_button"
	desc = "cwp_ccw_great_retreat_button_desc"

	visible = {
		c:ROC ?= this
		NOT = { has_variable = cwp_ccw_great_retreat }
	}

	possible = {
		cwp_ccw_war_running = yes
		has_state_in_state_region = STATE_FORMOSA
		cwp_ccw_roc_holds_formosa = yes
		OR = {
			custom_tooltip = {
				text = cwp_ccw_balance_communist_25_tt
				global_var:cwp_ccw_balance >= 25
			}
			custom_tooltip = {
				text = cwp_ccw_capital_occupied_tt
				capital = { NOT = { controller ?= root } }
			}
		}
	}

	effect = {
		set_variable = cwp_ccw_great_retreat
		set_capital = STATE_FORMOSA
		s:STATE_FORMOSA.region_state:ROC ?= {
			add_modifier = {
				name = cwp_ccw_relocation
				years = 10
				is_decaying = yes
			}
		}
		hidden_effect = {
			c:PRC ?= { trigger_event = { id = cwp_ccw.8 popup = yes } }
			c:USA ?= { trigger_event = { id = cwp_ccw.8 popup = yes } }
		}
	}

	ai_chance = {
		value = 5
		if = {
			limit = { global_var:cwp_ccw_balance >= 40 }
			add = 40
		}
		if = {
			limit = { capital = { NOT = { controller ?= root } } }
			add = 50
		}
		if = {
			limit = {
				any_scope_war = { has_war_support = { target = root value < 40 } }
			}
			add = 25
		}
	}
}

### Both Chinas

cwp_ccw_propose_armistice_button = {
	name = "cwp_ccw_propose_armistice_button"
	desc = "cwp_ccw_propose_armistice_button_desc"

	visible = {
		cwp_ccw_is_belligerent = yes
		cwp_ccw_war_running = yes
		cwp_ccw_roc_holds_mainland = yes
	}

	possible = {
		custom_tooltip = {
			text = cwp_ccw_armistice_cooldown_tt
			NOT = { has_global_variable = cwp_ccw_armistice_cd }
		}
		OR = {
			custom_tooltip = {
				text = cwp_ccw_war_support_low_tt
				any_scope_war = { has_war_support = { target = root value < 25 } }
			}
			custom_tooltip = {
				text = cwp_ccw_exhaustion_tt
				global_var:cwp_ccw_exhaustion >= 26
			}
		}
		custom_tooltip = {
			text = cwp_ccw_balance_undecided_tt
			global_var:cwp_ccw_balance > -40
			global_var:cwp_ccw_balance < 40
		}
		capital = { controller ?= root }
	}

	effect = {
		set_global_variable = { name = cwp_ccw_armistice_cd months = 12 }
		if = {
			limit = { c:PRC ?= this }
			set_global_variable = cwp_ccw_armistice_by_prc
			c:ROC ?= { trigger_event = { id = cwp_ccw.5 popup = yes } }
		}
		else = {
			remove_global_variable = cwp_ccw_armistice_by_prc
			c:PRC ?= { trigger_event = { id = cwp_ccw.5 popup = yes } }
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				any_scope_war = { has_war_support = { target = root value < 25 } }
			}
			add = 30
		}
		if = {
			limit = { global_var:cwp_ccw_exhaustion >= 52 }
			add = 10
		}
		# A side that is winning does not sue for peace, however tired (observer 2026-11-18: the PRC proposed at +25)
		if = {
			limit = {
				OR = {
					AND = {
						c:PRC ?= this
						global_var:cwp_ccw_balance >= 15
					}
					AND = {
						c:ROC ?= this
						global_var:cwp_ccw_balance <= -15
					}
				}
			}
			multiply = 0
		}
	}
}

cwp_ccw_break_ceasefire_button = {
	name = "cwp_ccw_break_ceasefire_button"
	desc = "cwp_ccw_break_ceasefire_button_desc"

	visible = {
		cwp_ccw_is_belligerent = yes
		has_global_variable = cwp_ccw_ceasefire
	}

	possible = {
		cwp_ccw_war_running = yes
	}

	effect = {
		remove_global_variable = cwp_ccw_ceasefire
		every_country = {
			limit = { has_modifier = cwp_ccw_ceasefire }
			remove_modifier = cwp_ccw_ceasefire
		}
		c:ROC ?= {
			if = {
				limit = { has_modifier = cwp_ccw_us_arms_embargo }
				remove_modifier = cwp_ccw_us_arms_embargo
			}
		}
		hidden_effect = {
			c:PRC ?= { trigger_event = { id = cwp_ccw.13 popup = yes } }
			c:ROC ?= { trigger_event = { id = cwp_ccw.13 popup = yes } }
			c:USA ?= { trigger_event = { id = cwp_ccw.13 popup = yes } }
		}
	}

	# The side the balance favours is the one that wants to fight on.
	ai_chance = {
		value = 2
		if = {
			limit = {
				OR = {
					AND = {
						c:PRC ?= this
						global_var:cwp_ccw_balance >= 0
					}
					AND = {
						c:ROC ?= this
						global_var:cwp_ccw_balance <= -10
					}
				}
			}
			add = 10
		}
		if = {
			limit = {
				OR = {
					AND = {
						c:PRC ?= this
						global_var:cwp_ccw_balance >= 10
					}
					AND = {
						c:ROC ?= this
						global_var:cwp_ccw_balance <= -20
					}
				}
			}
			add = 30
		}
	}
}
