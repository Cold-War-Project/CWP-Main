# Cold War Project: the French constituent assemblies. Country scope = FRA.

# The MRP's price: a second chamber to check the assembly (the Council of the Republic of the October draft).
cwp_fra_council_of_republic_button = {
	name = "cwp_fra_council_of_republic_button"
	desc = "cwp_fra_council_of_republic_button_desc"

	visible = {
		NOT = { has_variable = cwp_fra_council_conceded_var }
		NOT = { has_variable = cwp_fra_gaullist_draft_var }
	}

	possible = { always = yes }

	effect = {
		set_variable = cwp_fra_council_conceded_var
		if = {
			limit = { NOT = { has_variable = cwp_fra_compromise_var } }
			set_variable = { name = cwp_fra_compromise_var value = 0 }
		}
		change_variable = { name = cwp_fra_compromise_var add = 4 }
		custom_tooltip = cwp_fra_council_of_republic_tt
		ig:ig_trade_unions ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 1
			}
		}
		ig:ig_devout ?= {
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 1
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { var:cwp_fra_draft_current_var < 52 }
			add = 25
		}
	}
}

# De Gaulle's Bayeux programme: a president above the parties. A different republic, if the voters want it.
cwp_fra_strong_presidency_button = {
	name = "cwp_fra_strong_presidency_button"
	desc = "cwp_fra_strong_presidency_button_desc"

	visible = {
		NOT = { has_variable = cwp_fra_gaullist_draft_var }
	}

	possible = {
		ig:ig_armed_forces ?= { is_in_government = yes }
	}

	effect = {
		set_variable = cwp_fra_gaullist_draft_var
		custom_tooltip = cwp_fra_strong_presidency_tt
		if = {
			limit = { NOT = { has_variable = cwp_fra_compromise_var } }
			set_variable = { name = cwp_fra_compromise_var value = 0 }
		}
		change_variable = { name = cwp_fra_compromise_var add = -6 }
		ig:ig_trade_unions ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 2
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 2
			}
		}
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 2
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				ig:ig_armed_forces ?= { is_powerful = yes }
				ig:ig_trade_unions ?= { is_powerful = no }
			}
			add = 10
		}
	}
}

# Communists, Socialists and Christian Democrats govern together and keep the assembly working.

	}

}

cwp_fra_campaign_yes_button = {
	name = "cwp_fra_campaign_yes_button"
	desc = "cwp_fra_campaign_yes_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_iq_campaign_cap_tt
			OR = {
				NOT = { has_variable = cwp_fra_campaign_var }
				var:cwp_fra_campaign_var < 8
			}
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_variable = cwp_fra_campaign_var } }
			set_variable = { name = cwp_fra_campaign_var value = 0 }
		}
		change_variable = { name = cwp_fra_campaign_var add = 4 }
		custom_tooltip = cwp_fra_campaign_yes_tt
	}

	cooldown = { months = 4 }

	ai_chance = { value = 15 }
}
