# Cold War Project: IMF buttons (international organization template)
# Button root = the member country pressing it; the AI presses them through ai_chance.

# Ask the Fund for a stand-by arrangement: a loan in return for policy conditions.
cwp_imf_request_standby_button = {
	name = "cwp_imf_request_standby_button"
	desc = "cwp_imf_request_standby_button_desc"
	visible = { cwp_io_is_member = { ORG = imf } }
	possible = {
		custom_tooltip = {
			text = cwp_imf_in_difficulty_tt
			OR = {
				in_default = yes
				scaled_debt >= 0.5
				weeks_until_bankruptcy < 52
			}
		}
		custom_tooltip = {
			text = cwp_imf_no_request_tt
			NOT = { has_variable = cwp_io_applicant_imf_loan }
			NOT = { has_variable = cwp_io_rejected_imf_loan }
		}
	}
	effect = {
		custom_tooltip = cwp_imf_request_standby_tt
		hidden_effect = {
			set_variable = cwp_io_applicant_imf_loan
		}
	}
	ai_chance = {
		value = 20
		if = {
			limit = { in_default = yes }
			add = 60
		}
	}
}

cwp_imf_withdraw_button = {
	name = "cwp_imf_withdraw_button"
	desc = "cwp_imf_withdraw_button_desc"
	visible = { cwp_io_is_member = { ORG = imf } }
	possible = {
		NOT = { has_journal_entry = je_cwp_imf_programme }
	}
	effect = {
		cwp_io_remove_member = { ORG = imf }
		je:je_cwp_imf ?= { remove_involved_country = ROOT }
		set_variable = { name = cwp_io_rejected_imf years = 5 }
	}
	ai_chance = {
		value = 0
		if = {
			# Poland (1950), Czechoslovakia (1954) and Cuba (1964) left after turning east.
			limit = { cwp_io_in_eastern_camp = yes }
			add = 30
		}
	}
}

cwp_imf_admission_vote_yes_button = {
	name = "cwp_imf_admission_vote_yes_button"
	desc = "cwp_imf_admission_vote_yes_button_desc"
	visible = { cwp_io_is_member = { ORG = imf } }
	selected = { cwp_io_voted = { PROP = imf_admission VOTE = yes } }
	possible = {
		custom_tooltip = {
			text = cwp_io_can_vote_tt
			cwp_io_can_vote = { ORG = imf PROP = imf_admission }
		}
	}
	effect = { cwp_io_cast_vote = { PROP = imf_admission VOTE = yes } }
	ai_chance = {
		value = 40
		if = {
			limit = { global_var:cwp_imf_admission_target ?= { relations:root >= relations_threshold:cordial } }
			add = 30
		}
		if = {
			limit = { global_var:cwp_imf_admission_target ?= { relations:root <= relations_threshold:poor } }
			multiply = 0
		}
	}
}
