# Cold War Project: Belgium's Royal Question. Country scope = BEL.

cwp_rq_campaign_return_button = {
	name = "cwp_rq_campaign_return_button"
	desc = "cwp_rq_campaign_return_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_rq_campaign_cap_tt
			OR = {
				NOT = { has_variable = cwp_rq_campaign_var }
				var:cwp_rq_campaign_var < 9
			}
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_variable = cwp_rq_campaign_var } }
			set_variable = { name = cwp_rq_campaign_var value = 0 }
		}
		change_variable = { name = cwp_rq_campaign_var add = 3 }
		custom_tooltip = cwp_rq_campaign_return_tt
		ig:ig_trade_unions ?= {
			add_modifier = {
				name = cwp_rq_campaigned_against
				years = 1
			}
		}
	}

	cooldown = { months = 3 }

	ai_chance = {
		value = 0
		if = {
			limit = { ig:ig_devout ?= { is_in_government = yes } }
			add = 20
		}
	}
}

cwp_rq_campaign_against_button = {
	name = "cwp_rq_campaign_against_button"
	desc = "cwp_rq_campaign_against_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_rq_campaign_cap_tt
			OR = {
				NOT = { has_variable = cwp_rq_campaign_var }
				var:cwp_rq_campaign_var > -9
			}
		}
	}

	effect = {
		if = {
			limit = { NOT = { has_variable = cwp_rq_campaign_var } }
			set_variable = { name = cwp_rq_campaign_var value = 0 }
		}
		change_variable = { name = cwp_rq_campaign_var add = -3 }
		custom_tooltip = cwp_rq_campaign_against_tt
		ig:ig_devout ?= {
			add_modifier = {
				name = cwp_rq_campaigned_against
				years = 1
			}
		}
	}

	cooldown = { months = 3 }

	ai_chance = {
		value = 0
		if = {
			limit = {
				ig:ig_trade_unions ?= { is_in_government = yes }
				ig:ig_devout ?= { is_in_government = no }
			}
			add = 20
		}
	}
}

# The compromise: the King gives way to his son without a vote.
cwp_rq_abdicate_for_son_button = {
	name = "cwp_rq_abdicate_for_son_button"
	desc = "cwp_rq_abdicate_for_son_button_desc"

	visible = { always = yes }

	possible = {
		custom_tooltip = {
			text = cwp_rq_time_tt
			scope:journal_entry = { journal_entry_age >= 365 }
		}
	}

	effect = {
		trigger_event = { id = cwp_rq.4 popup = yes }
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				var:cwp_rq_return_current_var < 45
				ig:ig_trade_unions ?= { is_powerful = yes }
			}
			add = 10
		}
	}
}
