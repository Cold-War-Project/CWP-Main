# Grant Command to Ruler
grant_command_to_ruler = {
	icon = "gfx/interface/character_panel/grant_command.dds"
	clicksound = "event:/SFX/UI/Military/command_grant"

	potential = {
		is_ruler_of_own_country = yes
		NOR = {
			has_role_of_type = general
			has_role_of_type = admiral
		}
		owner ?= scope:actor
	}

	possible = {
		age >= define:NCharacters|ADULT_AGE
		OR = {
			is_female = no
			owner ?= { has_law_or_variant = law_type:law_womens_suffrage }
		}
		owner ?= { ruler_can_have_command = yes }
		exists = owner.capital.land_hq
	}

	effect = {
		add_character_role = general
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				AND = {
					age <= 50
					OR = {
						has_trait = reckless
						has_trait = direct
						has_trait = imposing
					}
					NOR = {
						has_trait = psychological_affliction
						has_trait = wounded
						has_trait = shellshocked
					}
				}
			}
			add = 100
		}
	}
}
