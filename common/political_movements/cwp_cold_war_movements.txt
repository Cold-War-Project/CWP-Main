# Additive Cold War movements using the installed vanilla political movement schema.
# Creation/disband triggers run in country scope; character and pop weights use their native scopes.
# Pop support factors describe the implemented weights; they do not supply the weights themselves.

movement_cwp_orthodox_communist = {
	category = movement_category_ideological
	ideology = ideology_cwp_orthodox_communist_movement
	character_ideologies = {
		ideology_cwp_marxist_leninist
	}

	creation_trigger = {
		game_date >= 1946.1.1
		OR = {
			pop_type_percent_country = { pop_type = laborers percent > 0.10 }
			has_law_or_variant = law_type:law_council_republic
		}
		NOT = {
			AND = {
				has_law_or_variant = law_type:law_council_republic
				has_law_or_variant = law_type:law_single_party_state
				has_law_or_variant = law_type:law_command_economy
				OR = {
					has_law_or_variant = law_type:law_corporatized_unions
					has_law_or_variant = law_type:law_factory_councils
				}
			}
		}
	}
	creation_weight = { value = 25 }
	on_created = { add_movement_enthusiasm_modifier = yes }
	disband_trigger = {
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_single_party_state
		has_law_or_variant = law_type:law_command_economy
		OR = {
			has_law_or_variant = law_type:law_corporatized_unions
			has_law_or_variant = law_type:law_factory_councils
		}
	}

	can_pressure_interest_group = {
		OR = {
			is_interest_group_type = ig_armed_forces
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_trade_unions
		}
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_cwp_marxist_leninist
			has_ideology = ideology:ideology_vanguardist
			has_ideology = ideology:ideology_communist
			has_ideology = ideology:ideology_cwp_destalinizer
			has_ideology = ideology:ideology_cwp_castroist
			has_ideology = ideology:ideology_cwp_hoxhaist
			has_ideology = ideology:ideology_cwp_juche
			has_ideology = ideology:ideology_cwp_soviet_conservative
		}
	}
	character_support_weight = { value = 300 }

	pop_support_trigger = {
		NOR = {
			is_pop_type = capitalists
			is_pop_type = aristocrats
			is_pop_type = slaves
		}
	}
	pop_support_factors = {
		movement_support_laborers
		movement_support_machinists
		movement_support_academics
		movement_support_bureaucrats
		movement_support_high_urbanization
		movement_support_below_expected_sol
		movement_support_radicalization
	}
	pop_support_weight = {
		add = { value = 5 desc = "POP_BASE_SUPPORT" }
		if = {
			limit = { is_pop_type = laborers }
			add = { value = 20 desc = "POP_LABORERS" }
		}
		if = {
			limit = { is_pop_type = machinists }
			add = { value = 25 desc = "POP_MACHINISTS" }
		}
		if = {
			limit = { is_pop_type = academics }
			add = { value = 10 desc = "POP_ACADEMICS" }
		}
		if = {
			limit = { is_pop_type = bureaucrats }
			add = { value = 10 desc = "POP_BUREAUCRATS" }
		}
		add = { value = state.state_urbanization_rate multiply = 10 desc = "POP_HIGH_URBANIZATION" }
		if = {
			limit = { standard_of_living < state.average_expected_sol }
			add = { value = state.average_expected_sol subtract = standard_of_living max = 10 desc = "IS_BELOW_EXPECTED_SOL" }
		}
		multiply = { value = pop_radical_fraction add = 1 desc = "POP_RADICALS" }
		if = {
			limit = { exists = scope:political_movement scope:political_movement ?= { is_being_suppressed = yes } }
			add = {
				value = suppression_base_value
				multiply = { value = -0.5 add = owner.modifier:country_suppression_attraction_factor }
				desc = "concept_suppressed"
			}
		}
		else_if = {
			limit = { exists = scope:political_movement scope:political_movement ?= { is_being_bolstered = yes } }
			add = {
				value = bolstering_base_value
				multiply = { value = 0.5 add = owner.modifier:country_bolster_attraction_factor }
				desc = "concept_bolstered"
			}
		}
		min = 0
	}

	revolution = {
		possible = { political_movement_support >= 0.10 }
		weight = { value = 100 }
	}
}
