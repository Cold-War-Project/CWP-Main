# Base game political movements that needed fixing for the mod
REPLACE:movement_hindu_nationalist = {
	category = movement_category_religious

	ideology = ideology_traditionalist_minoritarian_movement

	character_ideologies = {
		ideology_fascist
		ideology_theocrat
	}

	creation_trigger = {
		# Can only ever spawn once
		NOT = { any_political_movement = { is_political_movement_type = movement_hindu_nationalist } }
	}

	can_cause_obstinance = yes

	creation_weight = {
		value = 100
	}

	on_created = {}

	religion_selection_trigger = {
		this = rel:hindu
	}

	religion_selection_weight = {
		value = 100
	}

	character_support_trigger = {
		religion = rel:hindu
	}

	character_support_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_traditionalist_minoritarian
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_theocrat
				}
			}
			multiply = { value = 5 }
		}
	}

	pop_support_trigger = {
		religion = rel:hindu
	}

	pop_support_factors = {
		movement_support_hindu_religion
		movement_support_low_literacy
		movement_support_academics
		movement_support_clergymen
		movement_support_bureaucrats
		movement_support_clerks
		movement_support_corporatism_tech
		movement_support_je_communal_divides
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}

			if = {
				limit = { scope:political_movement = { is_being_suppressed = yes } }
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = {
			limit = { religion = rel:hindu }
			add = {
				value = 9
				desc = "POP_HINDU"
			}

			if = {
				limit = { is_pop_type = academics }
				add = {
					value = 12
					desc = "POP_ACADEMICS"
				}
			}

			if = {
				limit = { is_pop_type = clergymen }
				add = {
					value = 25
					desc = "POP_CLERGYMEN"
				}
			}

			if = {
				limit = { is_pop_type = bureaucrats }
				add = {
					value = 9
					desc = "POP_BUREAUCRATS"
				}
			}

			if = {
				limit = { is_pop_type = clerks }
				add = {
					value = 12
					desc = "POP_CLERKS"
				}
			}

			add = { # Weakened by increasing literacy rates
				desc = "POP_LITERACY"
				this = {
					value = literacy_rate
					multiply = -2
				}
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		interest_group_can_join = {
			OR = {
				is_interest_group_type = ig_devout
				leader = { religion = scope:political_movement.religion } # Muslims should never join this
			}
		}

		state_weight = {
			value = 1
		}
	}

	secession = {
		possible = { 
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		interest_group_can_join = {
			OR = {
				is_interest_group_type = ig_devout
				leader = { religion = scope:political_movement.religion } # Muslims should never join this
			}
		}

		forced_tags = {}

		state_weight = {
			value = 1
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = { owner = { is_subject = yes } }
			add = {
				desc = "LIBERTY_DESIRE"
				value = owner.liberty_desire
				divide = 500
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_hindu_nationalist
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}

		if = {
			limit = { this.days_since_movement_defeated <= very_long_modifier_time }
			add = { # (x-y)/y - ([1, 7300]-7300)/7300
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
}

	}

		value = 100

		}
	}

			religion = rel:ibadi
		}
	}

	pop_support_factors = {
		movement_support_muslim_religion
		movement_support_low_literacy
		movement_support_academics
		movement_support_clergymen
		movement_support_bureaucrats
		movement_support_clerks
		movement_support_corporatism_tech
		movement_support_je_communal_divides
	}

	pop_support_weight = {
		if = {
			limit = {

						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}

				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"

					religion = rel:sunni
					religion = rel:shiite
					religion = rel:ibadi
				}
			}
			add = { # Base acceptance values higher due to being a minority
				value = 25
				desc = "POP_MUSLIM"

			if = {
				limit = { is_pop_type = academics }
				add = {
					value = 20
					desc = "POP_ACADEMICS"
				}
			}

			if = {
				limit = { is_pop_type = clergymen }
				add = {
					value = 40
					desc = "POP_ACADEMICS"
				}
			}

			if = {
				limit = { is_pop_type = bureaucrats }
				add = {
					value = 20
					desc = "POP_BUREAUCRATS"
				}
			}

			if = {

		}

		state_weight = {
			value = 1
			if = {
				limit = { NOT = { state_is_homeland_of_indian_cultures = yes } }
				multiply = 0
			}

			add = {
				desc = "LIBERTY_DESIRE"

				divide = 500
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }

								is_insurrectionary = yes
							}

					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}

		if = {
			limit = {
				owner = {
					any_political_movement = {
						NOT = { this = root } # Insurrectionary movements should not make themselves more insurrectionary
						is_insurrectionary = yes
					}
				}
			}
			add = {
				desc = "SIMULTANEOUS_INSURRECTIONARY_MOVEMENTS"
				owner = {
					every_political_movement = {
						limit = {
							NOT = { this = root } # Insurrectionary movements should not make themselves more insurrectionary
							is_insurrectionary = yes
						}
						if = {
							limit = { exists = this.political_movement_identity_support }
							add = { # Identity support can get a bit high
								value = this.political_movement_identity_support
								multiply = 0.5
							}
						}
						else = {
							add = this.political_movement_support
						}
					}
				}
				min = 0.05
				max = 0.25
			}
		}

		if = {
			limit = { this.days_since_movement_defeated <= very_long_modifier_time }
			add = { # (x-y)/y - ([1, 7300]-7300)/7300
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = { value = very_long_modifier_time }
			}
		}
	}
