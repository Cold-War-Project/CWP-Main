# Cold War Project: bars for the German question, chapter 3 (Die Wende).
# Journal entry: common/journal_entries/cwp_die_wende.txt

# How far the peaceful revolution in the GDR has gone
cwp_dw_revolution_bar = {
	name = "cwp_dw_revolution_bar"
	desc = "cwp_dw_revolution_bar_desc"

	default_green = yes

	start_value = 30
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = {
				c:DDR ?= { radical_fraction = { value >= 0.1 } }
			}
			add = { desc = "cwp_dw_revolution_bar_protests" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_dw_wall_open }
			add = { desc = "cwp_dw_revolution_bar_wall_open" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_tg_support_withdrawn }
			add = { desc = "cwp_dw_revolution_bar_no_tanks" value = 1 }
		}
		progress_condition = {
			limit = { c:DDR ?= { has_modifier = cwp_dw_round_table } }
			add = { desc = "cwp_dw_revolution_bar_round_table" value = 1 }
		}
		progress_condition = {
			limit = { c:DDR ?= { has_modifier = cwp_dw_crackdown } }
			add = { desc = "cwp_dw_revolution_bar_crackdown" value = -3 }
		}
		progress_condition = {
			limit = {
				NOT = { has_global_variable = cwp_tg_support_withdrawn }
				exists = c:RUS
			}
			add = { desc = "cwp_dw_revolution_bar_soviet_troops" value = -1 }
		}
	}
}
