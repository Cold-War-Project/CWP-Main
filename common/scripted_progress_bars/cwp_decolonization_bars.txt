# Cold War Project: decolonization progress bars

### The Winds of Change: the anti-colonial tide (0 to 100)
# Rises only with what happens in play: independences, the voice of free former colonies, colonial wars,
# and the superpowers' stances. Falls when a metropole crushes a rising.
cwp_tide_bar = {
	name = "cwp_tide_bar"
	desc = "cwp_tide_bar_desc"
	default = yes

	start_value = 5
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_new_independences
				global_var:cwp_tide_new_independences > 0
			}
			add = {
				desc = "cwp_tide_bar_new_independences"
				value = global_var:cwp_tide_new_independences
				multiply = 2
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_free_former_colonies
				global_var:cwp_free_former_colonies > 0
			}
			add = {
				desc = "cwp_tide_bar_free_former_colonies"
				value = global_var:cwp_free_former_colonies
				multiply = 0.01
				max = 0.25
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_colonial_wars
				global_var:cwp_colonial_wars > 0
			}
			add = {
				desc = "cwp_tide_bar_colonial_wars"
				value = global_var:cwp_colonial_wars
				multiply = 0.3
			}
		}
		progress_condition = {
			limit = { c:USA ?= { has_variable = cwp_champions_self_determination } }
			add = {
				desc = "cwp_tide_bar_usa_champion"
				value = 0.1
			}
		}
		progress_condition = {
			limit = { c:USA ?= { has_variable = cwp_guards_allied_empires } }
			add = {
				desc = "cwp_tide_bar_usa_guardian"
				value = -0.2
			}
		}
		progress_condition = {
			limit = { c:RUS ?= { has_variable = cwp_champions_self_determination } }
			add = {
				desc = "cwp_tide_bar_rus_champion"
				value = 0.1
			}
		}
		progress_condition = {
			limit = { c:RUS ?= { has_variable = cwp_guards_allied_empires } }
			add = {
				desc = "cwp_tide_bar_rus_guardian"
				value = -0.2
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_setbacks
				global_var:cwp_tide_setbacks > 0
			}
			add = {
				desc = "cwp_tide_bar_setbacks"
				value = global_var:cwp_tide_setbacks
				multiply = -4
			}
		}
	}
}

### India: communal tension (independence itself reads India's liberty desire)

cwp_india_communal_bar = {
	name = "cwp_india_communal_bar"
	desc = "cwp_india_communal_bar_desc"
	default_bad = yes

	start_value = 55
	min_value = 0
	max_value = 100

	monthly_progress = {
		add = {
			desc = "cwp_india_communal_two_nations"
			value = 0.6
		}
		c:HND ?= {
			progress_condition = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_muslim_nationalist
						political_movement_support >= 0.05
					}
				}
				add = {
					desc = "cwp_india_communal_muslim_movement"
					value = 0.5
				}
			}
			progress_condition = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_hindu_nationalist
						political_movement_support >= 0.05
					}
				}
				add = {
					desc = "cwp_india_communal_hindu_movement"
					value = 0.25
				}
			}
			progress_condition = {
				limit = { has_variable = cwp_india_unity_talks }
				add = {
					desc = "cwp_india_communal_unity_talks"
					value = -1.5
				}
			}
			progress_condition = {
				limit = { has_variable = cwp_india_direct_action }
				add = {
					desc = "cwp_india_communal_direct_action"
					value = 3
				}
			}
		}
		c:GBR ?= {
			progress_condition = {
				limit = { has_variable = cwp_india_cabinet_mission }
				add = {
					desc = "cwp_india_communal_cabinet_mission"
					value = -1.5
				}
			}
			progress_condition = {
				limit = { has_variable = cwp_india_restore_order }
				add = {
					desc = "cwp_india_communal_divide_and_rule"
					value = 1
				}
			}
		}
	}
}

### Algeria: insurgency (0 to 100); the metropole's side runs through the Armed Forces IG and a strongman

cwp_algeria_insurgency_bar = {
	name = "cwp_algeria_insurgency_bar"
	desc = "cwp_algeria_insurgency_bar_desc"
	default_bad = yes

	start_value = 25
	min_value = 0
	max_value = 100

	monthly_progress = {
		add = {
			desc = "cwp_algeria_bar_fln"
			value = 0.75
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_tier
				global_var:cwp_tide_tier >= 1
			}
			add = {
				desc = "cwp_bar_tide_rising"
				value = 0.25
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_tier
				global_var:cwp_tide_tier >= 3
			}
			add = {
				desc = "cwp_bar_tide_strong"
				value = 0.25
			}
		}
		progress_condition = {
			limit = {
				OR = {
					c:MOR ?= { is_subject = no }
					c:TUN ?= { is_subject = no }
				}
			}
			add = {
				desc = "cwp_algeria_bar_sanctuaries"
				value = 0.5
			}
		}
		progress_condition = {
			limit = { c:EGY ?= { is_subject = no } }
			add = {
				desc = "cwp_algeria_bar_cairo"
				value = 0.25
			}
		}
		c:FRA ?= {
			progress_condition = {
				limit = { has_variable = cwp_france_lost_indochina }
				add = {
					desc = "cwp_algeria_bar_indochina_lost"
					value = 0.5
				}
			}
			progress_condition = {
				limit = { has_variable = cwp_algeria_pacification }
				add = {
					desc = "cwp_algeria_bar_pacification"
					value = -1
				}
			}
			progress_condition = {
				limit = { has_variable = cwp_algeria_constantine_plan }
				add = {
					desc = "cwp_algeria_bar_constantine_plan"
					value = -0.5
				}
			}
		}
	}
}
