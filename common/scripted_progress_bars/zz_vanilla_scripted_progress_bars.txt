REPLACE:divided_monarchists_bonapartist_progress_bar = {
	name = "divided_monarchists_bonapartist_progress_bar"
	desc = "divided_monarchists_bonapartist_progress_bar_desc"

	default = yes

	monthly_progress = {
		if = {
			limit = {
				NOT = {
					has_variable = fra_divided_monarchists_freeze #This variable stops iteration, and is granted by reaching 360 with one dynasty. It's reset when this JE is completed, or when je_cement_the_rightful_dynasty fails.
				}
			}
			# From Characters - combined prominence of every Bonapartist character (prominence is 0/100, so divide by 200).
			progress_condition = {
				limit = {
					any_scope_character = {
						has_ideology = ideology:ideology_authoritarian
					}
				}
				add = {
					desc = "bonapartist_progress_from_characters"
					every_scope_character = {
						limit = {
							has_ideology = ideology:ideology_authoritarian
						}
						add = this.prominence
					}
					divide = 200
				}
			}
			# From Interest Groups - clout of every Bonapartist-led IG, weighted double if in government.
			progress_condition = {
				limit = {
					any_interest_group = {
						leader = {
							has_ideology = ideology:ideology_authoritarian
						}
					}
				}
				add = {
					desc = "bonapartist_progress_from_igs"
					every_interest_group = {
						limit = {
							leader = {
								has_ideology = ideology:ideology_authoritarian
							}
						}
						add = {
							value = this.ig_clout
							progress_condition = {
								limit = {
									is_in_government = yes
								}
								multiply = 2
							}
						}
					}
				}
			}
			# From Buildings - sum of occupied military building levels (0.0033 per level).
			progress_condition = {
				limit = {
					any_scope_building = {
						is_building_group = bg_military # Derives from barracks and naval bases.
						occupancy >= 0.5
					}
				}
				add = {
					desc = "bonapartist_progress_from_military"
					every_scope_building = {
						limit = {
							is_building_group = bg_military # Derives from barracks and naval bases.
							occupancy >= 0.5
						}
						add = this.level
					}
					divide = 300
				}
			}
			# From Movements - support of any Bonapartist movement.
			progress_condition = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_bonapartist
					}
				}
				add = {
					desc = "bonapartist_progress_from_movements"
					every_political_movement = {
						limit = {
							is_political_movement_type = movement_bonapartist
						}
						add = {
							value = this.political_movement_support
							multiply = 5
						}
					}
				}
			}
			# From Legitimacy - balanced around 0.5; a Bonapartist ruler benefits from high legitimacy, otherwise low legitimacy.
			progress_condition = {
				limit = {
					ruler_is_bonapartist = yes
				}
				add = {
					desc = "bonapartist_progress_from_legitimacy"
					value = government_legitimacy
					divide = 100
					subtract = 0.5
				}
				else = {
					add = {
						desc = "bonapartist_progress_from_legitimacy"
						value = government_legitimacy
						divide = 100
						subtract = 0.5
						multiply = -0.5
					}
				}
			}
			# From Effects - scripted bonuses/maluses granted manually by events.
			progress_condition = {
				limit = {
					has_variable = bonapartist_progress_from_effects
				}
				add = {
					desc = "bonapartist_progress_from_effects"
					value = var:bonapartist_progress_from_effects
				}
			}
		}
	}

							weekly_profit > 0
						}
						add = this.level
					}
					divide = 200
				}
			}
			# From Movements - support of any Legitimist movement.
			progress_condition = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_legitimist
					}
				}
				add = {
					desc = "legitimist_progress_from_movements"
					every_political_movement = {
						limit = {
							is_political_movement_type = movement_legitimist
						}
						add = {
							value = this.political_movement_support
							multiply = 5
						}
					}
				}
			}
			# From Legitimacy - balanced around 0.5; a Legitimist ruler benefits from high legitimacy, otherwise low legitimacy.
			progress_condition = {
				limit = {
					ruler_is_legitimist = yes
				}
				add = {
					desc = "legitimist_progress_from_legitimacy"
					value = government_legitimacy
					divide = 100
					subtract = 0.5
				}
				else = {
					add = {
						desc = "legitimist_progress_from_legitimacy"
						value = government_legitimacy
						divide = 100
						subtract = 0.5
						multiply = -0.5
					}
				}
			}
			# From Effects - scripted bonuses/maluses granted manually by events.
			progress_condition = {
				limit = {
					has_variable = legitimist_progress_from_effects
				}
				add = {
					desc = "legitimist_progress_from_effects"
					value = var:legitimist_progress_from_effects
				}
			}
		}
	}
