# Cold War Project: bars for the German question, chapter 1 (occupation and division).
# Journal entry: common/journal_entries/cwp_german_question.txt
# Every factor reads simulation state or a choice a country made; none reads the date.

# How far the western zones have moved toward a West German state
cwp_gq_western_bar = {
	name = "cwp_gq_western_bar"
	desc = "cwp_gq_western_bar_desc"

	default_green = yes

	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = { has_global_variable = cwp_gq_bizone }
			add = { desc = "cwp_gq_western_bar_bizone" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_trizone }
			add = { desc = "cwp_gq_western_bar_trizone" value = 1 }
		}
		progress_condition = {
			limit = {
				any_country = {
					cwp_gq_is_west_german = yes
					has_modifier = cwp_gq_marshall_aid
				}
			}
			add = { desc = "cwp_gq_western_bar_marshall_aid" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_currency_reform }
			add = { desc = "cwp_gq_western_bar_currency_reform" value = 1 }
		}
		progress_condition = {
			limit = {
				any_country = {
					cwp_gq_is_west_german = yes
					has_modifier = cwp_gq_self_government
				}
			}
			add = { desc = "cwp_gq_western_bar_self_government" value = 0.5 }
		}
		progress_condition = {
			limit = {
				exists = c:DDR
				any_country = {
					cwp_gq_is_west_german = yes
					average_sol > c:DDR.average_sol
				}
			}
			add = { desc = "cwp_gq_western_bar_recovery" value = 0.5 }
		}
		progress_condition = {
			limit = {
				je:je_cwp_german_question ?= { "scripted_bar_progress(cwp_gq_soviet_bar)" >= 50 }
			}
			add = { desc = "cwp_gq_western_bar_fear_of_the_east" value = 0.5 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_airlift }
			add = { desc = "cwp_gq_western_bar_airlift" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_french_reservations }
			add = { desc = "cwp_gq_western_bar_french_reservations" value = -1 }
		}
		progress_condition = {
			limit = {
				je:je_cwp_german_question ?= { "scripted_bar_progress(cwp_gq_unity_bar)" >= 60 }
			}
			add = { desc = "cwp_gq_western_bar_four_power_hopes" value = -0.5 }
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_gq_frg_proclaimed
				NOT = { has_global_variable = cwp_gq_frg_sovereign }
			}
			add = { desc = "cwp_gq_western_bar_occupation_statute" value = -1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_berlin_abandoned }
			add = { desc = "cwp_gq_western_bar_berlin_abandoned" value = -1 }
		}
		progress_condition = {
			limit = {
				any_country = {
					cwp_gq_is_west_german = yes
					radical_fraction = { value >= 0.2 }
				}
			}
			add = { desc = "cwp_gq_western_bar_radicals" value = -0.5 }
		}
	}
}

# How firmly the Socialist Unity Party holds the Soviet zone

			add = { desc = "cwp_gq_soviet_bar_land_reform" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_sed_formed }
			add = { desc = "cwp_gq_soviet_bar_sed" value = 1.5 }

		progress_condition = {
			limit = { c:DDR ?= { has_modifier = cwp_gq_dismantling } }
			add = { desc = "cwp_gq_soviet_bar_dismantling" value = 0.5 }
		}
		progress_condition = {
			limit = { c:DDR ?= { has_modifier = cwp_gq_antifascist_bloc } }
			add = { desc = "cwp_gq_soviet_bar_antifascist_bloc" value = 0.5 }
		}
		progress_condition = {
			limit = {

			}

		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_currency_reform }

		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_blockade }
			add = { desc = "cwp_gq_soviet_bar_blockade" value = 0.5 }
		}
		progress_condition = {
			limit = {
				c:DDR ?= { radical_fraction = { value >= 0.2 } }
			}
			add = { desc = "cwp_gq_soviet_bar_radicals" value = -1 }
		}
		progress_condition = {
			limit = {
				exists = c:DDR

			}
			add = { desc = "cwp_gq_soviet_bar_not_held" value = -2 }
		}
	}
}

# Whether the four powers still try to govern Germany together

	desc = "cwp_gq_unity_bar_desc"

	default = yes

	max_value = 100

			limit = {

		progress_condition = {
			limit = { has_global_variable = cwp_gq_bizone }

			limit = { has_global_variable = cwp_gq_currency_reform }
			add = { desc = "cwp_gq_unity_bar_currency_reform" value = -1 }

}
