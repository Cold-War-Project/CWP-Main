# Cold War Project: bars for je_cwp_divided_peninsula

# Balance on the peninsula. Negative: the South holds the advantage. Positive: the North does.
cwp_korea_balance_bar = {
	name = "cwp_korea_balance_bar"
	desc = "cwp_korea_balance_bar_desc"
	second_desc = "cwp_korea_balance_bar_second_desc"

	progress_texture = "gfx/interface/progressbar/progressbar_bad.dds"
	background_texture = "gfx/interface/progressbar/progressbar_gold.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_bad_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_bad_glow_line.dds"
	marker = yes

	start_value = 5
	min_value = -50
	max_value = 50

	monthly_progress = {
		# Occupation troops
		progress_condition = {
			limit = { cwp_korea_balance_active = yes NOT = { has_global_variable = cwp_korea_us_withdrawn } }
			add = { desc = "cwp_korea_balance_us_troops" value = -1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes NOT = { has_global_variable = cwp_korea_soviet_withdrawn } }
			add = { desc = "cwp_korea_balance_soviet_troops" value = 1 }
		}
		# Patrons
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK ?= { has_modifier = cwp_korea_soviet_arms } }
			add = { desc = "cwp_korea_balance_soviet_arms" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:STK ?= { has_modifier = cwp_korea_us_aid } }
			add = { desc = "cwp_korea_balance_us_aid" value = -1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK ?= { has_modifier = cwp_korea_returned_divisions } }
			add = { desc = "cwp_korea_balance_returned_divisions" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes
				exists = c:USA
				c:USA ?= {
					any_scope_treaty = {
						binds = c:STK
						any_scope_article = {
							OR = {
								has_type = defensive_pact
								has_type = alliance
							}
						}
					}
				}
			}
			add = { desc = "cwp_korea_balance_us_treaty" value = -3 }
		}
		# Internal
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:STK ?= { has_modifier = cwp_korea_southern_insurgency } }
			add = { desc = "cwp_korea_balance_southern_insurgency" value = 1 }
		}
		# Armies
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK.army_power_projection > c:STK.army_power_projection }
			add = { desc = "cwp_korea_balance_north_army" value = 1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK.army_power_projection < c:STK.army_power_projection }
			add = { desc = "cwp_korea_balance_south_army" value = -1 }
		}
	}
}
