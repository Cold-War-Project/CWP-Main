# Cold War Project: bars for je_cwp_divided_peninsula

# The US-Soviet Joint Commission on a unified provisional Korean government.
# 100: the commission agrees (neutral unified Korea). 0: the occupiers go separate ways.
cwp_korea_commission_bar = {
	name = "cwp_korea_commission_bar"
	desc = "cwp_korea_commission_bar_desc"

	default_green = yes

	start_value = 40
	min_value = 0
	max_value = 100

	monthly_progress = {
		# Superpower relations
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:USA ?= { "relations(c:RUS)" >= 50 } }
			add = { desc = "cwp_korea_commission_relations_good" value = 0.5 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:USA ?= { "relations(c:RUS)" >= 0 "relations(c:RUS)" < 50 } }
			add = { desc = "cwp_korea_commission_relations_cool" value = -0.5 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:USA ?= { "relations(c:RUS)" >= -50 "relations(c:RUS)" < 0 } }
			add = { desc = "cwp_korea_commission_relations_poor" value = -1 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:USA ?= { "relations(c:RUS)" < -50 } }
			add = { desc = "cwp_korea_commission_relations_hostile" value = -1.5 }
		}
		# Korean opinion
		progress_condition = {
			limit = { cwp_korea_commission_active = yes }
			add = { desc = "cwp_korea_commission_korean_opposition" value = -0.5 }
		}
		# Occupier choices
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:USA ?= { has_modifier = cwp_korea_commission_concessions } }
			add = { desc = "cwp_korea_commission_us_concessions" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:RUS ?= { has_modifier = cwp_korea_commission_concessions } }
			add = { desc = "cwp_korea_commission_soviet_concessions" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes has_global_variable = cwp_korea_un_referral }
			add = { desc = "cwp_korea_commission_un_referral" value = -3 }
		}
		progress_condition = {
			limit = { cwp_korea_commission_active = yes c:RUS ?= { has_modifier = cwp_korea_peoples_committees } }
			add = { desc = "cwp_korea_commission_peoples_committees" value = -1.5 }
		}
	}
}

# Balance on the peninsula. Negative: the South holds the advantage. Positive: the North does.
cwp_korea_balance_bar = {
	name = "cwp_korea_balance_bar"
	desc = "cwp_korea_balance_bar_desc"
	second_desc = "cwp_korea_balance_bar_second_desc"

	progress_texture = "gfx/interface/progressbar/progressbar_bad.dds"
	background_texture = "gfx/interface/progressbar/progressbar_gold.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_bad_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_bad_glow_line.dds"
	marker = yes

	start_value = 5
	min_value = -50
	max_value = 50

	monthly_progress = {
		# Occupation troops
		progress_condition = {
			limit = { cwp_korea_balance_active = yes NOT = { has_global_variable = cwp_korea_us_withdrawn } }
			add = { desc = "cwp_korea_balance_us_troops" value = -1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes NOT = { has_global_variable = cwp_korea_soviet_withdrawn } }
			add = { desc = "cwp_korea_balance_soviet_troops" value = 1 }
		}
		# Patrons
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK ?= { has_modifier = cwp_korea_soviet_arms } }
			add = { desc = "cwp_korea_balance_soviet_arms" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:STK ?= { has_modifier = cwp_korea_us_aid } }
			add = { desc = "cwp_korea_balance_us_aid" value = -1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK ?= { has_modifier = cwp_korea_returned_divisions } }
			add = { desc = "cwp_korea_balance_returned_divisions" value = 2 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes
				exists = c:USA
				c:USA ?= {
					any_scope_treaty = {
						binds = c:STK
						any_scope_article = {
							OR = {
								has_type = defensive_pact
								has_type = alliance
							}
						}
					}
				}
			}
			add = { desc = "cwp_korea_balance_us_treaty" value = -3 }
		}
		# Internal
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:STK ?= { has_modifier = cwp_korea_southern_insurgency } }
			add = { desc = "cwp_korea_balance_southern_insurgency" value = 1 }
		}
		# Armies
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK.army_power_projection > c:STK.army_power_projection }
			add = { desc = "cwp_korea_balance_north_army" value = 1 }
		}
		progress_condition = {
			limit = { cwp_korea_balance_active = yes c:NRK.army_power_projection < c:STK.army_power_projection }
			add = { desc = "cwp_korea_balance_south_army" value = -1 }
		}
	}
}
