# Cold War Project: bars for je_cwp_two_koreas (Korea chapter 2)

# Tension along the demilitarized zone. 75 and above is a crisis; 80 and above makes a new war possible.
cwp_tkor_tension_bar = {
	name = "cwp_tkor_tension_bar"
	desc = "cwp_tkor_tension_bar_desc"

	default_bad = yes

	start_value = 60
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = { always = yes }
			add = { desc = "cwp_tkor_tension_fading" value = -0.5 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_modifier = cwp_tkor_provocations } }
			add = { desc = "cwp_tkor_tension_provocations" value = 3 }
		}
		progress_condition = {
			limit = { c:STK ?= { has_modifier = cwp_tkor_retaliation } }
			add = { desc = "cwp_tkor_tension_retaliation" value = 1 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_variable = cwp_nuc_programme_var } }
			add = { desc = "cwp_tkor_tension_nuclear_programme" value = 1 }
		}
		progress_condition = {
			limit = { c:NRK ?= { cwp_is_nuclear_power = yes } }
			add = { desc = "cwp_tkor_tension_nuclear_power" value = 1 }
		}
		progress_condition = {
			limit = {
				exists = c:STK
				NOT = { cwp_tkor_us_forces = yes }
				NOT = { cwp_tkor_us_treaty = yes }
			}
			add = { desc = "cwp_tkor_tension_no_guarantee" value = 1 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_modifier = cwp_tkor_restraint } }
			add = { desc = "cwp_tkor_tension_restraint" value = -2 }
		}
		progress_condition = {
			limit = {
				c:NRK ?= { has_modifier = cwp_tkor_dialogue }
				c:STK ?= { has_modifier = cwp_tkor_dialogue }
			}
			add = { desc = "cwp_tkor_tension_dialogue_both" value = -2 }
		}
		progress_condition = {
			limit = {
				OR = {
					c:NRK ?= { has_modifier = cwp_tkor_dialogue }
					c:STK ?= { has_modifier = cwp_tkor_dialogue }
				}
				NOT = {
					c:NRK ?= { has_modifier = cwp_tkor_dialogue }
					c:STK ?= { has_modifier = cwp_tkor_dialogue }
				}
			}
			add = { desc = "cwp_tkor_tension_dialogue_one" value = -0.5 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_modifier = cwp_tkor_engagement } }
			add = { desc = "cwp_tkor_tension_engagement" value = -1 }
		}
	}
}

# Reconciliation between the two Koreas. At 100 both governments may agree to reunify.
cwp_tkor_reconciliation_bar = {
	name = "cwp_tkor_reconciliation_bar"
	desc = "cwp_tkor_reconciliation_bar_desc"

	default_green = yes

	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = {
				c:NRK ?= { has_modifier = cwp_tkor_dialogue }
				c:STK ?= { has_modifier = cwp_tkor_dialogue }
			}
			add = { desc = "cwp_tkor_reconciliation_dialogue_both" value = 2 }
		}
		progress_condition = {
			limit = {
				OR = {
					c:NRK ?= { has_modifier = cwp_tkor_dialogue }
					c:STK ?= { has_modifier = cwp_tkor_dialogue }
				}
				NOT = {
					c:NRK ?= { has_modifier = cwp_tkor_dialogue }
					c:STK ?= { has_modifier = cwp_tkor_dialogue }
				}
			}
			add = { desc = "cwp_tkor_reconciliation_dialogue_one" value = 0.5 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_modifier = cwp_tkor_engagement } }
			add = { desc = "cwp_tkor_reconciliation_engagement" value = 1 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_tkor_summit }
			add = { desc = "cwp_tkor_reconciliation_summit" value = 2 }
		}
		progress_condition = {
			limit = {
				exists = c:USA
				exists = c:RUS
				c:USA = { "relations(c:RUS)" >= 50 }
			}
			add = { desc = "cwp_tkor_reconciliation_detente" value = 0.5 }
		}
		progress_condition = {
			limit = { exists = c:NRK cwp_tkor_north_isolated = yes }
			add = { desc = "cwp_tkor_reconciliation_north_isolated" value = 0.5 }
		}
		progress_condition = {
			limit = { c:NRK ?= { has_modifier = cwp_tkor_provocations } }
			add = { desc = "cwp_tkor_reconciliation_provocations" value = -1 }
		}
		progress_condition = {
			limit = {
				scope:journal_entry ?= { "scripted_bar_progress(cwp_tkor_tension_bar)" >= 75 }
			}
			add = { desc = "cwp_tkor_reconciliation_crisis" value = -2 }
		}
	}
}
