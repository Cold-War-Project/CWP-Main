# How close the four powers are to a State Treaty for Austria
cwp_at_treaty_bar = {
	name = "cwp_at_treaty_bar"
	desc = "cwp_at_treaty_bar_desc"

	default_green = yes

	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = { has_global_variable = cwp_at_neutrality_pledged }
			add = { desc = "cwp_at_treaty_bar_neutrality" value = 1 }
		}
		progress_condition = {
			limit = { c:RUS ?= { has_variable = cwp_at_consent } }
			add = { desc = "cwp_at_treaty_bar_soviet_consent" value = 1 }
		}
		progress_condition = {
			limit = { c:USA ?= { has_variable = cwp_at_consent } }
			add = { desc = "cwp_at_treaty_bar_western_consent" value = 0.5 }
		}
		progress_condition = {
			limit = {
				any_country = {
					OR = {
						c:USA ?= this
						c:GBR ?= this
						c:FRA ?= this
						c:RUS ?= this
					}
					has_modifier = cwp_gq_four_power_talks
				}
			}
			add = { desc = "cwp_at_treaty_bar_talks" value = 0.5 }
		}
		progress_condition = {
			limit = { has_global_variable = cwp_gq_frg_sovereign }
			add = { desc = "cwp_at_treaty_bar_germany_settled" value = 0.5 }
		}
		progress_condition = {
			limit = { c:RUS ?= { has_modifier = cwp_at_usia } }
			add = { desc = "cwp_at_treaty_bar_usia" value = -0.5 }
		}
		progress_condition = {
			limit = { c:AUS ?= { is_in_power_bloc = yes } }
			add = { desc = "cwp_at_treaty_bar_in_bloc" value = -1 }
		}
	}
}
