# Cold War Project: constitutional transitions. Bars chase a pivot script value (pattern: vanilla je_two_spains_tension_bar).

# Projected share for a republic in a referendum on the monarchy.
cwp_iq_vote_bar = {
	name = "cwp_iq_vote_bar"
	desc = "cwp_iq_vote_bar_desc"

	default = yes

	monthly_progress = {
		progress_condition = {
			limit = {
				owner = {
					has_variable = cwp_iq_vote_current_var
				}
			}
			add = {
				desc = "cwp_iq_vote_bar_drift_tt"
				value = owner.cwp_iq_republic_pivot
				subtract = owner.var:cwp_iq_vote_current_var
				multiply = 0.15
			}
		}
	}

	start_value = 50
	min_value = 0
	max_value = 100
}

# Pressure on the government to settle the question; at 100 the vote can no longer be put off.
cwp_iq_pressure_bar = {
	name = "cwp_iq_pressure_bar"
	desc = "cwp_iq_pressure_bar_desc"

	default_bad = yes

	monthly_progress = {
		add = {
			desc = "cwp_iq_pressure_base_tt"
			value = 1
		}
		progress_condition = {
			limit = {
				owner = { cwp_iq_republican_clout_in_government > 0 }
			}
			add = {
				desc = "cwp_iq_pressure_republicans_tt"
				value = owner.cwp_iq_pressure_republicans
			}
		}
		progress_condition = {
			limit = {
				owner = { cwp_iq_under_soviet_pressure = yes }
			}
			add = {
				desc = "cwp_iq_pressure_occupier_tt"
				value = 3
			}
		}
		progress_condition = {
			limit = {
				owner = { cwp_communists_in_government = yes }
			}
			add = {
				desc = "cwp_iq_pressure_communists_tt"
				value = 2
			}
		}
		progress_condition = {
			limit = {
				owner = { cwp_iq_monarchist_clout_in_government > 0 }
			}
			add = {
				desc = "cwp_iq_pressure_monarchists_tt"
				value = owner.cwp_iq_pressure_monarchists
				multiply = -1
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 100
}

# Japan: how hard the occupation authorities push for a new constitution.
cwp_jap_occupation_bar = {
	name = "cwp_jap_occupation_bar"
	desc = "cwp_jap_occupation_bar_desc"

	default_bad = yes

	monthly_progress = {
		progress_condition = {
			limit = {
				owner = { cwp_occupied_by_democracy = yes }
			}
			add = {
				desc = "cwp_jap_occupation_democratic_occupier_tt"
				value = 6
			}
		}
		progress_condition = {
			limit = {
				owner = {
					is_subject_type = subject_type_military_occupation
					cwp_occupied_by_democracy = no
				}
			}
			add = {
				desc = "cwp_jap_occupation_other_occupier_tt"
				value = 1
			}
		}
		progress_condition = {
			limit = {
				owner = {
					has_variable = cwp_jap_reform_current_var
					var:cwp_jap_reform_current_var >= 50
				}
			}
			add = {
				desc = "cwp_jap_occupation_japanese_reformers_tt"
				value = 2
			}
		}
	}

	start_value = 30
	min_value = 0
	max_value = 100
}
