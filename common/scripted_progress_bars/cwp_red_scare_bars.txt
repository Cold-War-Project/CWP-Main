# How afraid the public is of communist subversion. Fed by the threat at home and abroad, and fades on its own.

	name = "cwp_rs_fear_bar"
	desc = "cwp_rs_fear_bar_desc"

	default_bad = yes

	monthly_progress = {
		progress_condition = {
			limit = { owner = { cwp_rs_threat_at_home = yes } }
			add = {
				desc = "cwp_rs_fear_home_tt"
				value = 1
			}
		}
		progress_condition = {
			limit = { owner = { cwp_rs_rival_is_communist = yes } }
			add = {
				desc = "cwp_rs_fear_rival_tt"
				value = 1
			}
		}
		progress_condition = {
			limit = { owner = { is_at_war = yes } }
			add = {
				desc = "cwp_rs_fear_war_tt"
				value = 1
			}
		}
		add = {
			desc = "cwp_rs_fear_fades_tt"
			value = -1.5
		}
		progress_condition = {
			limit = { owner = { has_law_or_variant = law_type:law_protected_speech } }
			add = {
				desc = "cwp_rs_fear_free_press_tt"
				value = -0.5
			}
		}
	}

	start_value = 30
	min_value = 0
	max_value = 100
