# Cold War Project: decolonization in Africa, progress bars. The Gold Coast (a checklist) and the Congo (a preparation
# counter) have none on purpose (Jack, 2026-11-18: not every journal entry a bar).

### Kenya: settler power (0 to 100). The rising itself is Kenya's liberty desire, not a bar.
cwp_kenya_settler_bar = {
	name = "cwp_kenya_settler_bar"
	desc = "cwp_kenya_settler_bar_desc"
	default = yes

	start_value = 85
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = {
				c:GBR ?= { cwp_has_anticolonial_leaning_government = yes }
			}
			add = {
				desc = "cwp_bar_metropole_anticolonial_government_will"
				value = -0.3
			}
		}
		progress_condition = {
			limit = {
				c:GBR ?= { cwp_has_imperialist_leaning_government = yes }
			}
			add = {
				desc = "cwp_bar_metropole_imperialist_government_will"
				value = 0.2
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_tier
				global_var:cwp_tide_tier >= 2
			}
			add = {
				desc = "cwp_bar_tide_rising"
				value = -0.3
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_tide_tier
				global_var:cwp_tide_tier >= 3
			}
			add = {
				desc = "cwp_bar_tide_strong"
				value = -0.3
			}
		}
		progress_condition = {
			limit = {
				has_global_variable = cwp_kenya_rising
			}
			add = {
				desc = "cwp_kenya_settler_insurgency"
				value = -0.2
			}
		}
		progress_condition = {
			limit = {
				c:GBR ?= { has_variable = cwp_kenya_emergency }
			}
			add = {
				desc = "cwp_kenya_settler_emergency_scandal"
				value = -0.3
			}
		}
		progress_condition = {
			limit = {
				c:GBR ?= { has_variable = cwp_kenya_land_reform }
			}
			add = {
				desc = "cwp_kenya_settler_land_reform"
				value = -0.3
			}
		}
	}
}

### Portuguese Africa: Portugal's exhaustion (0 to 100), fed by each colony actually in revolt
cwp_portugal_exhaustion_bar = {
	name = "cwp_portugal_exhaustion_bar"
	desc = "cwp_portugal_exhaustion_bar_desc"
	default_bad = yes

	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		progress_condition = {
			limit = {
				c:OVM ?= {
					is_subject_of = c:POR
					liberty_desire >= 50
				}
			}
			add = {
				desc = "cwp_portugal_front_angola"
				value = 0.3
			}
		}
		progress_condition = {
			limit = {
				c:MKU ?= {
					is_subject_of = c:POR
					liberty_desire >= 50
				}
			}
			add = {
				desc = "cwp_portugal_front_mozambique"
				value = 0.3
			}
		}
		progress_condition = {
			limit = {
				c:GNB ?= {
					is_subject_of = c:POR
					liberty_desire >= 50
				}
			}
			add = {
				desc = "cwp_portugal_front_guinea"
				value = 0.3
			}
		}
		progress_condition = {
			limit = {
				c:POR ?= { has_variable = cwp_portugal_counter_insurgency }
			}
			add = {
				desc = "cwp_portugal_exhaustion_conscription"
				value = 0.2
			}
		}
		progress_condition = {
			limit = {
				c:POR ?= { scaled_debt > 0.5 }
			}
			add = {
				desc = "cwp_bar_metropole_debt"
				value = 0.3
			}
		}
		progress_condition = {
			limit = {
				c:POR ?= {
					OR = {
						has_law = law_type:law_autocracy
						has_law = law_type:law_single_party_state
					}
				}
			}
			add = {
				desc = "cwp_portugal_exhaustion_censorship"
				value = -0.2
			}
		}
	}
}
