# Cold War Project: Newfoundland's future.

# Projected vote for Confederation with Canada, against a return to Responsible Government.
cwp_nf_confederation_bar = {
	name = "cwp_nf_confederation_bar"
	desc = "cwp_nf_confederation_bar_desc"

	default = yes

	monthly_progress = {
		progress_condition = {
			limit = {
				owner = { has_variable = cwp_nf_vote_current_var }
			}
			add = {
				desc = "cwp_nf_confederation_bar_drift_tt"
				value = owner.cwp_nf_confederation_pivot
				subtract = owner.var:cwp_nf_vote_current_var
				multiply = 0.15
			}
		}
	}

	start_value = 38
	min_value = 0
	max_value = 100
}
