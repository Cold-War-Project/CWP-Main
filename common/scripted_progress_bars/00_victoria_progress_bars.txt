 divided_monarchists_bonapartist_progress_bar = {
	name = "divided_monarchists_bonapartist_progress_bar"
	desc = "divided_monarchists_bonapartist_progress_bar_desc"

	default = yes

	monthly_progress = {
		if = {
			limit = {
				NOT = {
					has_variable = fra_divided_monarchists_freeze #This variable stops iteration, and is granted by reaching 360 with one dynasty. It's reset when this JE is completed, or when je_cement_the_rightful_dynasty fails.
				}
			}
			# From Characters - combined prominence of every Bonapartist character (prominence is 0/100, so divide by 200).
			progress_condition = {
				limit = {
					any_scope_character = {
						has_ideology = ideology:ideology_bonapartist
					}
				}
				add = {
					desc = "bonapartist_progress_from_characters"
					every_scope_character = {
						limit = {
							has_ideology = ideology:ideology_bonapartist
						}
						add = this.prominence
					}
					divide = 200
				}
			}
			# From Interest Groups - clout of every Bonapartist-led IG, weighted double if in government.
			progress_condition = {
				limit = {
					any_interest_group = {
						leader = {
							has_ideology = ideology:ideology_bonapartist
						}
					}
				}
				add = {
					desc = "bonapartist_progress_from_igs"
					every_interest_group = {
						limit = {
							leader = {
								has_ideology = ideology:ideology_bonapartist
							}
						}
						add = {
							value = this.ig_clout
							progress_condition = {
								limit = {
									is_in_government = yes
								}
								multiply = 2
							}
						}
					}
				}
			}
			# From Buildings - sum of occupied military building levels (0.0033 per level).
			progress_condition = {
				limit = {
					any_scope_building = {
						is_building_group = bg_military # Derives from barracks and naval bases.
						occupancy >= 0.5
					}
				}
				add = {
					desc = "bonapartist_progress_from_military"
					every_scope_building = {
						limit = {
							is_building_group = bg_military # Derives from barracks and naval bases.
							occupancy >= 0.5
						}
						add = this.level
					}
					divide = 300
				}
			}
			# From Movements - support of any Bonapartist movement.
			progress_condition = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_bonapartist
					}
				}
				add = {
					desc = "bonapartist_progress_from_movements"
					every_political_movement = {
						limit = {
							is_political_movement_type = movement_bonapartist
						}
						add = {
							value = this.political_movement_support
							multiply = 5
						}
					}
				}
			}
			# From Legitimacy - balanced around 0.5; a Bonapartist ruler benefits from high legitimacy, otherwise low legitimacy.
			progress_condition = {
				limit = {
					ruler_is_bonapartist = yes
				}
				add = {
					desc = "bonapartist_progress_from_legitimacy"
					value = government_legitimacy
					divide = 100
					subtract = 0.5
				}
				else = {
					add = {
						desc = "bonapartist_progress_from_legitimacy"
						value = government_legitimacy
						divide = 100
						subtract = 0.5
						multiply = -0.5
					}
				}
			}
			# From Effects - scripted bonuses/maluses granted manually by events.
			progress_condition = {
				limit = {
					has_variable = bonapartist_progress_from_effects
				}
				add = {
					desc = "bonapartist_progress_from_effects"
					value = var:bonapartist_progress_from_effects
				}
			}
		}
	}

	start_value = 10
	min_value = 0
	max_value = 480 # divided_monarchists_upper_bound

 }

 communism_1_progress_bar = {
	name = "communism_1_progress_bar"
	desc = "communism_1_progress_bar_desc"

	default_bad = yes

	monthly_progress = {
		progress_condition = {
			limit = {
				owner = {
					any_scope_state = {
						is_incorporated = yes
						loyalists >= 0.10
					}
				}
			}
			owner = {
				subtract = {
					desc = "communism_progress_from_loyalists_tt"
					every_scope_state = {
						limit = {
							is_incorporated = yes
							loyalists >= 0.10
						}
						add = this.loyalists
					}
					divide = num_incorporated_states
					multiply = 3 # Maxes out at -3
				}
			}
		}
		progress_condition = {
			limit = {
				any_political_movement = {
					movement_is_socialist = yes
				}
			}
			add = {
				desc = "communism_progress_from_movement_activism_tt"
				every_political_movement = {
					limit = {
						movement_is_socialist = yes
					}
					add = {
						value = this.political_movement_radicalism
						multiply = {
							value = this.political_movement_support
							multiply = 5
						}
					}
				}
			}
		}
		progress_condition = {
			limit = {
				owner = {
					any_scope_character = {
						has_socialist_ideology = yes
					}
				}
			}
			add = {
				desc = "communism_progress_from_characters_tt"
				owner = {
					every_scope_character = {
						limit = {
							has_socialist_ideology = yes
						}
						add = this.popularity
						divide = 100
					}
				}
			}
		}
		progress_condition = {
			limit = {
				owner = {
					any_interest_group = {
						law_stance = {
							law = law_type:law_factory_councils
							value > neutral
						}
					}
				}
			}
			add = {
				desc = "communism_progress_from_interest_groups_tt"
				owner = {
					every_interest_group = {
						limit = {
							law_stance = {
								law = law_type:law_factory_councils
								value > neutral
							}
						}
						add = {
							value = this.ig_clout
							multiply = 5
						}
					}
				}
			}
		}
		progress_condition = {
			limit = {
				exists = owner.var:communism_progress_from_laws
			}
			progress_condition = {
				limit = {
					owner.var:communism_progress_from_laws = 0
				}
				add = {
					desc = "communism_progress_from_no_laws_tt"
					value = 0.25
				}
				else ={
					add = {
						desc = "communism_progress_from_laws_tt"
						value = owner.var:communism_progress_from_laws
					}
				}
			}
		}
		progress_condition = {
			limit = {
				exists = owner.var:communism_progress_var_add
				NOT = {
					owner.var:communism_progress_var_add = 0
				}
			}
			add = {
				desc = "communism_progress_from_events_tt"
				value = owner.var:communism_progress_var_add
			}
		}
	}

	start_value = 10
	min_value = 0
	max_value = 100

 }

	default = yes

						is_political_movement_type = movement_modernizer

						is_political_movement_type = movement_positivist

						}
					}
					add = this.political_movement_support
					multiply = 10
				}
			}
		}
		progress_condition = {

				limit = {
					owner = {
						ig:ig_petty_bourgeoisie ?= {
							ig_counts_as_marginal = yes
						}
					}
				}
				value = -10
				else = {
					limit = {
						owner = {
							ig:ig_petty_bourgeoisie ?= {
								is_powerful = yes
							}
						}
					}
					value = 3
					else = {
						value = 0
					}
				}
			}
		}
		progress_condition = {
			limit = {
				exists = owner.var:populism_progress_from_laws
				NOT = {
					owner.var:populism_progress_from_laws = 0
				}
