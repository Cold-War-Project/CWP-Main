# Cold War Project: the energy crisis (je_cwp_energy_crisis).

# How angry the public is about fuel prices and shortages. Follows the price in our market.
cwp_ec_anger_bar = {
	name = "cwp_ec_anger_bar"
	desc = "cwp_ec_anger_bar_desc"

	default_bad = yes

	monthly_progress = {
		progress_condition = {
			limit = { owner = { cwp_ec_fuel_pricier = { PCT = 0.5 } } }
			add = {
				desc = "cwp_ec_anger_price_tt"
				value = 2
			}
		}
		progress_condition = {
			limit = { owner = { cwp_ec_fuel_pricier = { PCT = 1 } } }
			add = {
				desc = "cwp_ec_anger_price_high_tt"
				value = 2
			}
		}
		progress_condition = {
			limit = { owner = { has_modifier = cwp_ec_rationing } }
			add = {
				desc = "cwp_ec_anger_rationing_tt"
				value = 1
			}
		}
		progress_condition = {
			limit = { owner = { has_modifier = cwp_ec_price_controls } }
			add = {
				desc = "cwp_ec_anger_controls_tt"
				value = -2
			}
		}
		progress_condition = {
			limit = { owner = { cwp_ec_fuel_pricier = { PCT = 0.5 } } }
			add = {
				desc = "cwp_ec_anger_fades_tt"
				value = -1
			}
			else = {
				add = {
					desc = "cwp_ec_anger_relief_tt"
					value = -4
				}
			}
		}
	}

	start_value = 20
	min_value = 0
	max_value = 100
}
