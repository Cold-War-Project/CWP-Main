# Progress Bar for the Coup Journal Entry
je_cwp_coup_progress_bar = {
	name = "je_cwp_coup_progress_bar"
	desc = "je_cwp_coup_progress_bar_desc"

	default = yes

	weekly_progress = {
		owner = {
			if = {	# Does the commander actually want to coup the government?
				limit = {
					any_scope_character = {
						has_variable = cwp_coup_character_leader_variable
						loyalty >= define:NCharacters|LOYALTY_NEUTRAL_THRESHOLD
					}
				}
				subtract = {
					desc = "cwp_appeased_cwp_coup_character_interest_group_tt"
					every_scope_character = {
						limit = { has_variable = cwp_coup_character_leader_variable }
						value = this.loyalty
						divide = 10
					}
				}
				add = {
					desc = "cwp_coup_character_coup_strength_tt"
					every_scope_character = {
						limit = { has_variable = cwp_coup_character_leader_variable }
						value = cwp_commander_coup_strength
					}
				}
				
				if = {	# Has the coup been aborted?
					limit = { has_variable = cwp_orchestrate_coup_aborted }
					subtract = {
						desc = "cwp_orchestrate_coup_aborted_tt"
						value = 5
					}
				}

				subtract = {	# What is the country's resistance to the coup?
					desc = "cwp_country_coup_resistance_tt"
					value = owner.cwp_country_coup_resistance
				}
			}
		}
	}

	start_value = 10
	min_value = 0
	max_value = 120
}
