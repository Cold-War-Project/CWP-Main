crown_land_republic_effect = {
	if = {
		limit = {
			any_subject_or_below = {
				is_subject_type = subject_type_crown_land
				is_direct_subject_of = ROOT
			}
		}
		every_subject_or_below = {
			limit = {
				is_subject_type = subject_type_crown_land
				is_direct_subject_of = ROOT
			}
			change_subject_type = subject_type_puppet
		}
	}
}
