# Cold War Project: the French constituent assemblies.

cwp_fra_constitution_cleanup = {
	remove_variable = cwp_fra_draft_current_var
	remove_variable = cwp_fra_round_var
	remove_variable = cwp_fra_compromise_var
	remove_variable = cwp_fra_campaign_var
	remove_variable = cwp_fra_swing_var
	remove_variable = cwp_fra_vote_pending_var
	remove_variable = cwp_fra_result_var
	remove_variable = cwp_fra_tripartisme_var
	remove_variable = cwp_fra_council_conceded_var
}

# The draft is approved. A Gaullist draft keeps a strong presidency; otherwise the assembly rules.
cwp_fra_adopt_constitution = {
	set_variable = cwp_fra_constitution_settled_var
	set_variable = cwp_fra_constitution_adopted_var
	if = {
		limit = {
			NOT = { has_variable = cwp_fra_gaullist_draft_var }
			NOT = { has_law = law_type:law_parliamentary_republic }
		}
		activate_law = law_type:law_parliamentary_republic
		# The parliamentary republic's on_activate counts French republics; this is still the Fourth.
		set_variable = { name = fra_republic_counter value = 4 }
	}
	add_modifier = {
		name = cwp_iq_republic_proclaimed
		years = 5
	}
}
