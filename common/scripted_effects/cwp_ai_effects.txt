# Construction backstop. The vanilla AI sizes its construction sector from money-spending ratios,
# so private capital can pile up with nothing to build it: in observer runs the USA's investment
# pool reached 9 billion (about 900 weeks of spending) by 1989. When investors are waiting on
# construction capacity and the budget is sound, a rational government adds capacity. Each level
# is mostly paid back by the investment pool once private projects use it.
cwp_ai_construction_backstop = {
	if = {
		limit = {
			is_ai = yes
			in_default = no
			taking_loans = no
			scaled_debt < 0.25
			has_sufficient_construction_capacity_for_investment = no
			# Count queued private levels instead of construction_queue_private_duration: the duration
			# estimate logs "total construction speed is zero" for countries whose sectors produce nothing.
			construction_queue_num_queued_private_levels >= 10
			construction_queue_num_queued_government_levels < 25
			# a state to build in (an empty ordered list logs "position bigger than the list")
			any_scope_state = {
				is_incorporated = yes
				state_population >= 250000
				can_construct_building = building_construction_sector
			}
		}
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				state_population >= 250000
				can_construct_building = building_construction_sector
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			start_building_construction = building_construction_sector
		}
		# Large economies add capacity in a second state as well.
		if = {
			limit = {
				total_population >= 40000000
				any_scope_state = {
					count >= 2
					is_incorporated = yes
					state_population >= 250000
					can_construct_building = building_construction_sector
				}
			}
			ordered_scope_state = {
				limit = {
					is_incorporated = yes
					state_population >= 250000
					can_construct_building = building_construction_sector
				}
				order_by = state_population
				position = 1
				check_range_bounds = no
				start_building_construction = building_construction_sector
			}
		}
	}
}

# Construction workforce. Technology sets a flat per-state cap on construction sector levels (30 by
# 1960), the same for California as for Nevada, so large economies run out of room to build: in
# observer runs every US state sat at the cap from the mid-1950s while the US and UK investment
# pools grew to several times annual GDP. A state can host more construction with more people.
# Run yearly from on_yearly_pulse_state (root = state).
cwp_construction_workforce_update = {
	# Only ever raise the bonus. Removing it, even to re-add it at once, lowers the cap for a moment
	# and the engine demolishes the levels above it; a shrinking population keeps what it built.
	if = {
		limit = {
			state_population >= 400000
			OR = {
				NOT = { has_variable = cwp_construction_workforce_levels }
				var:cwp_construction_workforce_levels < cwp_construction_workforce_target
			}
		}
		remove_modifier = cwp_construction_workforce
		set_variable = {
			name = cwp_construction_workforce_levels
			value = cwp_construction_workforce_target
		}
		add_modifier = {
			name = cwp_construction_workforce
			multiplier = var:cwp_construction_workforce_levels
		}
	}
}

# $TYPE$: building type. remove_building takes a literal key, hence one call per type.
cwp_ai_close_empty_building = {
	if = {
		limit = {
			any_scope_building = {
				is_building_type = $TYPE$
				level >= 1	# ownership triggers error on buildings with no built levels yet
				occupancy < 0.05
				country_ownership_fraction >= 0.5
			}
		}
		if = {
			limit = { has_variable = cwp_empty_$TYPE$ }
			change_variable = { name = cwp_empty_$TYPE$ add = 1 }
		}
		else = {
			set_variable = { name = cwp_empty_$TYPE$ value = 1 }
		}
		if = {
			limit = { var:cwp_empty_$TYPE$ >= 3 }
			remove_building = $TYPE$
			remove_variable = cwp_empty_$TYPE$
		}
	}
	else_if = {
		limit = { has_variable = cwp_empty_$TYPE$ }
		remove_variable = cwp_empty_$TYPE$
	}
}

# Institution austerity. Below a bureaucracy ratio of 1.0 the engine treats new administration as a
# must-have that ignores the budget, so a broke state that runs short of bureaucracy builds offices
# it can't pay for (observer 20261120-0529-d5350057e: North Korea 8 -> 24 admin levels on 3-6k/week
# income, spending 10x it, in default from 1950; Iceland, Albania the same). A rational government in
# that position trims an institution instead, which frees the bureaucracy without new costs. At most
# one level per half year, never below 1, least essential first (judiciary, voting and corporate affairs
# cost authority, legitimacy and company slots); administration is never cut. Bankruptcy
# resets every institution to 1 anyway, so this is the cheaper path to the same place.
# Monthly from on_monthly_pulse_country (root = country).
cwp_ai_institution_austerity = {
	if = {
		limit = {
			is_ai = yes
			bureaucracy < 0
			# One cut per half year: a cut takes effect gradually, so let it land before cutting again.
			NOT = { has_variable = cwp_austerity_cooldown }
			OR = {
				in_default = yes
				AND = {
					taking_loans = yes
					scaled_debt >= 0.5
				}
			}
		}
		if = {
			limit = {
				has_institution = institution_church_and_state
				NOT = { expanding_institution = institution_church_and_state }
				institution_investment_level = { institution = institution_church_and_state value >= 2 }
			}
			change_institution_investment_level = { institution = institution_church_and_state investment = -1 }
		}
		else_if = {
			limit = {
				has_institution = institution_citizenship
				NOT = { expanding_institution = institution_citizenship }
				institution_investment_level = { institution = institution_citizenship value >= 2 }
			}
			change_institution_investment_level = { institution = institution_citizenship investment = -1 }
		}

# Superpower-led blocs hold together (cwp_superpower_bloc). "Superpower" here means a bloc leader of
# great-power rank or above locked in rivalry with another bloc leader: the two camps of the Cold War.
# (The engine's superpower rank is prestige-based and in CWP runs only the USA reaches it; the USSR sits
# at great power.) Refreshed monthly for two months, and removed as soon as the leader no longer
# qualifies, so there is no yearly gap and no lingering bonus. Monthly from on_monthly_pulse_country
# (root = country, AI or not).
cwp_ai_superpower_bloc_update = {
	# Scope (not root) so it also works from on_game_started_after_lobby, which has no root.
	save_scope_as = cwp_sp_leader
	if = {
		limit = { is_power_bloc_leader = yes }
		power_bloc = {
			if = {
				limit = { has_modifier = cwp_superpower_bloc }
				remove_modifier = cwp_superpower_bloc
			}
		}
		if = {
			limit = {
				country_rank >= rank_value:great_power
				# a Cold War camp, not a colonial empire (Britain's bloc picked it up through rivalries)
				power_bloc ?= { NOT = { has_identity = identity:identity_sovereign_empire } }
				any_country = {
					is_power_bloc_leader = yes
					NOT = { this = scope:cwp_sp_leader }
					OR = {
						has_diplomatic_pact = { who = scope:cwp_sp_leader type = rivalry }
						scope:cwp_sp_leader = { has_diplomatic_pact = { who = prev type = rivalry } }
					}
				}
			}
			power_bloc = {
				add_modifier = { name = cwp_superpower_bloc months = 2 }
			}
		}
	}
}

# Planned-economy construction. Under a command economy there is no investment pool, so only the
# government builds industry, and the vanilla AI does not: in observer base 831affc99 the USSR's
# government queue held no production building in any sample 1950-59 (administration, barracks,
# airports, 37 power bloc statues), 8 items in 1952 against ~1,000 construction sectors, and its
# industry shrank from 2,000 to 1,919 levels by 1962. A planner builds what the economy is short of.
# Each month, for an AI command economy with real construction capacity (50+ sector levels) that
# owns its market (a customs-union member would react to the leader's prices), is solvent and has a
# short queue: walk the key industrial goods in order and queue one level of the building for each
# good priced 25%+ above base whose main input is not itself short, at most 3 levels a month, in the
# most populous incorporated state with spare infrastructure. Electricity is left out (a local good,
# priced per state). Independent review 2026-11-20 shaped these gates.
# Monthly from on_monthly_pulse_country (root = country).
cwp_ai_planned_economy = {
	if = {
		limit = {
			is_ai = yes
			has_law_or_variant = law_type:law_command_economy
			market ?= { owner = root }
			country_has_building_type_levels = {
				target = bt:building_construction_sector
				value >= 50
			}
			in_default = no
			taking_loans = no
			scaled_debt < 0.25
			construction_queue_num_queued_government_levels < 20
		}
		set_local_variable = { name = cwp_planned value = 0 }
		cwp_ai_plan_industry = { GOODS = steel BUILDING = building_steelworks INPUT = iron }
		cwp_ai_plan_industry = { GOODS = coal BUILDING = building_coal_mine INPUT = tools }
		cwp_ai_plan_industry = { GOODS = iron BUILDING = building_iron_mine INPUT = tools }
		cwp_ai_plan_industry = { GOODS = tools BUILDING = building_tool_manufactories INPUT = wood }
		cwp_ai_plan_industry = { GOODS = cement BUILDING = building_cement_plants INPUT = coal }
		cwp_ai_plan_industry = { GOODS = chemicals BUILDING = building_chemical_plants INPUT = sulfur }
		cwp_ai_plan_industry = { GOODS = engines BUILDING = building_motor_industry INPUT = steel }
		cwp_ai_plan_industry = { GOODS = glass BUILDING = building_glassworks INPUT = coal }
		cwp_ai_plan_industry = { GOODS = explosives BUILDING = building_explosives_industry INPUT = chemicals }
		cwp_ai_plan_industry = { GOODS = aluminum BUILDING = building_aluminum_refinery INPUT = bauxite }
		cwp_ai_plan_industry = { GOODS = refined_fuel BUILDING = building_oil_refineries INPUT = oil }
		cwp_ai_plan_industry = { GOODS = paper BUILDING = building_paper_manufactories INPUT = wood }
		cwp_ai_plan_industry = { GOODS = clothes BUILDING = building_textile_manufactories INPUT = fabric }
		cwp_ai_plan_industry = { GOODS = groceries BUILDING = building_food_industry INPUT = grain }
	}
}

# $GOODS$ short in our market (25%+ above base) and its main input $INPUT$ not short -> queue one level
# of $BUILDING$, while fewer than 3 were queued this month.
cwp_ai_plan_industry = {
	if = {
		limit = {
			local_var:cwp_planned < 3
			market ?= {
				mg:$GOODS$ ?= { market_goods_pricier > 0.25 }
				NOT = { mg:$INPUT$ ?= { market_goods_pricier > 0.25 } }
			}
			any_scope_state = {
				is_incorporated = yes
				infrastructure > infrastructure_usage
				can_construct_building = $BUILDING$
			}
		}
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				infrastructure > infrastructure_usage
				can_construct_building = $BUILDING$
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			start_building_construction = $BUILDING$
		}
		change_local_variable = { name = cwp_planned add = 1 }
	}
}
