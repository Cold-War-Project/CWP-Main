# Cold War Project: regime change aftermath (docs/agents/areas/regime-change.md).
# Root = the enforcer, scope:target_country = the target. Called from the regime_change war goal's on_enforced
# (common/war_goal_types/zz_vanilla_war_goal_types.txt) and after regime_change in force_regime_change.
# A regime change imposes the enforcer's governance law, so a crowned enforcer would install a monarchy.
# cwp_regime_change.1 replaces it with a republic a day later unless the target has a monarchy tradition.
cwp_regime_change_mark_target_effect = {
	scope:target_country ?= {
		set_variable = { name = cwp_regime_change_window_var days = 5 }
		set_variable = { name = cwp_regime_change_enforcer_var value = root days = 5 }
		if = {
			limit = { has_variable = cwp_crown_pretender_var }
			set_variable = { name = cwp_regime_change_had_pretender_var days = 5 }
		}
		trigger_event = { id = cwp_regime_change.1 days = 1 }
	}
}
