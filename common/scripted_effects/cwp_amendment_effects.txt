# Cold War Project: constitutional amendments in force on 1946.1.1.
# Called once from common/history/global/cwp_amendments_1946.txt, after every country has its laws.

# Country scope. Adds $AMENDMENT$ to the active law of $LAWGROUP$, sponsored by $IG$ when it exists.
# The amendment's own allowed_laws and possible decide whether it can attach; callers guard the law.
cwp_add_start_amendment = {
	if = {
		limit = { exists = ig:$IG$ }
		ig:$IG$ = { save_temporary_scope_as = cwp_amendment_sponsor }
	}
	else = {
		random_interest_group = {
			limit = { is_marginal = no }
			save_temporary_scope_as = cwp_amendment_sponsor
		}
	}
	if = {
		limit = { exists = scope:cwp_amendment_sponsor }
		active_law:$LAWGROUP$ ?= {
			add_amendment = {
				type = $AMENDMENT$
				sponsor = scope:cwp_amendment_sponsor
			}
		}
	}
}

# Country scope. Tradition of Free Elections needs vanilla's free_elections_var so that a later change
# of voting law carries it over (inherit_free_elections_effect in the distribution-of-power laws).
cwp_add_start_free_elections = {
	if = {
		limit = {
			OR = {
				has_law = law_type:law_oligarchy
				has_law = law_type:law_landed_voting
				has_law = law_type:law_wealth_voting
				has_law = law_type:law_census_voting
				has_law = law_type:law_universal_suffrage
			}
		}
		set_variable = free_elections_var
		cwp_add_start_amendment = { AMENDMENT = amendment_tradition_of_free_elections LAWGROUP = lawgroup_distribution_of_power IG = ig_intelligentsia }
	}
}
