# Cold War Project: 1946 starting literacy tiers, used by common/history/population.
# Vanilla's effect_starting_pop_literacy_* (00_starting_pop_literacy.txt) are 1836 levels and stay untouched.

# Western Europe, North America, the Dominions, Japan, Czechoslovakia: near-universal adult literacy
effect_starting_pop_literacy_1946_universal = {
	every_scope_pop = {
		set_pop_literacy = {
			literacy_rate = {
				value = 0.93
				# Professions requiring reading
				if = {
					limit = {
						OR = {
							is_pop_type = bureaucrats
							is_pop_type = clergymen
							is_pop_type = engineers
						}
					}
					add = 0.07
				}
				# Upper class
				if = {
					limit = { strata = upper }
					add = 0.07
				}
				# Subsistence farmers lag behind
				if = {
					limit = {
						OR = {
							pop_employment_building_group = bg_subsistence_agriculture
							pop_employment_building_group = bg_subsistence_ranching
						}
					}
					multiply = 0.7
				}
				# Unincorporated territories at half
				if = {
					limit = { state = { is_incorporated = no } }
					multiply = 0.5
				}
				max = 1
			}
		}
	}
}
