# Cold War Project: effects for the Austrian State Treaty (common/journal_entries/cwp_austria_saar.txt).

# Run in Austria's scope: it agrees to pay the Soviet Union for the German assets in the Soviet zone.
cwp_at_buy_back_assets = {
	set_global_variable = cwp_at_assets_bought
	add_modifier = {
		name = cwp_at_reparation_deliveries
		days = long_modifier_time
	}
	c:RUS ?= {
		add_modifier = {
			name = cwp_at_reparation_income
			days = long_modifier_time
		}
		if = {
			limit = { has_modifier = cwp_at_usia }
			remove_modifier = cwp_at_usia
		}
	}
	if = {
		limit = { has_modifier = cwp_at_usia_loss }
		remove_modifier = cwp_at_usia_loss
	}
}
