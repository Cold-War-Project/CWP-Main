# Cold War Project: the Red Scare (je_cwp_red_scare). Country scope.

# Move the fear bar from an event option. Values can be negative.
# Guarded: an event option can resolve after the JE has already concluded.
cwp_rs_move_fear = {
	if = {
		limit = { has_journal_entry = je_cwp_red_scare }
		je:je_cwp_red_scare = {
			add_progress = { value = $FEAR$ name = cwp_rs_fear_bar }
		}
	}
}

# Shared ending for the two good outcomes: a clean recovery if nothing repressive is left in force.
cwp_rs_conclude = {
	set_variable = cwp_rs_done_var
	if = {
		limit = { cwp_rs_measures_in_force < 1 }
		add_modifier = {
			name = cwp_rs_scare_subsided
			years = 5
		}
	}
	else = {
		add_modifier = {
			name = cwp_rs_chilled_society
			years = 5
		}
	}
	trigger_event = { id = cwp_rs.20 popup = yes }
}

# Lift every repressive measure still in force (event choices after the scare).
cwp_rs_lift_measures = {
	ig:ig_bureaucrats ?= { remove_modifier = cwp_rs_loyalty_review }
	ig:ig_intelligentsia ?= {
		remove_modifier = cwp_rs_loyalty_review
		remove_modifier = cwp_rs_blacklisted
	}
	ig:ig_trade_unions ?= { remove_modifier = cwp_rs_unions_investigated }
	every_interest_group = {
		limit = { has_modifier = cwp_rs_party_banned_ig }
		remove_modifier = cwp_rs_party_banned_ig
	}
	remove_modifier = cwp_rs_party_banned
}
