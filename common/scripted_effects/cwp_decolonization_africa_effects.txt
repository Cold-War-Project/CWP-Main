# Cold War Project: decolonization in Africa, scripted effects

# Scope: none, scope:journal_entry set. Mirrors Kenya's insurgency bar into cwp_kenya_insurgency_level (0..2).
cwp_kenya_set_level = {
	if = {
		limit = { scope:journal_entry = { "scripted_bar_progress(cwp_kenya_insurgency_bar)" >= 50 } }
		set_global_variable = { name = cwp_kenya_insurgency_level value = 2 }
		set_global_variable = cwp_kenya_rising_seen
	}
	else_if = {
		limit = { scope:journal_entry = { "scripted_bar_progress(cwp_kenya_insurgency_bar)" >= 25 } }
		set_global_variable = { name = cwp_kenya_insurgency_level value = 1 }
		set_global_variable = cwp_kenya_rising_seen
	}
	else = {
		set_global_variable = { name = cwp_kenya_insurgency_level value = 0 }
	}
}

# Scope: none. The Congo becomes independent. The less Belgium trained an African administration
# (cwp_congo_cadres on BEL, one per press of the cadres button), the more unprepared the new state, and
# with fewer than two rounds of training Katanga breaks away.
cwp_congo_independence_effect = {
	c:CNG = {
		cwp_decolonization_negotiated_independence = { METROPOLE = c:BEL }
		add_modifier = {
			name = cwp_congo_unprepared
			years = 10
			is_decaying = yes
			multiplier = {
				value = 3
				if = {
					limit = { c:BEL ?= { has_variable = cwp_congo_cadres } }
					subtract = c:BEL.var:cwp_congo_cadres
				}
				min = 0
			}
		}
	}
	if = {
		limit = {
			NOT = { exists = c:KTA }
			exists = s:STATE_KATANGA.region_state:CNG
			NOT = {
				c:BEL ?= {
					has_variable = cwp_congo_cadres
					var:cwp_congo_cadres >= 2
				}
			}
		}
		create_country = {
			tag = KTA
			origin = c:CNG
			state = s:STATE_KATANGA.region_state:CNG
			on_created = {
				set_variable = cwp_former_colony
				change_relations = {
					country = c:BEL
					value = 30
				}
			}
		}
		s:STATE_KATANGA = { add_claim = c:CNG }
		c:CNG = {
			change_relations = {
				country = c:KTA
				value = -50
			}
			trigger_event = { id = cwp_africa.22 popup = yes }
		}
		c:BEL ?= { trigger_event = { id = cwp_africa.22 popup = yes } }
	}
}

# Scope: none. Portugal lets Angola, Mozambique and Portuguese Guinea go.
cwp_portugal_release_effect = {
	every_country = {
		limit = {
			OR = {
				c:OVM ?= this
				c:MKU ?= this
				c:GNB ?= this
			}
			is_subject_of = c:POR
		}
		cwp_decolonization_negotiated_independence = { METROPOLE = c:POR }
		trigger_event = { id = cwp_africa.31 popup = yes }
	}
}
