# Cold War Project: decolonization in Africa, scripted effects

# Scope: none. Monthly while the Portuguese JE runs: the guerrillas' foreign bases raise each colony's liberty
# desire; every colony at 50+ is a front, and each front sours Portugal's Armed Forces.
cwp_portugal_fronts_pulse = {
	set_global_variable = { name = cwp_portugal_fronts value = 0 }
	every_country = {
		limit = {
			OR = {
				c:OVM ?= this
				c:MKU ?= this
				c:GNB ?= this
			}
			is_subject_of = c:POR
			liberty_desire >= 50
		}
		change_global_variable = { name = cwp_portugal_fronts add = 1 }
	}
	c:POR ?= {
		ig:ig_armed_forces ?= {
			remove_modifier = cwp_portugal_officers_discontent
			if = {
				limit = { global_var:cwp_portugal_fronts > 0 }
				add_modifier = {
					name = cwp_portugal_officers_discontent
					multiplier = global_var:cwp_portugal_fronts
				}
			}
		}
	}
	if = {
		limit = { c:CNG ?= { is_subject = no } }
		c:OVM ?= {
			if = {
				limit = { is_subject_of = c:POR }
				add_liberty_desire = 1
			}
		}
	}
	if = {
		limit = { c:TNZ ?= { is_subject = no } }
		c:MKU ?= {
			if = {
				limit = { is_subject_of = c:POR }
				add_liberty_desire = 1
			}
		}
	}
	if = {
		limit = { c:GLC ?= { is_subject = no } }
		c:GNB ?= {
			if = {
				limit = { is_subject_of = c:POR }
				add_liberty_desire = 0.5
			}
		}
	}
}

# Scope: none. The Congo becomes independent. The less Belgium trained an African administration
# (global cwp_congo_cadres, one per press of the cadres button), the more unprepared the new state, and
# with fewer than two rounds of training Katanga breaks away.
cwp_congo_independence_effect = {
	c:CNG = {
		cwp_decolonization_negotiated_independence = { METROPOLE = c:BEL }
		add_modifier = {
			name = cwp_congo_unprepared
			years = 10
			is_decaying = yes
			multiplier = {
				value = 3
				subtract = global_var:cwp_congo_cadres
				min = 0
			}
		}
	}
	if = {
		limit = {
			NOT = { exists = c:KTA }
			exists = s:STATE_KATANGA.region_state:CNG
			global_var:cwp_congo_cadres < 2
		}
		create_country = {
			tag = KTA
			origin = c:CNG
			state = s:STATE_KATANGA.region_state:CNG
			on_created = {
				set_variable = cwp_former_colony
				change_relations = {
					country = c:BEL
					value = 30
				}
			}
		}
		s:STATE_KATANGA = { add_claim = c:CNG }
		c:CNG = {
			change_relations = {
				country = c:KTA
				value = -50
			}
			trigger_event = { id = cwp_africa.22 popup = yes }
		}
		c:BEL ?= { trigger_event = { id = cwp_africa.22 popup = yes } }
	}
}

# Scope: none. Portugal lets Angola, Mozambique and Portuguese Guinea go.
cwp_portugal_release_effect = {
	every_country = {
		limit = {
			OR = {
				c:OVM ?= this
				c:MKU ?= this
				c:GNB ?= this
			}
			is_subject_of = c:POR
		}
		cwp_decolonization_negotiated_independence = { METROPOLE = c:POR }
		trigger_event = { id = cwp_africa.31 popup = yes }
	}
}

# Scope: none. Clears the officers' modifier when the Portuguese JE ends.
cwp_portugal_cleanup = {
	c:POR ?= {
		ig:ig_armed_forces ?= { remove_modifier = cwp_portugal_officers_discontent }
	}
}
