# Cold War Project: effects for the German question, chapter 1 (occupation and division).
# Journal entry: common/journal_entries/cwp_german_question.txt

# Shift one of the journal entry's bars by VALUE (any scope; vanilla precedent: add_progress in ip4_on_actions.txt).
# BAR = cwp_gq_western_bar / cwp_gq_soviet_bar
cwp_gq_shift_bar = {
	je:je_cwp_german_question ?= {
		add_progress = {
			name = $BAR$
			value = $VALUE$
		}
	}
}

# Pick the zone that carries the Federal Republic: a zone a player holds, else the
# American, British, French zone in that order. Saves scope:cwp_gq_frg_core.
cwp_gq_save_frg_core = {
	if = {
		limit = {
			any_country = {
				cwp_gq_is_western_zone = yes
				is_player = yes
			}
		}
		random_country = {
			limit = {
				cwp_gq_is_western_zone = yes
				is_player = yes
			}
			save_scope_as = cwp_gq_frg_core
		}
	}
	else_if = {
		limit = { exists = c:AOZ }
		c:AOZ = { save_scope_as = cwp_gq_frg_core }
	}
	else_if = {
		limit = { exists = c:BOZ }
		c:BOZ = { save_scope_as = cwp_gq_frg_core }
	}
	else_if = {
		limit = { exists = c:FOZ }
		c:FOZ = { save_scope_as = cwp_gq_frg_core }
	}
}

# Run in the core zone's scope. The other western zones that still answer to a western power
# merge into it, and it becomes the Federal Republic (tag GER). A player holding another zone
# is annexed with it (said in the tooltip). Each zone is written out so that nothing is annexed
# inside a list it is iterating.
cwp_gq_merge_zone = {
	if = {
		limit = {
			exists = c:$ZONE$
			NOT = { c:$ZONE$ = root }
			c:$ZONE$ = {
				OR = {
					is_subject = no
					overlord = { cwp_gq_is_western_occupier = yes }
				}
			}
		}
		annex_with_incorporation = c:$ZONE$
	}
}

cwp_gq_proclaim_frg = {
	custom_tooltip = cwp_gq_proclaim_frg_tt
	hidden_effect = {
		cwp_gq_merge_zone = { ZONE = AOZ }
		cwp_gq_merge_zone = { ZONE = BOZ }
		cwp_gq_merge_zone = { ZONE = FOZ }
		if = {
			limit = { NOT = { exists = c:GER } }
			change_tag = GER
		}
		if = {
			limit = { NOT = { is_country_type = recognized } }
			set_country_type = recognized
		}
		if = {
			limit = { has_state_in_state_region = STATE_NORTH_RHINE }
			set_capital = STATE_NORTH_RHINE
		}
		set_global_variable = cwp_gq_frg_proclaimed
		# The Federal Republic's road to sovereignty starts here, whatever the zones had built up
		je:je_cwp_german_question ?= {
			set_bar_progress = { name = cwp_gq_western_bar value = 60 }
		}
	}
	# The Basic Law: a parliamentary federal republic with free elections and independent courts
	activate_law = law_type:law_parliamentary_republic
	activate_law = law_type:law_universal_suffrage
	activate_law = law_type:law_independent_judiciary
	random_interest_group = {
		limit = { is_marginal = no }
		save_temporary_scope_as = cwp_gq_basic_law_sponsor
	}
	if = {
		limit = { exists = scope:cwp_gq_basic_law_sponsor }
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = cwp_amendment_basic_law
				sponsor = scope:cwp_gq_basic_law_sponsor
			}
		}
	}
}

# Run in the Federal Republic's scope: the occupation statute ends.
cwp_gq_frg_sovereignty = {
	if = {
		limit = { is_subject = yes }
		make_independent = yes
	}
	set_global_variable = cwp_gq_frg_sovereign
}

# Run in the GDR's scope: Moscow recognises its sovereignty; it stays in the Eastern bloc.
cwp_gq_gdr_sovereignty = {
	if = {
		limit = { is_subject = yes }
		make_independent = yes
	}
	if = {
		limit = {
			is_in_power_bloc = no
			exists = c:RUS
			c:RUS = { is_power_bloc_leader = yes }
		}
		join_power_bloc = c:RUS
	}
	set_global_variable = cwp_gq_gdr_sovereign
}

# Run in the Federal Republic's scope: unification as a neutral state (the Stalin Note accepted).
cwp_gq_neutral_unification = {
	if = {
		limit = { exists = c:DDR }
		annex_with_incorporation = c:DDR
	}
	if = {
		limit = { is_subject = yes }
		make_independent = yes
	}
	if = {
		limit = { is_in_power_bloc = yes }
		leave_power_bloc = yes
	}
	add_modifier = {
		name = cwp_gq_neutral_germany
		days = long_modifier_time
	}
	set_global_variable = cwp_gq_neutral_unification
}

# The Soviet blockade of West Berlin ends (any scope).
cwp_gq_end_blockade = {
	remove_global_variable = cwp_gq_blockade
	remove_global_variable = cwp_gq_blockade_months
	set_global_variable = cwp_gq_berlin_settled
	c:RUS ?= {
		if = {
			limit = { has_modifier = cwp_gq_blockade_strain }
			remove_modifier = cwp_gq_blockade_strain
		}
	}
	if = {
		limit = { has_global_variable = cwp_gq_airlift }
		custom_tooltip = cwp_gq_airlift_won_tt
		remove_global_variable = cwp_gq_airlift
		every_country = {
			limit = { has_modifier = cwp_gq_airlift }
			remove_modifier = cwp_gq_airlift
			add_modifier = {
				name = cwp_gq_airlift_victory
				days = normal_modifier_time
			}
		}
		c:RUS ?= {
			add_modifier = {
				name = cwp_gq_blockade_failed
				days = normal_modifier_time
			}
		}
		cwp_gq_shift_bar = { BAR = cwp_gq_western_bar VALUE = 5 }
	}
	hidden_effect = {
		every_country = {
			limit = {
				OR = {
					cwp_gq_is_occupier = yes
					cwp_gq_is_german_party = yes
				}
			}
			trigger_event = { id = cwp_german_question.8 days = 1 popup = yes }
		}
	}
}
