# Institutions as state capacity (docs/agents/areas/institutions.md).
#
# 1946 state-capacity tiers, called from common/history/countries after the starting law package (the
# package sets laws and the levels its constitution implies; the tier says how much of that the
# state could actually staff and fund in 1946: civil service, literacy, tax take). A tier caps
# every institution, and from tier 3 also sets a floor on the core capacity institutions
# (Administration, Judiciary, Police, Economic System, Schools, Health System) that the laws allow.
#
#   tier  cap  core floor   e.g.
#   5     5    3            USA, GBR, Nordics, USSR, Japan
#   4     4    2            Italy, Czechoslovakia, Argentina, occupied Germany
#   3     3    1            Greece, Turkey, Mexico, Philippines, Ceylon
#   2     2    -            India, Egypt, Nationalist China, Colombia
#   1     1    -            most colonies, Indonesia, Ethiopia, Afghanistan
#
# The tier table per country is in docs/agents/areas/institutions.md; the history thread owns it.

cwp_capacity_cap_institution = {
	if = {
		limit = {
			has_institution = institution_$INST$
			institution_investment_level = { institution = institution_$INST$ value > $CAP$ }
		}
		set_institution_investment_level = { institution = institution_$INST$ level = $CAP$ }
	}
}

cwp_capacity_floor_institution = {
	if = {
		limit = {
			has_institution = institution_$INST$
			institution_investment_level = { institution = institution_$INST$ value < $FLOOR$ }
		}
		set_institution_investment_level = { institution = institution_$INST$ level = $FLOOR$ }
	}
}
