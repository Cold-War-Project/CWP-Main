# Cold War Project: military cost tiers by GDP ranking (docs/agents/areas/military.md).

cwp_military_cost_tier_update = {
	remove_modifier = cwp_military_global_power_projection
	remove_modifier = cwp_military_overseas_commitments
	remove_modifier = cwp_military_labour_intensive_forces
	if = {
		limit = { gdp_ranking = 1 }
		add_modifier = { name = cwp_military_global_power_projection }
	}
	else_if = {
		limit = { gdp_ranking <= 5 }
		add_modifier = { name = cwp_military_overseas_commitments }
	}
	# Poor countries field cheaper, infantry-heavy armies (India 1946: 9% of GDP, 59% of income without this).
	else_if = {
		limit = { gdp_per_capita_ranking > 90 }
		add_modifier = { name = cwp_military_labour_intensive_forces }
	}
}

# Start-force floor (start audit H1/H2): remember the 1946 army and navy, the army law and the population, so the AI
# keeps at least 0.8x its start forces until the floor lapses (cwp_military_start_floor_valid).
cwp_military_store_start_forces = {
	set_variable = { name = cwp_start_army_size value = army_size }
	set_variable = { name = cwp_start_ships value = num_ships }
	set_variable = { name = cwp_start_population value = total_population }
	if = {
		limit = { has_law_or_variant = law_type:law_peasant_levies }
		set_variable = { name = cwp_start_army_law value = 1 }
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_national_militia }
		set_variable = { name = cwp_start_army_law value = 2 }
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_professional_army }
		set_variable = { name = cwp_start_army_law value = 3 }
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_mass_conscription }
		set_variable = { name = cwp_start_army_law value = 4 }
	}
	else = {
		set_variable = { name = cwp_start_army_law value = 5 }
	}
}
