# Cold War Project: military cost tiers by GDP ranking (docs/agents/areas/military.md).

cwp_military_cost_tier_update = {
	remove_modifier = cwp_military_global_power_projection
	remove_modifier = cwp_military_overseas_commitments
	remove_modifier = cwp_military_labour_intensive_forces
	if = {
		limit = { gdp_ranking = 1 }
		add_modifier = { name = cwp_military_global_power_projection }
	}
	else_if = {
		limit = { gdp_ranking <= 5 }
		add_modifier = { name = cwp_military_overseas_commitments }
	}
	# Poor countries field cheaper, infantry-heavy armies (India 1946: 9% of GDP, 59% of income without this).
	else_if = {
		limit = { gdp_per_capita_ranking > 90 }
		add_modifier = { name = cwp_military_labour_intensive_forces }
	}
}
