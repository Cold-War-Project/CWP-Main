# Cold War Project: 1946 school system (common/history/countries/zz_cwp_1946_school_systems.txt).
# Country scope. Enacts $LAW$ where the country has no schools law providing the institution, then sets the
# Schools institution to $LEVEL$.
cwp_1946_school_system = {
	if = {
		limit = { NOT = { has_institution = institution_schools } }
		activate_law = law_type:$LAW$
	}
	if = {
		limit = { has_institution = institution_schools }
		set_institution_investment_level = { institution = institution_schools level = $LEVEL$ }
	}
}
