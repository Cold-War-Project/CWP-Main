je_tsarpoints_add_partial = {
	hidden_effect = {
		scope:journal_entry ?= {
			set_variable = {
				name = tp_partial_progress_var
				value = 0
			}
			while = {
				limit = {
					scripted_bar_progress = {
						name = $PROGRESS_BAR$
						value > var:tp_partial_progress_var
					}
				}
				change_variable = {
					name = tp_partial_progress_var
					add = 1
				}
			}
			set_variable = {
				name = tp_partial_progress_percent_var
				value = {
					value = var:tp_partial_progress_var
					divide = $PROGRESS_BAR_MAX$
					max = 1
				}
			}
			set_variable = {
				name = tp_partial_reward_var
				value = {
					value = $REWARD$
					multiply = var:tp_partial_progress_percent_var
					round = yes
				}
			}
		}
	}
	custom_tooltip = {
		text = je_tsarpoints_add_partial_tt
		set_variable = {
			name = tp_last_mission_reward
			value = scope:journal_entry.var:tp_partial_reward_var
		}
	}
}

					NOT = {

				trigger = {
					NOT = {
						has_trait = charismatic
					}

	else_if = {
		limit = {
			has_trait = experienced_political_operator

		limit = {

		}
		remove_trait = basic_political_operator

		limit = {
			any_scope_character = {
				has_role = character_role_favorite
			}
		}
		random_scope_character = {
			limit = {
				has_role = character_role_favorite
			}
			save_temporary_scope_as = tsarpoints_favourite_scope
			save_scope_as = fav_character_scope
		}
		scope:journal_entry = {
			set_variable = {
				name = je_fav_character_var
				value = scope:tsarpoints_favourite_scope
			}
		}
	}
}
