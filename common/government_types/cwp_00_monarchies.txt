# Cold War Project: law-combination government types ported from dev. Loaded before the cwp_generic_* fallbacks;
# regencies, colonial administrations, chartered companies and military occupations are excluded here so their own types still apply.
# Cold War Project: a constitutional monarchy under Allied military occupation (Japan, Romania, Bulgaria in 1946):
# the occupation authorities supervise, a Prime Minister governs, the monarch reigns in the background.
# Sorts before cwp_generic_subject_government_types.txt, so it is checked before gov_military_occupation.
gov_cwp_occupied_constitutional_monarchy = {
	transfer_of_power = parliamentary_elective
	new_leader_on_reform_government = yes

	male_ruler = "RULER_TITLE_PRIME_MINISTER"
	female_ruler = "RULER_TITLE_PRIME_MINISTER"

	possible = {
		has_law = law_type:law_monarchy
		is_subject_type = subject_type_military_occupation
		cwp_crown_has_elected_parliament = yes
		cwp_country_has_parliamentary_crown = yes
	}

	on_government_type_change = { cwp_crown_change_to_parliamentary = yes }
	on_post_government_type_change = { cwp_crown_post_change_to_parliamentary = yes }
}
