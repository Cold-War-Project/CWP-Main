# Cold War Project: Cold War regime types. Loaded first (cwp_000_), before the law-combination
# types in cwp_00_*.txt and the cwp_generic_* fallbacks, so the more specific regime wins.
# Every type excludes chartered companies and military occupations; monarchies also exclude regencies.
# Vanilla keys (gov_junta, gov_military_dictatorship, gov_corporate_state, gov_sultanate, gov_sheikhdom)
# are reused so their vanilla loc applies and old vanilla references resolve.

# Communist one-party state outside the USSR
gov_cwp_peoples_republic = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = yes

	male_ruler = "RULER_CHAIRMAN"
	female_ruler = "RULER_CHAIRMAN"

	possible = {
		NOR = {
			is_subject_type = subject_type_chartered_company
			is_subject_type = subject_type_military_occupation
		}
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_single_party_state
		NOT = { c:RUS ?= this }
	}

	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}

# Authoritarian corporatist regime (Estado Novo Portugal, Francoist Spain)
gov_corporate_state = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = yes

	male_ruler = "RULER_TITLE_CWP_HEAD_OF_STATE"
	female_ruler = "RULER_TITLE_CWP_HEAD_OF_STATE"

	possible = {
		OR = {
			# The restored corporatist governance law always lands here.
			has_law_or_variant = law_type:law_corporate_state
			# A republic run as a corporatist dictatorship without the formal law.
			AND = {
				NOR = {
					is_subject_type = subject_type_chartered_company
					is_subject_type = subject_type_military_occupation
				}
				OR = {
					has_law_or_variant = law_type:law_presidential_republic
					has_law_or_variant = law_type:law_parliamentary_republic
				}
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_corporatized_unions
				OR = {
					has_law_or_variant = law_type:law_national_supremacy
					has_law_or_variant = law_type:law_ethnostate
				}
			}
		}
	}


	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}
