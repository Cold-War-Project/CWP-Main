# Cold War Project: law-combination government types ported from dev. Loaded before the cwp_generic_* fallbacks;
# regencies, colonial administrations and chartered companies are excluded here so their own types still apply.
gov_papal_commune = {
	transfer_of_power = dictatorial

	male_ruler = "RULER_PONTIFEX_POPULI"
	female_ruler = "RULER_PONTIFEX_POPULI"

	possible = {
		NOT = { is_subject_type = subject_type_chartered_company }
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_state_religion
		exists = c:PAP
		c:PAP = ROOT
	}

	on_government_type_change = {
		change_to_dictatorial = yes
	}
	on_post_government_type_change = {
		post_change_to_dictatorial = yes
	}
}
