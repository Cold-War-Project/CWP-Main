# Cold War Project: Crown dominions and colonies under a monarchy law whose ruler is a governor-general (Jack, 2026-11-17).
# The file name sorts before cwp_000_cold_war_regimes.txt, so this type is checked first (first valid type wins).
# Without it these countries are plain hereditary monarchies and the engine crowns a generated king.
# dictatorial: the governor-general is the ruler and is never an IG leader; the prime minister leads an IG.
# Elections still follow the distribution-of-power law. A country leaves this type when it stops being a
# subject or drops the monarchy law, and then falls through to the normal monarchy or republic types.
# Ruler characters: common/character_templates/cwp_1946_<tag>.txt. Notes: docs/agents/areas/characters.md.

gov_cwp_dominion_governor_general = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_GOVERNOR_GENERAL"
	female_ruler = "RULER_TITLE_GOVERNOR_GENERAL"

	possible = {
		has_law_or_variant = law_type:law_monarchy
		is_subject = yes
		NOT = { is_subject_type = subject_type_military_occupation }
		OR = {
			c:CAN ?= this
			c:AST ?= this
			c:NZL ?= this
			c:SAF ?= this
			c:HND ?= this
		}
	}

	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}
