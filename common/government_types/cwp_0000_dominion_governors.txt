# Cold War Project: realms that share the British crown, and crown colonies (Jack, 2026-11-17; 2026-11-20).
# The file name sorts before cwp_000_cold_war_regimes.txt, so these types are checked first (first valid type wins).
# gov_cwp_dominion: a self-governing dominion. The Prime Minister is the ruler (parliamentary_elective); the
# governor-general represents the monarch in the background (character_role_cwp_governor_general, cwp_gg_var),
# as the British monarch is a background head of state under gov_democratic_constitutional_monarchy.
# The other two types: the governor (-general) still governs, so is the ruler (dictatorial, never an IG leader).
# A country leaves these types when it drops the monarchy law, and then falls through to the republic types.
# Crown roles and effects: common/character_roles/cwp_crown_roles.txt, common/scripted_effects/cwp_crown_effects.txt.
# Notes: docs/agents/areas/constitutional-monarchy.md.

# Southern Rhodesia: a self-governing colony with responsible government since 1923, not a dominion.
gov_cwp_self_governing_colony = {
	transfer_of_power = parliamentary_elective
	new_leader_on_reform_government = yes

	male_ruler = "RULER_TITLE_PRIME_MINISTER"
	female_ruler = "RULER_TITLE_PRIME_MINISTER"

	possible = {
		c:MTB ?= this
		is_subject = yes
		cwp_crown_is_self_governing_realm = yes
	}

	on_government_type_change = { cwp_crown_change_to_parliamentary = yes }
	on_post_government_type_change = { cwp_crown_post_change_to_parliamentary = yes }
}

gov_cwp_dominion = {
	transfer_of_power = parliamentary_elective
	new_leader_on_reform_government = yes

	male_ruler = "RULER_TITLE_PRIME_MINISTER"
	female_ruler = "RULER_TITLE_PRIME_MINISTER"

	possible = {
		cwp_crown_is_self_governing_realm = yes
	}

	on_government_type_change = { cwp_crown_change_to_parliamentary = yes }
	on_post_government_type_change = { cwp_crown_post_change_to_parliamentary = yes }
}

# India under the Viceroy until the transfer of power; also a realm of the shared crown that has lost its
# elected parliament or is under occupation (the governor-general governs).
gov_cwp_dominion_governor_general = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_GOVERNOR_GENERAL"
	female_ruler = "RULER_TITLE_GOVERNOR_GENERAL"

	possible = {
		has_law_or_variant = law_type:law_monarchy
		OR = {
			AND = {
				c:HND ?= this
				is_subject = yes
				NOT = { is_subject_type = subject_type_military_occupation }
			}
			cwp_crown_is_dominion_realm = yes
		}
	}

	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { cwp_crown_post_change_to_governor = yes }
}

# Crown colonies and mandates under a monarchy law: Burma, Ceylon, South West Africa,
# Newfoundland (Commission of Government). Same mechanics, title Governor.
gov_cwp_crown_colony_governor = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_GOVERNOR"
	female_ruler = "RULER_TITLE_GOVERNOR"

	possible = {
		has_law_or_variant = law_type:law_monarchy
		is_subject = yes
		NOT = { is_subject_type = subject_type_military_occupation }
		OR = {
			c:MYM ?= this
			c:SRI ?= this
			c:NAM ?= this
			c:NEW ?= this
		}
	}

	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}
