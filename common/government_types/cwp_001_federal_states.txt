# Cold War Project: federal states. Loaded after cwp_000_cold_war_regimes.txt (so juntas, corporatist
# and communist regimes still win) and before the law-combination types in cwp_00_*.txt.
# Federalism has no law in the game, so these are tied to the countries that were federations.

# Federal presidential republics (United States, Brazil, Mexico, Argentina, Venezuela)
gov_cwp_federal_presidential_republic = {
	transfer_of_power = presidential_elective
	new_leader_on_reform_government = yes

	male_ruler = "RULER_TITLE_PRESIDENT"
	female_ruler = "RULER_TITLE_PRESIDENT"

	possible = {
		NOR = {
			is_subject_type = subject_type_chartered_company
			is_subject_type = subject_type_military_occupation
		}
		country_has_voting_franchise = yes
		has_law_or_variant = law_type:law_presidential_republic
		OR = {
			c:USA ?= this
			c:BRZ ?= this
			c:MEX ?= this
			c:ARG ?= this
			c:VNZ ?= this
		}
	}

	on_government_type_change = { change_to_presidential_elective = yes }
	on_post_government_type_change = { post_change_to_presidential_elective = yes }
}
