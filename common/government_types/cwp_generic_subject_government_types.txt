# Generic Government Type - Colonial Administration
gov_colonial_administration_fallback = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_GOVERNOR_GENERAL"
	female_ruler = "RULER_TITLE_GOVERNOR_GENERAL"

	possible = {
		has_law = law_type:law_colonial_administration
		NOR = {
			is_subject_type = subject_type_dominion
			is_subject_type = subject_type_protectorate
			is_subject_type = subject_type_military_occupation
		}
	}

	on_government_type_change = {
		change_to_dictatorial = yes
	}
	on_post_government_type_change = {
		post_change_to_dictatorial = yes
	}
}
