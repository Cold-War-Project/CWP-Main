cwp_monarchy_clothes_trigger = {
	exists = scope:character
	scope:character = {
		exists = owner
		owner = { has_law = law_type:law_monarchy }
		OR = {
			is_ruler_of_own_country = yes
			is_heir_of_own_country = yes
		}
	}
}

cwp_republic_clothes_trigger = {
	exists = scope:character
	scope:character = {
		exists = owner
		owner = { NOT = { has_law = law_type:law_monarchy } }
	}
}

	exists = scope:character
	scope:character = {
		AND = {
			has_role_of_type = politician
			is_ruler_of_own_country = no
		}
	}
