# Cold War Project: triggers for interest group attraction (pop_weight / pop_potential).

# Pop scope. Peasants, or farm labour, living under landlord tenure (serfdom, manorialism, tenancy,
# latifundia), or working a plantation anywhere land is not collectivized.
cwp_pop_under_rural_patronage = {
	OR = {
		is_pop_type = peasants
		pop_employment_building_group = bg_agriculture
		pop_employment_building_group = bg_ranching
		pop_employment_building_group = bg_plantations
	}
	OR = {
		owner = {
			OR = {
				has_law_or_variant = law_type:law_serfdom
				has_law_or_variant = law_type:law_tenant_farmers
			}
		}
		AND = {
			pop_employment_building_group = bg_plantations
			owner = { NOT = { has_law_or_variant = law_type:law_collectivized_agriculture } }
		}
	}
}

# Country scope. A party-state: council republic or single-party state.
cwp_is_party_state = {
	OR = {
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_single_party_state
	}
}
