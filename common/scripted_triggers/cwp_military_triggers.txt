# Cold War Project: military scripted triggers (docs/agents/areas/military.md).

# The start-force floor (start audit H1/H2) holds while the country is solvent, keeps its 1946 army law, keeps at
# least half its 1946 population, and its forces cost under ~30% of its income (estimated at 1,500 a week per
# battalion and 800 per ship: the engine exposes no military-spending trigger).
cwp_military_start_floor_valid = {
	has_variable = cwp_start_army_size
	in_default = no
	total_population >= {
		value = var:cwp_start_population
		multiply = 0.5
	}
	OR = {
		AND = { var:cwp_start_army_law = 1 has_law_or_variant = law_type:law_peasant_levies }
		AND = { var:cwp_start_army_law = 2 has_law_or_variant = law_type:law_national_militia }
		AND = { var:cwp_start_army_law = 3 has_law_or_variant = law_type:law_professional_army }
		AND = { var:cwp_start_army_law = 4 has_law_or_variant = law_type:law_mass_conscription }
		AND = { var:cwp_start_army_law = 5 has_law_or_variant = law_type:law_disarmed_nation }
	}
	total_income > {
		value = army_size
		multiply = 1500
		add = {
			value = num_ships
			multiply = 800
		}
		divide = 0.3
	}
}
