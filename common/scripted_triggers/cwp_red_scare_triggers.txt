# Cold War Project: the Red Scare (je_cwp_red_scare). Country scope unless noted.

# A communist power that matters to us: communist leadership, at least a major power, diplomatically relevant to root.
cwp_rs_threat_abroad = {
	any_country = {
		NOT = { THIS = ROOT }
		country_rank >= rank_value:major_power
		cwp_country_has_communist_leadership = yes
		has_diplomatic_relevance = root
	}
}

# Communists at home: a communist movement above marginal support, or a communist leading a non-marginal interest group.
cwp_rs_threat_at_home = {
	OR = {
		any_political_movement = {
			OR = {
				is_political_movement_type = movement_communist
				is_political_movement_type = movement_cwp_orthodox_communist
			}
			political_movement_support >= 0.05
		}
		any_interest_group = {
			is_marginal = no
			leader ?= {
				OR = {
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_cwp_marxist_leninist
				}
			}
		}
	}
}

# A country with a free vote that is not itself communist.
cwp_rs_is_open_society = {
	country_has_voting_franchise = yes
	cwp_country_has_communist_leadership = no
	NOT = { has_law_or_variant = law_type:law_council_republic }
	NOT = { has_law_or_variant = law_type:law_single_party_state }
}

# Not before 1947, and not while communists still sit in the government: in western Europe the scare followed the
# communists' exit from the postwar coalitions (France and Italy, May 1947).
cwp_rs_is_candidate = {
	NOT = { has_variable = cwp_rs_done_var }
	game_date >= 1947.1.1
	cwp_communists_in_government = no
	cwp_rs_is_open_society = yes
	cwp_rs_threat_abroad = yes
	cwp_rs_threat_at_home = yes
}

# A rival with communist leadership: the foreign threat is personal.
cwp_rs_rival_is_communist = {
	any_rival_country = {
		cwp_country_has_communist_leadership = yes
	}
}

# The four repressive measures. Each is visible in the game as a modifier or law-like flag.
cwp_rs_loyalty_review_active = {
	ig:ig_bureaucrats ?= { has_modifier = cwp_rs_loyalty_review }
}
cwp_rs_unions_investigated_active = {
	ig:ig_trade_unions ?= { has_modifier = cwp_rs_unions_investigated }
}
cwp_rs_blacklist_active = {
	ig:ig_intelligentsia ?= { has_modifier = cwp_rs_blacklisted }
}
cwp_rs_party_banned_active = {
	has_modifier = cwp_rs_party_banned
}
