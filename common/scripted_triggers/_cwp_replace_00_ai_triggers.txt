	OR = {
		has_law_commitment = scope:law.type
		any_political_movement = {
			law_enactment_stance = {
				law = scope:law.type
				value > neutral
			}
			political_movement_radicalism >= 0.75
		}
		AND = {
			has_journal_entry = je_government_petition
			var:desired_law_var	?= scope:law
		}
		# AND = {

	}
}
