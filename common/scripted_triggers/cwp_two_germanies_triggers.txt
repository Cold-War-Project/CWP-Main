# Cold War Project: triggers for the German question, chapter 2 (the two Germanies).
# Journal entry: common/journal_entries/cwp_two_germanies.txt

# One of the two German states, or the superpower behind it (country scope)
cwp_tg_is_party = {
	OR = {
		c:GER ?= this
		c:DDR ?= this
		c:USA ?= this
		c:RUS ?= this
	}
}

# East Germans can still leave for the West, and have a reason to (no scope needed)
cwp_tg_emigration_active = {
	exists = c:DDR
	exists = c:GER
	NOT = { has_global_variable = cwp_tg_wall }
	c:DDR.average_sol < c:GER.average_sol
}
