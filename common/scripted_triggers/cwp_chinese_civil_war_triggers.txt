		state_region = s:STATE_FUJIAN
		state_region = s:STATE_GUANGXI
		state_region = s:STATE_ZHEJIANG
		state_region = s:STATE_SUZHOU
		state_region = s:STATE_NANJING
		state_region = s:STATE_HUNAN
		state_region = s:STATE_JIANGXI
		state_region = s:STATE_YUNNAN
		state_region = s:STATE_EASTERN_HUBEI
		state_region = s:STATE_WESTERN_HUBEI
		state_region = s:STATE_SHANDONG

		cwp_ccw_is_manchurian_state = yes

# Scope: state. The Soviet-occupied north-east of 1946.
cwp_ccw_is_manchurian_state = {
	OR = {
		state_region = s:STATE_SHENGJING
		state_region = s:STATE_SOUTHERN_MANCHURIA
		state_region = s:STATE_NORTHERN_MANCHURIA
		state_region = s:STATE_HINGGAN
	}
}

# Scope: any. Both Chinas exist and are at war with each other.
cwp_ccw_war_running = {
	exists = c:PRC
	exists = c:ROC
	c:PRC = { has_war_with = c:ROC }
}

# Scope: any. The Nationalists still control some part of the mainland.
cwp_ccw_roc_holds_mainland = {
	exists = c:ROC
	any_state = {
		controller ?= c:ROC
		cwp_ccw_is_mainland_state = yes
	}
}

# Scope: any. The Nationalists control Formosa.
cwp_ccw_roc_holds_formosa = {
	exists = c:ROC
	any_state = {
		state_region = s:STATE_FORMOSA
		controller ?= c:ROC
	}
}

# Scope: any. The Soviet Union still holds part of Manchuria.
cwp_ccw_soviets_in_manchuria = {
	exists = c:RUS
	c:RUS = {
		any_scope_state = {
			cwp_ccw_is_manchurian_state = yes
		}
	}
}

# Scope: any. The Nationalists have moved their government to Formosa.
cwp_ccw_roc_retreated = {
	exists = c:ROC
	c:ROC = { has_variable = cwp_ccw_great_retreat }
}

# Scope: country. One of the two Chinas.
cwp_ccw_is_belligerent = {
	OR = {
		c:PRC ?= this
		c:ROC ?= this
	}
}

# Scope: any. $WHO$ (a tag) controls every part of state region $REGION$.
cwp_ccw_controls_region = {
	exists = c:$WHO$
	any_state = { state_region = s:$REGION$ }
	NOT = {
		any_state = {
			state_region = s:$REGION$
			NOT = { controller ?= c:$WHO$ }
		}
	}
}
