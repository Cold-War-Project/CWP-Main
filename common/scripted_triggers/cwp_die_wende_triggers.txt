# Cold War Project: triggers for the German question, chapter 3 (Die Wende).
# Journal entry: common/journal_entries/cwp_die_wende.txt

# One of the four powers with rights over Germany as a whole (country scope)
cwp_dw_is_four_power = {
	OR = {
		c:USA ?= this
		c:GBR ?= this
		c:FRA ?= this
		c:RUS ?= this
	}
}

# Takes part in chapter 3 (country scope)
cwp_dw_is_party = {
	OR = {
		cwp_dw_is_four_power = yes
		c:GER ?= this
		c:DDR ?= this
	}
}

# Every one of the four powers that still exists consents to unity (no scope needed)
cwp_dw_four_powers_consent = {
	custom_tooltip = {
		text = cwp_dw_usa_consents_tt
		OR = {
			NOT = { exists = c:USA }
			c:USA = { has_variable = cwp_dw_consent }
		}
	}
	custom_tooltip = {
		text = cwp_dw_gbr_consents_tt
		OR = {
			NOT = { exists = c:GBR }
			c:GBR = { has_variable = cwp_dw_consent }
		}
	}
	custom_tooltip = {
		text = cwp_dw_fra_consents_tt
		OR = {
			NOT = { exists = c:FRA }
			c:FRA = { has_variable = cwp_dw_consent }
		}
	}
	custom_tooltip = {
		text = cwp_dw_rus_consents_tt
		OR = {
			NOT = { exists = c:RUS }
			c:RUS = { has_variable = cwp_dw_consent }
		}
	}
}
