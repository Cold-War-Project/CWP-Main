# Cold War Project: nuclear weapons. Other content (objectives, organisations, crises) should test nuclear status with this trigger.

cwp_is_nuclear_power = {
	has_variable = cwp_nuclear_power_var
}

cwp_nuc_can_start_programme = {
	NOT = { cwp_is_nuclear_power = yes }
	NOT = { has_variable = cwp_nuc_programme_var }
	is_subject = no
	country_rank >= rank_value:major_power
}

# A nuclear power outside our power bloc: the threat that drives a programme (bloc rivalry, not day-to-day relations).
cwp_nuc_faces_outside_nuclear_power = {
	any_country = {
		NOT = { this = root }
		cwp_is_nuclear_power = yes
		NOT = { is_in_same_power_bloc = root }
	}
}

# Sheltered under a nuclear bloc leader's umbrella: little reason to pay for a bomb of one's own.
cwp_nuc_under_nuclear_umbrella = {
	is_in_power_bloc = yes
	is_power_bloc_leader = no
	power_bloc ?= {
		power_bloc_leader ?= { cwp_is_nuclear_power = yes }
	}
}

# A friendly nuclear power stands behind us: an alliance, a defensive pact or a guarantee of our independence,
# or relations of amicable or better. Protection, not bloc membership, is what removes the need for a bomb.
cwp_nuc_has_nuclear_protector = {
	any_country = {
		NOT = { this = root }
		cwp_is_nuclear_power = yes
		OR = {
			relations:root >= relations_threshold:amicable
			root = { has_treaty_alliance_with = { TARGET = prev } }
			root = { has_treaty_defensive_pact_with = { TARGET = prev } }
			root = { has_treaty_guarantee_independence_with = { TARGET = prev } }
		}
	}
}

# Uranium: our own mines (or a subject's), or ore bought abroad through the JE button.
cwp_nuc_has_uranium = {
	OR = {
		has_variable = cwp_nuc_uranium_var
		any_scope_building = { is_building_type = building_uranium_mine }
		any_subject_or_below = {
			any_scope_building = { is_building_type = building_uranium_mine }
		}
	}
}

# A weapons laboratory: a university town big enough to hide a Los Alamos or an Arzamas-16.
cwp_nuc_has_laboratory = {
	any_scope_building = {
		is_building_type = building_university
		level >= 4
	}
}

# The physics is done: by years of work, or shortened by espionage or an ally's help.
cwp_nuc_design_complete = {
	OR = {
		has_variable = cwp_nuc_espionage_var
		has_variable = cwp_nuc_foreign_help_var
		je:je_cwp_nuclear_programme ?= { journal_entry_age >= 1460 }
	}
}
