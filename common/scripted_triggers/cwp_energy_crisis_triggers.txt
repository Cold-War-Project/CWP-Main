# Cold War Project: the energy crisis (je_cwp_energy_crisis). Country scope.

# Fuel in our market costs at least $PCT$ more than its base price (0.5 = 50%).
cwp_ec_fuel_pricier = {
	market_capital ?= {
		market = {
			mg:refined_fuel = { market_goods_pricier > $PCT$ }
		}
	}
}

# Opens only for a country that runs its own market (not subjects or junior members of someone else's market),
# after the postwar grace period, when fuel costs at least 75% over its base price there.
cwp_ec_is_candidate = {
	NOT = { has_variable = cwp_ec_cooldown_var }
	game_date >= 1948.1.1
	country_rank >= rank_value:minor_power
	market ?= { owner ?= ROOT }
	cwp_ec_fuel_pricier = { PCT = 0.75 }
}
