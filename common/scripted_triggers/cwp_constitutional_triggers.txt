# Cold War Project: constitutional transitions.

# The country is occupied by, or a puppet of, a council republic (the Soviet zone pattern of 1945-47).
cwp_iq_under_soviet_pressure = {
	OR = {
		is_subject_type = subject_type_military_occupation
		is_subject_type = subject_type_puppet
	}
	overlord ?= {
		has_law = law_type:law_council_republic
	}
}

# The occupier is a democracy that wants a democratic constitution (SCAP in Japan).
cwp_occupied_by_democracy = {
	is_subject_type = subject_type_military_occupation
	overlord ?= {
		OR = {
			has_law = law_type:law_universal_suffrage
			has_law = law_type:law_census_voting
		}
		NOT = { has_law = law_type:law_council_republic }
	}
}

# A communist-led interest group sits in government.
cwp_communists_in_government = {
	custom_tooltip = {
		text = cwp_communists_in_government_tt
		any_interest_group = {
			is_in_government = yes
			OR = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_cwp_neocommunist
					}
				}
				# Under a Soviet occupier the labour movement in government is the communists' vehicle.
				AND = {
					is_interest_group_type = ig_trade_unions
					owner = { cwp_iq_under_soviet_pressure = yes }
				}
			}
		}
	}
}

# Countries whose monarchy faced a referendum or forced abdication after 1945 and start the game as monarchies.
# Greece voted on the return of the exiled George II rather than on abolition; the same question in reverse.
cwp_iq_is_candidate = {
	OR = {
		c:ITA ?= this
		c:BUL ?= this
		c:ROM ?= this
		c:GRE ?= this
	}
	country_has_monarchy_law = yes
	NOT = { has_variable = cwp_iq_settled_var }
}

# The government can turn the referendum into a formality: occupied by a council republic and holding a secret police.
cwp_iq_can_rig = {
	custom_tooltip = {
		text = cwp_iq_can_rig_tt
		cwp_iq_under_soviet_pressure = yes
		has_law = law_type:law_secret_police
	}
}

cwp_jap_constitution_is_candidate = {
	c:JAP ?= this
	country_has_monarchy_law = yes
	NOT = { has_variable = cwp_jap_constitution_settled_var }
}

# Interest group scope: counts as republican in the institutional question.
cwp_iq_ig_is_republican = {
	OR = {
		law_stance = {
			law = law_type:law_monarchy
			value <= disapprove
		}
		AND = {
			OR = {
				is_interest_group_type = ig_trade_unions
				is_interest_group_type = ig_intelligentsia
			}
			NOT = {
				law_stance = {
					law = law_type:law_monarchy
					value >= approve
				}
			}
		}
	}
}

# The government can no longer put off the vote. Each line is one way it gets forced; the JE status shows them.
cwp_iq_vote_forced = {
	NOT = { has_variable = cwp_iq_grace_var }
	OR = {
		# the parties were promised a vote and republicans sit in the cabinet to collect
		AND = {
			has_variable = cwp_iq_vote_promised_var
			cwp_iq_republican_clout_in_government >= 0.15
		}
		# a powerful republican movement outside the government takes to the streets
		any_interest_group = {
			cwp_iq_ig_is_republican = yes
			is_powerful = yes
			is_in_government = no
		}
		# the Soviet occupier and its communist ministers want the matter closed
		AND = {
			cwp_iq_under_soviet_pressure = yes
			cwp_communists_in_government = yes
		}
	}
}

# Japan: the occupier's demands for a new constitution (SCAP's draft of February 1946), each shown as a line.
cwp_jap_scap_demands_met = {
	custom_tooltip = {
		text = cwp_jap_demand_franchise_tt
		has_law = law_type:law_universal_suffrage
		has_law = law_type:law_womens_suffrage
	}
	custom_tooltip = {
		text = cwp_jap_demand_shinto_tt
		OR = {
			has_law = law_type:law_total_separation
			has_law = law_type:law_freedom_of_conscience
		}
	}
	custom_tooltip = {
		text = cwp_jap_demand_army_tt
		OR = {
			has_law = law_type:law_disarmed_nation
			has_law = law_type:law_national_militia
		}
	}
	custom_tooltip = {
		text = cwp_jap_demand_throne_tt
		has_variable = cwp_jap_throne_redefined_var
	}
}
