# Cold War Project: sovietization.

cwp_sov_is_candidate = {
	cwp_iq_under_soviet_pressure = yes
	NOT = { has_law = law_type:law_single_party_state }
	NOT = { has_variable = cwp_sov_settled_var }
	# Germany and Austria are four-power questions with content of their own.
	NOR = {
		c:DDR ?= this
		c:AUS ?= this
	}
}

# The strongest interest group that rejects a council republic, saved as scope:cwp_sov_target_ig.
cwp_sov_has_opposition_target = {
	custom_tooltip = {
		text = cwp_sov_has_opposition_target_tt
		any_interest_group = {
			is_marginal = no
			cwp_sov_ig_is_opposition = yes
		}
	}
}

# Interest group scope: belongs to the non-communist opposition. The peasant parties (PSL, Smallholders,
# National Peasants) were the main opposition everywhere, whatever their ideology's stance on council republics.
cwp_sov_ig_is_opposition = {
	OR = {
		law_stance = {
			law = law_type:law_council_republic
			value <= disapprove
		}
		AND = {
			is_interest_group_type = ig_rural_folk
			NOT = {
				law_stance = {
					law = law_type:law_council_republic
					value >= approve
				}
			}
		}
	}
}
