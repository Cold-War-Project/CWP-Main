# Country scope. A working electoral democracy: some franchise and no autocratic, oligarchic, technocratic or
# single-party distribution of power, and not a council republic. Used to keep mainstream interest groups in
# mainstream parties (postwar democracies' fascist and communist parties drew narrow bases).
cwp_country_is_established_democracy = {
	country_has_voting_franchise = yes
	NOR = {
		has_law_or_variant = law_type:law_autocracy
		has_law_or_variant = law_type:law_oligarchy
		has_law_or_variant = law_type:law_technocracy
		has_law_or_variant = law_type:law_elder_council
		has_law_or_variant = law_type:law_single_party_state
		has_law_or_variant = law_type:law_council_republic
	}
}
