# This file contains new triggers introduced in CWP
# Do not add vanilla triggers here - use `_cwp_replace_...` files for those

### scope is COUNTRY ###
coa_parties_banned_trigger = {
	OR = {
		has_law = law_type:law_autocracy
		has_law = law_type:law_oligarchy
		has_law = law_type:law_single_party_state
	}
}

coa_executive_flag_trigger = {
	has_law = law_type:law_presidential_republic
}

coa_corporatist_trigger = {
	has_government_type = gov_corporate_state
}

coa_controls_part_of_greater_rhineland = {
	OR = {
		owns_entire_state_region = STATE_WESTPHALIA
		owns_entire_state_region = STATE_HESSE
	}
}

coa_controls_part_of_north_germany = {
	OR = {
		owns_entire_state_region = STATE_SCHLESWIG_HOLSTEIN
		owns_entire_state_region = STATE_VORPOMMERN
	}
}

### scope is COUNTRY DEFINITION ###
coa_def_military_occupation_trigger = {
	OR = {
		NOT = {
			exists = scope:actor
		}
		AND = {
			exists = scope:actor
			scope:actor = {
				is_subject_type = subject_type_military_occupation
			}
		}
	}
}
