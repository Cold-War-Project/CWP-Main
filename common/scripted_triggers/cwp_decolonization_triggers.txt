# Cold War Project: decolonization scripted triggers
# Used by je_cwp_winds_of_change and the per-colony journal entries in common/journal_entries/cwp_decolonization.txt.

# Countries that ruled colonies in 1946 (country scope).
cwp_is_colonial_metropole_tag = {
	OR = {
		c:GBR ?= this
		c:FRA ?= this
		c:NET ?= this
		c:BEL ?= this
		c:POR ?= this
		c:SPA ?= this
		c:ITA ?= this
		c:USA ?= this
		c:AST ?= this
		c:NZL ?= this
		c:SAF ?= this
	}
}

# Settler dominions: subjects of a metropole that are not colonies in the decolonization sense.
cwp_is_settler_dominion_tag = {
	OR = {
		c:CAN ?= this
		c:AST ?= this
		c:NZL ?= this
		c:SAF ?= this
		c:NEW ?= this
	}
}

# A subject that the anti-colonial tide acts on (country scope).
cwp_qualifies_as_colony = {
	is_subject = yes
	overlord ?= { cwp_is_colonial_metropole_tag = yes }
	NOT = { is_subject_type = subject_type_military_occupation }
	NOT = { cwp_is_settler_dominion_tag = yes }
}

# Country scope: still a colony that the tide tracks.
cwp_is_tracked_colony = {
	has_variable = cwp_colony
	is_subject = yes
}

# Country scope: rules at least one tracked colony.
cwp_has_tracked_colonies = {
	any_direct_subject = {
		cwp_is_tracked_colony = yes
	}
}

# Country scope: government includes the labour movement or the intelligentsia, the groups that most often
# accepted the end of empire in the metropoles.
cwp_has_anticolonial_leaning_government = {
	OR = {
		ig:ig_trade_unions ?= { is_in_government = yes }
		ig:ig_intelligentsia ?= { is_in_government = yes }
	}
}

# Country scope: government leans on the army or the landed interest, the groups that most often fought for empire.
cwp_has_imperialist_leaning_government = {
	OR = {
		ig:ig_armed_forces ?= { is_in_government = yes }
		ig:ig_landowners ?= { is_in_government = yes }
	}
}

### Indonesia
# Country scope: holds a state in the Indonesian archipelago outside western New Guinea.
cwp_holds_indonesian_state = {
	OR = {
		has_state_in_state_region = STATE_WEST_JAVA
		has_state_in_state_region = STATE_CENTRAL_JAVA
		has_state_in_state_region = STATE_EAST_JAVA
		has_state_in_state_region = STATE_NORTH_SUMATRA
		has_state_in_state_region = STATE_SOUTH_SUMATRA
		has_state_in_state_region = STATE_ACEH
		has_state_in_state_region = STATE_WEST_BORNEO
		has_state_in_state_region = STATE_EAST_BORNEO
		has_state_in_state_region = STATE_CELEBES
		has_state_in_state_region = STATE_MOLUCCAS
		has_state_in_state_region = STATE_SUNDA_ISLANDS
	}
}

# No scope: the Netherlands or one of its subjects still holds part of Indonesia.
cwp_dutch_hold_indonesia = {
	any_country = {
		OR = {
			c:NET ?= this
			is_subject_of = c:NET
		}
		NOT = { c:IDS ?= this }
		cwp_holds_indonesian_state = yes
	}
}

### Indochina
# Country scope: France or one of its subjects.
cwp_is_french_side = {
	OR = {
		c:FRA ?= this
		is_subject_of = c:FRA
	}
	NOT = { c:DAI ?= this }
}

# No scope: the French side still holds part of Vietnam.
cwp_french_hold_vietnam = {
	any_country = {
		cwp_is_french_side = yes
		OR = {
			has_state_in_state_region = STATE_TONKIN
			has_state_in_state_region = STATE_ANNAM
			has_state_in_state_region = STATE_MEKONG
		}
	}
}

### Opening the journal entries

# No scope. je_cwp_winds_of_change may open (checked monthly by cwp_decolonization_monthly_pulse).
cwp_woc_can_open = {
	NOT = { has_global_variable = cwp_woc_active }
	NOT = { has_global_variable = cwp_woc_concluded }
}

# No scope. je_cwp_india_transfer_of_power may open (checked monthly by cwp_decolonization_monthly_pulse).
cwp_india_can_open = {
	NOT = { has_global_variable = cwp_india_active }
	c:HND ?= { is_subject_of = c:GBR }
	NOT = { has_global_variable = cwp_india_je_cooldown }
}

# No scope. je_cwp_indonesian_revolution may open (checked monthly by cwp_decolonization_monthly_pulse).
cwp_indonesia_can_open = {
	NOT = { has_global_variable = cwp_indonesia_active }
	exists = c:NET
	c:IDS ?= { is_subject = no }
	cwp_dutch_hold_indonesia = yes
	NOT = { has_global_variable = cwp_indonesia_je_cooldown }
}

# No scope. je_cwp_indochina_war may open (checked monthly by cwp_decolonization_monthly_pulse).
cwp_indochina_can_open = {
	NOT = { has_global_variable = cwp_indochina_active }
	exists = c:FRA
	c:DAI ?= { is_subject = no }
	cwp_french_hold_vietnam = yes
	NOT = { has_global_variable = cwp_indochina_je_cooldown }
}

# No scope. je_cwp_algerian_question may open (checked monthly by cwp_decolonization_monthly_pulse).
cwp_algeria_can_open = {
	NOT = { has_global_variable = cwp_algeria_active }
	exists = c:FRA
	exists = s:STATE_ALGIERS.region_state:FRA
	NOT = { has_global_variable = cwp_algeria_je_cooldown }
	OR = {
		AND = {
			has_global_variable = cwp_tide_tier
			global_var:cwp_tide_tier >= 2
		}
		c:FRA = { has_variable = cwp_france_lost_indochina }
		c:MOR ?= { is_subject = no }
		c:TUN ?= { is_subject = no }
	}
}
