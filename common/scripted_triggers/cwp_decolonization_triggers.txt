# Cold War Project: decolonization scripted triggers
# Used by je_cwp_winds_of_change and the per-colony journal entries in common/journal_entries/cwp_decolonization.txt.

# Countries that ruled colonies in 1946 (country scope).
cwp_is_colonial_metropole_tag = {
	OR = {
		c:GBR ?= this
		c:FRA ?= this
		c:NET ?= this
		c:BEL ?= this
		c:POR ?= this
		c:SPA ?= this
		c:ITA ?= this
		c:USA ?= this
		c:AST ?= this
		c:NZL ?= this
		c:SAF ?= this
	}
}

# Settler dominions: subjects of a metropole that are not colonies in the decolonization sense.
cwp_is_settler_dominion_tag = {
	OR = {
		c:CAN ?= this
		c:AST ?= this
		c:NZL ?= this
		c:SAF ?= this
		c:NEW ?= this
	}
}

# A subject that the anti-colonial tide acts on (country scope).
cwp_qualifies_as_colony = {
	is_subject = yes
	overlord ?= { cwp_is_colonial_metropole_tag = yes }
	NOT = { is_subject_type = subject_type_military_occupation }
	NOT = { cwp_is_settler_dominion_tag = yes }
}

# Country scope: still a colony that the tide tracks.
cwp_is_tracked_colony = {
	has_variable = cwp_colony
	is_subject = yes
}

# Country scope: rules at least one tracked colony.
cwp_has_tracked_colonies = {
	any_direct_subject = {
		cwp_is_tracked_colony = yes
	}
}

# Country scope: government includes the labour movement or the intelligentsia, the groups that most often
# accepted the end of empire in the metropoles.
cwp_has_anticolonial_leaning_government = {
	OR = {
		ig:ig_trade_unions ?= { is_in_government = yes }
		ig:ig_intelligentsia ?= { is_in_government = yes }
	}
}

# Country scope: government leans on the army or the landed interest, the groups that most often fought for empire.
cwp_has_imperialist_leaning_government = {
	OR = {
		ig:ig_armed_forces ?= { is_in_government = yes }
		ig:ig_landowners ?= { is_in_government = yes }
	}
}
