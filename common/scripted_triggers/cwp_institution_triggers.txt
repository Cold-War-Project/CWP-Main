# Institutions as state capacity (docs/agents/areas/institutions.md). Country scope.

# The country can pay for more state: not in default, not borrowing with debt at a quarter of its
# credit limit or more (government administration's should_auto_expand test), and not carrying half its
# credit limit at all. taking_loans is a weekly flag, so the last test keeps a deeply indebted state that
# runs a brief surplus from switching straight back to full spending.
cwp_ai_state_affordable = {
	in_default = no
	OR = {
		taking_loans = no
		scaled_debt < 0.25
	}
	scaled_debt < 0.5
}

# Administration (tax capacity) would raise revenue: some incorporated state taxes past its capacity.
# Where no state does, another level only costs bureaucracy.
cwp_ai_tax_capacity_binds = {
	any_scope_state = {
		is_incorporated = yes
		tax_capacity < tax_capacity_usage
	}
}
