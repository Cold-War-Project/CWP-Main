# Cold War Project: triggers for the German question, chapter 1 (occupation and division).
# Journal entry: common/journal_entries/cwp_german_question.txt

# One of the three western occupation zones (country scope)
cwp_gq_is_western_zone = {
	OR = {
		c:AOZ ?= this
		c:BOZ ?= this
		c:FOZ ?= this
	}
}

# The West German state, once proclaimed (country scope)
cwp_gq_is_frg = {
	has_global_variable = cwp_gq_frg_proclaimed
	c:GER ?= this
}

# A western zone or, after the Basic Law, the Federal Republic (country scope)
cwp_gq_is_west_german = {
	OR = {
		cwp_gq_is_western_zone = yes
		cwp_gq_is_frg = yes
	}
}

# One of the four occupying powers (country scope)
cwp_gq_is_occupier = {
	OR = {
		c:USA ?= this
		c:GBR ?= this
		c:FRA ?= this
		c:RUS ?= this
	}
}

# A western occupying power (country scope)
cwp_gq_is_western_occupier = {
	OR = {
		c:USA ?= this
		c:GBR ?= this
		c:FRA ?= this
	}
}

# Any German state taking part in the question (country scope)
cwp_gq_is_german_party = {
	OR = {
		cwp_gq_is_west_german = yes
		c:DDR ?= this
	}
}

# The Soviet zone still answers to Moscow (no scope needed)
cwp_gq_soviet_zone_held = {
	exists = c:DDR
	exists = c:RUS
	c:DDR = { is_subject_of = c:RUS }
}

# At least one western zone or the Federal Republic exists (no scope needed)
cwp_gq_west_exists = {
	OR = {
		exists = c:AOZ
		exists = c:BOZ
		exists = c:FOZ
		AND = {
			has_global_variable = cwp_gq_frg_proclaimed
			exists = c:GER
		}
	}
}

# Every occupying power that still exists backs an all-German government (no scope needed)
cwp_gq_four_power_accord = {
	custom_tooltip = {
		text = cwp_gq_usa_backs_tt
		OR = {
			NOT = { exists = c:USA }
			c:USA = { has_variable = cwp_gq_backs_central_government }
		}
	}
	custom_tooltip = {
		text = cwp_gq_gbr_backs_tt
		OR = {
			NOT = { exists = c:GBR }
			c:GBR = { has_variable = cwp_gq_backs_central_government }
		}
	}
	custom_tooltip = {
		text = cwp_gq_fra_backs_tt
		OR = {
			NOT = { exists = c:FRA }
			c:FRA = { has_variable = cwp_gq_backs_central_government }
		}
	}
	custom_tooltip = {
		text = cwp_gq_rus_backs_tt
		OR = {
			NOT = { exists = c:RUS }
			c:RUS = { has_variable = cwp_gq_backs_central_government }
		}
	}
}

# The all-German checklist: the four powers agree before the zones have grown apart (no scope needed)
cwp_gq_all_german_conditions = {
	cwp_gq_four_power_accord = yes
	custom_tooltip = {
		text = cwp_gq_one_currency_tt
		NOT = { has_global_variable = cwp_gq_currency_reform }
	}
	custom_tooltip = {
		text = cwp_gq_no_separate_state_tt
		NOT = { has_global_variable = cwp_gq_frg_proclaimed }
		NOT = { has_global_variable = cwp_gq_gdr_proclaimed }
	}
	custom_tooltip = {
		text = cwp_gq_soviet_zone_exists_tt
		exists = c:DDR
		cwp_gq_soviet_zone_held = yes
	}
}
