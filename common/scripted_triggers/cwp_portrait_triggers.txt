# Cold War Project portrait triggers (1946-2046).
# Used only by gfx/portraits/portrait_modifiers/cwp_*.txt. They replace the vanilla clothing
# triggers in 00_clothes_triggers.txt, which test Victorian tags, techs and cultures.
#
# Naming:
#   cwp_portrait_culture_*   culture scope   (call inside culture = { })
#   cwp_portrait_country_*   country scope
#   cwp_portrait_character_* no scope needed, reads scope:character, false when it is absent
#   cwp_portrait_pop_*       no scope needed, reads scope:pop, false when it is absent
#   cwp_portrait_era_*       no scope needed, compares the game date

############
### Eras ###
############

# Hats as everyday wear: fedoras, flat caps and bowlers fade out in the 1960s.
cwp_portrait_era_hats_trigger = {
	game_date < 1965.1.1
}

# Traditional dress among the urban middle and upper classes outside the West gives way
# to Western clothing over the 1970s.
cwp_portrait_era_traditional_dress_trigger = {
	game_date < 1980.1.1
}

# Rural traditional dress lasts longer.
cwp_portrait_era_rural_dress_trigger = {
	game_date < 2000.1.1
}

# Winter months (the month trigger counts 0-11): November to February.
cwp_portrait_era_winter_trigger = {
	OR = {
		month < 2
		month > 9
	}
}

cwp_portrait_character_child = {
	scope:character ?= {
		is_adult = no
	}
}

cwp_portrait_character_adult_male = {
	scope:character ?= {
		is_adult = yes
		is_female = no
	}
}

cwp_portrait_character_adult_female = {
	scope:character ?= {
		is_adult = yes
		is_female = yes
	}
}

# Generals, admirals, Armed Forces politicians and military rulers.
cwp_portrait_character_military = {
	scope:character ?= {
		is_adult = yes
		OR = {
			has_role_of_type = general
			has_role_of_type = admiral
			AND = {
				is_interest_group_type = ig_armed_forces
				OR = {
					has_role_of_type = politician
					is_ruler_of_own_country = yes
				}
			}
		}
	}
}

cwp_portrait_character_senior_commander = {
	scope:character ?= {
		OR = {
			has_role_of_type = general
			has_role_of_type = admiral
		}
		commander_rank >= 2
	}
}

cwp_portrait_character_monarch = {
	scope:character ?= {
		cwp_crown_is_royal = yes # reigning or in line; not a Prime Minister who holds the ruler slot (cwp_crown_triggers.txt)
		owner ?= {
			cwp_portrait_country_monarchy = yes
		}
	}
}

cwp_portrait_character_clergy = {
	scope:character ?= {
		OR = {
			AND = {
				is_interest_group_type = ig_devout
				is_ruler_of_own_country = no
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
				}
			}
			AND = {
				is_ruler_of_own_country = yes
				owner ?= {
					has_law_or_variant = law_type:law_theocracy
				}
			}
		}
	}
}

cwp_portrait_character_business = {
	scope:character ?= {
		OR = {
			has_role_of_type = executive
			has_role_of_type = magnate
			AND = {
				has_role_of_type = politician
				is_interest_group_type = ig_industrialists
			}
		}
	}
}

# Leaders who wear a regional dress: traditional-zone cultures, until the 1980s, and outside
# communist states, where the party suit took over. Business leaders always wear suits.
cwp_portrait_character_regional_dress = {
	scope:character ?= {
		culture = {
			cwp_portrait_culture_traditional_dress_zone = yes
		}
	}
	cwp_portrait_character_business = no
	cwp_portrait_character_communist_country = no
	OR = {
		cwp_portrait_era_traditional_dress_trigger = yes
		scope:character ?= {
			OR = {
				is_interest_group_type = ig_devout
				is_interest_group_type = ig_rural_folk
				is_interest_group_type = ig_landowners
			}
		}
	}
}

### Uniform style (both) ###
############################

# Which uniform a military character or pop wears. Exactly one of these is true for anyone
# in uniform; the order of precedence is USA, Britain, communist states, everyone else.
cwp_portrait_uniform_usa = {
	OR = {
		AND = {
			cwp_portrait_character_military = yes
			cwp_portrait_character_usa = yes
		}
		AND = {
			cwp_portrait_pop_military = yes
			cwp_portrait_pop_usa = yes
		}
	}
}

cwp_portrait_uniform_britain = {
	OR = {
		AND = {
			cwp_portrait_character_military = yes
			cwp_portrait_character_britain = yes
		}
		AND = {
			cwp_portrait_pop_military = yes
			cwp_portrait_pop_britain = yes
		}
	}
}

cwp_portrait_uniform_communist = {
	OR = {
		AND = {
			cwp_portrait_character_military = yes
			cwp_portrait_character_communist_country = yes
		}
		AND = {
			cwp_portrait_pop_military = yes
			cwp_portrait_pop_communist_country = yes
		}
	}
	cwp_portrait_uniform_usa = no
	cwp_portrait_uniform_britain = no
}

cwp_portrait_uniform_generic = {
	OR = {
		cwp_portrait_character_military = yes
		cwp_portrait_pop_military = yes
	}
	cwp_portrait_uniform_usa = no
	cwp_portrait_uniform_britain = no
	cwp_portrait_uniform_communist = no
}

# Officers' uniform: officer pops and military characters (as opposed to soldier pops).
cwp_portrait_uniform_officer = {
	OR = {
		cwp_portrait_character_military = yes
		cwp_portrait_pop_officer = yes
	}
}

### Body weight ###
###################
# Same thresholds as vanilla; standard_of_living and pop_weight_modifier_scale are engine values.

cwp_portrait_pop_slightly_underweight = {
	scope:pop ?= {
		OR = {
			AND = {
				pop_weight_modifier_scale < 25
				standard_of_living >= 10
			}
			AND = {
				standard_of_living >= 5
				standard_of_living < 10
				pop_weight_modifier_scale >= 25
				pop_weight_modifier_scale < 50
			}
			AND = {
				standard_of_living < 5
				pop_weight_modifier_scale >= 50
			}
		}
	}
}

cwp_portrait_pop_underweight = {
	scope:pop ?= {
		OR = {
			AND = {
				pop_weight_modifier_scale < 25
				standard_of_living >= 5
				standard_of_living < 10
			}
			AND = {
				standard_of_living < 5
				pop_weight_modifier_scale >= 25
				pop_weight_modifier_scale < 50
			}
		}
	}
}

cwp_portrait_pop_very_underweight = {
