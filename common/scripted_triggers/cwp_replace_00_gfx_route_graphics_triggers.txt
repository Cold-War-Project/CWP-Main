# This file contains only vanilla triggers that need to be modified to work with CWP
# Do not add any non-vanilla triggers here


#################################################################
#####				SHIP CULTURE TRIGGERS					#####
#################################################################

gfx_marmara_ship_trigger = {
	OR = {
		OR = {
			country_has_primary_culture = cu:turkish
			country_has_primary_culture = cu:greek
		}
		AND = {
			is_subject = yes
			any_overlord_or_above = {
				OR = {
					country_has_primary_culture = cu:turkish
					country_has_primary_culture = cu:greek
				}
			}
		}
		power_bloc ?= {
			power_bloc_leader = {
				OR = {
					country_has_primary_culture = cu:turkish
					country_has_primary_culture = cu:greek
				}
			}
		}
	}
}

gfx_dhow_ship_trigger = {
	NOT = {
		country_has_primary_culture = cu:turkish
	}
	OR = {
		any_primary_culture = {
			OR = {
				has_discrimination_trait = arabic_heritage
				has_discrimination_trait = south_asian_heritage
			}
		}
		AND = {
			is_subject = yes
			any_overlord_or_above = {
				any_primary_culture = {
					OR = {
						has_discrimination_trait = arabic_heritage
						has_discrimination_trait = south_asian_heritage
					}
				}
			}
		}
		power_bloc ?= {
			power_bloc_leader = {
				any_primary_culture = {
					OR = {
						has_discrimination_trait = arabic_heritage
						has_discrimination_trait = south_asian_heritage
					}
				}
			}
		}
	}
}
