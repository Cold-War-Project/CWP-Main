# Cold War Project: triggers for the constitutional amendments (common/amendments/cwp_amendments.txt).

# Country scope. Communists hold power: a Marxist-Leninist ruler, or a planned economy with no free vote.
cwp_country_has_communist_leadership = {
	OR = {
		ruler ?= {
			OR = {
				has_ideology = ideology:ideology_communist
				has_ideology = ideology:ideology_vanguardist
				has_ideology = ideology:ideology_cwp_marxist_leninist
				has_ideology = ideology:ideology_cwp_maoist
			}
		}
		AND = {
			has_law_or_variant = law_type:law_command_economy
			country_has_voting_franchise = no
		}
	}
}
