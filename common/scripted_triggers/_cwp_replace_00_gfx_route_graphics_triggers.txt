#################################################################
#####				SHIP CULTURE TRIGGERS					#####
#################################################################

gfx_marmara_ship_trigger = {
	OR = {
		OR = {
			country_has_primary_culture = cu:turkish
			country_has_primary_culture = cu:greek
		}
		AND = {
			is_subject = yes
			any_overlord_or_above = {
				OR = {
					country_has_primary_culture = cu:turkish
					country_has_primary_culture = cu:greek
				}
			}
		}
		power_bloc ?= {
			power_bloc_leader = {
				OR = {
					country_has_primary_culture = cu:turkish
					country_has_primary_culture = cu:greek
				}
			}
		}
	}
}

gfx_dhow_ship_trigger = {
	NOT = {
		country_has_primary_culture = cu:turkish
	}
	OR = {
		any_primary_culture = {
			OR = {
				has_discrimination_trait = arabic_heritage
				has_discrimination_trait = south_asian_heritage
			}
		}
		AND = {
			is_subject = yes
			any_overlord_or_above = {
				any_primary_culture = {
					OR = {
						has_discrimination_trait = arabic_heritage
						has_discrimination_trait = south_asian_heritage
					}
				}
			}
		}
		power_bloc ?= {
			power_bloc_leader = {
				any_primary_culture = {
					OR = {
						has_discrimination_trait = arabic_heritage
						has_discrimination_trait = south_asian_heritage
					}
				}
			}
		}
	}
}

#################################################################
#####				ROUTE GRAPHICS TRIGGERS					#####

gfx_car_condition_trigger = {
	scope:state = {
		market = {
			mg:automobiles = {
				market_goods_sell_orders >= 10
			}
		}
	}
}

gfx_truck_condition_trigger = {
	scope:state = {
		any_scope_building = {
			OR = {
				has_active_production_method = pm_public_bus_system
				has_active_production_method = pm_subway_systems
				has_active_production_method = pm_integrated_transit
				has_active_production_method = pm_automated_transit
			}
		}
	}
}

gfx_tracktor_condition_trigger = {
	scope:state = {
		any_scope_building = {
			OR = {
				has_active_production_method = pm_mechanized_harvesting
				has_active_production_method = pm_advanced_agricultural_machinery
				has_active_production_method = pm_iot_sensors
				has_active_production_method = pm_ai_analyzed_harvesting
			}
		}
	}
}

gfx_experimental_locomotive_trigger = {
	scope:state = {
		any_scope_building = {
			is_building_type = building_railway
			level < 5
		}
	}
}

gfx_steam_locomotive_trigger = {
	scope:state = {
		any_scope_building = {

			level >= 5

		}
	}
}
