secession_progress_low = {
	any_civil_war = {
		civil_war_progress >= 0
		NOT = { civil_war_progress >= 0.34 }
		is_civil_war_type = secession
	}
}

secession_progress_medium = {
	any_civil_war = {
		civil_war_progress >= 0.34
		NOT = { civil_war_progress >= 0.67 }
		is_civil_war_type = secession
	}
}

secession_progress_high = {
	any_civil_war = {
		civil_war_progress >= 0.67
		is_civil_war_type = secession
	}
}

civil_war_progress_low = {
	civil_war_progress < 0.34
}

civil_war_progress_medium = {
	civil_war_progress >= 0.34
	civil_war_progress < 0.67
}

civil_war_progress_high = { # In a civil_war scope
	civil_war_progress >= 0.67
}
