# Cold War Project: decolonization in Africa, scripted triggers (design: docs/agents/areas/decolonization.md)

# No scope. Each JE opens from cwp_decolonization_monthly_pulse (common/on_actions/cwp_decolonization_on_actions.txt).
cwp_gold_can_open = {
	NOT = { has_global_variable = cwp_gold_active }
	NOT = { has_global_variable = cwp_gold_je_cooldown }
	c:GLC ?= { is_subject_of = c:GBR }
}

cwp_kenya_can_open = {
	NOT = { has_global_variable = cwp_kenya_active }
	c:KKY ?= { is_subject_of = c:GBR }
}

# Belgian paternalism holds until Africa starts to move.
cwp_congo_can_open = {
	NOT = { has_global_variable = cwp_congo_active }
	NOT = { has_global_variable = cwp_congo_je_cooldown }
	c:CNG ?= { is_subject_of = c:BEL }
	has_global_variable = cwp_tide_tier
	global_var:cwp_tide_tier >= 1
}

# The Estado Novo calls its colonies overseas provinces; the wars begin once the tide is running strongly.
cwp_portugal_can_open = {
	NOT = { has_global_variable = cwp_portugal_active }
	NOT = { has_global_variable = cwp_portugal_je_cooldown }
	cwp_portugal_holds_africa = yes
	has_global_variable = cwp_tide_tier
	global_var:cwp_tide_tier >= 2
}

# No scope: Portugal still rules Angola, Mozambique or Portuguese Guinea.
cwp_portugal_holds_africa = {
	exists = c:POR
	OR = {
		c:OVM ?= { is_subject_of = c:POR }
		c:MKU ?= { is_subject_of = c:POR }
		c:GNB ?= { is_subject_of = c:POR }
	}
}
