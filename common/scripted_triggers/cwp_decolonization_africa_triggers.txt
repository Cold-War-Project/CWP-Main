# Cold War Project: decolonization in Africa, scripted triggers (design: docs/agents/areas/decolonization.md)

# No scope. Each JE opens from cwp_decolonization_monthly_pulse (common/on_actions/cwp_decolonization_on_actions.txt).
cwp_gold_can_open = {
	NOT = { has_global_variable = cwp_gold_active }
	c:GLC ?= { is_subject_of = c:GBR }
}

cwp_kenya_can_open = {
	NOT = { has_global_variable = cwp_kenya_active }
	c:KKY ?= { is_subject_of = c:GBR }
}

# Opens at the start: Belgian paternalism (cwp_congo_paternalism) holds the Congo down until the tide reaches tier 2.
cwp_congo_can_open = {
	NOT = { has_global_variable = cwp_congo_active }
	c:CNG ?= { is_subject_of = c:BEL }
}

# The Estado Novo calls its colonies overseas provinces; the wars begin once the tide is running strongly.
cwp_portugal_can_open = {
	NOT = { has_global_variable = cwp_portugal_active }
	cwp_portugal_holds_africa = yes
	has_global_variable = cwp_tide_tier
	global_var:cwp_tide_tier >= 3
}

# No scope: Portugal still rules Angola, Mozambique or Portuguese Guinea.
cwp_portugal_holds_africa = {
	exists = c:POR
	OR = {
		c:OVM ?= { is_subject_of = c:POR }
		c:MKU ?= { is_subject_of = c:POR }
		c:GNB ?= { is_subject_of = c:POR }
	}
}

# No scope: the Gold Coast's three milestones, each shown in the tooltip.
cwp_gold_milestones_met = {
	custom_tooltip = {
		text = cwp_gold_milestone_constitution_tt
		has_global_variable = cwp_gold_constitution_adopted
	}
	custom_tooltip = {
		text = cwp_gold_milestone_election_tt
		has_global_variable = cwp_gold_election_won
	}
	custom_tooltip = {
		text = cwp_gold_milestone_precedent_tt
		c:HND ?= { is_subject = no }
		has_global_variable = cwp_tide_tier
		global_var:cwp_tide_tier >= 2
	}
}

# No scope: the secession of Katanga is under way.
cwp_katanga_can_open = {
	NOT = { has_global_variable = cwp_katanga_active }
	NOT = { has_global_variable = cwp_katanga_concluded }
	exists = c:CNG
	c:KTA ?= { is_subject = no }
}

# No scope: the officers are ready to overthrow the regime in Lisbon (each condition shown in the tooltip).
cwp_portugal_coup_ready = {
	c:POR ?= {
		custom_tooltip = {
			text = cwp_portugal_coup_regime_tt
			OR = {
				has_law = law_type:law_autocracy
				has_law = law_type:law_single_party_state
			}
		}
		custom_tooltip = {
			text = cwp_portugal_coup_officers_tt
			ig:ig_armed_forces ?= { ig_approval < 0 }
		}
	}
	custom_tooltip = {
		text = cwp_portugal_coup_fronts_tt
		has_global_variable = cwp_portugal_fronts
		global_var:cwp_portugal_fronts >= 2
	}
	custom_tooltip = {
		text = cwp_portugal_coup_campaigns_tt
		has_global_variable = cwp_portugal_campaigns
		global_var:cwp_portugal_campaigns >= 3
	}
}
