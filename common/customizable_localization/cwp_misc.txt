country_power = {
	type = country
	random_valid = no

	text = {
		trigger = { country_rank >= rank_value:superpower}
		localization_key = custom_loc_superpower
	}

	text = {
		trigger = { country_rank >= rank_value:great_power }
		localization_key = custom_loc_great_power
	}

	text = {
		trigger = { country_rank = rank_value:major_power }
		localization_key = custom_loc_major_power
	}

	text = {
		trigger = { country_rank = rank_value:minor_power }
		localization_key = custom_loc_minor_power
	}

	text = {
		trigger = { country_rank = rank_value:insignificant_power }
		localization_key = custom_loc_insignificant_power
	}
}

enactment_phase_0 = {
	type = country

	text = {
		localization_key = enactment_phase_0_tec
		trigger = { has_law_or_variant = law_type:law_technocracy }
	}

	text = {
		localization_key = enactment_phase_0_courep
		trigger = { has_law_or_variant = law_type:law_council_republic }
	}

	text = {
		localization_key = enactment_phase_0_autocracy
		trigger = {
			has_law_or_variant = law_type:law_autocracy
			NOT = { has_law_or_variant = law_type:law_council_republic }
		}
	}

	text = {
		localization_key = enactment_phase_0_demrep
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_parliamentary_republic
				has_law_or_variant = law_type:law_presidential_republic
			}
			OR = {
				has_law_or_variant = law_type:law_landed_voting
				has_law_or_variant = law_type:law_census_voting
				has_law_or_variant = law_type:law_wealth_voting
				has_law_or_variant = law_type:law_universal_suffrage
			}
		}
	}

	text = {
		localization_key = enactment_phase_0_conmon
		trigger = {
			country_has_monarchy_law = yes
			OR = {
				has_law_or_variant = law_type:law_census_voting
				has_law_or_variant = law_type:law_wealth_voting
				has_law_or_variant = law_type:law_universal_suffrage
			}
		}
	}

	text = {
		localization_key = enactment_phase_0_mon
		trigger = {
			country_has_monarchy_law = yes
			OR = {
				has_law_or_variant = law_type:law_landed_voting
				has_law_or_variant = law_type:law_oligarchy
				has_law_or_variant = law_type:law_single_party_state
			}
		}
	}

	text = {
		localization_key = enactment_phase_0_generic
		trigger = { always = yes }
	}
}
