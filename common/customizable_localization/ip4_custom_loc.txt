GetJoinUSALine = {
	type = country
	text = {
		trigger = {
			c:CUB ?= {
				liberty_desire > 25
				liberty_desire_weekly_progress >= -0.15
				has_variable = independencia_current
				var:independencia_current <= je_cuba_low_bracket
			}
		}
		localization_key = je_cuba_Pro_USA_status_desc
	}
	text = {
		trigger = {
			c:CUB ?= {
				liberty_desire > 25
				liberty_desire_weekly_progress >= -0.15
				has_variable = independencia_current
				var:independencia_current > je_cuba_low_bracket
				var:independencia_current < je_cuba_high_bracket
			}
		}
		localization_key = je_cuba_neutral_status_desc
	}
	text = {
		trigger = {
			c:CUB ?= {
				liberty_desire > 25
				liberty_desire_weekly_progress >= -0.15
				has_variable = independencia_current
				var:independencia_current >= je_cuba_high_bracket
			}
		}
		localization_key = je_cuba_independencia_status_desc
	}
	text = {
		trigger = {
			c:CUB ?= {
				OR = {
					liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
					liberty_desire_weekly_progress < -0.15
				}
			}
		}
		localization_key = je_cuba_stay_status_desc
	}
}
