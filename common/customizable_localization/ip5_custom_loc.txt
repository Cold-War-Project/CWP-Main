my_rcw_side_loc = {
	type = country

	text = {
		trigger = {
			has_law_or_variant = law_type:law_council_republic
			any_country = {
				country_has_primary_culture = cu:russian
				 NOT = { has_law_or_variant = law_type:law_council_republic }
			}
		}
		localization_key = rcw_reds_custom
	}
	text = {
		trigger = {
			NOT = { has_law_or_variant = law_type:law_council_republic }
			any_country = {
				country_has_primary_culture = cu:russian
				has_law_or_variant = law_type:law_council_republic
			}
		}
		localization_key = rcw_whites_custom
	}
	text = {
		trigger = {
			always = yes
		}
		localization_key = rcw_generic_custom
	}
}
