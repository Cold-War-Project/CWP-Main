ruler_or_heir = {
	type = country
	text = {
		trigger = {
			exists = ruler
			ruler ?= {
				age > define:NCharacters|ADULT_AGE
				NOT = { has_variable = is_married }
			}
		}
		localization_key = ruler_marriage_string_tt
	}
	text = {
		trigger = {
			exists = heir
			heir ?= {
				age > define:NCharacters|ADULT_AGE
				NOT = { has_variable = is_married }
			}
		}
		localization_key = heir_marriage_string_tt
	}
}
