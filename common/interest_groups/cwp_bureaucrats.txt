ig_bureaucrats = {
	color = hsv{ 0.65 0.54 0.52 }
	texture = "gfx/interface/icons/ig_icons/petty_bourgeoisie.dds"
	layer = "revolution_dynamic_petty_bourgeoisie"
	index = 8

	ideologies = {
		ideology_managerial
		ideology_state_paternalism
		ideology_statist
		# ideology_colonial_administrative
	}

	character_ideologies = {
		ideology_moderate
	}

	enable = {
		always = yes
	}

	traits = {
		ig_trait_corruption
		ig_trait_civil_service
		ig_trait_accommodating_government
	}

	on_enable = {
		ig:ig_bureaucrats = {
			set_ig_trait = ig_trait:ig_trait_corruption
			set_ig_trait = ig_trait:ig_trait_civil_service_2
			set_ig_trait = ig_trait:ig_trait_accommodating_government
		}
	}

	on_disable = {  }

	priority_cultures = {  }

	on_character_ig_membership = {  }

	pop_potential = {
		NOR = {
			is_pop_type = peasants
			is_pop_type = soldiers
		}
	}

	pop_weight = {
		value = 0

		# Academics: 50 if working in a nationalized building
		add = {
			desc = "POP_ACADEMICS"
			if = {
				limit = {
					is_pop_type = academics
					is_employed = yes
				}
				value = {
					value = 50
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
		
		# Aristocrats: 50 if employed in an administrative building
		add = {
			desc = "POP_ARISTOCRATS"
			if = {
				limit = {
					is_pop_type = aristocrats
					is_employed = yes
					workplace = { is_building_group = bg_government }
				}
				value = 50
			}
		}
		
		# Bureaucrats: 250, with an additional 50 if the politizied administration law is active
		add = {
			desc = "POP_BUREAUCRATS"
			if = {
				limit = { 
					is_pop_type = bureaucrats
					NOT = { owner = { has_law_or_variant = law_type:law_politicized_bureaucrats } }
				}
				value = 250
			}
			else_if = {
				limit = { 
					is_pop_type = bureaucrats
					owner = { has_law_or_variant = law_type:law_politicized_bureaucrats }
				}
				value = 300
			}
		}
		
		# Capitalists: 25 if the national champions law is active
		add = {
			desc = "POP_CAPITALISTS"
			if = {
				limit = { owner = { has_law_or_variant = law_type:law_national_champions } }
				value = 25
			}
		}
		
		# Clergymen: 50 if employed in an administrative building
		add = {
			desc = "POP_CLERGYMEN"
			if = {
				limit = {
					is_pop_type = clergymen
					is_employed = yes
					workplace = { is_building_group = bg_government }
				}
				value = 50
			}
		}
		
		# Clerks: 100 if employed in an administrative building; 50 if the politizied administration law is active; 50 if working in a nationalized building
		add = {
			desc = "POP_CLERKS"
			add = {
				if = {
					limit = {
						is_pop_type = clerks
						is_employed = yes
						workplace = { is_building_group = bg_government }
					}
					value = 100
				}
			}
			add = {
				if = {
					limit = { owner = { has_law_or_variant = law_type:law_politicized_bureaucrats } }
					value = 50
				}
			}
			add = {
				if = {
					limit = {
						is_pop_type = clerks
						is_employed = yes
						NOT = { workplace = { is_building_group = bg_government } }
					}
					value = {
						value = 50
						multiply = workplace.country_ownership_fraction
					}
				}
			}
		}
		
		# Engineers: 50 if working in a nationalized building
		add = {
			desc = "POP_ENGINEERS"
			if = {
				limit = {
					is_pop_type = engineers
					is_employed = yes
				}
				value = {
					value = 50
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
		
		# Farmers: 25 if working in a nationalized building
		add = {
			desc = "POP_FARMERS"
			if = {
				limit = {
					is_pop_type = farmers
					is_employed = yes
				}
				value = {
					value = 25
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
		
		# Laborers: 15 if working in a nationalized building
		add = {
			desc = "POP_LABORERS"
			if = {
				limit = {
					is_pop_type = laborers
					is_employed = yes
				}
				value = {
					value = 15
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
		
		# Machinists: 25 if working in a nationalized building
		add = {
			desc = "POP_MACHINISTS"
			if = {
				limit = {
					is_pop_type = machinists
					is_employed = yes
				}
				value = {
					value = 25
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
		
		# Officers: 15 if the home affairs institution is enabled
		add = {
			desc = "POP_OFFICERS"
			if = {
				limit = { owner = { has_institution = institution_home_affairs } }
				value = 15
			}
		}
		
		# Shopkeepers: 15 if working in a nationalized building
		add = {
			desc = "POP_SHOPKEEPERS"
			if = {
				limit = {
					is_pop_type = shopkeepers
					is_employed = yes
				}
				value = {
					value = 15
					multiply = workplace.country_ownership_fraction
				}
			}
			else = {
				value = 0
			}
		}
	}
}
