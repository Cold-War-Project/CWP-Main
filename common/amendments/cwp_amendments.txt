# USA: the first ten amendments (1791). Speech, press, assembly, religion, due process.
cwp_amendment_bill_of_rights = {
	parent = law_protected_speech

	allowed_laws = {
		law_right_of_assembly
		law_protected_speech
	}

	modifier = {
		country_legitimacy_base_add = 5
		state_radicals_from_political_movements_mult = -0.05
		country_coup_resistance_add = 0.25
	}

	possible = {
		c:USA ?= this
	}

	would_sponsor = {
		AND = {
			exists = scope:approval
			scope:approval > 0
		}
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.25
}

# Presidential term limits: Mexico's "no reelección" (1917), the bars on immediate re-election in
# Argentina, Chile, Colombia, Peru and Uruguay, and what became the US 22nd Amendment (1951).
# Parent is the parliamentary republic: groups that want a weaker presidency back it.
cwp_amendment_no_reelection = {
	parent = law_parliamentary_republic

	allowed_laws = {
		law_presidential_republic
	}

	modifier = {
		country_legitimacy_base_add = 5
		country_authority_mult = -0.05
		country_enactment_success_chance_law_autocracy_add = -0.10
	}

	possible = {
		has_law = law_type:law_presidential_republic
		country_has_voting_franchise = yes
	}

	# Mostly the opposition: the party that has not held the presidency wants it limited.
	would_sponsor = {
		OR = {
			is_in_government = no
			AND = {
				exists = scope:approval
				scope:approval > 0
			}
		}
	}

	ai_will_revoke = {
		country_has_voting_franchise = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

# The army as guardian of the nation above civilian politics: Brazil's poder moderador, Turkey's 1935

	parent = law_professional_army

	allowed_laws = {
		law_peasant_levies
		law_professional_army
		law_national_militia
		law_mass_conscription
	}

	modifier = {
		interest_group_ig_armed_forces_pol_str_mult = 0.25
		interest_group_ig_armed_forces_approval_add = 1
		country_coup_resistance_add = -0.25
	}

	possible = {
		always = yes
	}

	would_sponsor = {
		is_interest_group_type = ig_armed_forces
	}

	ai_will_revoke = {
		NOT = {
			any_interest_group = {
				is_interest_group_type = ig_armed_forces
				is_powerful = yes
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}
