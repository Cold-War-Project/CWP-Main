# Japan 1947, chapter I: the Emperor is the symbol of the State, without powers related to government.
cwp_amendment_symbol_emperor = {
	parent = law_monarchy

	allowed_laws = {
		law_monarchy
	}

	modifier = {
		country_legitimacy_base_add = 10
		country_authority_mult = -0.10
		interest_group_ig_armed_forces_pol_str_mult = -0.20
	}

	possible = {
		has_law = law_type:law_monarchy
	}

	would_sponsor = {
		always = no
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}
