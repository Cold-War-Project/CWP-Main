amendment_reinstated_fueros = {
	parent = law_subjecthood

	allowed_laws = {
		law_subjecthood
	}

	modifier = {
		country_basque_cultural_acceptance_add = 20
		country_fervor_target_basque_add = -10
		interest_group_ig_landowners_approval_add = 1
		interest_group_ig_rural_folk_approval_add = 1
		country_two_spains_conservative_drift_add = 0.1
	}

	possible = {
		has_variable = is_carlist_spain
	}

	would_sponsor = {
		AND = {
			exists = scope:approval
			scope:approval > 0
		}
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	on_activate = {
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_BASQUE_COUNTRY
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_BASQUE_COUNTRY
				}
				add_modifier = spc_state_fueros_restored
			}
		}
		add_primary_culture = cu:basque
	}

	on_deactivate = {
		if = {
			limit = {
				NAND = {
					has_variable_list = spain_accepted_cultures
					any_in_list = {
						variable = spain_accepted_cultures
						cu:basque ?= this
					}
				}
			}
			remove_primary_culture = cu:basque
		}
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_BASQUE_COUNTRY
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_BASQUE_COUNTRY
				}
				remove_modifier = spc_state_fueros_restored
			}
			add_radicals = {
				culture = cu:basque
				value = {
					add = huge_radicals
				}
			}
		}
	}

	amendment_activism_multiplier = 0.50
}
