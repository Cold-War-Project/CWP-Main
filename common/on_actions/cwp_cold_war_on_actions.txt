# Cold War Project: the global Cold War. Chained into the code on_action, so no shared file changes.
# The contextless je_cwp_cold_war opens on the first monthly pulse and stays until one camp collapses.

on_monthly_pulse = {
	on_actions = { cwp_cold_war_monthly_pulse }
}

cwp_cold_war_monthly_pulse = {
	effect = {
		if = {
			limit = {
				NOR = {
					has_global_variable = cwp_cw_active
					has_global_variable = cwp_cw_outcome_west
					has_global_variable = cwp_cw_outcome_east
				}
			}
			add_contextless_journal_entry = je_cwp_cold_war
		}
	}
}
