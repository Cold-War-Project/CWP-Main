# Cold War Project: the Korean question (je_cwp_divided_peninsula). Chained into the code on_action, so no shared file changes.

on_monthly_pulse = {
	on_actions = { cwp_korea_monthly_pulse }
}

cwp_korea_monthly_pulse = {
	effect = {
		# Open the situation (first month of a new game, or the first month after the mod is added to a save)
		if = {
			limit = {
				NOT = { has_global_variable = cwp_korea_active }
				NOT = { has_global_variable = cwp_korea_concluded }
				exists = c:NRK
				exists = c:STK
			}
			add_contextless_journal_entry = je_cwp_divided_peninsula
		}
		if = {
			limit = { has_global_variable = cwp_korea_active }
			cwp_korea_monthly_effect = yes
		}
	}
}
