# Fires after everyone enters the game from the lobby for the first time for each campaign
# At this point we DO know who the players are
# No root scope set
on_game_started_after_lobby = {
	effect = {
		# Game Rule: No Dynamic Naming
		if = {
			limit = {
				has_game_rule = no_dynamic_naming
			}
			on_game_started_after_lobby_reset_dynamic_naming_effect = yes
		}
		# The Ripper
		on_game_started_after_lobby_jack_the_ripper_effect = yes
		# Electoral Confidence
		on_game_started_after_lobby_electoral_confidence_effect = yes
		# Country lists
		assemble_country_list_effects = yes
		# Liberalism sliding scale
		calculate_global_democracy_index_effect = yes
		set_global_variable = { # Ensures that Abraham Lincoln is not a valid leader of the Chinese Literati in 1836
			name = ruler_selector_pool_invalidation_global_var
			days = 30
		}
		c:JAP ?= {
			on_game_started_after_lobby_japan_setup_effect = yes
		}
		c:RUS ?= {
			on_game_started_after_lobby_russia_setup_effect = yes
		}
		c:CHI ?= {
			set_variable = {
				name = opium_wars_unresolved
				days = long_modifier_time
			}
		}
	}
}

# No root scope set
on_monthly_pulse = {
	on_actions = {
		exile_pool_culling_on_action
	}
	effect = {
		on_monthly_pulse_prison_sentences_effect = yes

		if = {
			limit = {
				year < 1930
				NOT = { has_global_variable = germany_formed }
				c:GER ?= { is_country_alive = yes }
			}
			set_global_variable = germany_formed
		}

		# USA gets a positive/negative modifier depending on whether their capital is in Washington DC
		c:USA ?= {
			on_monthly_pulse_usa_capital_modifier_effect = yes
		}

		### Scripted country lists

		assemble_country_list_effects = yes
		# Executes the below:
		# assemble_aes_country_list_effect
		# assemble_german_confederation_list_effect
		# assemble_great_power_list_effect

		### Potato Famine
		if = {
			limit = {
				NOT = { has_global_variable = hungry_forties_happened }
				NOT = { has_global_variable = hungry_forties_var }
				any_state_in_europe = {
					state_region = {
						has_harvest_condition = disease_outbreak
					}
					count >= 15
				}
			}
			set_global_variable = hungry_forties_var
		}

		# Gründerzeit JE - has to happen here in order to not kill performance
		if = {
			limit = {
				has_dlc_feature = ip3_content
				has_global_variable_list = german_confederation_country_list
			}
			on_monthly_pulse_grunderzeit_effect = yes
		}

		# Reform or Revolution JE trigger
		if = {
			limit = {
				has_global_variable = marx_happened
			}
			on_monthly_pulse_reform_or_revolution_effect = yes
		}

		if = {
			limit = {
				exists = global_var:krakatoa_temperature_drop
				exists = global_var:krakatoa_temperature_recovery_rate
			}
			change_global_variable = {
				name = krakatoa_temperature_drop
				subtract = {
					value = global_var:krakatoa_temperature_recovery_rate
				}
			}
		}

		if = {
			limit = {
				NAND = {
					exists = c:MON
					has_global_variable = montenegro_defeat_var #do only once
				}
			}
			on_monthly_pulse_montenegro_raid_cleanup_effect = yes
		}

		### Checking for the Portugal is Not a Small Country achievement conditions
		if = {
			limit = {
				c:POR ?= {
					is_player = yes
				}
			}
			c:POR ?= {
				on_monthly_pulse_portugal_achievement_effect = yes
			}
		}
	}
}

on_monthly_pulse_character = {
	effect = {
		on_monthly_pulse_character_flip_traditionalist_effect = yes

		if = { # Updates biannual unit share cache
			limit = {
				OR = {
					has_role_of_type = admiral
					has_role_of_type = general
				}
			}
			set_variable = { name = monthly_cached_units_share value = this.num_units_share }
			set_variable = { name = monthly_cached_commander_coup_strength value = this.commander_coup_strength }
		}

		if = { # Temporary Immortality
			limit = {
				is_immortal = yes
				has_variable = temporary_immortality_var
				var:temporary_immortality_var != 0
			}
			change_variable = { name = temporary_immortality_var subtract = 1 }
		}
		else_if = {
			limit = {
				is_immortal = yes
				has_variable = temporary_immortality_var
				var:temporary_immortality_var = 0
			}
			set_character_immortal = no
		}
	}
}

				}

					state_region = s:STATE_CATALONIA
				}

					state = { always = no }
				}

					state = { always = no }
				}
				owner ?= {
					trigger_event = { id = sagrada_familia.3 days = 3 }

# Fires when a diplomatic play is started
# root = The diplomatic play that has been started
# scope:initiator = Initiator of the play
# scope:target = Target of the play
on_diplomatic_play_started = {
	effect = {
		on_diplomatic_play_started_annex_unification_supporters_effect = yes
		if = {
			limit = {
				is_diplomatic_play_type = dp_annex_subject
				scope:initiator = {
					has_variable = selected_federal_solution_var
				}
				scope:target ?= {
					has_variable = indian_federation_member
				}
			}
			on_diplomatic_play_started_indian_federation_annex_effect = yes
		}
		if = {
			limit = {
				scope:initiator = {
					c:TUR ?= this
					NOT = { has_variable = egyptian_crisis }
				}
				scope:target = {
					c:EGY ?= this
				}
				OR = {
					is_diplomatic_play_type = dp_conquer_state
					is_diplomatic_play_type = dp_return_state
					is_diplomatic_play_type = dp_return_state_to_subject
					is_diplomatic_play_type = dp_reduce_autonomy
					is_diplomatic_play_type = dp_make_tributary
					is_diplomatic_play_type = dp_make_protectorate
					is_diplomatic_play_type = dp_annex_subject
				}
			}
			save_scope_as = egyptian_crisis_scope
			scope:initiator = {
				trigger_event = { id = egyptian_crisis_events.1 popup = yes }
			}
		}
		# A subject whose own overlord turns on it starts looking for the exit
		if = {
			limit = {
				scope:target ?= {
					is_direct_subject_of = scope:initiator
				}
			}
			scope:target ?= {
				add_liberty_desire = define:NDiplomacy|LIBERTY_DESIRE_FROM_OVERLORD_DIPLOMATIC_PLAY
			}
		}
	}
}

# Root = Character
on_character_death = {
	effect = {
		#JE about Hawaii triggered by an event
		if = {
			limit = {
				is_ruler_of_own_country = yes
				always = no
				exists = c:HAW
				owner = c:HAW
				owner = {
					has_law = law_type:law_monarchy
					NOT = { has_variable = death_of_kamehameha }
					NOT = { has_global_variable = gp_hawaii_interest }
				}
			}
			owner = {
				set_variable = death_of_kamehameha
				add_journal_entry = { type = je_hawaii }
			}
		}

		if = {
			limit = {
				has_variable = expedition_leader_var
				owner = {
					NOT = { has_variable = tibet_expedition_progress_var }
				}
			}
			owner = {
				trigger_event = expedition_events.110
			}
		}

		### Force-generating French monarchs of the correct dynasty if something goes wrong
		if = {
			limit = {
				is_ruler_of_own_country = yes
				owner = {
					has_law = law_type:law_monarchy
					NOT = {
						exists = heir
					}
				}
			}
			on_character_death_french_monarch_fallback_effect = yes
		}
		### Caudillo Kingmaker
		if = {
			limit = {
				NOT = { has_variable = caudillo_kingmaker }
				is_interest_group_leader = yes
				interest_group ?= {
					is_interest_group_type = ig_armed_forces
					ig_counts_as_marginal = no
				}
				owner = {
					in_election_campaign = no
					country_has_voting_franchise = yes
					OR = {
						has_journal_entry = je_caudillo
						has_modifier = age_of_caudillos_modifier
					}
				}
			}
			owner = {
				trigger_event = { id = caudillo.5 days = 1 }
			}
		}

		### Garibaldi exist check
		if = {
			limit = {
				always = no
			}
			remove_global_variable = garibaldi_is_alive_global_var
		}
		### Espartero exist check
		if = {
			limit = {
				always = no
			}
			set_global_variable = espartero_is_dead_var
		}
		# Ethiopian emperor succession pre-unification
		if = {
			limit = {
				is_ruler_of_own_country = yes
				owner ?= {
					has_law = law_type:law_monarchy
					any_scope_character = {
						is_heir_of_own_country = yes
					}
				}
				has_variable = ethiopian_emperor
			}
			owner ?= {
				random_scope_character = {
					limit = {
						is_heir_of_own_country = yes
					}
					set_variable = ethiopian_emperor
				}
			}
			# non-ETH rulers in the ethiopian_emperor succession will continue to style themselves 'Emperor' even if another tag forms ETH. This is fine! The other tag is obviously an upstart, and we've created two competing claims for the title. This is what wwe in the biz call 'dynamic storytelling'.
		}
		if = { # if Franz Joseph I of Austria dies before The Mayerling Incident in 1889, making Rudolf Emperor
			limit = {
				owner = {
					c:AUS ?= this
				}
				is_ruler_of_own_country = yes
				always = no
				AND = {
					game_date > 1875.6.29
					game_date < 1889.1.30
				}
			}
			owner = {
				create_character = {
					template = aus_franz_ferdinand
				}
			}
		}

		if = {
			limit = {
				owner = {
					c:MON ?= this
					heir ?= {
						always = no
					}
				}
				is_ruler_of_own_country = yes
				always = no
			}
			trigger_event = { id = mon_state_formation.6 popup = yes }
		}

		if = { # In case Milan dies really quickly
			limit = {
				always = no
				owner ?= {
					c:SER ?= this
					ruler ?= {
						always = no
					}
				}
			}
			owner ?= {
				create_character = {
					template = SER_mihailo_obrenovic
				}
			}
		}

		if = { # Hijacks the deaths of colonial governors
			limit = {
				is_ruler_of_own_country = yes
				has_variable = colonial_governor_term
				owner ?= {
					is_country_type = colonial
					has_law = law_type:law_colonial_administration
				}
			}
			on_character_death_colonial_governor_succession_effect = yes
		}

		if = {
			limit = {
				owner ?= {
					c:JAP ?= this
					country_has_monarchy_law = yes
					country_has_non_monarchy_law = no # To handle an execution order issue
				}
			}
			on_character_death_japan_succession_effect = yes
		}
		if = {
			limit = {
				has_variable = daimyo_var
				var:daimyo_var ?= { # If the daimyo's owner has a state in the daimyo's domain
					any_scope_state = {
						owner ?= ROOT.owner # owner = state owner, ROOT.owner = daimyo owner
					}
				}
			}
			JAP_character_generate_new_daimyo = yes
		}
		if = {
			limit = {
				has_variable = former_shogun_var # designates Taikun
				owner ?= {
					country_is_taikunate = yes
				}
			}
			owner ?= {
				#debug_log = "[TimeKeeper.GetCurrentDate.GetString]: Taikun died: new Taikun designated - [THIS.GetCountry.GetNameNoFormatting]"
				country_designate_new_taikun = yes
			}
		}
		if = { # clear je_tp_fav_rasputin_complete_var if Alexei dies
			limit = {
				always = no
				owner ?= {
					has_variable = je_tp_fav_rasputin_complete_var
				}
			}
			owner ?= {
				remove_variable = je_tp_fav_rasputin_complete_var
			}
		}
	}
}

# Root = Character
on_new_ruler = {
	effect = {
		if = {
			limit = {
				owner = { # Assigning a regent should never fire this on_action
					has_gov_regency = no
				}
			}
			if = {
				limit = {
					owner = {
						NOT = {
							has_variable = hide_ruler_change_notification
						}
					}
					is_adult = yes
				}
				post_notification = new_ruler
			}
			if = {
				limit = {
					owner = {
						has_law_or_variant = law_type:law_theocracy
					}
					NOT = { religion = owner.religion }
				}
				change_character_religion = owner.religion
			}

			# Setup regencies
			if = {
				limit = {
					owner = {
						has_law = law_type:law_monarchy
					}
					is_adult = no
					NOT = { has_variable = regency_complete }
					NOT = { always = no } # Has his own event
				}
				owner = {
					trigger_event = {
						id = regency.1
						popup = yes
					}
				}
			}

			# For handling the Sikh Maharaja achievement
			if = {
				limit = {
					owner = {
						c:PAN ?= this
						has_journal_entry = je_sikh_sovereignty
					}
				}
				on_new_ruler_sikh_maharaja_effect = yes
			}

			# For transitioning Zanzibar out of Arab rule
			if = {
				limit = {
					owner = {
						c:ZZB ?= this
						any_primary_culture = {
							cu:arab = this
						}
					}
					culture = {
						NOT = { cu:arab = this }
					}
					NOT = {
						has_global_variable = decided_future_of_zanzibar
					}
				}
				owner = {
					trigger_event = { id = zanzibar.6 popup = yes }
				}
			}

			### Monarchial succession - add all successions here.
			if = {
				limit = {
					is_ruler_of_own_country = yes
					owner ?= {
						country_has_monarchy_law = yes
						country_has_non_monarchy_law = no # To handle an execution order issue
					}
				}
				on_new_ruler_monarch_succession_chains_effect = yes
				if = {
					limit = {
						c:JAP ?= owner
					}
					on_new_ruler_japan_succession_effect = yes
				}
			}

			if = { # For handling the Succession Rights treaty article in the case of ruler death
				limit = {
					exists = c:BRZ
					exists = c:POR
					owner = {
						c:POR ?= this
						NOT = {
							has_diplomatic_pact = {
								who = c:BRZ
								type = personal_union
							}
						}
						any_scope_treaty = {
							binds = c:BRZ
							any_scope_article = {
								has_type = amend_succession
								source_country = c:BRZ
							}
						}
					}
				}
				on_new_ruler_brazil_succession_rights_effect = yes
			}

			# For handling the assignment of emperor_var, shogun_var, and emperor traits
			if = {
				limit = {
					c:JAP ?= owner
				}
				on_new_ruler_japan_effect = yes
			}

			# every time Russia gets a new Tsar, the alignment variables are halved, making new things matter more.
			if = {
				limit = {
					owner = {
						c:RUS ?= this
						country_has_monarchy_law = yes
						has_russian_revolution_alignment_variables = yes
					}
				}
				owner ?= {
					change_variable = { name = rus_gov_con divide = 2 }
					change_variable = { name = rus_gov_lib divide = 2 }
					change_variable = { name = rus_gov_soc divide = 2 }
				}
			}
		}

		if = { # Handles total years in power, not consecutive years. Mostly for Japan.
			limit = {
				NOT = {
					has_variable = year_in_power
				}
			}
			set_variable = { name = year_in_power value = 1 } # In case all else fails
		}

		# Dismiss any existing favorites when a new ruler takes power
		owner = {
			every_scope_character = {
				limit = {
					has_role = character_role_favorite
				}
				remove_character_role = favorite
			}
		}
	}
}

# Root = Treaty
on_treaty_entered_into_force = {
	effect = {
		post_notification = treaty_entered_into_force
		if = {
			limit = {
				is_enforced = no
			}
			root.first_country = {
				create_diplomatic_catalyst = {
					type = catalyst_treaty_signed
					target = root.second_country
				}
			}
			root.second_country = {
				create_diplomatic_catalyst = {
					type = catalyst_treaty_signed
					target = root.first_country
				}
			}
		}
		if = { #add balkan war allies to the list
			limit = {
				OR = {
					#balkan ally
					OR = {
						root.first_country = {
							has_journal_entry = je_the_balkan_league
						}
						root.second_country = {
							has_journal_entry = je_the_balkan_league
						}
					}
					any_scope_article = {
						OR = {
							has_type = alliance
							has_type = support_independence
						}
					}
					#Ottoman ally during balkan war
					any_country_in_balkans = {
						has_journal_entry = je_the_balkan_league
					}
					OR = {
						root.first_country = {
							c:TUR ?= this
						}
						root.second_country = {
							c:TUR ?= this
						}
					}
					any_scope_article = {
						OR = {
							has_type = alliance
							has_type = defensive_pact
						}
					}
				}
			}
			generate_balkan_league_side_list = yes
			generate_ottoman_side_list = yes
		}

		# clear ownership transfer fleet(s)
		if = {
			limit = {
				any_scope_article_option = {
					has_type = ship_transfer
					target_country = root.first_country
				}
			}
			root.first_country = {
				clear_ownership_transfer_fleet = yes
			}
		}
		if = {
			limit = {
				any_scope_article_option = {
					has_type = ship_transfer
					target_country = root.second_country
				}
			}
			root.second_country = {
				clear_ownership_transfer_fleet = yes
			}
		}
	}
}

		# Negotiation achievement

			limit = {
				any_interest_group = {
					law_enactment_stance = {
						law = owner.currently_enacting_law.type
						value < neutral
					}
					ig_counts_as_marginal = no
					count >= 2
				}
			}
			set_variable = unanimity_achievement_possible
		}
	}

# Root = owner Country of the Law
# Executed when an enacting law is passed by reaching the final enactment phase
#
# NOTE: at the point in time when this on-action is triggered, the law
# is still considered being enacted, therefore the `currently_enacting_law` link
# will work as expected, even though the enactment is _just_ about to end
on_law_enactment_pass = {
	effect = {
		if = {
			limit = {
				c:MON ?= this
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_religious_schools
					ROOT.currently_enacting_law.type ?= law_type:law_public_schools
					ROOT.currently_enacting_law.type ?= law_type:law_private_schools
				}
			}
			trigger_event = {
				id = mon_state_formation.11
				days = 1
			}
		}
		if = {
			limit = {
				is_player = yes
				NOT = { has_variable = achievement_vox_populi_var }
				any_political_movement = {
					law_stance = {
						law = root.currently_enacting_law.type
						value > neutral
					}
					any_supporting_character = {
						has_role_of_type = agitator
					}
				}
			}
			set_variable = achievement_vox_populi_var
		}
		if = {
			limit = {
				is_player = yes
				NOT = {
					has_variable = achievement_amish_paradise_var
				}
				is_enacting_law = law_type:law_industry_banned
				year < 1846
			}
			set_variable = achievement_amish_paradise_var
		}
		if = {
			limit = {
				has_variable = unanimity_achievement_possible
				any_interest_group = {
					law_enactment_stance = {
						law = ROOT.currently_enacting_law.type
						value < neutral
					}
					ig_counts_as_marginal = no
					count = 0
				}
			}
			set_variable = unanimity_achievement_var
		}
		if = {
			limit = {
				has_journal_entry = je_pedro_republic
				is_enacting_law = law_type:law_presidential_republic
			}
			set_variable = { name = hide_ruler_change_notification days = 3 }
			set_variable = { name = keep_current_ruler_var days = 7 }
			trigger_event = { id = pedro.10 popup = yes days = 1 }
		}
		if = {
			limit = {
				has_modifier = brazil_slaving_pariah_modifier
				has_law_or_variant = law_type:law_slave_trade
				OR = {
					is_enacting_law = law_type:law_debt_slavery
					is_enacting_law = law_type:law_legacy_slavery
					is_enacting_law = law_type:law_colonial_slavery
					is_enacting_law = law_type:law_slavery_banned
				}
			}
			remove_modifier = brazil_slaving_pariah_modifier
		}
		if = {
			limit = {
				OR = {
					has_modifier = unrestricted_slave_trade_modifier
					has_modifier = restricted_slave_trade_modifier
					has_modifier = slave_shipments_intercepted_small
					has_modifier = slave_shipments_intercepted_large
				}
				has_law_or_variant = law_type:law_slave_trade
				OR = {
					is_enacting_law = law_type:law_debt_slavery
					is_enacting_law = law_type:law_legacy_slavery
					is_enacting_law = law_type:law_colonial_slavery
				}
			}
			on_law_enactment_pass_clear_slave_trade_modifiers_effect = yes
		}
		if = {
			limit = {
				c:CHI ?= {
					has_modifier = opium_ban_authority
					is_enacting_law = law_type:law_free_trade
				}
			}
			remove_modifier = opium_ban_authority
		}
	}
}

# owner = owner country of the Law
# root = law
# executed when the law is activated for any reason. It's usually because it
# has been enacted, but script can activate laws directly without going through
# the enactment process. Normally, it's immediately preceded by
# on_law_enactment_ended, but if a law is directly set by script, thus
# bypassing the enactment process, this will execute while
# on_law_enactment_ended will not
#
# NOTE: unlike the other on_law_<something> on-actions, at the point in time
# when this is triggered the law that was enacted is no longer considered being
# enacted. Therefore using the `currently_enacting_law` link will return the
# null law
on_law_activated = {
	effect = {
		if = {
			limit = {
				c:CHI ?= owner
				owner = {
					has_variable = opium_wars_unresolved
					has_law_or_variant = law_type:law_free_trade
				}
			}
			owner = {
				remove_variable = opium_wars_unresolved
			}
		}
		if = {
			limit = {
				c:BRZ ?= {
					has_journal_entry = je_matter_of_slavery
					OR = {
						has_modifier = unrestricted_slave_trade_modifier
						has_modifier = restricted_slave_trade_modifier
						has_modifier = slave_shipments_intercepted_small
						has_modifier = slave_shipments_intercepted_large
					}
					OR = {
						has_law_or_variant = law_type:law_debt_slavery
						has_law_or_variant = law_type:law_legacy_slavery
						has_law_or_variant = law_type:law_colonial_slavery
					}
				}
			}
			on_law_activated_clear_brazil_slave_modifiers_effect = yes
		}
		if = {
			limit = {
				c:BRZ ?= {
					has_journal_entry = je_matter_of_slavery
					OR = {
						has_law_or_variant = law_type:law_debt_slavery
						has_law_or_variant = law_type:law_legacy_slavery
						has_law_or_variant = law_type:law_colonial_slavery
					}
					any_interest_group = {
						has_modifier = restricted_slave_trade_ig_modifier
					}
				}
			}
			on_law_activated_clear_brazil_ig_slave_modifiers_effect = yes
		}
		owner = {
			if = { # If a currently enacting law is invalidated by an activated law, stop enacting it
				limit = {
					exists = currently_enacting_law
					currently_enacting_law ?= {
						law_is_available = no
					}
				}
				cancel_enactment = yes
			}
		}
		owner = {
			if = {
				limit = {
					game_date > 1836.1.1
					c:JAP ?= this
					NOR = {
						has_law = law_type:law_isolationism
						has_law = law_type:law_closed_borders
					}
				}
				trigger_event = {
					id = japan_religion.10
					years = 10
					popup = yes
				}
			}
		}

		# Dismiss favorites if the new law makes them incompatible
		owner = {
			if = {
				limit = {
					NOT = { can_have_favorites = yes }
				}
				every_scope_character = {
					limit = {
						has_role = character_role_favorite
					}
					remove_character_role = favorite
				}
			}
		}
	}
}

# Fires when a goal is taken in such a manner that it cannot be reversed - from diplomatic demands, capitulation, peace deals and so on
# Also fires for all wargoals taken via occupation timer and not reversed before the end of war, at the conclusion of peace, in addition to prior on_wargoal_enforced_by_timer
# Root = Country
# scope:actor = the country that enforced the wargoal
# scope:target = the country the wargoal was enforced against
# scope:diplomatic_play = the relevant diplomatic play
# scope:wargoal_impact = the impact of the wargoal enforced
# scope:war_goal_enforced = the wargoal that was enforced
on_wargoal_enforced = {
	effect = {
		# Any war victory over the country that reneged on our inherited treaty ports settles the grudge
		if = {
			limit = {
				has_variable = wants_treaty_port_revenge_against
				var:wants_treaty_port_revenge_against ?= scope:target
			}
			remove_variable = wants_treaty_port_revenge_against
		}

		if = {
			limit = {
				OR = {
					AND = {
						has_journal_entry = je_anti_lobby_demand
						var:anti_lobby_demand_target ?= scope:target
					}
					AND = {
						has_journal_entry = je_pro_lobby_demand
						var:pro_lobby_demand_target ?= scope:target
					}
				}
			}
			set_variable = { name = enforced_wargoal_on_enemy days = 90 }
		}
		if = {
			limit = {
				has_journal_entry = je_tp_fav_war
				var:tp_fav_war_target ?= scope:target
			}
			set_variable = {
				name = tp_fav_war_won
			}
		}
		set_variable = {
            name = recently_won_war
            days = 1825
            value = yes
        }
        if = { # recently crushed a secession
			limit = {
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_secession
				}
			}
			ROOT ?= {
				set_variable = {
					name = crushed_secession
					days = 365
				}
			}
		}
		if = {
			limit = {
				owner ?= {
					is_country_type = recognized
				}
			}
			scope:target = {
				set_variable = wargoal_enforced_by_recognized_power
			}
		}
		on_wargoal_enforced_effects = yes
		# Terms imposed by its own overlord push a subject towards independence, scaled by how harsh the war goal was
		if = {
			limit = {
				scope:target ?= {
					is_direct_subject_of = scope:actor
				}
			}
			scope:target ?= {
				add_liberty_desire = {
					value = scope:wargoal_impact
					max = define:NDiplomacy|LIBERTY_DESIRE_FROM_OVERLORD_WAR_GOAL_MAX
				}
			}
		}
	}
}

	}
}

						name = tut_commander_target_initial_rank
						value = root.commander_rank
					}
				}
			}
		}
	}
}

# Root = country
on_become_subject = {
	effect = {
		every_scope_state = {
			evaluate_and_assign_state_hub_dynamic_names = yes # All dynamic naming is handled by this scripted effect
		}
		if = { # revert back to ig_armed_forces
			limit = {
				has_variable = ig_guardia_civil_active
				c:PHI ?= this
				top_overlord = {
		            NOT = {
		            	c:SPA ?= this
		            }
		        }
			}
			ig:ig_armed_forces ?= {
				set_interest_group_name = ig_armed_forces
			}
			remove_variable = ig_guardia_civil_active
		}
		if = {
			limit = {
				has_modifier = modifier_not_recognized_by_spain
				top_overlord ?= {
					c:SPA ?= this
				}
			}
			remove_modifier = modifier_not_recognized_by_spain
		}
	}
}

# Root = country
on_become_independent = {
	effect = {
		every_scope_state = {
			evaluate_and_assign_state_hub_dynamic_names = yes # All dynamic naming is handled by this scripted effect
		}
		if = {
			limit = {
				always = no
				NOT = {
					country_has_primary_culture = cu:sinhala
				}
			}
			on_become_independent_ceylon_effect = yes
		}
		if = {
			limit = {
				c:BUL ?= this
			}
			trigger_event = {
				id = bulgaria_events.2
				popup = yes
				days = 7
			}
		}
		if = {
			limit = {
				has_law = law_type:law_colonial_administration
				NOR = {
					c:BIC ?= this # has bic_breakup.1
					always = no # has dei_breakup.1
					always = no # has alk_breakup.1
					has_variable = senegal_subject_var # has colonial_administration_events.2
					has_variable = zanj_subject_var # has colonial_administration_events.2
					has_variable = southern_africa_subject_var # has colonial_administration_events.2
					has_variable = congo_subject_var # has colonial_administration_events.2
				}
			}
			trigger_event = { id = independence.2 days = 1 popup = yes }
		}
		if = {  # if player gains independence as Philippines before researching Nationalism or without je_philippines_main
			limit = {
				has_dlc_feature = ip4_content
				c:PHI ?= this
			}
			add_primary_culture = cu:filipino
       		remove_primary_culture = cu:spanish
       		post_notification = filipino_mestizo_now_primary_culture
		}
		if = { # revert to ig_armed_forces
			limit = {
				has_variable = ig_guardia_civil_active
				c:PHI ?= this
			}
			ig:ig_armed_forces ?= {
				set_interest_group_name = ig_armed_forces
			}
			remove_variable = ig_guardia_civil_active
		}
		if = {
			limit = {
				c:POL ?= this
			}
			on_become_independent_poland_effect = yes
		}
	}
}

# This on-action is triggered in cases where there is no overriding on_impose effect on a law type. In case an on_impose effect exists on the law type, this will NOT be executed.
# Root = country being imposed upon
# scope:initiator = country doing the imposition
# scope:law = law being imposed
on_impose_law = {
	effect = {
		add_modifier = {
			name = impose_law_cooldown
			multiplier = 365
			days = 365
			is_decaying = yes
		}
		scope:law = {
			save_scope_as = foreign_imposition_law
		}
		save_scope_as = country_imposed_upon
		scope:initiator = {
			post_notification = imposition_demand_sent
			save_scope_as = foreign_imposition_country
		}
		if = {
			limit = {
				c:BIC ?= this
				is_subject = yes
				has_dlc_feature = ip2_content
			}
			trigger_event = { id = british_dictates.3 days = 0 popup = yes }
		}
		else = {
			trigger_event = { id = law_imposition_events.1 days = 0 popup = yes }
		}
	}
}

# This on-action is triggered every time a Power Bloc's Power Struggle starts.
# Root = Country that started the Power Struggle (other interesting scopes, such as Power Bloc leader, can be inferred from root)
on_power_bloc_struggle_started = {
	effect = {
		if = {
			limit = {
				is_player = yes
			}
			set_variable = achievement_im_the_captain_now_challanger_var
		}
	}
}

# This on-action is triggered every time a Power Bloc's Power Struggle concludes (after the changes in leadership have been made).
# Root = Power Bloc whose leadership position is being fought over
# scope:successful_contender = country that won the Power Struggle
# scope:failed_contender = country that lost the Power Struggle
on_power_bloc_struggle_ended = {
	effect = {
		if = {
			limit = {
				scope:successful_contender ?= {
					has_variable = achievement_im_the_captain_now_challanger_var
					is_player = yes
					NOT = { has_variable = achievement_im_the_captain_now }
				}
			}
			scope:successful_contender = {
				set_variable = achievement_im_the_captain_now
			}
		}
		if = {
			limit = {
				scope:successful_contender ?= {
					has_variable = achievement_im_the_captain_now_challanger_var
				}
			}
			scope:successful_contender = {
				remove_variable = achievement_im_the_captain_now_challanger_var
			}
		}
		if = {
			limit = {
				scope:failed_contender ?= {
					has_variable = achievement_im_the_captain_now_challanger_var
				}
			}
			scope:failed_contender = {
				remove_variable = achievement_im_the_captain_now_challanger_var
			}
		}
	}
}
