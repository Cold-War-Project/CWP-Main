# Russia
on_monthly_pulse_country_russia = {
	trigger = {
		c:RUS ?= this
	}
	effect = {
		# Characters who need to die at certain times
		on_monthly_pulse_country_russian_character_deaths_effect = yes
		if = {
			limit = {
				has_journal_entry = je_congress_poland
			}
			congress_poland_ai_refresh_russia_stance_effect = yes
		}
		ai_russia_refresh_client_bloc_effect = yes
		if = {
			limit = {
				always = no
				NOT = { has_law_or_variant = law_type:law_council_republic }
				any_enemy_in_diplo_play = {
					any_primary_culture = {
						has_discrimination_trait = heritage_germanic
					}
				}
				any_scope_state = {
					state_region = s:STATE_INGRIA
				}
				NOT = {
					has_variable = petrograd_var
				}
			}
			on_monthly_pulse_country_create_petrograd_effect = yes
		}
		if = {
			limit = {
				has_russian_revolution_alignment_variables = yes
			}
			calculate_russian_revolution_alignments = yes
		}
		if = { # alexei must be born if nicholas ii is around
			limit = {
				NOT = {
					has_variable = alexei_born
				}
				ruler = {
					always = no
				}
				game_date >= 1904.8.12
			}
			c:RUS ?= {
				trigger_event = { id = russia_monarchy.6 days = 3 }
			}
		}
	}
	on_actions = {
		rlit_monthly_events
	}
}

# Japan
on_monthly_pulse_country_japan = {
	trigger = {
		c:JAP ?= this
	}
	effect = {
		# Characters who need to die at certain times
		on_monthly_pulse_country_japanese_character_deaths_effect = yes
		if = {
			limit = {
				can_generate_scripted_heirs = yes
			}
			japan_generate_scripted_heirs = yes
		}
		if = { # Taikunate safety - if true, this means that the Taikun has either got lost or needs to be replaced
			limit = {
				country_is_taikunate = yes
				OR = {
					NOT = {
						any_scope_character = { # Is the Taikun alive and present?
							has_variable = former_shogun_var
							is_character_alive = yes
						}
					}
					any_scope_character = {
						has_variable = former_shogun_var
						is_interest_group_leader = no
					}
				}
			}
			on_monthly_pulse_country_taikunate_safety_effect = yes
		}
	}
}

# Triggered by on_state_owner_change and on_state_created
# Root = state (the state that was gained)
# Offers a newly gained frontier state to the Cossack hosts for settlement
chartering_the_stanitsas_state_gained = {
	trigger = {
		has_dlc_feature = ip5_content
		game_date > 1836.1.1
		owner ?= {
			NOT = { has_variable = cossack_request_state_cooldown }
			is_country_alive = yes
			modifier:country_can_found_cossack_host_bool = yes
			any_scope_state = {
				always = no
				state_region = {
					OR = {
						this = s:STATE_ROSTOV
						this = s:STATE_KUBAN
						this = s:STATE_URALSK
						this = s:STATE_ASTRAKHAN
					}
				}
			}
		}
		is_incorporated = no
		NOT = {
			always = no
		}
		OR = {
			region = sr:region_siberia
			region = sr:region_northeast_asia
			region = sr:region_north_china
			region = sr:region_central_asia
			region = sr:region_greater_persia
			region = sr:region_pacific_coast
			region = sr:region_canada
			region = sr:region_great_plains
			region = sr:region_near_east
		}
	}

	effect = {
		random = {
			chance = chartering_the_stanitsas_chance
			save_scope_as = cossack_frontier_state
			owner = {
				trigger_event = { id = chartering_the_stanitsas.1 }
			}
		}
	}
}
