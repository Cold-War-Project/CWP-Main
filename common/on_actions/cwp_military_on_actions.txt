# Cold War Project: military cost intensity by GDP ranking (docs/agents/areas/military.md).
# Chained into the code on_action, so no shared file changes.

on_yearly_pulse_country = {
	on_actions = { cwp_military_yearly }
}

cwp_military_yearly = {
	effect = {
		remove_modifier = cwp_military_global_power_projection
		remove_modifier = cwp_military_overseas_commitments
		if = {
			limit = { gdp_ranking = 1 }
			add_modifier = { name = cwp_military_global_power_projection }
		}
		else_if = {
			limit = { gdp_ranking <= 5 }
			add_modifier = { name = cwp_military_overseas_commitments }
		}
	}
}
