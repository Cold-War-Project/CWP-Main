# Cold War Project: military cost intensity by GDP ranking (docs/agents/areas/military.md).
# Chained into the code on_action, so no shared file changes.

on_yearly_pulse_country = {
	on_actions = { cwp_military_yearly }
}

cwp_military_yearly = {
	effect = {
		cwp_military_cost_tier_update = yes
	}
}

# Also at game start, so the 1946 budgets are right before the first yearly pulse.
on_game_started = {
	on_actions = { cwp_military_game_start }
}

cwp_military_game_start = {
	effect = {
		every_country = {
			cwp_military_cost_tier_update = yes
			cwp_military_store_start_forces = yes
		}
	}
}
