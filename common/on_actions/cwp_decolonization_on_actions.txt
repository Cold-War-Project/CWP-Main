# Cold War Project: decolonization. Chained into the code on_action, so no shared file changes.
# Contextless journal entries do not open from their `possible` block; each is added here, once its
# cwp_<x>_can_open trigger holds (common/scripted_triggers/cwp_decolonization_triggers.txt).
# Each JE sets cwp_<x>_active in its immediate and clears it in every outcome.

on_monthly_pulse = {
	on_actions = { cwp_decolonization_monthly_pulse }
}

cwp_decolonization_monthly_pulse = {
	effect = {
		if = {
			limit = { cwp_woc_can_open = yes }
			add_contextless_journal_entry = je_cwp_winds_of_change
		}
		if = {
			limit = { cwp_india_can_open = yes }
			add_contextless_journal_entry = je_cwp_india_transfer_of_power
		}
		if = {
			limit = { cwp_indonesia_can_open = yes }
			add_contextless_journal_entry = je_cwp_indonesian_revolution
		}
		if = {
			limit = { cwp_indochina_can_open = yes }
			add_contextless_journal_entry = je_cwp_indochina_war
		}
		if = {
			limit = { cwp_algeria_can_open = yes }
			add_contextless_journal_entry = je_cwp_algerian_question
		}
	}
}
