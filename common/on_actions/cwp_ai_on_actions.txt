# Cold War Project: AI behaviour hooks (docs/agents/areas/ai.md). Chained into the code on_action, so no shared file changes.

on_monthly_pulse_country = {
	on_actions = { cwp_ai_monthly }
}

cwp_ai_monthly = {
	trigger = { is_ai = yes }
	effect = {
		cwp_ai_construction_backstop = yes
		cwp_ai_institution_austerity = yes
	}
}

on_yearly_pulse_state = {
	on_actions = { cwp_ai_yearly_state }
}

cwp_ai_yearly_state = {
	effect = {
		cwp_construction_workforce_update = yes
		cwp_ai_empty_building_cleanup = yes
	}
}

on_monthly_pulse_country = {
	on_actions = { cwp_ai_monthly_all }
}

cwp_ai_monthly_all = {
	effect = {
		cwp_ai_superpower_bloc_update = yes
	}
}
