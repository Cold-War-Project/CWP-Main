# Cold War Project: regime change monarchy guard (common/scripted_effects/cwp_regime_change_effects.txt).

on_monthly_pulse_country = {
	on_actions = { cwp_regime_change_monthly_pulse_country }
}

# Root = country: a monarchy that was not just imposed by a regime change has a monarchy tradition from now on.
cwp_regime_change_monthly_pulse_country = {
	trigger = {
		has_law = law_type:law_monarchy
		NOT = { has_variable = cwp_monarchy_tradition_var }
		NOT = { has_variable = cwp_regime_change_window_var }
	}
	effect = {
		set_variable = cwp_monarchy_tradition_var
	}
}
