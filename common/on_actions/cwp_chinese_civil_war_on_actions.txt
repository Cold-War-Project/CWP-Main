chinese_civil_war_battle_score = {
	trigger = {
		is_involved_in_journal_entry = je_chinese_civil_war
		has_global_variable = chinese_civil_war_guerilla_phase
		scope:enemy_country ?= { is_involved_in_journal_entry = je_chinese_civil_war }
	}

	effect = {
		if = {
			limit = { c:PRC ?= this }
			change_global_variable = { name = prc_chinese_civil_war_score add = 1 }
			je:je_chinese_civil_war ?= { add_progress = { value = -1 name = je_chinese_civil_war_score_bar } }
		}
		else_if = {
			limit = { c:ROC ?= this }
			change_global_variable = { name = roc_chinese_civil_war_score add = 1 }
			je:je_chinese_civil_war ?= { add_progress = { value = 1 name = je_chinese_civil_war_score_bar } }
		}
	}
}
