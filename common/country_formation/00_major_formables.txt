GER = {
	geographic_region = geographic_region_greater_germany

	is_major_formation = yes

	unification_play = dp_unify_germany
	leadership_play = dp_leadership_germany

	required_states_fraction = 0.73

	ai_will_do = { has_technology_researched = nationalism }

	possible = {
		OR = {
			has_technology_researched = pan-nationalism
			custom_tooltip = {
				text = je_german_unification_idea_trigger_desc
				has_variable = je_german_unification_idea
			}
		}
		custom_tooltip = {
			text = no_dual_or_triple_monarchy_tt
			NOR = {
				has_government_type = gov_constitutional_kingdom
			}
		}
		NOT = {
			c:KUK ?= this
		}
	}

	max_num_formation_candidates = 3
	can_be_formation_candidate = {
		country_rank >= rank_value:major_power
	}

	can_be_unification_target = {
		NOR = {
			has_government_type = gov_constitutional_kingdom
		}
	}

	should_target_state_in_unification_play = {}
}

ITA = {
	use_culture_states = yes

	is_major_formation = yes

	unification_play = dp_unify_italy
	leadership_play = dp_leadership_italy

	required_states_fraction = 0.7

	ai_will_do = { has_technology_researched = nationalism }

	potential = {
        NOR = {
            c:AUS ?= this
            c:KUK ?= this
            c:GER ?= this
        }
	}

	possible = {
		is_in_geographic_region = geographic_region_italy_old
		OR = {
			country_has_primary_culture = cu:italian
		}
		is_country_type = recognized
        NOR = {
            c:AUS ?= this
            c:KUK ?= this
            c:GER ?= this
        }
	}

	max_num_formation_candidates = 3
	can_be_formation_candidate = {
		country_rank >= rank_value:minor_power
        NOR = {
            c:AUS ?= this
            c:KUK ?= this
            c:GER ?= this
        }
		is_country_type = recognized
		any_country_in_italy_old = {
			has_technology_researched = nationalism
			percent = 1
		}
	}

	can_be_unification_target = {
		NAND = { # United States of Greater Austria
			country_has_primary_culture = cu:italian
			OR = {
				has_journal_entry = je_danubian_federation
				c:KUK ?= this
          	  	c:GER ?= this
			}
		}
	}

	should_target_state_in_unification_play = {}
}

YUG = {
	geographic_region = geographic_region_yugoslavia

	is_major_formation = yes

	unification_play = dp_unify_yugoslavia
	leadership_play = dp_leadership_yugoslavia

	required_states_fraction = 0.75

	ai_will_do = { always = yes }

	potential = {
        NOR = {
            c:AUS ?= this
            c:KUK ?= this
        }
		trigger_if = {
			limit = {
				c:BUL ?= this
			}
			has_variable = bul_pursued_greater_yugoslavism
		}
		trigger_else = {
			always = yes
		}
	}

	possible = {
        NOR = { # Not Danubian State or HRE
            c:AUS ?= this
            c:KUK ?= this
            c:GER ?= this
        }
		has_technology_researched = pan-nationalism
		trigger_if = {
			limit = {
				c:BUL ?= this
			}
			custom_tooltip = {
				text = bulgaria_pursued_greater_yugoslavism_tt
				has_variable = bul_pursued_greater_yugoslavism
			}
		}
	}

	max_num_formation_candidates = 3 # For the odd Croatia, Serbia and Bulgaria FFA
	can_be_formation_candidate = {
		country_rank >= rank_value:major_power
        NOR = { # Not Danubian State or HRE
            c:AUS ?= this
            c:KUK ?= this
            c:GER ?= this
        }
		trigger_if = {
			limit = {
				c:BUL ?= this
			}
			custom_tooltip = {
				text = bulgaria_pursued_greater_yugoslavism_tt
				has_variable = bul_pursued_greater_yugoslavism
			}
		}
	}

	can_be_unification_target = {
		is_in_geographic_region = geographic_region_balkans
		NAND = { # United States of Greater Austria
			OR = {
				country_has_primary_culture = cu:slovene
				country_has_primary_culture = cu:croat
				country_has_primary_culture = cu:serb
			}
			OR = {  # Not Danubian State or HRE
				has_journal_entry = je_danubian_federation
				c:KUK ?= this
            	c:GER ?= this
			}
		}
		trigger_if = {
			limit = {
				c:BUL ?= this
			}
			custom_tooltip = {
				text = bulgaria_pursued_greater_yugoslavism_tt
				has_variable = bul_pursued_greater_yugoslavism
			}
		}
		trigger_else = {
			always = yes
		}
	}

	should_target_state_in_unification_play = {}
}
