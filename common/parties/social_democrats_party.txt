				value = -3
				desc = "from_early_trade_unions"
			}
		}

		if = { # against free trade
			limit = {
				owner = { has_law_or_variant = law_type:law_free_trade }

					has_ideology = ideology:ideology_socialist

			}

				value = 3
				desc = "from_free_trade"
			}
		}

		if = { # against l-f
			limit = {
				owner = { has_law_or_variant = law_type:law_laissez_faire }
				OR = {
					has_ideology = ideology:ideology_agrarian
					has_ideology = ideology:ideology_socialist

						OR = {
							cwp_vanguardist_party_character = yes
							has_ideology = ideology:ideology_protectionist
							has_ideology = ideology:ideology_luddite
						}
					}
				}
			}
			add = {
				value = 3
				desc = "from_anti_laissez_faire"
			}
		}

				value = 2
				desc = "from_archetype_party"
			}
		}

		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				is_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 3 }

				value = 0.05

			}
		}
		else_if = {
			limit = { scope:number > 3 }

				value = 0.0

			}

					OR = {
						any_active_law = { has_amendment = amendment_type:amendment_socialist_party_dominance }
