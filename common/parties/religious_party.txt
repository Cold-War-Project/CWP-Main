# Fallback party for Devout
# When devout are powerful, other religious IG's might join

religious_party = {
	color = { 100 100 100 }
	name = {
		first_valid = {
			# 1946-era names (cwp); see localization/english/cwp/cwp_party_names_1946_l_english.yml
			triggered_desc = {
				desc = party_cwp_aus_rel
				trigger = { exists = c:AUS THIS = c:AUS }
			}
			triggered_desc = {
				desc = party_cwp_bel_rel
				trigger = { exists = c:BEL THIS = c:BEL }
			}
			triggered_desc = {
				desc = party_cwp_chl_rel
				trigger = { exists = c:CHL THIS = c:CHL }
			}
			triggered_desc = {
				desc = party_cwp_czh_rel
				trigger = { exists = c:CZH THIS = c:CZH }
			}
			triggered_desc = {
				desc = party_cwp_ddr_rel
				trigger = { exists = c:DDR THIS = c:DDR }
			}
			triggered_desc = {
				desc = party_cwp_egy_rel
				trigger = { exists = c:EGY THIS = c:EGY }
			}
			triggered_desc = {
				desc = party_cwp_fra_rel
				trigger = { exists = c:FRA THIS = c:FRA }
			}
			triggered_desc = {
				desc = party_cwp_ger_rel
				trigger = { exists = c:GER THIS = c:GER }
			}
			triggered_desc = {
				desc = party_cwp_hnd_rel
				trigger = { exists = c:HND THIS = c:HND }
			}
			triggered_desc = {
				desc = party_cwp_hun_rel
				trigger = { exists = c:HUN THIS = c:HUN }
			}
			triggered_desc = {
				desc = party_cwp_ita_rel
				trigger = { exists = c:ITA THIS = c:ITA }
			}
			triggered_desc = {
				desc = party_cwp_lux_rel
				trigger = { exists = c:LUX THIS = c:LUX }
			}
			triggered_desc = {
				desc = party_cwp_mex_rel
				trigger = { exists = c:MEX THIS = c:MEX }
			}
			triggered_desc = {
				desc = party_cwp_net_rel
				trigger = { exists = c:NET THIS = c:NET }
			}
			triggered_desc = {
				desc = party_cwp_nor_rel
				trigger = { exists = c:NOR THIS = c:NOR }
			}
			triggered_desc = {
				desc = party_cwp_pol_rel
				trigger = { exists = c:POL THIS = c:POL }
			}
			triggered_desc = {
				desc = party_cwp_swi_rel
				trigger = { exists = c:SWI THIS = c:SWI }
			}
			triggered_desc = {
				desc = party_cwp_vnz_rel
				trigger = { exists = c:VNZ THIS = c:VNZ }
			}
			triggered_desc = {
				desc = party_shinto
				trigger = {
					country_has_state_religion = rel:shinto
				}
			}
			triggered_desc = {
				desc = party_buddhist
				trigger = {
					root.religion = {
						has_discrimination_trait = heritage_dharmic
					}
				}
			}
			triggered_desc = {
				desc = party_sikh
				trigger = {
					country_has_state_religion = rel:sikh
				}
			}
			triggered_desc = {
				desc = party_hindu
				trigger = {
					country_has_state_religion = rel:hindu
				}
			}
			triggered_desc = {
				desc = party_jewish
				trigger = {
					country_has_state_religion = rel:jewish
				}
			}
			triggered_desc = {
				desc = party_muslim
				trigger = {
					root.religion = {
						has_discrimination_trait = heritage_islamic
					}
				}
			}
			triggered_desc = {
				desc = party_catholic
				trigger = {
					country_has_state_religion = rel:catholic
				}
			}
			triggered_desc = {
				desc = party_christian
				trigger = {
					root.religion = {
						has_discrimination_trait = heritage_christian
					}
				}
			}
			triggered_desc = {
				desc = party_religious
				trigger = { always = yes }
			}
		}
	}

	icon = {
		default = "/gfx/interface/icons/political_parties/religious.dds"
	}

	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		NOT = { c:USA ?= this } # USA: two-party system (Democrats / Republicans)
	}

	available_for_interest_group = {
		trigger_if = {
			limit = {
				ig_is_member_of_dominant_party = { PARTY = religious }
			}
			always = yes
		}
		trigger_else = {
			owner = { country_has_voting_franchise = yes }
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}

		if = { # postwar democracies: Christian democracy anchored in the churches
			limit = {
				owner = { cwp_country_is_established_democracy = yes }
				OR = {
					is_interest_group_type = ig_devout
				}
			}
			add = {
				value = 5
				desc = "from_cwp_established_democracy"
			}
		}

		if = { # ITA: Christian Democracy's Catholic and middle-class base
			limit = {
				owner = { c:ITA ?= this }
				OR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_petty_bourgeoisie
				}
			}
			add = {
				value = 30
				desc = "from_cwp_governing_party_1946"
			}
		}

		if = { # FRA: the MRP's Catholic and middle-class base
			limit = {
				owner = { c:FRA ?= this }
				OR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_petty_bourgeoisie
				}
			}
			add = {
				value = 30
				desc = "from_cwp_governing_party_1946"
			}
		}

		if = {
			limit = {
				is_interest_group_type = ig_devout
				owner = { has_law_or_variant = law_type:law_theocracy }
			}
			add = {
				value = 5
				desc = "from_theocracy"
			}
		}

		if = { # nobody will join if the devout don't join
			limit = {
				NOT = { is_interest_group_type = ig_devout }
				OR = {
					NOT = { exists = owner.py:religious_party }
					AND = {
						exists = owner.py:religious_party
						owner = { ig:ig_devout ?= { NOT = { is_member_of_party = owner.py:religious_party } } }
					}
				}
			}
			add = {
				value = -20
				desc = "from_no_devout_in_party" # this should probably never actually display in game because the party won't exist
			}
		}

		if = { # a party for powerful religious people
			limit = {
				is_interest_group_type = ig_devout
				is_powerful = yes
			}
			add = {
				value = 10
				desc = "from_powerful_devout"
			}
		}

		if = { # religiously inclined people might join if devout are powerful
			limit = {
				is_powerful = no
				owner = { ig:ig_devout ?= { is_powerful = yes } }
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_traditionalist
						has_ideology = ideology:ideology_theocrat
						has_ideology = ideology:ideology_corporatist_leader
					}
				}
			}
			add = {
				value = 10
				desc = "from_powerful_devout_other"
			}
		}

		if = { # antireligious characters and IG's will never join
			limit = {
				OR = {
					has_ideology = ideology:ideology_anti_clerical
					leader ?= { has_ideology = ideology:ideology_nihilist }
				}
			}
			add = {
				value = -15
				desc = "from_antireligious"
			}
		}

		if = { # Corporatism
			limit = {
				OR = {
					has_ideology = ideology:ideology_corporatist
					leader ?= { has_ideology = ideology:ideology_corporatist_leader }
				}
			}
			add = {
				value = 10
				desc = "from_corporatism"
			}
		}

		if = { # patriarchs more likely to join conservatives with women's suffrage
			limit = {
				has_ideology = ideology:ideology_patriarchal
				owner = { has_law_or_variant = law_type:law_womens_suffrage }
			}
			add = {
				value = 3
				desc = "from_patriarchy"
			}
		}

		if = {
			limit = {
				is_interest_group_type = ig_devout
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}

		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				is_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 3 }
			multiply = {
				value = 0.05
				desc = "from_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number > 3 }
			multiply = {
				value = 0.0
				desc = "from_more_than_three_prior_members"
			}
		}
		if = { # 1.15.2: the dominant party of a single-party state keeps its members
			limit = {
				ig_is_member_of_dominant_party = { PARTY = religious }
			}
			add = {
				value = 1000
				desc = "amendment_religious_party_dominance"
			}
		}
	}
}
