# Fallback party for Rural Folk
# Occasionally used for an anti-serfdom alliance
# More popular in Nordic countries

agrarian_party = {
	color = { 64 173 59 }
	name = {
		first_valid = {
			# 1946-era names (cwp); see localization/english/cwp/cwp_party_names_1946_l_english.yml
			triggered_desc = {
				desc = party_cwp_ast_agr
				trigger = { exists = c:AST THIS = c:AST }
			}
			triggered_desc = {
				desc = party_cwp_bul_agr
				trigger = { exists = c:BUL THIS = c:BUL }
			}
			triggered_desc = {
				desc = party_cwp_chl_agr
				trigger = { exists = c:CHL THIS = c:CHL }
			}
			triggered_desc = {
				desc = party_cwp_ddr_agr
				trigger = { exists = c:DDR THIS = c:DDR }
			}
			triggered_desc = {
				desc = party_cwp_fin_agr
				trigger = { exists = c:FIN THIS = c:FIN }
			}
			triggered_desc = {
				desc = party_cwp_hun_agr
				trigger = { exists = c:HUN THIS = c:HUN }
			}
			triggered_desc = {
				desc = party_cwp_icl_agr
				trigger = { exists = c:ICL THIS = c:ICL }
			}
			triggered_desc = {
				desc = party_cwp_ire_agr
				trigger = { exists = c:IRE THIS = c:IRE }
			}
			triggered_desc = {
				desc = party_cwp_ita_agr
				trigger = { exists = c:ITA THIS = c:ITA }
			}
			triggered_desc = {
				desc = party_cwp_jap_agr
				trigger = { exists = c:JAP THIS = c:JAP }
			}
			triggered_desc = {
				desc = party_cwp_nor_agr
				trigger = { exists = c:NOR THIS = c:NOR }
			}
			triggered_desc = {
				desc = party_cwp_pol_agr
				trigger = { exists = c:POL THIS = c:POL }
			}
			triggered_desc = {
				desc = party_cwp_rom_agr
				trigger = { exists = c:ROM THIS = c:ROM }
			}
			triggered_desc = {
				desc = party_cwp_swe_agr
				trigger = { exists = c:SWE THIS = c:SWE }
			}
			triggered_desc = {
				desc = party_cwp_swi_agr
				trigger = { exists = c:SWI THIS = c:SWI }
			}
			triggered_desc = {
				desc = party_agrarian
				trigger = {
					always = yes
				}
			}
		}
	}
	
	icon = {
		default = "/gfx/interface/icons/political_parties/agrarians.dds"
	}

	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		NOT = { c:USA ?= this } # USA: two-party system (Democrats / Republicans)
	}

	# THIS IS RUN ONLY ONCE AT GAME START
	available_for_interest_group = {
		trigger_if = {
			limit = {
				ig_is_member_of_dominant_party = { PARTY = agrarian }
			}
			always = yes
		}
		trigger_else = {
			owner = {
				country_has_voting_franchise = yes
				ig:ig_rural_folk ?= { is_marginal = no }
			}
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}

		if = { # a party for powerful farmers
			limit = {
				has_ideology = ideology:ideology_agrarian
				is_powerful = yes
			}
			add = {
				value = 10
				desc = "from_peasants"
			}
		}

		if = { # weak anti-serfdom types might fall in line behind strong farmers
			limit = {
				is_powerful = no
				owner = { has_law_or_variant = law_type:law_serfdom }
				law_stance = {
					law = law_type:law_serfdom
					value < neutral
				}
			}
			add = {
				value = 10
				desc = "from_powerful_farmers"
			}
		}

		if = { # pacifist anti-colonial alliance
			limit = { leader ?= { has_ideology = ideology:ideology_pacifist } }
			add = {
				value = 6
				desc = "from_pacifism"
			}
		}

		if = { # anti-immigration alliance
			limit = {
				OR = {
					has_ideology = ideology:ideology_isolationist
					has_ideology = ideology:ideology_populist
					has_ideology = ideology:ideology_reactionary
				}
				owner = { has_law_or_variant = law_type:law_no_migration_controls }
			}
			add = {
				value = 6
				desc = "from_migration_law"
			}
		}

		if = { # limited suffrage Rural Folk - prevents forming of impotent parties
			limit = {
				is_interest_group_type = ig_rural_folk
				owner = { NOT = { has_law_or_variant = law_type:law_universal_suffrage } }
			}
			add = {
				desc = "from_early_rural_folk"
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_landed_voting
								has_law_or_variant = law_type:law_wealth_voting
							}
						}
					}
					add = -10
				}
				if = {
					limit = { owner = { has_law_or_variant = law_type:law_census_voting } }
					add = -5
				}
			}
		}

		if = { # against arbitrary or militarized policing
			limit = {
				OR = {
					has_ideology = ideology:ideology_liberal
					has_ideology = ideology:ideology_particularist
				}
				owner = {
					OR = {
						has_law_or_variant = law_type:law_local_police
						has_law_or_variant = law_type:law_militarized_police
					}
				}
			}
			add = {
				value = 6
				desc = "from_bad_police"
			}
		}

		if = { # against land-based taxation
			limit = {
				OR = {
					has_ideology = ideology:ideology_meritocratic
					has_ideology = ideology:ideology_agrarian
					leader ?= { cwp_social_democrat_party_character = yes }
				}
				owner = { has_law_or_variant = law_type:law_land_based_taxation }
			}
			add = {
				value = 6
				desc = "from_land_tax"
			}
		}

		if = { # against free trade - peasant/socialist alliance
			limit = {
				OR = {
					has_ideology = ideology:ideology_proletarian
					has_ideology = ideology:ideology_agrarian
					has_ideology = ideology:ideology_paternalistic
				}
				owner = { has_law_or_variant = law_type:law_free_trade }
				NOT = { leader ?= { cwp_market_liberal_party_character = yes } }
			}
			add = {
				value = 6
				desc = "from_anti_free_trade"
			}
		}

		if = { # against laissez-faire - peasant/socialist alliance
			limit = {
				OR = {
					has_ideology = ideology:ideology_proletarian
					has_ideology = ideology:ideology_agrarian
					leader ?= { has_ideology = ideology:ideology_protectionist }
				}
				owner = { has_law_or_variant = law_type:law_laissez_faire }
				NOT = { leader ?= { cwp_market_liberal_party_character = yes } }
			}
			add = {
				value = 6
				desc = "from_anti_free_trade"
			}
		}

		if = { # opposition to peasant levies
			limit = {
				OR = {
					has_ideology = ideology:ideology_jingoist
					has_ideology = ideology:ideology_particularist
					leader ?= {
						OR = {
							has_ideology = ideology:ideology_fascist
							has_ideology = ideology:ideology_anarchist
							has_ideology = ideology:ideology_pacifist
							has_ideology = ideology:ideology_jingoist_leader
						}
					}
				}
				owner = { has_law_or_variant = law_type:law_peasant_levies }
			}
			add = {
				value = 6
				desc = "from_peasant_levies"
			}
		}

		if = { # anti-serfdom
			limit = {
				owner = { has_law_or_variant = law_type:law_serfdom }
				OR = {
					has_ideology = ideology:ideology_egalitarian
					has_ideology = ideology:ideology_laissez_faire
					has_ideology = ideology:ideology_particularist
					leader ?= {
						has_ideology = ideology:ideology_abolitionist
						cwp_social_democrat_party_character = yes
						cwp_communist_party_character = yes
					}
				}
			}
			add = {
				value = 10
				desc = "from_serfdom"
			}
		}

		if = { # nobody will join if the rural folk don't join
			limit = {
				NOT = { is_interest_group_type = ig_rural_folk }
				OR = {
					NOT = { exists = owner.py:agrarian_party }
					AND = {
						exists = owner.py:agrarian_party
						owner = { ig:ig_rural_folk ?= { NOT = { is_member_of_party = owner.py:agrarian_party } } }
					}
				}
			}
			add = {
				value = -20
				desc = "from_no_rural_folk_in_party" # this should probably never actually display in game because the party won't exist
			}
		}

		if = {
			limit = {
				is_interest_group_type = ig_rural_folk
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}

		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				is_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 3 }
			multiply = {
				value = 0.05
				desc = "from_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number > 3 }
			multiply = {
				value = 0.0
				desc = "from_more_than_three_prior_members"
			}
		}
		if = { # 1.15.2: the dominant party of a single-party state keeps its members
			limit = {
				ig_is_member_of_dominant_party = { PARTY = agrarian }
			}
			add = {
				value = 1000
				desc = "amendment_agrarian_party_dominance"
			}
		}
	}
}
