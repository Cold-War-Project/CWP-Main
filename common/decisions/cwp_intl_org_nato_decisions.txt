# Cold War Project: NATO and Warsaw Pact decisions (international organization template)

# Call a conference on a North Atlantic security pact. Possible from 1948, when the Brussels
# Treaty was signed, and only for a country that feels the eastern camp as a threat.
cwp_nato_call_conference_decision = {
	is_shown = {
		NOT = { has_global_variable = cwp_nato_founded }
		cwp_nato_eligible = yes
		country_rank >= rank_value:major_power
	}
	possible = {
		game_date >= 1948.1.1
		cwp_nato_can_call_conference = yes
		custom_tooltip = {
			text = cwp_nato_threat_tt
			any_country = {
				is_power_bloc_leader = yes
				cwp_io_in_eastern_camp = yes
				relations:root < relations_threshold:cordial
			}
		}
	}
	when_taken = {
		custom_tooltip = cwp_nato_call_conference_tt
		hidden_effect = {
			set_global_variable = { name = cwp_nato_conf_caller value = root }
			every_country = {
				limit = { cwp_nato_eligible = yes }
				set_variable = cwp_io_member_nato_conf
			}
			cwp_io_open_proposal = { PROP = nato_conf }
			cwp_io_cast_vote = { PROP = nato_conf VOTE = yes }
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = 10
		}
		if = {
			limit = { is_power_bloc_leader = yes }
			add = 10
		}
	}
}

cwp_nato_apply_decision = {
	is_shown = {
		has_global_variable = cwp_nato_founded
		NOT = { cwp_io_is_member = { ORG = nato } }
		cwp_nato_eligible = yes
	}
	possible = {
		custom_tooltip = {
			text = cwp_io_not_applicant_tt
			NOT = { has_variable = cwp_io_applicant_nato }
		}
		custom_tooltip = {
			text = cwp_io_not_rejected_tt
			NOT = { has_variable = cwp_io_rejected_nato }
		}
	}
	when_taken = {
		custom_tooltip = cwp_nato_apply_tt
		hidden_effect = {
			cwp_io_apply = { ORG = nato }
		}
	}
	ai_chance = {
		value = 2
		if = {
			limit = {
				power_bloc ?= {
					power_bloc_leader ?= { has_variable = cwp_io_member_nato }
				}
			}
			add = 30
		}
		if = {
			limit = {
				any_country = {
					is_power_bloc_leader = yes
					cwp_io_in_eastern_camp = yes
					relations:root <= relations_threshold:poor
				}
			}
			add = 20
		}
		if = {
			limit = { is_in_power_bloc = no }
			multiply = 0.2
		}
	}
}

# The eastern answer: only once a North Atlantic alliance exists and has taken in a new member
# (historically the Federal Republic of Germany, 1955).
cwp_wp_call_conference_decision = {
	is_shown = {
		NOT = { has_global_variable = cwp_wp_founded }
		cwp_wp_eligible = yes
		is_power_bloc_leader = yes
	}
	possible = {
		custom_tooltip = {
			text = cwp_wp_threat_tt
			cwp_wp_can_call_conference = yes
		}
	}
	when_taken = {
		custom_tooltip = cwp_wp_call_conference_tt
		hidden_effect = {
			set_global_variable = { name = cwp_wp_conf_caller value = root }
			every_country = {
				limit = { cwp_wp_eligible = yes }
				set_variable = cwp_io_member_wp_conf
			}
			cwp_io_open_proposal = { PROP = wp_conf }
			cwp_io_cast_vote = { PROP = wp_conf VOTE = yes }
		}
	}
	ai_chance = {
		value = 20
	}
}
