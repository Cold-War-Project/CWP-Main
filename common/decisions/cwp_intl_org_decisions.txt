# Cold War Project: international organization decisions

# Apply for United Nations membership. Admission goes to a vote in je_cwp_un_admission,
# where a permanent member of the Security Council can veto it.
cwp_un_apply_for_membership_decision = {
	is_shown = {
		NOT = { cwp_io_is_member = { ORG = un } }
		is_subject = no
	}
	possible = {
		custom_tooltip = {
			text = cwp_un_not_applicant_tt
			NOT = { has_variable = cwp_io_applicant_un }
		}
		custom_tooltip = {
			text = cwp_un_not_rejected_tt
			NOT = { has_variable = cwp_io_rejected_un }
		}
	}
	when_taken = {
		custom_tooltip = cwp_un_apply_tt
		hidden_effect = {
			cwp_io_apply = { ORG = un }
		}
	}
	ai_chance = {
		value = 50
	}
}
