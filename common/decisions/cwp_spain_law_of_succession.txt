# Cold War Project: Spain's Law of Succession (referendum of 6 July 1947).
# Franco's regime declares Spain a kingdom with the Caudillo as head of state for life. Available, not forced:
# the AI takes it when the monarchist interest groups are strong enough to need placating.

cwp_spa_law_of_succession_decision = {
	is_shown = {
		c:SPA ?= this
		NOT = { country_has_monarchy_law = yes }
		NOT = { has_variable = cwp_spa_succession_var }
		has_law = law_type:law_autocracy
	}

	possible = {
		custom_tooltip = {
			text = cwp_spa_monarchists_matter_tt
			any_interest_group = {
				is_marginal = no
				law_stance = {
					law = law_type:law_monarchy
					value >= approve
				}
			}
		}
		legitimacy >= 25
	}

	when_taken = {
		set_variable = cwp_spa_succession_var
		activate_law = law_type:law_monarchy
		add_modifier = {
			name = cwp_spa_kingdom_without_a_king
			years = 10
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_monarchy
					value >= approve
				}
			}
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 5
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { cwp_iq_monarchist_clout_in_government > 0.2 }
			add = 10
		}
	}
}
