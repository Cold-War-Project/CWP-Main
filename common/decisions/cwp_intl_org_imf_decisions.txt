# Cold War Project: IMF decisions (international organization template)

cwp_imf_apply_decision = {
	is_shown = {
		NOT = { cwp_io_is_member = { ORG = imf } }
		NOT = { has_law_or_variant = law_type:law_council_republic }
	}
	possible = {
		custom_tooltip = {
			text = cwp_io_not_applicant_tt
			NOT = { has_variable = cwp_io_applicant_imf }
		}
		custom_tooltip = {
			text = cwp_io_not_rejected_tt
			NOT = { has_variable = cwp_io_rejected_imf }
		}
	}
	when_taken = {
		custom_tooltip = cwp_imf_apply_tt
		hidden_effect = {
			cwp_io_apply = { ORG = imf }
		}
	}
	ai_chance = {
		value = 20
		if = {
			limit = { cwp_io_in_eastern_camp = yes }
			multiply = 0.1
		}
	}
}
