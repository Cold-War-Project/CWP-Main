return_manchuria_prc = {
	is_shown = {
		c:RUS ?= this
		exists = c:PRC
		exists = c:ROC
		NOT = { has_global_variable = chinese_civil_war_manchuria_resolved }
	}
	possible = { exists = c:PRC }
	when_taken = { trigger_event = { id = chinese_civil_war.2 popup = yes } }
	ai_chance = { value = 10 }
}

roc_great_retreat = {
	is_shown = {
		c:ROC ?= this
		exists = c:PRC
		NOT = { has_global_variable = chinese_civil_war_roc_retreat }
	}
	possible = {
		has_journal_entry = je_chinese_civil_war
		OR = {
			global_var:prc_chinese_civil_war_score >= 12
			NOT = { owns_entire_state_region = STATE_NANJING }
			NOT = { owns_entire_state_region = STATE_GUANGDONG }
		}
	}
	when_taken = { trigger_event = { id = chinese_civil_war.9 popup = yes } }
	ai_chance = { value = 0 }
}
