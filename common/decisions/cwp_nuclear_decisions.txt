# Cold War Project: nuclear weapons.

cwp_nuc_start_programme_decision = {
	is_shown = {
		cwp_nuc_can_start_programme = yes
	}

	possible = {
		custom_tooltip = {
			text = cwp_nuc_bomb_exists_tt
			any_country = { cwp_is_nuclear_power = yes }
		}
	}

	when_taken = {
		set_variable = cwp_nuc_programme_var
		add_journal_entry = { type = je_cwp_nuclear_programme }
	}

	ai_chance = {
		value = 0
		# Great powers facing a nuclear rival want their own bomb; others rarely.
		if = {
			limit = {
				country_rank >= rank_value:great_power
				any_country = {
					cwp_is_nuclear_power = yes
					relations:root <= relations_threshold:cold
				}
			}
			add = 50
		}
		else_if = {
			limit = { country_rank >= rank_value:great_power }
			add = 5
		}
	}
}
