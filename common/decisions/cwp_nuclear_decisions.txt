# Cold War Project: nuclear weapons.

cwp_nuc_start_programme_decision = {
	is_shown = {
		cwp_nuc_can_start_programme = yes
	}

	possible = {
		custom_tooltip = {
			text = cwp_nuc_bomb_exists_tt
			any_country = { cwp_is_nuclear_power = yes }
		}
	}

	when_taken = {
		set_variable = cwp_nuc_programme_var
		add_journal_entry = { type = je_cwp_nuclear_programme }
	}

	ai_chance = {
		value = 0
		# Bomb programmes follow bloc leadership and threat, not rank (the rank checks proved unreliable):
		# - a bloc leader facing a nuclear power outside its bloc that is not on good terms builds at once (USSR 1945);
		# - a bloc leader facing only friendly nuclear powers sometimes wants an independent deterrent (Britain, France);
		# - a great power outside every bloc facing a hostile nuclear power may build (China after the split, India);
		# - bloc members and everyone else leave it to their bloc leader.
		if = {
			limit = {
				is_power_bloc_leader = yes
				any_country = {
					NOT = { this = root }
					cwp_is_nuclear_power = yes
					NOT = { is_in_same_power_bloc = root }
					relations:root < relations_threshold:cordial
				}
			}
			add = 100
		}
		else_if = {
			limit = {
				is_power_bloc_leader = yes
				cwp_nuc_faces_outside_nuclear_power = yes
			}
			add = 3
		}
		else_if = {
			limit = {
				is_in_power_bloc = no
				country_rank >= rank_value:great_power
				any_country = {
					NOT = { this = root }
					cwp_is_nuclear_power = yes
					relations:root <= relations_threshold:poor
				}
			}
			add = 20
		}
	}
}
