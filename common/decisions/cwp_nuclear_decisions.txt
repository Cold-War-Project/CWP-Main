# Cold War Project: nuclear weapons.

cwp_nuc_start_programme_decision = {
	is_shown = {
		cwp_nuc_can_start_programme = yes
	}

	possible = {
		custom_tooltip = {
			text = cwp_nuc_bomb_exists_tt
			any_country = { cwp_is_nuclear_power = yes }
		}
	}

	when_taken = {
		set_variable = cwp_nuc_programme_var
		add_journal_entry = { type = je_cwp_nuclear_programme }
	}

	ai_chance = {
		value = 0
		# A superpower facing a nuclear power outside its bloc builds at once (the USSR in 1945). Other great
		# powers outside anyone's umbrella sometimes decide to (Britain 1947, France 1954); sheltered allies rarely do.
		if = {
			limit = {
				country_rank >= rank_value:superpower
				cwp_nuc_faces_outside_nuclear_power = yes
			}
			add = 100
		}
		else_if = {
			limit = {
				country_rank >= rank_value:great_power
				cwp_nuc_faces_outside_nuclear_power = yes
				cwp_nuc_under_nuclear_umbrella = no
			}
			add = 5
		}
		if = {
			limit = { cwp_nuc_under_nuclear_umbrella = yes }
			multiply = 0
		}
	}
}
