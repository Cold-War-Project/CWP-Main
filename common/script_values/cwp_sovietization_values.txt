# Cold War Project: sovietization of the Soviet-occupied countries. Country scope = the occupied country.

# Clout of the interest groups that reject a council republic: the peasant parties, the Church, the middle class.
cwp_sov_opposition_clout = {
	value = 0
	every_interest_group = {
		limit = {
			is_marginal = no
			cwp_sov_ig_is_opposition = yes
		}
		add = this.ig_clout
	}
}

# The opposition's drag weakens as the communists take the levers of the state: at full control it is gone.
cwp_sov_opposition_drag = {
	value = cwp_sov_opposition_clout
	multiply = -3
	if = {
		limit = { has_variable = cwp_sov_control_current_var }
		multiply = {
			value = 100
			subtract = var:cwp_sov_control_current_var
			divide = 100
		}
	}
}

cwp_sov_radical_pressure = {
	value = radical_fraction
	multiply = 5
}
