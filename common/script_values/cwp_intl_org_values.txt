# Cold War Project: international organization template (script values)

# Vote weight of one member, per organization. Country scope.
# The UN General Assembly is one state, one vote.
cwp_io_vote_weight_un = {
	value = 1
}

# Economic weight, Arkady's GDP plus prestige rule, for weighted organizations
# (the IMF's quota votes come in a later slice). Country scope.
cwp_io_economic_weight = {
	value = gdp
	divide = 1000000
	add = prestige
	divide = 100
	min = 1
}

# 1 while a resolution has passed, else 0. Drives scaling_modifiers_while_active on the UN JE,
# so a resolution's effect follows membership: new members gain it, leavers lose it.
cwp_un_human_rights_in_force = {
	value = 0
	if = {
		limit = { global_var:cwp_io_un_human_rights_result ?= flag:passed }
		value = 1
	}
}

cwp_un_law_of_the_sea_in_force = {
	value = 0
	if = {
		limit = { global_var:cwp_io_un_law_of_the_sea_result ?= flag:passed }
		value = 1
	}
}
