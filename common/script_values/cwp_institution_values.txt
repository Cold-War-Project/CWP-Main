# Institutions as state capacity (docs/agents/areas/institutions.md).

# Country scope, added to the AI's score for Schools, Health System and Economic System in
# ai_strategy_default. The institutions that grow the economy come once Administration (tax capacity,
# the means of funding them) is in place: held back below level 2 while the AI is pushing Administration
# (its taxes outrun capacity and it can afford more state; see institution_administration), favoured from
# level 3 while the country can afford it and has bureaucracy to spare. Political strategies add 100-200
# on top, so a level-1 state still keeps what it has.
cwp_ai_institution_after_administration = {
	value = 0
	if = {
		limit = {
			institution_investment_level = { institution = institution_administration value < 2 }
			cwp_ai_state_affordable = yes
			cwp_ai_tax_capacity_binds = yes
		}
		add = -120
	}
	else_if = {
		limit = {
			institution_investment_level = { institution = institution_administration value >= 3 }
			cwp_ai_state_affordable = yes
			approaching_bureaucracy_shortage = no
			bureaucracy >= 0
		}
		add = 60
	}
}
