# Cold War Project: the French constituent assemblies of 1945-46. Country scope = FRA.

cwp_fra_left_clout = {
	value = 0
	every_interest_group = {
		limit = {
			is_marginal = no
			OR = {
				is_interest_group_type = ig_trade_unions
				is_interest_group_type = ig_intelligentsia
			}
		}
		add = this.ig_clout
	}
}

cwp_fra_right_clout = {
	value = 0
	every_interest_group = {
		limit = {
			is_marginal = no
			OR = {
				is_interest_group_type = ig_devout
				is_interest_group_type = ig_industrialists
				is_interest_group_type = ig_petty_bourgeoisie
				is_interest_group_type = ig_landowners
				is_interest_group_type = ig_armed_forces
			}
		}
		add = this.ig_clout
	}
}

cwp_fra_draft_pivot = {
	add = {
		desc = "cwp_fra_pivot_base_tt"
		value = 48
	}
	add = {
		desc = "cwp_fra_pivot_left_tt"
		value = cwp_fra_left_clout
		multiply = 30
	}
	subtract = {
		desc = "cwp_fra_pivot_right_tt"
		value = cwp_fra_right_clout
		multiply = 25
	}
	if = {
		limit = { has_variable = cwp_fra_compromise_var }
		add = {
			desc = "cwp_fra_pivot_compromise_tt"
			value = var:cwp_fra_compromise_var
		}
	}
	if = {
		limit = { has_variable = cwp_fra_campaign_var }
		add = {
			desc = "cwp_fra_pivot_campaign_tt"
			value = var:cwp_fra_campaign_var
		}
	}
	min = 2
	max = 98
}

cwp_fra_counted_result = {
	value = var:cwp_fra_draft_current_var
	if = {
		limit = { has_variable = cwp_fra_swing_var }
		add = var:cwp_fra_swing_var
	}
	min = 1
	max = 99
}
