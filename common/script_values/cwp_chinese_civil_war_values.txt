# Cold War Project: the Chinese Civil War (je_cwp_chinese_civil_war)
# Root = country, scope:war = the war. Added per weekly beat to war support through war_support_from_journal_entries
# (the REPLACE in zz_vanilla_script_values.txt, owned by the decolonization thread, calls this value).
# Scale: occupation costs up to -5 a beat (NWar WAR_SUPPORT_CHANGE_ENEMY_OCCUPATION_FACTOR); vanilla's Carlist war adds +10.
cwp_china_war_support = {
	value = 0
	if = {
		limit = {
			has_global_variable = cwp_ccw_active
			cwp_ccw_is_belligerent = yes
			exists = c:PRC
			exists = c:ROC
			scope:war = {
				is_war_participant = c:PRC
				is_war_participant = c:ROC
			}
		}
		# People's war: neither side is losing yet. Base areas and walled cities change hands without deciding anything,
		# so the occupation penalty is cancelled as well (vanilla Carlist caps occupation at 0 and adds +10).
		# Observer 2026-11-17 with 1946 armies (#833): +5 alone let the ROC annex the PRC by November 1947.
		if = {
			limit = { has_global_variable = cwp_ccw_peoples_war_phase }
			add = {
				value = enemy_occupation
				multiply = 5
				add = 5
				desc = "cwp_ccw_war_support_peoples_war"
			}
			# Battles, casualties and devastation still wore the PRC down to capitulation in one 2026-11-17 run:
			# below 50 the people's war adds as much as vanilla's Carlist war (+10)
			if = {
				limit = { scope:war = { has_war_support = { target = root value < 50 } } }
				add = {
					value = 10
					desc = "cwp_ccw_war_support_peoples_war"
				}
			}
			# Gap fill from staging run 1121110313: the PRC still fell from 99 to 49 by 1947.8 and capitulated before
			# 1948.5 inside the people's war. Below 30 the Communists fall back on the base areas instead of giving up.
			if = {
				limit = {
					c:PRC = this
					scope:war = { has_war_support = { target = root value < 30 } }
				}
				add = {
					value = 10
					desc = "cwp_ccw_war_support_peoples_war"
				}
			}
			# Nationalist war-weariness: while hyperinflation runs, the Nationalist war effort loses support every beat.
			if = {
				limit = {
					c:ROC = this
					has_modifier = cwp_ccw_hyperinflation
				}
				add = {
					value = -3
					desc = "cwp_ccw_war_support_hyperinflation"
				}
			}
		}
		# The strait: the Nationalists on Formosa cannot be forced to capitulate without a landing
		if = {
			limit = {
				c:ROC = this
				cwp_ccw_roc_holds_formosa = yes
				OR = {
					NOT = { cwp_ccw_roc_holds_mainland = yes }
					cwp_ccw_roc_retreated = yes
				}
			}
			add = {
				value = enemy_occupation
				multiply = 5
				add = 2
				desc = "cwp_ccw_war_support_strait"
			}
			# Run 1122081642-batch-199e: after the retreat the ROC still fell to war support 5 (bankruptcy, devastation) and was
			# annexed with Formosa in 1949. Behind the strait it holds like the people's war does.
			if = {
				limit = { scope:war = { has_war_support = { target = root value < 50 } } }
				add = {
					value = 10
					desc = "cwp_ccw_war_support_strait"
				}
			}
		}
		# Decisive phase: the side the balance favours by 25 or more keeps its will to fight (victory is in sight), so its
		# own casualties do not make it sue for peace in the middle of its offensive (observer 2026-11-18)
		if = {
			limit = {
				NOT = { has_global_variable = cwp_ccw_peoples_war_phase }
				OR = {
					AND = {
						c:PRC = this
						global_var:cwp_ccw_balance >= 25
					}
					AND = {
						c:ROC = this
						global_var:cwp_ccw_balance <= -25
					}
				}
			}
			add = {
				value = 3
				desc = "cwp_ccw_war_support_winning"
			}
			# Batch 1122110222-batch-71b7: the PRC, leading at +31.5 with its base areas unoccupied, still fell to 12.5
			# war support (casualties, bankruptcy, the duration drain) and folded in 1949. The side that is winning holds
			# the way the people's war does.
			if = {
				limit = { scope:war = { has_war_support = { target = root value < 50 } } }
				add = {
					value = 10
					desc = "cwp_ccw_war_support_winning"
				}
			}
		}
		# Decisive phase: the balance of the war wears the losing side down (vanilla Russian Civil War exhaustion pattern)
		if = {
			limit = {
				NOT = { has_global_variable = cwp_ccw_peoples_war_phase }
				OR = {
					AND = {
						c:ROC = this
						global_var:cwp_ccw_balance >= 25
						cwp_ccw_roc_holds_mainland = yes
					}
					AND = {
						c:PRC = this
						global_var:cwp_ccw_balance <= -25
					}
				}
			}
			add = {
				value = enemy_occupation
				if = {
					limit = {
						OR = {
							global_var:cwp_ccw_balance >= 40
							global_var:cwp_ccw_balance <= -40
						}
					}
					multiply = -2.0
				}
				else = {
					multiply = -1.0
				}
				desc = "cwp_ccw_war_support_balance"
			}
		}
	}
}

# Called from REPLACE:battle_victory_score (zz_vanilla_script_values.txt). Same scope as that value (the battle's winning
# side, with a province link). In the people's war a won battle takes a quarter of the usual ground in either China,
# a tenth while the American-brokered ceasefire holds: the Carlist skirmish phase without freezing the map entirely.
cwp_china_battle_victory_factor = {
	value = 1
	# The Communist base areas hold as long as the Nationalists have not gained the upper hand (bar above -25), in either
	# phase: batch 1122100449-batch-a3b9 had the PRC launch its offensive at +26.5 and then lose every state to an army
	# three times its size, folding at once (AI_CAPITULATE_COMPLETELY_DEFEATED).
	if = {
		limit = {
			NOT = { has_global_variable = cwp_ccw_peoples_war_phase }
			has_global_variable = cwp_ccw_active
			global_var:cwp_ccw_balance > -25
			province ?= { owner ?= { c:PRC ?= this } }
		}
		multiply = {
			value = 0
			desc = "cwp_ccw_occupation_base_areas"
		}
	}
	if = {
		limit = {
			has_global_variable = cwp_ccw_peoples_war_phase
			province ?= {
				owner ?= { cwp_ccw_is_belligerent = yes }
			}
		}
		# The Communist base areas cannot be occupied in the people's war (the Carlist skirmish phase's 0). The AI
		# capitulates at once when 100% occupied (AI_CAPITULATE_COMPLETELY_DEFEATED), and the PRC's 9 small states were
		# fully overrun at 52 war support in run 1122091440-batch-3ea9 even at a quarter of the usual ground.
		if = {
			limit = {
				province ?= { owner ?= { c:PRC ?= this } }
			}
			multiply = {
				value = 0
				desc = "cwp_ccw_occupation_base_areas"
			}
		}
		else_if = {
			limit = { has_global_variable = cwp_ccw_ceasefire }
			multiply = {
				value = 0.1
				desc = "cwp_ccw_occupation_ceasefire"
			}
		}
		else = {
			multiply = {
				value = 0.25
				desc = "cwp_ccw_occupation_peoples_war"
			}
		}
	}
}
