base_impose_law_weight= {
	value = 0.01

	# Impose law is a fairly hostile act, AIs should use it mostly when angry at a subject
	if = {
		limit = {
			has_attitude = {
				who = scope:target_country
				attitude = domineering
			}
		}
		multiply = 0.5
	}

	# Non ideological unions make less use of this
	if = {
		limit = { NOT = { power_bloc ?= { has_identity = identity:identity_ideological_union } } }
		multiply = 0.5
	}

	# Less likely to do this to subjects unless LD is low
	if = {
		limit = {
			scope:target_country = {
				is_subject = yes
				liberty_desire > 25
			}
		}
		multiply = 0.5
	}

	# Shouldn't care much about meddling in really tiny countries
	if = {
		limit = { scope:target_country = { gdp < 100000 } }
		multiply = 0.25
	}
	else_if = {
		limit = { scope:target_country = { gdp < 250000 } }
		multiply = 0.5
	}
	else_if = {
		limit = { scope:target_country = { gdp < 500000 } }
		multiply = 0.75
	}

	# Impact of AI strategies
	if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_anti_imperialism
				has_strategy = ai_strategy_maintain_power_balance
			}
		}
		multiply = 2
	}
	else_if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_armed_isolationism
			}
		}
		multiply = 0.5
	}
}


wanted_army_size_script_value = {
	# How many levels of barracks should the AI have
	# Using this value in other strategies will function additively
	save_temporary_scope_as = army_planning_country
	value = 0

	# GDP is the economic basis of the standing army. The first two terms are
	# capped so that very large economies continue to grow their armies, but at
	# a much slower rate.
	add = {
		value = gdp
		divide = 250000
		max = 20
	}

	add = {
		value = gdp
		divide = 1000000
		max = 100
	}

	add = {
		value = gdp
		divide = 8000000
	}

	# Maritime countries reserve up to a quarter of their military spending for
	# the navy. This preserves the old coastal adjustment without making the
	# army itself population-driven.
	if = {
		limit = {
			cached_ai_total_population > 0
			cached_ai_coastal_population > 0
		}
		multiply = {
			value = 1
			subtract = {
				value = cached_ai_coastal_population
				divide = cached_ai_total_population
				divide = 4
			}
		}
	}

	# More advanced standing armies are expensive and should be smaller relative to population
	if = {	# Combat Unit Type - Armored Brigade & Mechanized Brigade
		limit = { has_technology_researched = armored_warfare }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Air Mobile Brigade & Air Mobile Howitzers
		limit = { has_technology_researched = airmobile_warfare }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Tactical Brigade
		limit = { has_technology_researched = fourth_generation_warfare }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Rocket Artillery
		limit = { has_technology_researched = rocketry }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Tactical Ballistic Missile Launchers
		limit = { has_technology_researched = tactical_ballistic_weapons }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Air Superiority Fighters
		limit = { has_technology_researched = jet_engine }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Fighter-Bombers
		limit = { has_technology_researched = multi_role_jets }
		multiply = 0.95
	}
	if = {	# Combat Unit Type - Stealth Aircraft
		limit = { has_technology_researched = stealth_aircraft }
		multiply = 0.95
	}

	# Diplomatic strategy sets the military posture. Expansionist and
	# interventionist strategies support a larger standing army, while
	# defensive strategies receive a smaller readiness bonus.
	if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_territorial_expansion
				has_strategy = ai_strategy_restore_natural_borders
			}
		}
		multiply = 1.25
	}
	else_if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_economic_imperialism
				has_strategy = ai_strategy_colonial_expansion
				has_strategy = ai_strategy_global_liberalism
				has_strategy = ai_strategy_international_socialism
			}
		}
		multiply = 1.15
	}
	else_if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_armed_isolationism
				has_strategy = ai_strategy_maintain_power_balance
				has_strategy = ai_strategy_anti_imperialism
			}
		}
		multiply = 1.10
	}

	# Hegemons and autarkic superpowers devote additional economic capacity to
	# their armed forces.
	if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_maintain_hegemony
				has_strategy = ai_strategy_superpower_industrial_complex
			}
		}
		multiply = 1.10
	}

	# Subjects preparing for independence prioritize land forces, while a
	# compliant subject is less likely to maintain an oversized army.
	if = {
		limit = { has_strategy = ai_strategy_subject_break_free }
		multiply = 1.25
	}
	else_if = {
		limit = { has_strategy = ai_strategy_subject_strengthen_self }
		multiply = 1.15
	}
	else_if = {
		limit = { has_strategy = ai_strategy_subject_maintain_autonomy }
		multiply = 1.05
	}
	else_if = {
		limit = { has_strategy = ai_strategy_subject_overlord_compliance }
		multiply = 0.90
	}

	# Autocratic and military-led regimes sustain larger standing armies.
	# Political strategies are included so the AI begins adapting before its
	# preferred distribution-of-power law has finished enacting.
	if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_autocratic_agenda
				has_strategy = ai_strategy_juntaist_agenda
				has_strategy = ai_strategy_hardline_communist
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_single_party_state
			}
		}
		multiply = 1.20
	}
	else_if = {
		limit = {
			OR = {
				has_strategy = ai_strategy_nationalist_agenda
				has_strategy = ai_strategy_reactionary_agenda
				has_strategy = ai_strategy_conservative_communist
				has_strategy = ai_strategy_security_agenda
				has_law_or_variant = law_type:law_oligarchy
			}
		}
		multiply = 1.10
	}

	# Rivalries are one-sided diplomatic pacts, so both directions must be
	# checked. A peer-or-higher rank, or a rival with at least 75% of this
	# country's GDP, creates additional military pressure.
	if = {
		limit = {
			OR = {
				any_rival_country = {
					OR = {
						country_rank >= scope:army_planning_country.country_rank
						gdp >= {
							value = scope:army_planning_country.gdp
							multiply = 0.75
						}
					}
				}
				any_rivaling_country = {
					OR = {
						country_rank >= scope:army_planning_country.country_rank
						gdp >= {
							value = scope:army_planning_country.gdp
							multiply = 0.75
						}
					}
				}
			}
		}
		multiply = 1.15
	}

	if = {
		limit = { has_law_or_variant = law_type:law_national_militia }
		multiply = 0.33
	}

	min = 1

	# Population is a true final ceiling, applied after every economic,
	# strategic, political, and rivalry modifier. The curve is adapted from the
	# old population-based target: incorporated population supplies most of the
	# manpower, while unincorporated population contributes at a reduced rate.
	max = {
		value = cached_ai_incorporated_population
		divide = 100000
		max = 10
		add = {
			value = cached_ai_incorporated_population
			divide = 250000
		}
		add = {
			value = cached_ai_unincorporated_population
			divide = 1000000
		}
		min = 1
	}
}

ship_mod_ai_price_ratio_reject     = { value = 1.75 }  # Weighted buy/sell ratio at/above which a mod is rejected (multiplier = 0)
ship_mod_ai_price_ratio_bonus      = { value = 0.75 }  # At/below which the full bonus multiplier applies
ship_mod_ai_price_bonus_multiplier = { value = 1.5 }   # Multiplier at the bonus ratio (linear interpolation in between)


# Scales each mod's base ai_weight by a function of the goods-weighted buy/sell
# ratio for its construction goods, against the country's market.
# Curve:
#   weighted_ratio >= ship_mod_ai_price_ratio_reject   -> multiplier = 0
#   weighted_ratio <= ship_mod_ai_price_ratio_bonus    -> multiplier = ship_mod_ai_price_bonus_multiplier
#   between                                            -> linear interpolation
# Linear form: multiplier = BONUS_MULT * (REJECT - ratio) / (REJECT - BONUS)
# At ratio = 1 (neutral market) this is BONUS_MULT * 0.5 / 0.5 = 1 with default
# thresholds (1.25 / 0.75 / 2.0), so a neutral-market mod keeps its base weight.
ship_mod_market_multiplier = {
	# Evaluate the trigger once and save the ratio as a temporary; we refer to
	# it three times below (two limit branches + the else arithmetic).
	value = "ship_modification_market_demand_ratio(scope:ship_modification_type)"
	save_temporary_value_as = temp_ship_mod_demand_ratio

	# Base multiplier = 1; each branch overrides via multiply.
	value = 1
	if = {
		limit = { scope:temp_ship_mod_demand_ratio >= ship_mod_ai_price_ratio_reject }
		multiply = 0.1 # We still want to score better modules a bit higher instead of all being 0
	}
	else_if = {
		limit = { scope:temp_ship_mod_demand_ratio <= ship_mod_ai_price_ratio_bonus }
		multiply = ship_mod_ai_price_bonus_multiplier
	}
	else = {
		multiply = {
			value = ship_mod_ai_price_bonus_multiplier
			multiply = {
				value = ship_mod_ai_price_ratio_reject
				subtract = scope:temp_ship_mod_demand_ratio
			}
			divide = {
				value = ship_mod_ai_price_ratio_reject
				subtract = ship_mod_ai_price_ratio_bonus
			}
		}
	}
}
