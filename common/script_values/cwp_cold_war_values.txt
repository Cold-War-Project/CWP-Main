# Cold War Project: the global Cold War. See docs/agents/areas/cold-war.md.

# The communist camp's share of the two camps' GDP, 0-100.
cwp_cw_east_gdp_share = {
	value = global_var:cwp_cw_east_gdp
	multiply = 100
	divide = {
		value = global_var:cwp_cw_east_gdp
		add = global_var:cwp_cw_west_gdp
		add = 1
	}
}

# The communist camp's share of the two camps' countries, 0-100.
cwp_cw_east_count_share = {
	value = global_var:cwp_cw_east_count
	multiply = 100
	divide = {
		value = global_var:cwp_cw_east_count
		add = global_var:cwp_cw_west_count
		add = 1
	}
}

# -5 while only the Western camp leader has the bomb.
cwp_cw_nuclear_term = {
	value = 0
	if = {
		limit = {
			global_var:cwp_cw_west_leader ?= { cwp_is_nuclear_power = yes }
			NOT = { global_var:cwp_cw_east_leader ?= { cwp_is_nuclear_power = yes } }
		}
		add = -5
	}
}

# The correlation of forces: 0 the West dominant, 100 socialism dominant.
cwp_cw_correlation_value = {
	value = cwp_cw_east_gdp_share
	multiply = 0.6
	add = {
		value = cwp_cw_east_count_share
		multiply = 0.4
	}
	add = global_var:cwp_cw_momentum
	add = cwp_cw_nuclear_term
	min = 0
	max = 100
}
