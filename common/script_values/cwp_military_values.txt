# Cold War Project: military script values (docs/agents/areas/military.md).

# Multiplier of the military cost tiers on this country (cwp_military_cost_tier_update): +150% Global Power Projection,
# +50% Overseas Commitments, -40% Labour-Intensive Forces, otherwise 1.
cwp_military_cost_tier_factor = {
	value = 1
	if = {
		limit = { has_modifier = cwp_military_global_power_projection }
		add = 1.5
	}
	else_if = {
		limit = { has_modifier = cwp_military_overseas_commitments }
		add = 0.5
	}
	else_if = {
		limit = { has_modifier = cwp_military_labour_intensive_forces }
		add = -0.4
	}
}

# Estimated weekly cost of the standing forces, for the start-force floor (the engine exposes no military-spending
# value). About 1,000 a week per regular battalion (infantry upkeep goods at base prices plus wages; army_size counts
# regular battalions, not conscripts) and 800 per ship, both times the cost tier.
cwp_military_force_cost_estimate = {
	value = army_size
	multiply = 1000
	add = {
		value = num_ships
		multiply = 800
	}
	multiply = cwp_military_cost_tier_factor
}
