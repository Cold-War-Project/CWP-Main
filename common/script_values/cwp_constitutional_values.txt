# Cold War Project: constitutional transitions (institutional question, Japanese constitution).
# Country scope unless noted. The pivots are what opinion drifts towards; the bars chase them (pattern: vanilla two_spains_tension_pivot).

# Clout of the interest groups that oppose the monarchy law, weighted by how strongly.
# The labour movement and the intelligentsia count as republican unless their stance says otherwise:
# most CWP ideologies carry no stance on the monarchy, but in 1946 the left and the liberals were its main opponents.
cwp_iq_republican_clout = {
	value = 0
	every_interest_group = {
		limit = {
			is_marginal = no
			OR = {
				law_stance = {
					law = law_type:law_monarchy
					value <= disapprove
				}
				AND = {
					OR = {
						is_interest_group_type = ig_trade_unions
						is_interest_group_type = ig_intelligentsia
					}
					NOT = {
						law_stance = {
							law = law_type:law_monarchy
							value >= approve
						}
					}
				}
			}
		}
		add = {
			value = this.ig_clout
			if = {
				limit = {
					law_stance = {
						law = law_type:law_monarchy
						value = strongly_disapprove
					}
				}
				multiply = 1.5
			}
		}
	}
}

# Where the republican vote share is heading (0-100).
cwp_iq_republic_pivot = {
	add = {
		desc = "cwp_iq_pivot_base_tt"
		value = 50
	}
	add = {
		desc = "cwp_iq_pivot_republican_igs_tt"
		value = cwp_iq_republican_clout
		multiply = 60
	}
	subtract = {
		desc = "cwp_iq_pivot_monarchist_igs_tt"
		value = cwp_iq_monarchist_clout
		multiply = 60
	}
	if = {
		limit = { has_variable = cwp_iq_crown_record_var }
		add = {
			desc = "cwp_iq_pivot_crown_record_tt"
			value = var:cwp_iq_crown_record_var
		}
	}
	if = {
		limit = { has_variable = cwp_iq_campaign_var }
		add = {
			desc = "cwp_iq_pivot_campaign_tt"
			value = var:cwp_iq_campaign_var
		}
	}
	min = 2
	max = 98
}

# Monthly pressure on the government to settle the question.
cwp_iq_pressure_republicans = {
	value = cwp_iq_republican_clout_in_government
	multiply = 8
}

cwp_iq_pressure_monarchists = {
	value = cwp_iq_monarchist_clout_in_government
	multiply = 6
}

# The counted result: the vote share on polling day plus the swing rolled in the referendum event.
cwp_iq_counted_result = {
	value = var:cwp_iq_vote_current_var
	if = {
		limit = { has_variable = cwp_iq_swing_var }
		add = var:cwp_iq_swing_var
	}
	if = {
		limit = { has_variable = cwp_iq_rigged_var }
		add = 30
	}
	min = 1
	max = 99
}
