# Cold War Project: decolonization war support, added to war_support_from_journal_entries
# (common/script_values/zz_vanilla_script_values.txt). Root = the country whose war support changes,
# scope:war = the war; the result is part of the per-beat war support change and shows in its tooltip.
# Since the 2026-11-18 rework the JEs have no war bars: the buttons and the state of the war act here directly.
# War support runs 0-100 (red band at 25); native per-beat factors reach -5 (occupation), so these are sized to match.

cwp_decolonization_war_support = {
	value = 0

	# Indochina: France and the Viet Minh state
	if = {
		limit = {
			has_global_variable = cwp_indochina_active
			OR = {
				c:FRA ?= this
				c:DAI ?= this
			}
			exists = c:FRA
			exists = c:DAI
			scope:war = {
				is_war_participant = c:FRA
				is_war_participant = c:DAI
			}
		}
		# Protracted war favours the side that can wait: only the DRV gets it. France's own support erodes (the
		# "dirty war" below) unless it keeps reinforcing and the Americans keep paying. Batches 199e and 3ea9 had
		# the +3 on both sides, and French war support stayed at 100 through a four-year war.
		if = {
			limit = {
				has_global_variable = cwp_indochina_peoples_war_phase
				c:DAI ?= this
			}
			add = {
				value = 3
				desc = "cwp_war_support_peoples_war"
			}
			if = {
				limit = { scope:war = { has_war_support = { target = root value < 30 } } }
				add = {
					value = 5
					desc = "cwp_war_support_peoples_war_resilience"
				}
			}
		}
		if = {
			limit = {
				c:FRA ?= this
				has_variable = cwp_indochina_reinforcements
			}
			add = {
				value = 1
				desc = "cwp_war_support_indochina_reinforcements"
			}
		}
		if = {
			limit = {
				c:FRA ?= this
				c:USA ?= { has_variable = cwp_indochina_american_aid }
			}
			add = {
				value = 1
				desc = "cwp_war_support_indochina_american_aid"
			}
		}
		if = {
			limit = {
				c:FRA ?= this
				has_global_variable = cwp_indochina_months
				global_var:cwp_indochina_months >= 24
			}
			add = {
				value = -1
				desc = "cwp_war_support_indochina_dirty_war"
			}
		}
		if = {
			limit = {
				c:FRA ?= this
				has_global_variable = cwp_indochina_months
				global_var:cwp_indochina_months >= 48
			}
			add = {
				value = -1
				desc = "cwp_war_support_indochina_endless_war"
			}
		}
		if = {
			limit = {
				c:DAI ?= this
				c:PRC ?= { has_variable = cwp_indochina_chinese_aid }
			}
			add = {
				value = 1
				desc = "cwp_war_support_indochina_chinese_aid"
			}
		}
	}

	# Indonesia: the Netherlands and the Republic
	if = {
		limit = {
			has_global_variable = cwp_indonesia_active
			OR = {
				c:NET ?= this
				c:IDS ?= this
			}
			exists = c:NET
			exists = c:IDS
			scope:war = {
				is_war_participant = c:NET
				is_war_participant = c:IDS
			}
		}
		if = {
			limit = {
				c:IDS ?= this
				cwp_indonesia_republic_holds = yes
				scope:war = { has_war_support = { target = root value < 30 } }
			}
			add = {
				value = 5
				desc = "cwp_war_support_indonesia_resilience"
			}
		}
		if = {
			limit = {
				c:IDS ?= this
				has_variable = cwp_indonesia_guerrilla
			}
			add = {
				value = 1
				desc = "cwp_war_support_indonesia_guerrilla"
			}
		}
		if = {
			limit = {
				c:NET ?= this
				has_variable = cwp_indonesia_police_action
			}
			add = {
				value = 2
				desc = "cwp_war_support_indonesia_police_action"
			}
		}
		if = {
			limit = {
				c:NET ?= this
				c:USA ?= { has_variable = cwp_indonesia_marshall_pressure }
			}
			add = {
				value = -2
				desc = "cwp_war_support_indonesia_pressure"
			}
		}
		if = {
			limit = {
				c:NET ?= this
				scaled_debt > 0.5
			}
			add = {
				value = -1
				desc = "cwp_war_support_metropole_debt"
			}
		}
	}
}

# Called from REPLACE:battle_victory_score (zz_vanilla_script_values.txt), scope = the battle's winning side with a
# province link. While the Indochina war is a people's war, a won battle in DRV territory takes a quarter of the usual
# ground: the DRV holds only Tonkin, and without this the French occupied it and forced capitulation within months.
cwp_indochina_battle_victory_factor = {
	value = 1
	if = {
		limit = {
			has_global_variable = cwp_indochina_peoples_war_phase
			province ?= {
				owner ?= { c:DAI ?= this }
			}
		}
		multiply = {
			value = 0.25
			desc = "cwp_indochina_occupation_peoples_war"
		}
	}
}

# Country scope (DAI). Weekly income from cwp_indochina_liberated_zone_taxes: 1,000 per million inhabitants (about
# 9,900 a week for the 9.9 million of 1946), so it shrinks as the DRV loses ground and grows if it gains it.
cwp_indochina_liberated_zone_taxes_value = {
	value = total_population
	divide = 1000000
	multiply = 1000
	round = yes
	min = 500
}

# No scope (global). One-off change to France's war support (0-100) when a new round of the Indochina war opens:
# -2 per month of France-DRV war fought so far (cwp_indochina_months), at most -60, so a late round opens near 40.
cwp_indochina_french_strain_value = {
	value = 0
	if = {
		limit = { has_global_variable = cwp_indochina_months }
		add = global_var:cwp_indochina_months
	}
	multiply = -2
	min = -60
}
