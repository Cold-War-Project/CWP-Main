# Cold War Project: Newfoundland's future (National Convention and referendum). Country scope = NEW.
# No bars: the poll is computed directly and shown in the JE text. Three options on the first ballot
# (Commission, Responsible Government, Confederation); if none wins half, the weakest drops out and a runoff follows.

# How far Canadian living standards outshine Newfoundland's: family allowances, pensions, the federal dole.
cwp_nf_canadian_prosperity = {
	value = 0
	if = {
		limit = {
			exists = c:CAN
			"c:CAN.average_sol" > "root.average_sol"
		}
		add = "c:CAN.average_sol"
		subtract = "root.average_sol"
		multiply = 2
	}
	max = 12
}

cwp_nf_confederation_pivot = {
	add = {
		desc = "cwp_nf_pivot_base_tt"
		value = 38
	}
	add = {
		desc = "cwp_nf_pivot_prosperity_tt"
		value = cwp_nf_canadian_prosperity
	}
	add = {
		desc = "cwp_nf_pivot_outports_tt"
		value = 0
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_rural_folk
					is_interest_group_type = ig_trade_unions
				}
			}
			add = this.ig_clout
		}
		multiply = 30
	}
	subtract = {
		desc = "cwp_nf_pivot_church_tt"
		value = 0
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_devout
				}
			}
			add = this.ig_clout
		}
		multiply = 30
	}
	subtract = {
		desc = "cwp_nf_pivot_merchants_tt"
		value = 0
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_industrialists
					is_interest_group_type = ig_petty_bourgeoisie
				}
			}
			add = this.ig_clout
		}
		multiply = 20
	}
	if = {
		limit = { has_variable = cwp_nf_campaign_var }
		add = {
			desc = "cwp_nf_pivot_campaign_tt"
			value = var:cwp_nf_campaign_var
		}
	}
	min = 2
	max = 98
}

cwp_nf_counted_result = {
	value = cwp_nf_confederation_pivot
	if = {
		limit = { has_variable = cwp_nf_swing_var }
		add = var:cwp_nf_swing_var
	}
	min = 1
	max = 99
}

# Commission of Government: the safe choice of those who distrusted self-government; a fixed minority.
cwp_nf_commission_share = {
	value = 14
}

cwp_nf_responsible_share = {
	value = 100
	subtract = cwp_nf_confederation_pivot
	subtract = cwp_nf_commission_share
	min = 1
}

# Runoff: most Commission voters, wanting stability and Canadian money over a return to 1933, go to Confederation.
cwp_nf_runoff_result = {
	value = var:cwp_nf_result_var
	add = {
		value = cwp_nf_commission_share
		multiply = 0.75
	}
	if = {
		limit = { has_variable = cwp_nf_swing2_var }
		add = var:cwp_nf_swing2_var
	}
	min = 1
	max = 99
}

cwp_nf_rg_counted_result = {
	value = 100
	subtract = var:cwp_nf_result_var
	subtract = var:cwp_nf_cm_result_var
	min = 1
}
