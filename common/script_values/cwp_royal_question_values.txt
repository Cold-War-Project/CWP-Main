cwp_rq_return_pivot = {
	add = {
		desc = "cwp_rq_pivot_base_tt"
		value = 50
	}
	add = {
		desc = "cwp_rq_pivot_catholics_tt"
		value = 0
		every_interest_group = {
			limit = {
				is_marginal = no
				OR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_rural_folk
				}
			}
			add = this.ig_clout
		}
		multiply = 35
	}
	subtract = {
		desc = "cwp_rq_pivot_left_tt"
		value = 0
		every_interest_group = {
			limit = {
				is_marginal = no
				OR = {
					is_interest_group_type = ig_trade_unions
					is_interest_group_type = ig_intelligentsia
				}
			}
			add = this.ig_clout
		}
		multiply = 30
	}
	if = {
		limit = { has_variable = cwp_rq_campaign_var }
		add = {
			desc = "cwp_rq_pivot_campaign_tt"
			value = var:cwp_rq_campaign_var
		}
	}
	min = 2
	max = 98
}
