			trigger = {
				has_technology_researched = combustion_gas_engine
			}
		}
		el_truck = {
			trigger = {
				has_technology_researched = combustion_gas_engine
			}
		}
		post_truck = {
			trigger = {
				has_technology_researched = combustion_gas_engine
			}
		}
		tractor = {
			trigger = {
				has_technology_researched = early_agriculture				# nice early out
				scope:state = {
 					any_scope_building = {
 						not = {
 							has_active_production_method = pm_manual_harvesting
 						}
 					}
 				}
			}
		}
	}


	route_entities = {
		car_01 = {
			tags = { european_car }
			entity = "european_car_01_entity"
			position = 0
			entity_length = 3
			
			count = {
				value = 0
				scope:state = {
					market = {
						mg:automobiles = {
							add = market_goods_sell_orders							# Must be widely available
						}
					}
				}
				divide = 100
				add = 1
				max = 3
			}
		}

		el_truck = {
			tags = { el_truck }
			entity = "el_truck_entity"
			position = 0
			entity_length = 3
			
			count = {
				value = 0
				scope:state = {
					add = infrastructure_usage
				}
				divide = 100
				add = 1
				max = 5
			}
		}

		post_truck = {
			tags = { post_truck }
			entity = "post_truck_entity"
			position = 0
			entity_length = 3
			
			count = {
				value = 0
				scope:state = {
					every_scope_building = {
						limit = { is_building_type = building_government_administration }
						add = level
					}
				}
				divide = 100
				add = 1
				max = 5
			}
		}

		tractor_01 = {
			tags = { tractor }
 			entity = "tractor_01_entity"
 			position = 0
 			entity_length = 3
 			
 			count = {
 				value = 0
 				scope:state = {
 					market = {
 						mg:grain = {
 							add = market_goods_sell_orders							# We have a lot of farm activity
 						}
 					}
 				}
 				divide = 1000

 			}
