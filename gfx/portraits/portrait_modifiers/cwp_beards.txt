# Cold War Project beards (1946-2046).
# Clean-shaven is the norm; moustaches are common until the 1980s; full beards belong to clergy,
# Sikhs and parts of the Muslim world. Historical characters are left alone so that their
# scripted DNA keeps the face they were given.
beards = {
	usage = game
	selection_behavior = weighted_random
	fallback = no_beard

	no_beard = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = beards
				template = no_beard
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100
				exists = scope:pop
			}
			modifier = {
				add = 100
				scope:character ?= {
					is_historical = no
				}
			}
		}
	}

	cwp_moustache = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = mustaches
				template = european_mustaches
				range = { 0.33 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 40
				cwp_portrait_era_traditional_dress_trigger = yes
				OR = {
					exists = scope:pop
					scope:character ?= {
						is_historical = no
						is_adult = yes
					}
				}
			}
			modifier = { # The officer's moustache
				add = 40
				OR = {
					cwp_portrait_pop_officer = yes
					AND = {
						cwp_portrait_character_military = yes
						scope:character ?= {
							is_historical = no
						}
					}
				}
			}
			modifier = { # Turkey and Iran
				add = 100
				OR = {
					scope:pop ?= {
						culture = {
							cwp_portrait_culture_turko_persian = yes
						}
					}
					scope:character ?= {
						is_historical = no
						is_adult = yes
						culture = {
							cwp_portrait_culture_turko_persian = yes
						}
					}
				}
			}
		}
	}

	cwp_arab_beards = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = beards
				template = arabic_beards
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100
				OR = {
					scope:pop ?= {
						culture = {
							OR = {
								cwp_portrait_culture_arab = yes
								cwp_portrait_culture_steppe = yes
							}
						}
					}
					scope:character ?= {
						is_historical = no
						age > 16
						culture = {
							OR = {
								cwp_portrait_culture_arab = yes
								cwp_portrait_culture_steppe = yes
							}
						}
					}
				}
			}
			modifier = { # Muslim clergy
				add = 2000
				OR = {
					AND = {
						cwp_portrait_pop_clergy = yes
						scope:pop ?= {
							religion = {
								has_discrimination_trait = heritage_islamic
							}
						}
					}
					AND = {
						cwp_portrait_character_clergy = yes
						scope:character ?= {
							is_historical = no
							religion = {
								has_discrimination_trait = heritage_islamic
							}
						}
					}
				}
			}
		}
	}

	cwp_african_beards = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = beards
				template = african_beards
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 50
				OR = {
					scope:pop ?= {
						culture = {
							cwp_portrait_culture_african = yes
						}
					}
					scope:character ?= {
						is_historical = no
						age > 16
						culture = {
							cwp_portrait_culture_african = yes
						}
					}
				}
			}
		}
	}

	cwp_asian_beards = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = beards
				template = asian_beards
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 30
				scope:character ?= {
					is_historical = no
					age > 50
					culture = {
						OR = {
							cwp_portrait_culture_chinese = yes
							cwp_portrait_culture_korean = yes
							has_discrimination_trait = heritage_vietnamese
						}
					}
				}
			}
		}
	}

	# Full beards: Orthodox and Jewish clergy, devout Jewish leaders, and Sikh men.
	cwp_full_facial_hair = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = beards
				template = full_beards
				range = { 0 1 }
			}
			accessory = {
				mode = replace
				gene = mustaches
				template = full_mustache
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 2000
				OR = {
					AND = {
						cwp_portrait_pop_clergy = yes
						scope:pop ?= {
							religion = {
								OR = {
									has_discrimination_trait = heritage_jewish
									rel:orthodox = this
									rel:oriental_orthodox = this
								}
							}
						}
					}
					AND = {
						cwp_portrait_character_clergy = yes
						scope:character ?= {
							is_historical = no
							religion = {
								OR = {
									has_discrimination_trait = heritage_jewish
									rel:orthodox = this
									rel:oriental_orthodox = this
								}
							}
						}
					}
				}
			}
			modifier = {
				add = 2000
				OR = {
					scope:pop ?= {
						religion = rel:sikh
					}
					scope:character ?= {
						is_historical = no
						age > 16
						religion = rel:sikh
					}
				}
			}
		}
	}

	cwp_slavic_peasant_beards = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = mustaches
				template = european_peasant_mustaches
				range = { 0 1 } # For the randomness to work correctly
			}
			accessory = {
				mode = replace
				gene = beards
				template = european_peasant_beards
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 50
				has_dlc_feature = ip5_cosmetics
				cwp_portrait_era_hats_trigger = yes
				cwp_portrait_pop_rural = yes
				scope:pop ?= {
					culture = {
						cwp_portrait_culture_east_slavic = yes
					}
				}
			}
		}
	}
}
