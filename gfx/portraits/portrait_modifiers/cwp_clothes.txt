# Cold War Project clothes (1946-2046).
# Built from the vanilla entries whose garments fit the period. Gene templates and accessories
# are vanilla (common/genes); every DLC gate matches the vanilla entry the garment was taken from.
# Triggers: common/scripted_triggers/cwp_portrait_triggers.txt.
#
# Weight tiers (weighted_random picks one valid entry in proportion to its weight):
#   200000 children | 30000 special leaders | 20000 uniforms | 15000 monarchs | 5000 clergy
#   1000-2000 civilian characters and pops | 1 fallback

	usage = game
	selection_behavior = weighted_random
	fallback = default_clothes
