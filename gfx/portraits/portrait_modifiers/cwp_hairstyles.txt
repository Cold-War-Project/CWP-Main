# Cold War Project hairstyles (1946-2046).
# The mod's simple short haircut stays the default. Additions: military short-back-and-sides,
# shaved heads for Buddhist monks. Historical characters keep the hair of their scripted DNA.
hairstyles = {
	usage = game
	selection_behavior = weighted_random
	fallback = european_peasant_hair

	european_peasant_hair = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = hairstyles
				template = european_peasant_hair
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100
				exists = scope:pop
			}
			modifier = {
				add = 100
				scope:character ?= {
					is_historical = no
				}
			}
		}
	}

	# The short cut vanilla uses under American military caps; avoids clipping with service caps.
	cwp_military_hair = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = hairstyles
				template = american_military_hairstyles
				range = { 0 1 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 1000
				OR = {
					cwp_portrait_pop_military = yes
					AND = {
						cwp_portrait_character_military = yes
						cwp_portrait_character_adult_male = yes
						scope:character ?= {
							is_historical = no
						}
					}
				}
			}
		}
	}

	cwp_buddhist_monk_hair = {
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = hairstyles
				template = no_hairstyles
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 1100
				OR = {
					AND = {
						cwp_portrait_pop_clergy = yes
						scope:pop ?= {
							religion = {
								has_discrimination_trait = heritage_dharmic
								NOT = { rel:hindu = this }
								NOT = { rel:sikh = this }
							}
						}
					}
					AND = {
						cwp_portrait_character_clergy = yes
						scope:character ?= {
							is_historical = no
							religion = {
								has_discrimination_trait = heritage_dharmic
								NOT = { rel:hindu = this }
								NOT = { rel:sikh = this }
							}
						}
					}
				}
			}
		}
	}
}
