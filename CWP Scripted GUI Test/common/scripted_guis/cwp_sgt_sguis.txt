### T06 to T08: one scripted GUI for every member row and all three answers
# scope:member is the row's country, scope:answer is 1 (approve), 2 (trim) or 3 (reject).

cwp_sgt_answer_sgui = {
	scope = country
	saved_scopes = { member answer }

	is_shown = {
		exists = scope:member
	}

	is_valid = {
		has_variable = cwp_sgt_ready
		scope:member = {
			has_variable = cwp_sgt_recovery
			NOT = { has_variable = cwp_sgt_answer_cd }
		}
	}

	effect = {
		scope:member = {
			set_variable = { name = cwp_sgt_answer value = scope:answer }
			# T08: a variable with a lifetime is the cooldown. It removes itself after 60 days.
			set_variable = { name = cwp_sgt_answer_cd value = yes days = 60 }
			if = {
				limit = { scope:answer = 1 }
				change_variable = { name = cwp_sgt_recovery add = 10 }
				change_variable = { name = cwp_sgt_alignment add = 5 }
			}
			else_if = {
				limit = { scope:answer = 2 }
				change_variable = { name = cwp_sgt_recovery add = 5 }
				change_variable = { name = cwp_sgt_alignment subtract = 3 }
			}
			else = {
				change_variable = { name = cwp_sgt_alignment subtract = 10 }
			}
			clamp_variable = { name = cwp_sgt_recovery min = 0 max = 100 }
			clamp_variable = { name = cwp_sgt_alignment min = 0 max = 100 }
		}
		debug_log = "CWP_SGT T06 answer;root=[THIS.GetCountry.GetNameNoFormatting];member=[SCOPE.sCountry('member').GetNameNoFormatting];answer=[SCOPE.sCountry('member').MakeScope.Var('cwp_sgt_answer').GetValue|0];recovery=[SCOPE.sCountry('member').MakeScope.Var('cwp_sgt_recovery').GetValue|0]"
	}
}

### T09: a tooltip built per row. The effect is not run, only described.

cwp_sgt_member_tt_sgui = {
	scope = country
	saved_scopes = { member }

	# The lines sit at root level and only the checks go into scope:member. Two or more lines inside
	# a scope:member block gain a heading line with the country's name.
	effect = {
		custom_tooltip_no_bullet = cwp_sgt_member_tt_head
		custom_tooltip = cwp_sgt_member_tt_score
		if = {
			limit = {
				scope:member = {
					has_variable = cwp_sgt_recovery
					var:cwp_sgt_recovery >= 50
				}
			}
			custom_tooltip = cwp_sgt_member_tt_good
		}
		else = {
			custom_tooltip = cwp_sgt_member_tt_bad
		}
		if = {
			limit = {
				scope:member = { has_variable = cwp_sgt_answer_cd }
			}
			custom_tooltip = cwp_sgt_member_tt_cooling
		}
	}
}

### T12, T13: one scripted GUI for every good and all three ration-book buttons
# scope:good is a goods scope, scope:action is 1 (ease), 2 (tighten) or 3 (ration).

cwp_sgt_ration_sgui = {
	scope = country
	saved_scopes = { good action }

	is_shown = {
		trigger_if = {
			limit = { scope:action = 1 }
			OR = {
				cwp_sgt_good_is_tight = yes
				cwp_sgt_good_is_standard = yes
			}
		}
		trigger_else_if = {
			limit = { scope:action = 2 }
			cwp_sgt_good_is_standard = yes
		}
		trigger_else = {
			cwp_sgt_good_is_tight = no
			cwp_sgt_good_is_standard = no
		}
	}

	is_valid = {
		cwp_sgt_good_on_cooldown = no
	}

	effect = {
		if = {
			limit = { scope:action = 1 }
			if = {
				limit = { cwp_sgt_good_is_tight = yes }
				remove_list_variable = { name = cwp_sgt_ration_tight target = scope:good }
				add_to_variable_list = { name = cwp_sgt_ration_standard target = scope:good }
			}
			else_if = {
				limit = { cwp_sgt_good_is_standard = yes }
				remove_list_variable = { name = cwp_sgt_ration_standard target = scope:good }
			}
		}
		else_if = {
			limit = {
				scope:action = 2
				cwp_sgt_good_is_standard = yes
			}
			remove_list_variable = { name = cwp_sgt_ration_standard target = scope:good }
			add_to_variable_list = { name = cwp_sgt_ration_tight target = scope:good }
		}
		else_if = {
			limit = {
				scope:action = 3
				cwp_sgt_good_is_tight = no
				cwp_sgt_good_is_standard = no
			}
			add_to_variable_list = { name = cwp_sgt_ration_standard target = scope:good }
		}
		# T13: a list entry with a lifetime is the per-good cooldown. Two months.
		add_to_variable_list = { name = cwp_sgt_ration_cd target = scope:good months = 2 }
		debug_log = "CWP_SGT T12 ration;[THIS.GetCountry.GetNameNoFormatting];good=[SCOPE.sGoods('good').GetNameNoFormatting]"
	}
}

# scope:level is 0 (free), 1 (standard) or 2 (tight). True when the good is at that level.
cwp_sgt_level_sgui = {
	scope = country
	saved_scopes = { good level }

	is_valid = {
		trigger_if = {
			limit = { scope:level = 2 }
			cwp_sgt_good_is_tight = yes
		}
		trigger_else_if = {
			limit = { scope:level = 1 }
			cwp_sgt_good_is_standard = yes
		}
		trigger_else = {
			cwp_sgt_good_is_tight = no
			cwp_sgt_good_is_standard = no
		}
	}
}

### T23: world numbers. scope:direction 1 raises Confidence, anything else lowers it, by scope:amount.

cwp_sgt_world_sgui = {
	scope = country
	saved_scopes = { direction amount }

	is_valid = {
		has_global_variable = cwp_sgt_confidence
		has_global_variable = cwp_sgt_liquidity
	}

	effect = {
		if = {
			limit = { scope:direction = 1 }
			change_global_variable = { name = cwp_sgt_confidence add = scope:amount }
			change_global_variable = { name = cwp_sgt_liquidity subtract = 5 }
		}
		else = {
			change_global_variable = { name = cwp_sgt_confidence subtract = scope:amount }
			change_global_variable = { name = cwp_sgt_liquidity add = 5 }
		}
		change_global_variable = { name = cwp_sgt_confidence max = 0 }
		change_global_variable = { name = cwp_sgt_confidence min = 100 }
		change_global_variable = { name = cwp_sgt_liquidity max = 0 }
		change_global_variable = { name = cwp_sgt_liquidity min = 100 }
		debug_log = "CWP_SGT T23 world;confidence=[GetGlobalVariable('cwp_sgt_confidence').GetValue|0];liquidity=[GetGlobalVariable('cwp_sgt_liquidity').GetValue|0]"
	}
}
