# CWP scripted GUI test. Country scope.

# T02: a script value read by the journal entry panel. Twice the counter.
cwp_sgt_counter_double = {
	value = 0
	if = {
		limit = { has_variable = cwp_sgt_counter }
		add = var:cwp_sgt_counter
		multiply = 2
	}
}

# T24: a number on a curve. 35 while Confidence is 100, rising with the square of the shortfall:
# 35 + (100 - Confidence)^2 / 100. Confidence 70 gives 44, Confidence 0 gives 135.
cwp_sgt_curve = {
	value = 0
	if = {
		limit = { has_global_variable = cwp_sgt_confidence }
		add = 100
		subtract = global_var:cwp_sgt_confidence
		multiply = {
			value = 100
			subtract = global_var:cwp_sgt_confidence
		}
		divide = 100
		add = 35
	}
}

# T07: a script value read once per ledger row, in the scope of the row's country.
cwp_sgt_member_score = {
	value = 0
	if = {
		limit = {
			has_variable = cwp_sgt_recovery
			has_variable = cwp_sgt_alignment
		}
		add = var:cwp_sgt_recovery
		add = var:cwp_sgt_alignment
		divide = 2
	}
}
