# CWP scripted GUI test: two repeatable, player-only decisions.
# "Start / Reset" builds the test data and adds the journal entry. "End" removes the journal entry.
# Every line the test writes to debug.log starts with CWP_SGT.

cwp_sgt_start_decision = {
	is_shown = {
		is_player = yes
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		custom_tooltip = cwp_sgt_start_decision_tt
		hidden_effect = {
			cwp_sgt_setup = yes
		}
	}

	ai_chance = {
		value = 0
	}
}

cwp_sgt_end_decision = {
	is_shown = {
		is_player = yes
		has_journal_entry = cwp_sgt_je
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		set_variable = {
			name = cwp_sgt_end
			value = yes
		}
	}

	ai_chance = {
		value = 0
	}
}
