# CWP scripted GUI test. Country scope (the player), with the good passed in as scope:good.

cwp_sgt_good_is_tight = {
	has_variable_list = cwp_sgt_ration_tight
	is_target_in_variable_list = { name = cwp_sgt_ration_tight target = scope:good }
}

cwp_sgt_good_is_standard = {
	has_variable_list = cwp_sgt_ration_standard
	is_target_in_variable_list = { name = cwp_sgt_ration_standard target = scope:good }
}

# The cooldown is a timed entry of a variable list: the entry removes itself when the time is up.
cwp_sgt_good_on_cooldown = {
	has_variable_list = cwp_sgt_ration_cd
	is_target_in_variable_list = { name = cwp_sgt_ration_cd target = scope:good }
}

# Hand-written "index" rows: the level is an ordinary variable, 0 to 2. scope:action 1 lowers it.
cwp_sgt_idx_can_step = {
	has_variable = $VAR$
	trigger_if = {
		limit = { scope:action = 1 }
		var:$VAR$ > 0
	}
	trigger_else = {
		var:$VAR$ < 2
	}
}
