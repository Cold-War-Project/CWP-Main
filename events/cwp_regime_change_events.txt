namespace = cwp_regime_change

# Root = a country that was the target of a regime change a day ago (cwp_regime_change_mark_target_effect).
# A monarchy is kept only where one existed in or after 1946: cwp_monarchy_tradition_var (set at game start and
# yearly for countries with law_monarchy), or an exiled pretender of a deposed crown.
cwp_regime_change.1 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = cwp_regime_change_window_var
		has_law = law_type:law_monarchy
		NOT = { has_variable = cwp_monarchy_tradition_var }
		NOT = { has_variable = cwp_regime_change_had_pretender_var }
	}

	immediate = {
		if = {
			limit = {
				var:cwp_regime_change_enforcer_var ?= { cwp_country_has_parliamentary_crown = yes }
			}
			activate_law = law_type:law_parliamentary_republic
		}
		else = {
			activate_law = law_type:law_presidential_republic
		}
		# The crown code may have seated and then deposed a monarch for the imposed day: forget that pretender.
		if = {
			limit = { has_variable = cwp_crown_pretender_var }
			remove_variable = cwp_crown_pretender_var
		}
		remove_variable = cwp_regime_change_window_var
		remove_variable = cwp_regime_change_enforcer_var
	}
}
