namespace = cwp_nuc

# The first test.
cwp_nuc.1 = {
	type = country_event
	placement = root

	title = cwp_nuc.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_nuc.1.d.RUS
				trigger = { c:RUS ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.GBR
				trigger = { c:GBR ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.FRA
				trigger = { c:FRA ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.CHI
				trigger = { c:CHI ?= this }
			}
			desc = cwp_nuc.1.d
		}
	}
	flavor = cwp_nuc.1.f

	event_image = { video = "unspecific_devastation" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		NOT = { cwp_is_nuclear_power = yes }
	}

	option = {
		name = cwp_nuc.1.a
		default_option = yes
		set_variable = cwp_nuclear_power_var
		add_modifier = { name = cwp_nuclear_power }
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = cwp_nuc_military_pride
				years = 2
			}
		}
		ai_chance = { base = 10 }
	}

	option = { # Say nothing and let the rivals' seismographs find it
		name = cwp_nuc.1.b
		set_variable = cwp_nuclear_power_var
		add_modifier = {
			name = cwp_nuclear_power
			multiplier = 0.5
		}
		custom_tooltip = cwp_nuc_secret_test_tt
		ai_chance = { base = 2 }
	}
}

# Retired 2026-11-18: the reaction events (cwp_rx.1 to .3) give other powers a real choice when a country
# tests; this notice had none. Kept as an orphan so saves with a pending copy still resolve.
cwp_nuc.2 = {
	type = country_event
	orphan = yes
	placement = scope:cwp_nuc_new_power.capital

	title = cwp_nuc.2.t
	desc = cwp_nuc.2.d
	flavor = cwp_nuc.2.f

	event_image = { video = "europenorthamerica_political_extremism" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	option = { # Alarm
		name = cwp_nuc.2.a
		trigger = {
			scope:cwp_nuc_new_power = { relations:root <= relations_threshold:cold }
		}
		default_option = yes
		add_modifier = {
			name = cwp_nuc_rival_bomb_fear
			years = 2
		}
	}

	option = { # Noted
		name = cwp_nuc.2.b
		trigger = {
			scope:cwp_nuc_new_power = { relations:root > relations_threshold:cold }
		}
		default_option = yes
	}
}

	duration = 3

		name = cwp_nuc.4.a

			}
		}
	}

	option = { # A steady program within the budget
		name = cwp_nuc.4.b
		custom_tooltip = cwp_nuc_steady_tt
		ai_chance = { base = 10 }
	}
