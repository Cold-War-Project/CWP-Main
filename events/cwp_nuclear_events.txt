namespace = cwp_nuc

# The first test.
cwp_nuc.1 = {
	type = country_event
	placement = root

	title = cwp_nuc.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_nuc.1.d.RUS
				trigger = { c:RUS ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.GBR
				trigger = { c:GBR ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.FRA
				trigger = { c:FRA ?= this }
			}
			triggered_desc = {
				desc = cwp_nuc.1.d.CHI
				trigger = { c:CHI ?= this }
			}
			desc = cwp_nuc.1.d
		}
	}
	flavor = cwp_nuc.1.f

	event_image = { video = "unspecific_devastation" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		NOT = { cwp_is_nuclear_power = yes }
	}

	option = {
		name = cwp_nuc.1.a
		default_option = yes
		set_variable = cwp_nuclear_power_var
		add_modifier = { name = cwp_nuclear_power }
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = cwp_nuc_military_pride
				years = 2
			}
		}
		hidden_effect = {
			save_scope_as = cwp_nuc_new_power
			every_country = {
				limit = {
					NOT = { this = root }
					country_rank >= rank_value:major_power
				}
				trigger_event = { id = cwp_nuc.2 days = 7 popup = yes }
			}
		}
	}
}

# The world hears of it. Root = another major country, scope:cwp_nuc_new_power = the tester.
cwp_nuc.2 = {
	type = country_event
	placement = scope:cwp_nuc_new_power.capital

	title = cwp_nuc.2.t
	desc = cwp_nuc.2.d
	flavor = cwp_nuc.2.f

	event_image = { video = "europenorthamerica_political_extremism" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	option = { # Alarm
		name = cwp_nuc.2.a
		trigger = {
			scope:cwp_nuc_new_power = { relations:root <= relations_threshold:cold }
		}
		default_option = yes
		add_modifier = {
			name = cwp_nuc_rival_bomb_fear
			years = 2
		}
	}

	option = { # Noted
		name = cwp_nuc.2.b
		trigger = {
			scope:cwp_nuc_new_power = { relations:root > relations_threshold:cold }
		}
		default_option = yes
	}
}
