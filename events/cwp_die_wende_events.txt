	type = country_event
	placement = root

	title = cwp_die_wende.1.t

	flavor = cwp_die_wende.1.f

	event_image = { video = "europenorthamerica_springtime_of_nations" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	duration = 3

	option = {
		name = cwp_die_wende.1.a
		default_option = yes
	}
	option = {
		name = cwp_die_wende.1.b_ddr
		trigger = { c:DDR ?= ROOT }
		if = { limit = { has_journal_entry = je_cwp_die_wende } je:je_cwp_die_wende = { add_progress = { value = -5 name = cwp_dw_revolution_bar } } }
		ig:ig_armed_forces ?= { add_modifier = { name = cwp_choice_grievance years = 2 } }
		ai_chance = {
			base = 2
		}
	}
	option = {
		name = cwp_die_wende.1.c_ddr
		trigger = { c:DDR ?= ROOT }
		if = { limit = { has_journal_entry = je_cwp_die_wende } je:je_cwp_die_wende = { add_progress = { value = -10 name = cwp_dw_revolution_bar } } }
		ig:ig_intelligentsia ?= { add_modifier = { name = cwp_choice_grievance years = 2 } }
		ig:ig_trade_unions ?= { add_modifier = { name = cwp_choice_grievance years = 2 } }
		ai_chance = {
			base = 1
		}
	}
}
