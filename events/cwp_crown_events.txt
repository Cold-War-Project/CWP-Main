namespace = cwp_crown

# Accession of a background monarch under a Prime-Minister government (common/scripted_effects/cwp_crown_effects.txt).
# One option: the event formalises a succession that has already happened.
cwp_crown.1 = {
	type = country_event
	placement = root

	title = cwp_crown.1.t
	desc = cwp_crown.1.d
	flavor = cwp_crown.1.f

	event_image = { video = "votp_monarch_holding_court" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	left_icon = scope:cwp_crown_scope_event_monarch

	duration = 3

	trigger = {
		exists = var:cwp_monarch_var
	}

	immediate = {
		var:cwp_monarch_var = { save_scope_as = cwp_crown_scope_event_monarch }
	}

	option = {
		name = cwp_crown.1.a
		default_option = yes
		add_modifier = {
			name = cwp_crown_accession
			months = 12
		}
	}
}
