# The Berlin blockade: the western leader chooses
cwp_german_question.5 = {
	type = country_event
	placement = root

	title = cwp_german_question.5.t
	desc = cwp_german_question.5.d
	flavor = cwp_german_question.5.f

	event_image = { video = "unspecific_airPlane" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		has_global_variable = cwp_gq_blockade
	}

	option = {
		name = cwp_german_question.5.a
		default_option = yes
		set_global_variable = cwp_gq_airlift
		add_modifier = {
			name = cwp_gq_airlift
			days = normal_modifier_time
		}
		c:GBR ?= {
			if = {
				limit = { NOT = { this = root } }
				add_modifier = {
					name = cwp_gq_airlift
					days = normal_modifier_time
				}
			}
		}
		custom_tooltip = cwp_german_question.5.a_tt
		ai_chance = { base = 90 }
	}

	option = {
		name = cwp_german_question.5.b
		cwp_gq_shift_bar = { BAR = cwp_gq_unity_bar VALUE = 10 }
		cwp_gq_shift_bar = { BAR = cwp_gq_western_bar VALUE = -10 }
		c:RUS ?= {
			trigger_event = { id = cwp_german_question.19 days = 7 popup = yes }
		}
		ai_chance = { base = 8 }
	}

	option = {
		name = cwp_german_question.5.c
		set_global_variable = { name = cwp_gq_berlin_abandoned days = long_modifier_time }
		add_modifier = {
			name = cwp_gq_berlin_abandoned
			days = normal_modifier_time
		}
		c:RUS ?= {
			add_modifier = {
				name = cwp_gq_berlin_won
				days = normal_modifier_time
			}
		}
		cwp_gq_shift_bar = { BAR = cwp_gq_soviet_bar VALUE = 10 }
		cwp_gq_end_blockade = yes
		ai_chance = { base = 2 }
	}
}
