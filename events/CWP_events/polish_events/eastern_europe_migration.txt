namespace = ee_migration
# Eastern Europe Migration events

ee_migration.1 = {
	type = country_event
	placement = scope:german_exile_state

	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	event_image = {
		video = "europenorthamerica_native_american"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = ee_migration.1.t
	desc = ee_migration.1.d
	flavor = ee_migration.1.f

	duration = 3
	
	trigger = {
		any_scope_state = {
			has_modifier = ongoing_exile
		}
		has_journal_entry = je_german_resettlement
		any_scope_state = {
			any_scope_pop = {
				culture = cu:north_german
			}
		}
		NOT = {
			has_variable = exile_event_happened
		}
		this = {
			is_ai = no
		}
	}

	immediate = {
		random_scope_state = {
			limit = {
				has_modifier = ongoing_exile
			}
			save_scope_as = german_exile_state
		}
		ROOT = {
			set_variable = {
				name = exile_event_happened
				months = 2
			}
		}
	}
	
	option = { # use force
		name = ee_migration.1.a
		default_option = yes
		scope:german_exile_state = {
			kill_population_percent_in_state = {
				percent = 0.05
				culture = cu:north_german
			}
		}
		c:DDR = {
			change_relations = {
				country = c:POL
				value = -10
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					c:POL = {
						has_attitude = {
							who = c:DDR
							attitude = disinterested
						}
					}
				}
				add = 1000
			}
		}
	}
	option = { # pay their way
		name = ee_migration.1.b
		scope:german_exile_state = {
			kill_population_percent_in_state = {
				percent = 0.025
				culture = cu:north_german
			}
		}
		add_modifier = {
			name = USA_paying_removal_costs
			multiplier = money_amount_multiplier_very_small
			months = 1
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					c:POL = {
						in_default = yes
					}
				}
				add = -50
			}
		}
	}
}

# German Great Migration
ee_migration.5 = {
	type = country_event
	placement = scope:german_migration_target_state
	
	title = sa_migration.5.t
	desc = sa_migration.5.d
	flavor = sa_migration.5.f
	
	event_image = {
		video = "southamerica_factory_opening"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"
	
	duration = 3
	
	trigger = {
		# triggered by on_monthly_pulse_country, guaranteed first month of the game
		c:POL ?= THIS
		NOT = { has_variable = german_migration_var }
	}

	immediate = {
		random_state = {
			limit = {
				owner = c:DDR
				NOT = {
					has_variable = german_migration_state_var
				}
			}
			save_scope_as = german_migration_target_state
			set_variable = {
				name = german_migration_state_var
			}
		}
		set_variable = german_migration_var
		c:POL = {
			save_scope_as = migration_country
		}
		cu:north_german = {
			save_scope_as = migration_culture
		}
	}

	option = {
		name = ee_migration.5.a
		default_option = yes
		scope:german_migration_target_state = {
			create_mass_migration = {
				origin = scope:migration_country
				culture = scope:migration_culture
			}
		}
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = ee_migration.5.b
		ai_chance = {
			base = 0
		}
	}
}
