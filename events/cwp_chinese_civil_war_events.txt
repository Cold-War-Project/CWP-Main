# An armistice is proposed (the other China decides)
cwp_ccw.5 = {
	type = country_event
	placement = ROOT

	title = cwp_ccw.5.t
	desc = cwp_ccw.5.d
	flavor = cwp_ccw.5.f

	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	trigger = {
		cwp_ccw_war_running = yes
	}

	option = {
		name = cwp_ccw.5.a
		cwp_ccw_front_line_settlement = yes
		hidden_effect = {
			if = {
				limit = { has_global_variable = cwp_ccw_armistice_by_prc }
				c:PRC ?= { trigger_event = { id = cwp_ccw.6 popup = yes } }
			}
			else = {
				c:ROC ?= { trigger_event = { id = cwp_ccw.6 popup = yes } }
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { any_scope_war = { has_war_support = { target = root value < 25 } } }
				add = 30
			}
			modifier = {
				trigger = { global_var:cwp_ccw_exhaustion >= 26 }
				add = 20
			}
			modifier = {
				trigger = {
					c:PRC ?= root
					has_global_variable = cwp_ccw_soviet_restraint
				}
				add = 30
			}
			modifier = {
				trigger = {
					c:ROC ?= root
					NOT = { has_modifier = cwp_ccw_us_aid }
				}
				add = 15
			}
		}
	}

	option = {
		name = cwp_ccw.5.b
		default_option = yes
		custom_tooltip = cwp_ccw_war_goes_on_tt
		hidden_effect = {
			if = {
				limit = { has_global_variable = cwp_ccw_armistice_by_prc }
				c:PRC ?= { trigger_event = { id = cwp_ccw.6 popup = yes } }
			}
			else = {
				c:ROC ?= { trigger_event = { id = cwp_ccw.6 popup = yes } }
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				trigger = {
					OR = {
						AND = {
							c:PRC ?= root
							global_var:cwp_ccw_balance >= 15
						}
						AND = {
							c:ROC ?= root
							global_var:cwp_ccw_balance <= -15
						}
					}
				}
				add = 40
			}
		}
	}

}
