namespace = cwp_interstate_highway_events

cwp_interstate_highway_events.1 = {
	type = country_event
	placement = ROOT

	title = cwp_interstate_highway_events.1.t
	desc = cwp_interstate_highway_events.1.d
	flavor = cwp_interstate_highway_events.1.f

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/building_icons/automobile_industry.dds"

	duration = 3

	option = {
		name = cwp_interstate_highway_events.1.a

		default_option = yes

		add_modifier = {
			name = cwp_interstate_highway_national_program
			months = 120
		}

		ai_chance = {
			factor = 1
		}
	}

	option = {
		name = cwp_interstate_highway_events.1.b

		add_modifier = {
			name = cwp_interstate_highway_connected_states
			months = 120
		}

		ai_chance = {
			factor = 1
		}
	}

	option = {
		name = cwp_interstate_highway_events.1.c

		ordered_primary_culture = {
			order_by = "root.culture_percent_country(this)"
			save_scope_as = cwp_highway_primary_culture
		}
		if = {
			limit = {
				exists = scope:cwp_highway_primary_culture
				scope:cwp_highway_primary_culture = {
					NOT = { has_cultural_obsession = automobiles }
				}
			}
			custom_tooltip = cwp_interstate_highway_automobile_obsession_tt
			hidden_effect = {
				scope:cwp_highway_primary_culture = {
					add_cultural_obsession = automobiles
				}
			}
		}
		if = {
			limit = {
				any_scope_state = {
					is_incorporated = yes
					NOT = { has_modifier = cwp_interstate_highway_suburban_corridors }
				}
			}
			custom_tooltip = cwp_interstate_highway_suburban_corridors_tt
			hidden_effect = {
				every_scope_state = {
					limit = {
						is_incorporated = yes
						NOT = { has_modifier = cwp_interstate_highway_suburban_corridors }
					}
					add_modifier = {
						name = cwp_interstate_highway_suburban_corridors
						months = 120
					}
				}
			}
		}

		ai_chance = {
			factor = 1
		}
	}
}
