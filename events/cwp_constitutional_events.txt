# The institutional question opens.
cwp_iq.1 = {
	type = country_event
	placement = root

	title = cwp_iq.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_iq.1.d.ITA
				trigger = { c:ITA ?= this }
			}
			triggered_desc = {
				desc = cwp_iq.1.d.BUL
				trigger = { c:BUL ?= this }
			}
			triggered_desc = {
				desc = cwp_iq.1.d.ROM
				trigger = { c:ROM ?= this }
			}
			triggered_desc = {
				desc = cwp_iq.1.d.GRE
				trigger = { c:GRE ?= this }
			}
			desc = cwp_iq.1.d
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = cwp_iq.1.f.ITA
				trigger = { c:ITA ?= this }
			}
			triggered_desc = {
				desc = cwp_iq.1.f.ROM
				trigger = { c:ROM ?= this }
			}
			desc = cwp_iq.1.f
		}
	}

	event_image = { video = "votp_monarch_holding_court" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	immediate = {
		cwp_crown_save_monarch_scope = { NAME = cwp_iq_monarch }
	}

	option = { # Put it to the people
		name = cwp_iq.1.a
		default_option = yes
		remove_variable = cwp_iq_grace_var
		set_variable = { name = cwp_iq_grace_var days = 90 }
		custom_tooltip = cwp_iq_grace_short_tt
		ai_chance = {
			base = 10
			modifier = {
				trigger = { cwp_iq_republican_clout_in_government > cwp_iq_monarchist_clout_in_government }
				add = 20
			}
		}
	}

	option = { # Reconstruction first
		name = cwp_iq.1.b
		remove_variable = cwp_iq_grace_var
		set_variable = { name = cwp_iq_grace_var days = 540 }
		custom_tooltip = cwp_iq_grace_long_tt
		ai_chance = {
			base = 10
			modifier = {
				trigger = { cwp_iq_monarchist_clout_in_government > cwp_iq_republican_clout_in_government }
				add = 20
			}
		}
	}
}

# Polling day.
cwp_iq.2 = {
	type = country_event
	placement = root

	title = cwp_iq.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_iq.2.d.rigged
				trigger = { has_variable = cwp_iq_rigged_var }
			}
			triggered_desc = {
				desc = cwp_iq.2.d.republic
				trigger = { var:cwp_iq_result_var > 50 }
			}
			desc = cwp_iq.2.d.crown
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = cwp_iq.2.f.republic
				trigger = { var:cwp_iq_result_var > 50 }
			}
			desc = cwp_iq.2.f.crown
		}
	}

	event_image = { video = "southamerica_election" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"

	duration = 3

	trigger = {
		has_journal_entry = je_cwp_institutional_question
	}

	immediate = {
		random_list = {
			20 = { set_variable = { name = cwp_iq_swing_var value = -6 } }
			20 = { set_variable = { name = cwp_iq_swing_var value = -3 } }
			20 = { set_variable = { name = cwp_iq_swing_var value = 0 } }
			20 = { set_variable = { name = cwp_iq_swing_var value = 3 } }
			20 = { set_variable = { name = cwp_iq_swing_var value = 6 } }
		}
		set_variable = { name = cwp_iq_result_var value = cwp_iq_counted_result }
		cwp_crown_save_monarch_scope = { NAME = cwp_iq_monarch }
	}

	option = { # A parliamentary republic
		name = cwp_iq.2.a
		trigger = { var:cwp_iq_result_var > 50 }
		default_option = yes
		cwp_iq_proclaim_republic = { LAW = law_parliamentary_republic }
		cwp_iq_result_aftermath = yes
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					OR = {
						cwp_iq_under_soviet_pressure = yes
						cwp_communists_in_government = yes
					}
				}
				factor = 0.1
			}
		}
	}

	option = { # A people's republic
		name = cwp_iq.2.b
		trigger = {
			var:cwp_iq_result_var > 50
			OR = {
				cwp_iq_under_soviet_pressure = yes
				cwp_communists_in_government = yes
			}
		}
		cwp_iq_proclaim_republic = { LAW = law_council_republic }
		cwp_iq_result_aftermath = yes
		ai_chance = { base = 10 }
	}

	option = { # A presidential republic with a strong executive
		name = cwp_iq.2.d
		trigger = { var:cwp_iq_result_var > 50 }
		cwp_iq_proclaim_republic = { LAW = law_presidential_republic }
		cwp_iq_result_aftermath = yes
		ig:ig_armed_forces ?= {
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 2
			}
		}
		ig:ig_trade_unions ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 2
			}
		}
		ai_chance = {
			base = 3
			modifier = {
				trigger = { ig:ig_armed_forces ?= { is_in_government = yes } }
				add = 5
			}
		}
	}

	option = { # The crown stays
		name = cwp_iq.2.c
		trigger = { NOT = { var:cwp_iq_result_var > 50 } }
		default_option = yes
		cwp_iq_keep_crown = yes
		cwp_iq_result_aftermath = yes
		ai_chance = { base = 10 }
	}

	option = { # The crown stays, but the court gives up its political role to calm the losers
		name = cwp_iq.2.e
		trigger = { NOT = { var:cwp_iq_result_var > 50 } }
		set_variable = cwp_iq_settled_var
		add_modifier = {
			name = cwp_iq_crown_confirmed
			years = 2
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_monarchy
					value >= approve
				}
			}
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 1
			}
		}
		cwp_iq_result_aftermath = yes
		ai_chance = { base = 5 }
	}
}

# The vote is forced (cwp_iq_vote_forced): the promise, the street or the occupier.
cwp_iq.3 = {
	type = country_event
	placement = root

	title = cwp_iq.3.t
	desc = cwp_iq.3.d
	flavor = cwp_iq.3.f

	event_image = { video = "unspecific_politicians_arguing" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	duration = 3

	trigger = {
		has_journal_entry = je_cwp_institutional_question
		NOT = { has_variable = cwp_iq_vote_pending_var }
	}

	immediate = {
		set_variable = cwp_iq_vote_pending_var
	}

	option = { # Set the date
		name = cwp_iq.3.a
		default_option = yes
		custom_tooltip = cwp_iq_call_referendum_tt
		trigger_event = { id = cwp_iq.2 days = 30 popup = yes }
		ai_chance = {
			base = 10
			# In the Soviet zone a vote the crown would still win, even rigged, is not worth holding.
			modifier = {
				trigger = {
					cwp_iq_under_soviet_pressure = yes
					has_law = law_type:law_secret_police
					var:cwp_iq_vote_current_var < 25
				}
				factor = 0
			}
		}
	}

	option = { # Force the King out instead
		name = cwp_iq.3.b
		trigger = {
			cwp_iq_under_soviet_pressure = yes
			has_law = law_type:law_secret_police
		}
		custom_tooltip = cwp_iq_force_abdication_tt
		trigger_event = { id = cwp_iq.4 popup = yes }
		ai_chance = {
			base = 0
			modifier = {
				trigger = { var:cwp_iq_vote_current_var < 55 }
				add = 15
			}
			modifier = {
				trigger = { var:cwp_iq_vote_current_var < 25 }
				add = 30
			}
		}
	}
}

	type = country_event
	placement = root

	title = cwp_iq.4.t

	flavor = cwp_iq.4.f

	event_image = { video = "ip4_abdication" }

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		has_journal_entry = je_cwp_institutional_question
		country_has_monarchy_law = yes
	}

	immediate = {
		set_variable = cwp_iq_vote_pending_var
		cwp_crown_save_monarch_scope = { NAME = cwp_iq_monarch }
	}

	option = {
		name = cwp_iq.4.a
		default_option = yes
		cwp_iq_proclaim_republic = { LAW = law_council_republic }
		add_modifier = {
			name = cwp_iq_forced_abdication
			years = 5
		}
		add_radicals = {
			value = 0.15
			interest_group = ig:ig_landowners
		}
		add_radicals = {
			value = 0.1
			interest_group = ig:ig_devout
		}
		ai_chance = { base = 10 }
	}

	option = { # A republic, but keep the coalition parties in it
		name = cwp_iq.4.b
		cwp_iq_proclaim_republic = { LAW = law_parliamentary_republic }
		add_modifier = {
			name = cwp_iq_forced_abdication
			years = 3
		}
		overlord ?= {
			change_relations = { country = root value = -20 }
		}
		ai_chance = {
			base = 3
			modifier = {
				trigger = { cwp_communists_in_government = no }
				add = 10
			}
		}
	}
}

# Japan: the cabinet's draft reaches the occupation authorities.

	type = country_event
	placement = root

	title = cwp_jap_constitution.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_jap_constitution.2.d.rejected
				trigger = {
					cwp_occupied_by_democracy = yes

				}
			}
			desc = cwp_jap_constitution.2.d.accepted
		}
	}
	flavor = cwp_jap_constitution.2.f

	duration = 3

	trigger = {
		has_journal_entry = je_cwp_japan_postwar_constitution
	}

	immediate = {
		if = {
			limit = {
				cwp_occupied_by_democracy = yes

			}
			set_variable = cwp_jap_matsumoto_rejected_var
		}
	}

	option = { # Rejected: the occupier will write its own

		trigger = { has_variable = cwp_jap_matsumoto_rejected_var }

	}

	option = {
		name = cwp_jap_constitution.2.c
		trigger = { has_variable = cwp_jap_matsumoto_rejected_var }
		remove_variable = cwp_jap_draft_pending_var
		custom_tooltip = cwp_jap_occupier_draft_tt
		trigger_event = { id = cwp_jap_constitution.3 days = 10 popup = yes }

			change_relations = { country = root value = -10 }

		ig:ig_landowners ?= {
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 2
			}
		}

	}

	option = { # Accepted: a revised Meiji constitution
		name = cwp_jap_constitution.2.b
		trigger = { NOT = { has_variable = cwp_jap_matsumoto_rejected_var } }
		default_option = yes

			add_modifier = {
				name = cwp_iq_lost_the_referendum
				years = 2
			}
		}
	}

	option = {
		name = cwp_jap_constitution.2.d
		trigger = { NOT = { has_variable = cwp_jap_matsumoto_rejected_var } }
		set_variable = cwp_jap_meiji_revised_var
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = cwp_iq_lost_the_referendum
				years = 2
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = cwp_iq_campaign_ally
				years = 2
			}
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = cwp_iq_campaigned_against
				years = 2
			}
		}
		ai_chance = { base = 3 }
	}

# Japan: the occupation authorities present their own draft.
cwp_jap_constitution.3 = {
	type = country_event
	placement = root

	title = cwp_jap_constitution.3.t
	desc = cwp_jap_constitution.3.d
	flavor = cwp_jap_constitution.3.f

	event_image = { video = "ep2_important_japanese_men_negotiating" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	trigger = {
		has_journal_entry = je_cwp_japan_postwar_constitution
	}

	immediate = {
		overlord ?= { save_scope_as = cwp_jap_occupier }
	}

	option = { # Accept the draft as written
		name = cwp_jap_constitution.3.a
		default_option = yes
		cwp_jap_adopt_constitution = { ARTICLE_9 = cwp_amendment_article_9 }
		ai_chance = { base = 10 }
	}

	option = { # Accept it, with the Ashida amendment
		name = cwp_jap_constitution.3.b
		trigger = { ig:ig_intelligentsia ?= { is_in_government = yes } }
		cwp_jap_adopt_constitution = { ARTICLE_9 = cwp_amendment_article_9_ashida }
		ai_chance = { base = 15 }
	}

	option = { # Refuse
		name = cwp_jap_constitution.3.c
		remove_variable = cwp_jap_draft_pending_var
		add_modifier = {
			name = cwp_jap_constitution_rejected
			years = 2
		}
		scope:cwp_jap_occupier ?= {
			change_relations = { country = root value = -30 }
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_armed_forces ?= { is_powerful = yes } }
				add = 5
			}
		}
	}
}
