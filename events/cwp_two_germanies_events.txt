# The Federal Republic offers a Basic Treaty: the GDR answers
cwp_two_germanies.3 = {
	type = country_event
	placement = root

	title = cwp_two_germanies.3.t
	desc = cwp_two_germanies.3.d
	flavor = cwp_two_germanies.3.f

	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	trigger = {
		exists = c:GER
		NOT = { has_global_variable = cwp_tg_basic_treaty }
	}

	option = {
		name = cwp_two_germanies.3.a
		default_option = yes
		set_global_variable = cwp_tg_basic_treaty
		add_modifier = {
			name = cwp_tg_basic_treaty
			days = long_modifier_time
		}
		c:GER = {
			add_modifier = {
				name = cwp_tg_basic_treaty
				days = long_modifier_time
			}
		}
		cwp_tg_shift_bar = { BAR = cwp_tg_legitimacy_bar VALUE = 10 }
		hidden_effect = {
			every_country = {
				limit = {
					cwp_tg_is_party = yes
					NOT = { this = root }
				}
				trigger_event = { id = cwp_two_germanies.4 days = 1 popup = yes }
			}
		}
		ai_chance = { base = 80 }
	}

	option = {
		name = cwp_two_germanies.3.b
		ai_chance = { base = 20 }
	}
}
