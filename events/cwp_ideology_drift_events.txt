namespace = cwp_ideo

# The unions make their peace with the market and the ballot box (Bad Godesberg, 1959; the British TUC's postwar settlement).
cwp_ideo.1 = {
	type = country_event
	placement = root

	title = cwp_ideo.1.t
	desc = cwp_ideo.1.d
	flavor = cwp_ideo.1.f

	event_image = { video = "asia_union_leader" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	duration = 3
	cooldown = { days = long_modifier_time }

	trigger = {
		has_law = law_type:law_universal_suffrage
		OR = {
			has_law = law_type:law_right_of_assembly
			has_law = law_type:law_protected_speech
		}
		average_sol >= 15
		ig:ig_trade_unions ?= {
			is_marginal = no
			has_ideology = ideology:ideology_proletarian
			NOT = { has_ideology = ideology:ideology_cwp_free_trade_unionism }
		}
	}

	immediate = {
		ig:ig_trade_unions = {
			save_scope_as = cwp_ideo_ig
			leader ?= { save_scope_as = cwp_ideo_leader }
		}
	}

	option = { # Reform, not revolution
		name = cwp_ideo.1.a
		default_option = yes
		scope:cwp_ideo_ig = {
			remove_ideology = ideology_proletarian
			add_ideology = ideology_cwp_free_trade_unionism
		}
		ai_chance = { base = 10 }
	}

	option = { # The old guard holds
		name = cwp_ideo.1.b
		scope:cwp_ideo_ig = {
			add_modifier = {
				name = cwp_ideo_internal_strife
				years = 2
			}
		}
		ai_chance = { base = 5 }
	}
}
