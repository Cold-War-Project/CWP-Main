### A first atomic test. scope:cwp_rx_tester = the new nuclear power.

# A nuclear power whose monopoly or lead is broken by a rival.
cwp_rx.1 = {
	type = country_event
	placement = root

	title = cwp_rx.1.t
	desc = cwp_rx.1.d
	flavor = cwp_rx.1.f

	event_image = { video = "unspecific_devastation" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		exists = scope:cwp_rx_tester
	}

	option = { # A thermonuclear program
		name = cwp_rx.1.a
		default_option = yes
		add_modifier = { name = cwp_rx_thermonuclear_program days = normal_modifier_time multiplier = cwp_rx_cost_rearmament }
		cwp_rx_ig_mood = { IG = ig_armed_forces MOD = cwp_rx_ig_budget_raised }
		cwp_rx_ig_mood = { IG = ig_intelligentsia MOD = cwp_rx_ig_overruled }
		if = {
			limit = { scope:cwp_rx_tester = { cwp_cw_is_communist_state = yes } }
			cwp_rs_move_fear = { FEAR = 10 }
		}
		ai_chance = {
			base = 3
		}
	}
	option = { # International control
		name = cwp_rx.1.b
		add_modifier = { name = cwp_rx_international_control days = short_modifier_time is_decaying = yes }
		cwp_rx_ig_mood = { IG = ig_armed_forces MOD = cwp_rx_ig_overruled }
		cwp_rx_ig_mood = { IG = ig_intelligentsia MOD = cwp_rx_ig_heeded }
		change_relations = { country = scope:cwp_rx_tester value = 10 }
		if = {
			limit = { scope:cwp_rx_tester = { cwp_cw_is_communist_state = yes } }
			cwp_rs_move_fear = { FEAR = 5 }
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_intelligentsia ?= { is_in_government = yes } }
				add = 1
			}
		}
	}
}

			limit = { cwp_cw_in_west = yes }

			limit = { cwp_cw_in_east = yes }

		}
	}

	option = { # Rely on the patron's deterrent
		name = cwp_rx.3.a
		trigger = {
			exists = scope:cwp_rx_patron
			NOT = { scope:cwp_rx_patron = root }

		}
		default_option = yes
		add_modifier = { name = cwp_rx_nuclear_umbrella days = short_modifier_time is_decaying = yes }

		cwp_rx_ig_mood = { IG = ig_armed_forces MOD = cwp_rx_ig_overruled }

			base = 2
		}
	}
	option = { # No nuclear ally: call for a ban on tests
		name = cwp_rx.3.c

			NOT = {
				scope:cwp_rx_patron ?= {
					NOT = { this = root }

			}
		}
		default_option = yes
		add_modifier = { name = cwp_rx_international_control days = short_modifier_time is_decaying = yes }
		cwp_rx_ig_mood = { IG = ig_armed_forces MOD = cwp_rx_ig_overruled }
		cwp_rx_ig_mood = { IG = ig_intelligentsia MOD = cwp_rx_ig_heeded }
		ai_chance = {
			base = 2
		}
	}

		add_modifier = { name = cwp_rx_defense_budget_raised days = normal_modifier_time multiplier = cwp_rx_cost_small }
