namespace = chinese_civil_war

chinese_civil_war.1 = {
	type = country_event
	placement = ROOT
	duration = 3
	title = chinese_civil_war.1.t
	desc = chinese_civil_war.1.d
	flavor = chinese_civil_war.1.f
	event_image = { video = "europenorthamerica_american_civil_war" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	trigger = {
		OR = { c:PRC ?= this c:ROC ?= this }
		NOT = { has_variable = cwp_ccw_opening_seen }
	}
	immediate = {
		set_variable = cwp_ccw_opening_seen
		c:PRC.ruler ?= { save_scope_as = communist_leader }
		c:ROC.ruler ?= { save_scope_as = nationalist_leader }
	}
	option = {
		name = chinese_civil_war.1.a
		default_option = yes
		if = {
			limit = { c:PRC ?= this }
			add_modifier = { name = chinese_civil_war_yanan_network years = 3 is_decaying = yes }
		}
		else_if = {
			limit = { c:ROC ?= this }
			add_modifier = { name = chinese_civil_war_nanjing_exhaustion years = 3 is_decaying = yes }
		}
	}
}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	trigger = {
		c:RUS ?= this

		NOT = { has_global_variable = chinese_civil_war_manchuria_resolved }
	}
	option = {
		name = chinese_civil_war.2.a
		default_option = yes
		chinese_civil_war_transfer_manchuria_to_prc = yes

		c:ROC ?= { trigger_event = { id = chinese_civil_war.13 popup = yes } }
	}
	option = {
		name = chinese_civil_war.2.b
		set_global_variable = chinese_civil_war_manchuria_resolved
		c:PRC ?= {
			add_modifier = { name = chinese_civil_war_soviet_advisors years = 2 is_decaying = yes }
			change_global_variable = { name = prc_chinese_civil_war_score add = 2 }
		}
		je:je_chinese_civil_war ?= { add_progress = { value = -2 name = je_chinese_civil_war_score_bar } }

	option = {
		name = chinese_civil_war.2.c

		c:ROC ?= {
			add_modifier = { name = chinese_civil_war_hold_the_line years = 2 is_decaying = yes }

		}
		je:je_chinese_civil_war ?= { add_progress = { value = 3 name = je_chinese_civil_war_score_bar } }
	}
