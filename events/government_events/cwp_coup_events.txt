# Start Event
cwp_coup_events.1 = {
	type = country_event
	placement = root

	duration = 3

	title = cwp_coup_events.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_coup_events.1.desc_1
				trigger = { var:cwp_coup_reason_variable ?= 1 }
			}
			triggered_desc = {
				desc = cwp_coup_events.1.desc_2
				trigger = { var:cwp_coup_reason_variable ?= 2 }
			}
			triggered_desc = {
				desc = cwp_coup_events.1.desc_3
				trigger = { var:cwp_coup_reason_variable ?= 3 }
			}
			triggered_desc = {
				desc = cwp_coup_events.1.desc_4
				trigger = { var:cwp_coup_reason_variable ?= 4 }
			}
			triggered_desc = {
				desc = cwp_coup_events.1.desc_8
				trigger = { always = yes }
			}
		}
	}
	flavor = cwp_coup_events.1.flavor

	is_popup = yes

	gui_window = event_window_1char_tabloid

	left_icon = scope:cwp_coup_character_leader

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	trigger = {
		NOT = { has_journal_entry = je_cwp_coup }
		NOT = { has_free_government_reform = yes }
		NOT = { has_variable = cwp_coup_event_active_variable }
		NOT = { has_variable = cwp_coup_cooldown_variable }

		any_scope_character = {
			OR = {
				has_role_of_type = general
				has_role_of_type = admiral
			}

			loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD

			cwp_commander_coup_strength >= {
				value = owner.cwp_country_coup_resistance
				multiply = 0.9
			}

			cwp_character_has_valid_reason_for_coup = yes
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:cwp_coup_interest_group }
			NOT = { exists = scope:cwp_coup_character_leader }
			scope:cwp_coup_character_leader ?= { is_character_alive = no }
			scope:cwp_coup_character_leader ?= { is_in_exile_pool = yes }
		}
	}

	immediate = {
		set_variable = {
			name = cwp_coup_event_active_variable
			days = 90
		}

		ordered_scope_character = {
			limit = {
				OR = {
					has_role_of_type = general
					has_role_of_type = admiral
				}

				loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD

				cwp_commander_coup_strength >= {
					value = owner.cwp_country_coup_resistance
					multiply = 0.9
				}

				cwp_character_has_valid_reason_for_coup = yes
			}
			order_by = cwp_commander_coup_strength
			position = 0
			check_range_bounds = no

			set_variable = cwp_coup_character_leader_variable

			set_variable = {
				name = monthly_cached_units_share
				value = this.num_units_share
			}
			set_variable = {
				name = monthly_cached_commander_coup_strength
				value = this.cwp_commander_coup_strength
			}

			save_scope_as = cwp_coup_character_leader

			interest_group ?= {
				set_variable = cwp_coup_interest_group_variable
				save_scope_as = cwp_coup_interest_group
			}

			cwp_character_get_and_save_coup_reason = yes
		}

		if = {
			limit = { exists = currently_enacting_law }
			currently_enacting_law = { save_scope_as = cwp_coup_current_law_scope }
		}

		generate_2_common_name = {
			CULTURE_1 = primary_culture
			CULTURE_2 = primary_culture
		}
	}

	option = {
		name = cwp_coup_events.1.a

		add_journal_entry = { type = je_cwp_coup }

		scope:cwp_coup_character_leader ?= {
			add_modifier = {
				name = coup_supported
				days = short_modifier_time
			}
		}

		clear_generated_names = yes
	}

	option = {
		name = cwp_coup_events.1.b
		default_option = yes

		add_journal_entry = { type = je_cwp_coup }

		add_modifier = {
			name = coup_resisted
			days = short_modifier_time
		}

		clear_generated_names = yes
	}

	after = {
		show_as_tooltip = {
			custom_tooltip = cwp_if_coup_is_successful_tt
			cwp_coup_victory_effect = yes
		}

		remove_variable = cwp_coup_event_active_variable
	}
}


# For characters realising that they're getting marginalised
cwp_coup_events.2 = {
	type = country_event
	placement = root

	duration = 3

	title = cwp_coup_events.2.title
	desc = cwp_coup_events.2.desc
	flavor = cwp_coup_events.2.flavor

	event_image = { video = "votp_conspiring" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	# Cooldown is handled by cwp_coup_cooldown_variable

	trigger = {
		# triggered by je_cwp_coup
		any_scope_character = {
			has_variable = cwp_coup_character_leader_variable
			has_variable = monthly_cached_units_share
			
			trigger_if = {
				limit = { has_variable = monthly_cached_units_share }
				this.num_units_share <= {
					value = this.var:monthly_cached_units_share # Their amount of controlled units has declined by at least 20%
					multiply = 0.8
				}
			}
			
			NOT = { has_variable = cwp_coup_cooldown_variable }
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:cwp_coup_character_leader }
			NOT = { has_journal_entry = je_cwp_coup }
		}
	}

	immediate = {
		random_interest_group = {
			limit = { has_variable = cwp_coup_interest_group_variable }
			save_scope_as = cwp_coup_interest_group
		}
		
		random_scope_character = {
			limit = {
				has_variable = cwp_coup_character_leader_variable
				this.num_units_share <= {
					value = this.var:monthly_cached_units_share # Their amount of controlled units has declined by at least 20%
					multiply = 0.8
				}
			}
			save_scope_as = cwp_coup_character_leader

			set_variable = {
				name = cwp_coup_cooldown_variable
				days = short_modifier_time
			}
		}

		generate_2_common_name = {
			CULTURE_1 = primary_culture
			CULTURE_2 = primary_culture
		}
	}

	option = {
		name = cwp_coup_events.2.a
		default_option = yes
		
		scope:cwp_coup_character_leader.interest_group ?= {
			add_modifier = {
				name = modifier_sabotaged_military_forces
				days = normal_modifier_time
				multiplier = 2
			}
		}
		
		custom_tooltip = {
			text = cwp_coup_progress_bar_add_50_tt
			je:je_cwp_coup ?= {
				add_progress = {
					value = 50
					name = je_cwp_coup_progress_bar
				}
			}
		}

		clear_generated_names = yes
	}

	option = {
		name = cwp_coup_events.2.b

		scope:cwp_coup_character_leader.interest_group ?= {
			add_modifier = {
				name = modifier_sabotaged_military_forces
				days = normal_modifier_time
				multiplier = 4
			}
		}
		
		add_radicals = {
			pop_type = officers
			value = large_radicals
		}
		
		add_radicals = {
			pop_type = soldiers
			value = large_radicals
		}

	}

# Successful Coup
cwp_coup_events.3 = {
	type = country_event
	placement = scope:cwp_pronunciamento_state

	duration = 3

	title = cwp_coup_events.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cwp_coup_events.3.desc_1
				trigger = { exists = scope:relevant_lobby }
			}
			triggered_desc = {
				desc = cwp_coup_events.3.desc_2
				trigger = { always = yes }
			}
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = cwp_coup_events.3.flavor_1
				trigger = { var:cwp_coup_reason_variable ?= 7 }
			}
			triggered_desc = {
				desc = cwp_coup_events.3.flavor_2
				trigger = { always = yes }
			}
		}
	}

	gui_window = event_window_1char_tabloid

	left_icon = scope:cwp_coup_character_leader

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	trigger = { }

	cancellation_trigger = { NOT = { exists = scope:cwp_coup_character_leader } }

	immediate = {
		random_scope_character = {
			limit = { has_variable = cwp_coup_character_leader_variable }
			save_scope_as = cwp_coup_character_leader
			interest_group ?= { save_scope_as = cwp_coup_interest_group }
		}

		scope:cwp_coup_character_leader = {
			if = {
				limit = { exists = commander_military_formation }
				commander_military_formation ?= {
					if = {
						limit = { exists = current_hq }
						current_hq ?= {	# Case 1: Commander has a formation and is stationed in a HQ
							region ?= {
								if = {	# Case 1a: Commander has a formation and is stationed in a HQ; ROOT ownes at least one state in this HQ
									limit = { any_scope_state = { owner = ROOT } }
									cwp_coup_get_owned_pronunciamento_state = yes
								}
								else = { # Case 1b: Commander has a formation and is stationed in a HQ; ROOT does not own at least one state in this HQ
									cwp_coup_get_unowned_pronunciamento_state = yes
								}
							}
						}
					}
					else_if = {
						limit = { exists = front }
						front ?= {	# Case 2: Commander has a formation and is either on or travelling to a front
							if = {	# Case 2a: Commander has a formation and is either on or travelling to a front; ROOT owns at least one state on this front
								limit = { any_scope_state = { owner = ROOT } }
								cwp_coup_get_owned_pronunciamento_state = yes
							}
							else_if = {	# Case 2b: Commander has a formation and is either on or travelling to a front; any allied country owns at least one state on this front
								limit = { any_scope_state = { owner = { is_diplomatic_play_ally_of = ROOT } } }
								cwp_coup_get_allied_pronunciamento_state = yes
							}
							else = { # Case 2c: Commander has a formation and is either on or travelling to a front; no allied country owns at least one state on this front
								ordered_scope_state = {
									order_by = gdp
									position = 0
									check_range_bounds = no
									save_scope_as = cwp_coup_pronunciamento_state
								}
							}
						}
					}
					else_if = {
						limit = { exists = home_hq }
						home_hq ?= {	# Case 3: Commander has a formation; somehow doesn't have either a front or a current HQ
							region ?= {
								if = {	# Case 3a: Commander has a formation; somehow doesn't have either a front or a current HQ; ROOT ownes at least one state in the home HQ
									limit = { any_scope_state = { owner = ROOT } }
									cwp_coup_get_owned_pronunciamento_state = yes
								}
								else = {	# Case 3b: Commander has a formation; somehow doesn't have either a front or a current HQ; ROOT does not own at least one state in the home HQ
									cwp_coup_get_unowned_pronunciamento_state = yes
								}
							}
						}
					}
					else = {	# Case 4: Commander has a formation; does not have a current HQ, front, or home HQ
						ROOT.capital = { save_scope_as = cwp_coup_pronunciamento_state }
					}
				}
			}
			else = {	# Case 5: Commander does not have a formation but is still somehow valid
				ROOT.capital = { save_scope_as = cwp_coup_pronunciamento_state }
			}
		}

		scope:cwp_coup_interest_group = {
			ordered_preferred_law = {
				limit = {
					law_is_available = yes
					law_is_valid_for_ig_petition = yes
					NOT = { ROOT = { has_law = prev.type } }
				}
				position = 0
				check_range_bounds = no
				save_scope_as = cwp_coup_desired_law
			}
		}

		random_political_lobby = {
			limit = { has_variable = cwp_coup_lobby_variable }
			save_scope_as = relevant_lobby
		}

		set_variable = {
			name = cwp_coup_event_active_variable
			days = 90
		}
	}

	option = {
		name = cwp_coup_events.3.a
		default_option = yes

		trigger = {
			scope:cwp_coup_character_leader ?= { NOT = { has_modifier = modifier_preempetive_coup } }
		}

		ai_chance = {
			base = 25
		}

		custom_tooltip = cwp_coup_victory_tt

		cwp_coup_victory_effect = yes
	}

	option = {
		name = cwp_coup_events.3.b
		default_option = yes
		highlighted_option = yes

		trigger = {
			OR = {
				custom_tooltip = {
					text = cwp_preemptive_coup_tt
					scope:cwp_coup_character_leader = { has_modifier = modifier_preempetive_coup }
				}
				root.cwp_country_coup_resistance >= {
					value = scope:cwp_coup_character_leader.cwp_commander_coup_strength
					multiply = 0.9
				}
			}
		}

		ai_chance = {
			base = 100	# If the AI can avert a coup, then it will try to
		}

		scope:cwp_pronunciamento_state ?= { add_devastation = 25 }

		random_list = {
			1 = {
				modifier = {
					add = { value = scope:cwp_coup_character_leader.cwp_commander_coup_strength }
					min = 0
				}

				custom_tooltip = cwp_coup_victory_tt

				cwp_coup_victory_effect = yes
			}
			1 = {
				modifier = {
					add = { value = root.cwp_country_coup_resistance }
					min = 0
				}

				custom_tooltip = cwp_coup_crushed_tt

				ROOT = { save_scope_as = cwp_coup_country }

				post_notification = failed_coup_in_country

				if = {
					limit = {
						has_variable = cwp_coup_sponsor
						
						var:cwp_coup_sponsor ?= { is_country_alive = yes }
						
						var:cwp_coup_sponsor ?= {
							has_diplomatic_pact = {
								who = ROOT
								type = orchestrate_coup
							}
						}
					}

					var:cwp_coup_sponsor ?= { post_notification = failed_orchestrated_coup_in_country }

					create_diplomatic_catalyst = {
						type = catalyst_orchestrate_coup_failed
						target = var:cwp_coup_sponsor
					}
				}

				scope:cwp_pronunciamento_state ?= {
					if = {
						limit = { scope:cwp_coup_character_leader.popularity >= 0 }
						add_radicals_in_state = {
							interest_group = scope:cwp_coup_interest_group
							value = {
								add = scope:cwp_coup_character_leader.popularity
								divide = 100
								max = 0.75
							}
						}
					}
					else = {
						add_radicals_in_state = { value = small_radicals }
					}
				}

				add_radicals = {
					pop_type = soldiers
					value = {
						add = scope:cwp_coup_character_leader.commander_rank
						divide = owner.cwp_country_collected_army_commander_ranks
					}
				}

				add_radicals = {
					pop_type = officers
					value = {
						add = scope:cwp_coup_character_leader.commander_rank
						divide = owner.cwp_country_collected_army_commander_ranks
					}
				}

				if = {
					limit = { has_variable = cwp_coup_character_exile_variable }
					scope:cwp_coup_character_leader = {
						show_as_tooltip = { exile_character = yes }
						hidden_effect = { exile_character_with_martyrdom = yes }
					}
					remove_variable = cwp_coup_character_exile_variable
				}
				else_if = {
					limit = { has_variable = cwp_coup_character_retire_variable }
					scope:cwp_coup_character_leader = {
						if = {
							limit = { has_role_of_type = general }
							remove_character_role = general
						}

						if = {
							limit = { has_role_of_type = admiral }
							remove_character_role = admiral
						}
					}

					remove_variable = cwp_coup_character_retire_variable
				}

				change_global_variable = {
					name = failed_coups
					add = 1
				}
			}
		}
	}

	after = {
		cwp_coup_cleanup_effects = yes
	}
}
