# zz_cwp_fix_08_subject_types.txt
# Proposed patch for vanilla common/subject_types/00_subject_types.txt (subject_types is NOT in replace_paths).
# Fixes initializefrombookmark.cpp:1430 "Historical diplomatic pact dominion between GBR and CYP/HND/IRQ/NZL/SAF cannot be maintained and will break soon".
# In CWP those subjects are country_type = recognized (CAN/AST are colonial and had no warning). Vanilla dominion only accepts colonial subjects of minor rank under a higher-ranked overlord.
# Edits: valid_subject_country_types += recognized unrecognized; valid_subject_ranks += great_power major_power; overlord_must_be_higher_rank = no.
# Alternative with fewer side effects: change those five countries to country_type = colonial in common/country_definitions instead.

REPLACE:subject_type_dominion = {
	allow_change_country_flag = yes
	use_overlord_map_color = yes
	use_overlord_ruler = no
	annex_on_country_formation = no
	can_start_own_diplomatic_plays = yes
	breaks_if_subject_not_protected = yes
	join_overlord_wars = yes
	can_have_subjects = yes
	overlord_must_be_higher_rank = no
	overlord_must_be_same_country_type = no
	use_for_release_country = no
	gives_prestige_to_overlord = yes

	diplomatic_action = dominion

	autonomy_level = 2
	category = same_as_personal_union

	lower_autonomy_subject_type_alternatives = {
		subject_type_colony
		subject_type_puppet
	}

	same_autonomy_subject_type_alternatives = {
		subject_type_protectorate
	}

	valid_overlord_country_types = {
		recognized
		colonial
		company
	}

	valid_subject_country_types = {
		colonial
		recognized
		unrecognized
	}

	valid_overlord_ranks = {
		great_power
		major_power
		minor_power
	}

	valid_subject_ranks = {
		great_power
		major_power
		minor_power
		insignificant_power
		unrecognized_regional_power
		unrecognized_power
	}

	# Multiplies both subject_value and become_subject_value in AI strategies
	# Higher value means AI is more interested in having this type of subject but less interested in *being* this type of subject
	# Cannot return a value lower than MIN_SUBJECT_TYPE_VALUE
	ai_value = {
		value = 1.25
	}
}
