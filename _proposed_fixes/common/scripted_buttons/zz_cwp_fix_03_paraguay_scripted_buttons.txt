# CWP fix patch for vanilla common/scripted_buttons/03_paraguay_scripted_buttons.txt
# Each block below is a faithful copy of the vanilla top-level block with dead 1836-era references repaired (see _proposed_fixes/_notes/06_on_actions_decisions.md).

REPLACE:ybycui_iron_works_button = {
	name = "ybycui_iron_works_button"
	desc = "ybycui_iron_works_button_desc"

	visible = {
		owns_entire_state_region = STATE_BAJO_PARAGUAY
		NOT = {
			has_global_variable = ybycui_iron_works_global_var
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_BAJO_PARAGUAY
            any_scope_building = {
                is_building_type = building_iron_mine
                level >= 3
                occupancy >= 0.5
            }
            any_scope_building = {
                is_building_type = building_logging_camp
                level >= 2
                occupancy >= 0.5
            }
            any_scope_building = {
            	is_building_type = building_small_arms_manufactories
                level >= 2
                occupancy >= 0.5
            }
        }
	}

	effect = {
		trigger_event = { id = paraguay.3 popup = yes }
		set_global_variable = ybycui_iron_works_global_var
		custom_tooltip = choose_one_of_tt
		show_as_tooltip = {
			random_scope_state = {
				limit = {
					state_region = s:STATE_BAJO_PARAGUAY
				}
				add_modifier = {
					name = ybycui_iron_works_modifier
					days = very_long_modifier_time
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_BAJO_PARAGUAY
				}
				if = {
					limit = {
						NOT = {
							any_scope_building = {
								is_building_type = building_steelworks
							}
						}
					}
					create_building = {
						building = building_steelworks
						level = 2
					}
					add_modifier = {
						name = ybycui_steelworks
						days = very_long_modifier_time
					}
				}
				else = {
					add_modifier = {
						name = ybycui_steelworks
						days = very_long_modifier_time
						multiplier = 1.5
					}
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_BAJO_PARAGUAY
				}
				add_modifier = {
					name = ybycui_prisoner_workforce_modifier
					days = very_long_modifier_time
				}
				add_radicals_in_state = {
					value = small_radicals
				}
			}
		}
	}
}

REPLACE:good_health_prg_button = {
	name = "good_health_prg_button"
	desc = "good_health_prg_button_desc"

	visible = {
		owns_entire_state_region = STATE_BAJO_PARAGUAY
		NOT = {
			has_global_variable = improve_living_conditions_global_var
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_BAJO_PARAGUAY
			any_scope_building = {
				is_building_type = building_urban_center
				level >= 2
			}
		}
		OR = {
			has_law_or_variant = law_type:law_charitable_health_system
			has_law_or_variant = law_type:law_private_health_insurance
			has_law_or_variant = law_type:law_public_health_insurance
			any_country = {
				country_rank = rank_value:great_power
				relations:root >= relations_threshold:amicable
			}
		}
	}

	effect = {
		trigger_event = { id = paraguay.5 popup = yes }
		set_global_variable = improve_living_conditions_global_var
		if = {
			limit = {
				any_country = {
					OR = {
						is_in_geographic_region = geographic_region_europe
						is_in_geographic_region = geographic_region_north_america
					}
					has_technology_researched = medical_science
					relations:root >= relations_threshold:amicable
				}
			}
			random_country = {
				limit = {
					OR = {
						is_in_geographic_region = geographic_region_europe
						is_in_geographic_region = geographic_region_north_america
					}
					has_technology_researched = medical_science
					relations:root >= relations_threshold:amicable
				}
				save_scope_as = paraguay_doctors_country_scope
			}
		}
		else = {
			save_scope_as = paraguay_doctors_country_scope
		}
		show_as_tooltip = {
			random_scope_state = {
				limit = {
					state_region = s:STATE_BAJO_PARAGUAY
				}
				add_modifier = {
					name = vaccinations_prg_modifier
					days = very_long_modifier_time
				}
			}
		}
	}
}
