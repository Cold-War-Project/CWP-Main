# zz_cwp_fix_08_prestige_goods.txt
# Proposed patch for vanilla common/prestige_goods/00_prestige_goods.txt (prestige_goods is NOT in replace_paths).
# base_good keys that no longer exist in the CWP goods list are mapped to the CWP equivalent. Where there is no equivalent (silk, porcelain, dye, opium, ironclads, fine_art)
# the block is kept valid by pointing base_good at an unrelated CWP good and its possible trigger is set to always = no, so the prestige good can never be selected.

REPLACE:prestige_good_canton_porcelain = {
	possible = {
		always = no
	}
	base_good = glass
	prestige_bonus = 0.1
	texture = "gfx/interface/icons/goods_icons/prestige_goods/canton_porcelain_prestige.dds"
}
