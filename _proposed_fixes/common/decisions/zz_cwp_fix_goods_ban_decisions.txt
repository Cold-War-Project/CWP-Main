# CWP fix patch for vanilla common/decisions/goods_ban_decisions.txt
# Each block below is a faithful copy of the vanilla top-level block with dead 1836-era references repaired (see _proposed_fixes/_notes/06_on_actions_decisions.md).

REPLACE:ban_liquor = {
	is_shown = {
		has_variable = prohibition_laws_var
		exists = market
		market.owner = root
		NOT = { is_banning_goods = g:liquor }
	}

	possible = {
		NOT = { has_law_or_variant = law_type:law_free_trade }
		exists = market
	}

	when_taken = {
		add_banned_goods = g:liquor
		add_modifier = {
			name = liqour_ban_authority
		}
	}

    ai_chance = {
		value = 0
		if = {
			limit = {
				OR = {
					has_strategy = ai_strategy_conservative_agenda
					has_strategy = ai_strategy_reactionary_agenda
					always = no
				}
			}
			add = 10
		}
    }
}

REPLACE:remove_opium_ban = {
	is_shown = {
		exists = market
		market.owner = root
		NOT = { has_journal_entry = je_opium_obsession }
		always = no
	}

	possible = {
		always = no
	}

	when_taken = {
		if = {
			limit = { has_modifier = opium_ban_authority }
			remove_modifier = opium_ban_authority
		}
		if = {
			limit = { has_modifier = china_welfare  }
			remove_modifier = china_welfare
		}
	}

   ai_chance = {
      value = 0
   }
}
