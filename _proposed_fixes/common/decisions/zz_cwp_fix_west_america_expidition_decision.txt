# CWP fix patch for vanilla common/decisions/west_america_expidition_decision.txt
# Each block below is a faithful copy of the vanilla top-level block with dead 1836-era references repaired (see _proposed_fixes/_notes/06_on_actions_decisions.md).

REPLACE:west_america_expedition = {
	is_shown = {
		NAND = {
			has_global_variable = buenaventura_river_disproven
			has_global_variable = oregon_trail_mapped
			has_global_variable = great_salt_lake_mapped
	    }
		always = no
		NOT = {  has_variable = on_expedition }
		OR = {
			is_ai = no
			NOT = { has_variable = ai_expedition_cooldown }
		}
	}

	possible = {
		OR = {
			has_interest_marker_in_region = sr:region_pacific_coast
			has_interest_marker_in_region = sr:region_great_plains
		}
		is_at_war = no
		has_technology_researched = antibiotics
		any_scope_character = {
			OR = {
				has_role_of_type = general
				has_role_of_type = admiral
			}
			count >= 3
		}
	}

	when_taken = {
		if = {
			limit = {
				is_player = yes
			}
			trigger_event = { id = expedition_events.200 popup = yes }
			set_variable = on_expedition
		}
		else = {
			set_variable = AI_EXPEDITION_WAMERICA
			ai_expedition_simulator_start = yes
		}
		hidden_effect = {
			set_variable = {
				name = ai_expedition_cooldown
				value = yes
				days = 1825
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				net_fixed_income > money_amount_multiplier_medium
				gold_reserves > 0
			}
			add = 10
		}
		if = {
			limit = {
				has_journal_entry = je_oregon
			}
			add = 500
		}
		if = {
			limit = {
				has_strategy = ai_strategy_armed_isolationism
			}
			multiply = 0
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}
	}
}
