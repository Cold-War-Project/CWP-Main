	category = movement_category_pan_national

	ideology = ideology_sovereignist_movement
	character_ideologies = {
		ideology_sovereignist_leader
		ideology_social_democrat
		ideology_humanitarian
	}

	can_cause_obstinance = yes

	creation_trigger = {
		c:BIC ?= this
		is_subject = yes
		any_scope_culture = {
			has_discrimination_trait_group = heritage_group_south_asian
			culture_current_fervor > 10
		}
		NOT = {
			any_primary_culture = {
				has_discrimination_trait_group = heritage_group_south_asian
			}
		}
		has_technology_researched = pan-nationalism
	}
	creation_weight = {
		value = 5000
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_sovereignist_leader
			has_ideology = ideology:ideology_humanitarian
			culture = { has_discrimination_trait_group = heritage_group_south_asian }
		}
	}
	character_support_weight = {
		value = 250
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_sovereignist_leader
					has_ideology = ideology:ideology_humanitarian
					has_ideology = ideology:ideology_social_democrat
				}
			}
			multiply = {
				value = 5
			}
		}
		if = {
			limit = {
				OR = {
					always = no
				}
			}
			multiply = {
				value = 10
			}
		}
	}

	pop_support_trigger = {
		culture = {
			has_discrimination_trait_group = heritage_group_south_asian
		}
	}

	pop_support_factors = {
		movement_support_national_awakening
		movement_support_cultural_fervor
		movement_support_high_urbanization
		movement_support_below_expected_sol
		movement_support_radicalization
		movement_support_cultural_discrimination
		movement_support_south_asian_heritage
		movement_support_academics
		movement_support_bureaucrats
		movement_support_clerks
		movement_support_capitalists
		movement_support_nationalism_tech
		movement_support_pan_nationalism_tech
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = {
			limit = {
				culture = {
					has_discrimination_trait_group = heritage_group_south_asian
				}
			}
			add = {
				value = 10
				desc = "POP_BASE_SUPPORT"
			}
			if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 30
					desc = "POP_ACADEMICS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 12
					desc = "POP_BUREAUCRATS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 15
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 6
					desc = "POP_CAPITALISTS"
				}
			}

			add = {
				desc = "URBAN_STATE"
				value = 10
				multiply = state.state_urbanization_rate
			}

			if = {
				limit = {
					standard_of_living < state.average_expected_sol
				}
				add = { # Increases likelihood of support if impoverished
					desc = "POP_POVERTY"
					value = state.average_expected_sol
					subtract = standard_of_living
					max = 10
				}
			}

			if = {
				limit = {
					state = {
						state_cultural_acceptance = {
							target = PREV.culture
							value < acceptance_status_5
						}
					}
				}
				add = { # From 1.21x at 79 to 2x at 0
					desc = "POP_DISCRIMINATED"
					value = 100
					subtract = pop_acceptance
					divide = 10
				}
			}
			else = { # Non-discriminated pops are less likely to form separatist movements
				multiply = {
					desc = "POP_ACCEPTANCE"
					value = 0.25
				}
			}

			multiply = { # Multiplied by 2x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				add = 1
			}

			if = {
				limit = {
					owner ?= {
						has_technology_researched = pan-nationalism
					}
				}
				multiply = {
					value = 1.5
					desc = "PAN_NATIONALISM_TECH"
				}
			}
			else_if = {
				limit = {
					owner ?= {
						has_technology_researched = nationalism
					}
				}
				multiply = {
					value = 1.25
					desc = "NATIONALISM_TECH"
				}
			}

			multiply = { # Range from x0.775 to 2.275
				desc = "POP_CULTURE_FERVOR"
				value = culture.culture_current_fervor
				subtract = 15
				multiply = 0.015
				add = 1
			}

			if = {
				limit = {
					culture ?= {
						culture_has_national_awakening = yes
					}
				}
				multiply = {
					value = 1.5
					desc = "POP_ACTIVE_NATIONAL_AWAKENING"
				}
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			add = {
				value = state_urbanization_rate
				multiply = 5
			}
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}

		forced_tags = {
			HND = {
				trigger = {
					always = yes
				}
				weight = 1000
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		forced_tags = {
			HND = {
				trigger = {
					always = yes
				}
				weight = 1000
			}
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= {
					is_subject = yes
				}
			}
			add = {
				desc = "INDIA_SUBJECT"
				if = {
					limit = {
						owner ?= {
							OR = {
								is_subject_type = subject_type_colony
								is_subject_type = subject_type_chartered_company
							}
						}
					}
					value = 0.25
				}
				if = {
					limit = {
						owner ?= {
							is_subject_type = subject_type_dominion
						}
					}
					value = 0.10
				}
			}
		}
		if = {
			limit = {
				owner ?= {
					is_subject = yes
				}
			}
			add = {
				desc = "LIBERTY_DESIRE"
				value = owner.liberty_desire
				divide = 500
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_india_pan_national
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		if = {
			limit = {
				this.days_since_movement_defeated <= very_long_modifier_time
			}
			add = { # (x-y)/y - ([1, 7300]-7300)/7300
				desc = "movement_suppressed_after_civil_war"
				value = this.days_since_movement_defeated
				subtract = very_long_modifier_time
				divide = {
					value = very_long_modifier_time
				}
			}
		}
	}
}
