REPLACE:amendment_cult_of_personality = {
	parent = law_autocracy

	allowed_laws = {
		law_autocracy
		law_oligarchy
		law_single_party_state
	}

	modifier = {
		country_legitimacy_headofstate_add = 10
		country_legitimacy_base_add = -5
		country_authority_mult = 0.10
		country_bureaucracy_mult = -0.10
	}

	possible = {
		always = yes
	}

	would_sponsor = {
		OR = {
			has_ideology = ideology:ideology_authoritarian
			has_ideology = ideology:ideology_reactionary
			leader = {
				OR = {
					has_ideology = ideology:ideology_authoritarian
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_integralist
					has_ideology = ideology:ideology_utopian
				}
			}
		}
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			law_stance = {
				law = law_type:law_autocracy
				value < neutral
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 1
}
