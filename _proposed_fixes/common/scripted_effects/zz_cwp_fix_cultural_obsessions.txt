# CWP fix: second_amendment_add_obsessions_effect was defined both by the mod file 00_starting_cultural_obsessions.txt and by vanilla
# 00_victoria_ip4_scripted_effects.txt (gamedatabase 'Duplicated key' error). The CWP definition (cu:american) is moved here and
# applied with REPLACE: so it overrides the vanilla one (which still references the removed cultures yankee/dixie).
# Requires the companion overwrite of common/scripted_effects/00_starting_cultural_obsessions.txt (emptied).

REPLACE:second_amendment_add_obsessions_effect = {
	if = { # Second Amendment - add obsessions
		limit = {
			owner ?= { country_has_primary_culture = cu:american }
			NOT = { cu:american = { has_cultural_obsession = small_arms } }
		}
		cu:american = { add_cultural_obsession = small_arms }
	}
}
