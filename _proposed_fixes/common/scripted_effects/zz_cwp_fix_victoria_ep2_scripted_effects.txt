REPLACE:JAP_character_generate_kishu_daimyo = { # Owner = country, ROOT = predecessor
	owner = {
		if = {
			limit = {
				s:STATE_KANSAI ?= {
					var:daimyo_chain_id = 0
				}
				game_date >= 1820.6.8
			}
		}
		else_if = {
			limit = {
				s:STATE_KANSAI ?= {
					var:daimyo_chain_id = 1
				}
			}
			if = {
				limit = {
					game_date >= 1846.7.16
					NOT = {
						any_scope_character = {
							always = no
						}
					}
				}
			}
			else_if = {
				limit = {
					game_date >= 1844.3.1
				}
			}
		}
		else_if = {
			limit = {
				s:STATE_KANSAI ?= {
					var:daimyo_chain_id = 2
				}
				game_date >= 1844.3.1
				NOT = {
					any_scope_character = {
						always = no
					}
				}
			}
		}
		else = { # Kishu people were typically Landowners
			create_character = {
				last_name = Tokugawa
				noble = yes
				culture = cu:japanese
				role = character_role_magnate
				role = character_role_politician
				home_region = STATE_KANSAI
				interest_group = ig_landowners
				on_created = {
					set_variable = house_tokugawa
					if = {
						limit = {
							NOT = {
								has_global_variable = japan_daimyo_abolished
							}
						}
						set_variable = { name = daimyo_var value = s:STATE_KANSAI }
					}
					designate_as_shinpan_daimyo = yes
				}
			}
		}
		s:STATE_KANSAI ?= {
			change_variable = { name = daimyo_chain_id add = 1 }
		}
	}
}

						always = no

				}

		}
		else = {

			role = character_role_politician

		}
	}
}
