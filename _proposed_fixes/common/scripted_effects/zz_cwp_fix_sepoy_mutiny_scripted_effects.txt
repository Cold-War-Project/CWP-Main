REPLACE:sepoy_mutiny_bombay_effect_medium = {
	# Medium Effect
	# Destroy a random non-capital barracks
	if = {
		limit = {
			any_scope_state = {
				region = sr:region_south_india
				NOT = {
					state_region = s:STATE_BOMBAY
				}
				any_scope_building = {
					is_building_type = building_barrack
				}
			}
		}
		random_scope_state = {
			limit = {
				region = sr:region_south_india
				NOT = {
					state_region = s:STATE_BOMBAY
				}
				any_scope_building = {
					is_building_type = building_barrack
				}
			}
			remove_building = building_barrack
			add_modifier = {
				name = sepoy_mutiny_barracks_destruction_modifier
				days = normal_modifier_time
			}
			add_radicals_in_state = {
				value = medium_radicals
			}
			save_scope_as = bengal_sepoy_barracks_scope
		}
	}
	# Add negative modifiers to every other barracks
	if = {
		limit = {
			any_scope_state = {
				region = sr:region_south_india
				NOT = {
					scope:bengal_sepoy_barracks_scope ?= this
				}
			}
		}
		every_scope_state = {
			limit = {
				region = sr:region_south_india
				NOT = {
					scope:bengal_sepoy_barracks_scope ?= this
				}
			}
			add_modifier = {
				name = sepoy_mutiny_barracks_unrest_modifier
				days = normal_modifier_time
			}
			add_radicals_in_state = {
				value = medium_radicals
			}
		}
	}
	# Maratha independence war diplomatic play
	if = {
		limit = {
			always = no
			NOT = {
				any_diplomatic_play = {
					target = { always = no }
					initiator = { c:BIC ?= this }
					is_diplomatic_play_type = dp_secession
				}
			}
		}
		# Add Britain
		random_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_sepoy_mutiny
				target = {
					always = no
				}
			}
			if = {
				limit = {
					initiator = {
						is_subject = yes
					}
				}
				add_initiator_backers = { initiator.top_overlord }
			}
		}
		# Add wargoals
		random_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_sepoy_mutiny
				target = {
					always = no
				}
			}
			# Bombay State
			if = {
				limit = {
					exists = s:STATE_BOMBAY.region_state:BIC
				}
			}
		}
	}
}
