REPLACE:STATE_SOUTH_CAMEROON_state_name_assign = {
    if = {
        # You only own Fernando Po
        limit = {
            owner = {
                p:x902089.state.owner ?= this
                p:x902089.state ?= {
                    NOT = {
                        num_provinces > 1
                    }
                }
            }
        }
        if = {
            limit = {
                owner = {
                    any_primary_culture = {
                        has_discrimination_trait = language_hispanophone
                    }
                }
            }
            set_state_name = STATE_SOUTH_CAMEROON_fernando_po_spanish
            set_hub_name = {
                type = farm
                name = HUB_NAME_STATE_SOUTH_CAMEROON_farm_spanish
            }
        }
        else_if = {
            limit = {
                owner = {
                    any_primary_culture = {
                        has_discrimination_trait = language_anglophone
                    }
                }
            }
            set_state_name = STATE_SOUTH_CAMEROON_fernando_po_english
            set_hub_name = {
                type = farm
                name = HUB_NAME_STATE_SOUTH_CAMEROON_farm_english
            }
        }
    }
    else_if = {
        # You only own Sao Tome
        limit = {
            owner = {
                p:xC02050.state.owner ?= this
                p:xC02050.state ?= {
                    NOT = {
                        num_provinces > 1
                    }
                }
            }
        }
        if = {
            limit = {
                owner = {
                    any_primary_culture = {
                        has_discrimination_trait = language_lusophone
                    }
                }
            }
            set_state_name = STATE_SOUTH_CAMEROON_sao_tome_portuguese
        }
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = { has_discrimination_trait = language_germanophone }
                NOT = { country_has_primary_culture = cu:french }
            }
        }

        set_state_name = STATE_SOUTH_CAMEROON_german

        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_SOUTH_CAMEROON_city_german
        }
        set_hub_name = {
            type = port
            name = HUB_NAME_STATE_SOUTH_CAMEROON_port_german
        }
        set_hub_name = {
            type = mine
            name = HUB_NAME_STATE_SOUTH_CAMEROON_mine_german
        }
    }
    else_if = {
        limit = {
            owner = {
                NOT = { any_primary_culture = { has_discrimination_trait = language_germanophone } }
                any_primary_culture = { has_discrimination_trait = language_francophone }
            }
        }

        set_state_name = STATE_SOUTH_CAMEROON_french

        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_SOUTH_CAMEROON_city_french
        }
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = language_lusophone
                }
            }
        }

        set_hub_name = {
            type = port
            name = HUB_NAME_STATE_SOUTH_CAMEROON_port_portuguese
        }
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = language_hispanophone
                }
            }
        }

        set_hub_name = {
            type = farm
            name = HUB_NAME_STATE_SOUTH_CAMEROON_farm_spanish
        }
    }
    else = { # By putting an else at the end that resets the names we ensure that the names are switched back to their defaults if anyone else takes the states
        reset_state_name = yes
        reset_hub_names = yes
    }
}

REPLACE:STATE_ZULULAND_state_name_assign = {
    if = {
        limit = {
            owner = {
                NOT = { country_has_primary_culture = cu:english }
                any_primary_culture = { 
                    has_discrimination_trait = language_dutch 
                }
            }
        }

        set_state_name = STATE_ZULULAND_boer

        set_hub_names = boer
    }
    else_if = {
        limit = {
            owner = {
                OR = {
                    c:GBR ?= this
                    c:SAF ?= this
                    AND = {
                        exists = c:GBR
                        exists = top_overlord
                        top_overlord = c:GBR
                    }
                }
            }
        }

        set_state_name = STATE_ZULULAND_boer
        set_hub_names = british
    }
    else = { # By putting an else at the end that resets the names we ensure that the names are switched back to their defaults if anyone else takes the states
        reset_state_name = yes
        reset_hub_names = yes
    }
}
