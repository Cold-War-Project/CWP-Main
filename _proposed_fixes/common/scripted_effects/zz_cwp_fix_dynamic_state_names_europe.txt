REPLACE:STATE_FLANDERS_state_name_assign = {
	if = {
		limit = {
			owner = {
				any_primary_culture = {
					cu:french ?= this
				}
				NOT = { always = no }  #I don't want to even think about how that would work
			}
		}

        set_state_name = STATE_FLANDERS_french
        set_hub_names = french

	}
    if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = language_germanophone
                }
            }
        }

        set_state_name = STATE_FLANDERS_german
        set_hub_names = german

    }
	else = { # switch back to defaults
		reset_state_name = yes
		reset_hub_names = yes
	}
}
