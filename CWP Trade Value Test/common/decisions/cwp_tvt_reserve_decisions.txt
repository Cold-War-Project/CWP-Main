# CWP reserve limit test: four repeatable, player-only decisions.
# remove_modifier on a country without the modifier is a script error, and the decision tooltip repeats it
# hundreds of times, so every removal that may find nothing is guarded with has_modifier.
# Take "Show" first, then one of the "Set" decisions, let a week pass, and take "Show" again.

cwp_tvt_reserve_show_decision = {
	is_shown = {
		is_player = yes
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		debug_log = "CWP_TVT RESERVE;[TimeKeeper.GetCurrentDate.GetString];[THIS.GetCountry.GetNameNoFormatting];gdp=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_gdp')|0];base=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_base')|0];limit=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_limit')|0];ratio=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_limit_ratio')|4];reserves=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserves')|0];scaled=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserves_scaled')|4];target_imports=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_target_imports')|0];mult_imports=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_mult_imports')|4];mult_fixed=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_mult_fixed')|4]"
		trigger_event = { id = cwp_tvt_reserve.1 }
	}

	ai_chance = {
		value = 0
	}
}

cwp_tvt_reserve_imports_decision = {
	is_shown = {
		is_player = yes
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		if = {
			limit = { has_modifier = cwp_tvt_reserve_limit }
			remove_modifier = cwp_tvt_reserve_limit
		}
		add_modifier = {
			name = cwp_tvt_reserve_limit
			multiplier = cwp_tvt_reserve_mult_imports
		}
		debug_log = "CWP_TVT RESERVE SET;imports;target=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_target_imports')|0];multiplier=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_mult_imports')|4]"
	}

	ai_chance = {
		value = 0
	}
}

cwp_tvt_reserve_fixed_decision = {
	is_shown = {
		is_player = yes
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		if = {
			limit = { has_modifier = cwp_tvt_reserve_limit }
			remove_modifier = cwp_tvt_reserve_limit
		}
		add_modifier = {
			name = cwp_tvt_reserve_limit
			multiplier = cwp_tvt_reserve_mult_fixed
		}
		debug_log = "CWP_TVT RESERVE SET;fixed;target=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_target_fixed')|0];multiplier=[THIS.GetCountry.MakeScope.ScriptValue('cwp_tvt_reserve_mult_fixed')|4]"
	}

	ai_chance = {
		value = 0
	}
}

cwp_tvt_reserve_clear_decision = {
	is_shown = {
		is_player = yes
		has_modifier = cwp_tvt_reserve_limit
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		remove_modifier = cwp_tvt_reserve_limit
		debug_log = "CWP_TVT RESERVE SET;cleared"
	}

	ai_chance = {
		value = 0
	}
}
