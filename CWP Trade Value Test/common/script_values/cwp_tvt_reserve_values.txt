# CWP reserve limit test. Country scope.
# Question: can a country's gold reserve limit be set to a figure that does not follow GDP?

# What the engine should use before modifiers: NEconomy GOLD_RESERVE_LIMIT_FACTOR (0.2) x annual GDP.
cwp_tvt_reserve_base = {
	value = gdp
	multiply = 0.2
	min = 1000
}

cwp_tvt_reserve_limit = {
	value = gold_reserves_limit
}

# Engine limit divided by the 20% of GDP figure. 1 means the base formula is right and no modifier applies.
cwp_tvt_reserve_limit_ratio = {
	value = gold_reserves_limit
	divide = cwp_tvt_reserve_base
}

cwp_tvt_reserves = {
	value = gold_reserves
}

cwp_tvt_reserves_scaled = {
	value = scaled_gold_reserves
}

# Target A: 26 weeks of the country's own imports (the "months of import cover" measure).
cwp_tvt_reserve_target_imports = {
	value = total_import_value
	multiply = 26
	min = 1000
}

# Target B: a fixed sum that ignores the economy altogether.
cwp_tvt_reserve_target_fixed = {
	value = 5000000
}

# Multiplier that turns the base into the target: target / base - 1. Negative when the target is below the base.
cwp_tvt_reserve_mult_imports = {
	value = cwp_tvt_reserve_target_imports
	divide = cwp_tvt_reserve_base
	subtract = 1
}

cwp_tvt_reserve_mult_fixed = {
	value = cwp_tvt_reserve_target_fixed
	divide = cwp_tvt_reserve_base
	subtract = 1
}
